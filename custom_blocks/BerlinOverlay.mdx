export const BerlinOverlay = ({ header, apiKey, height = 400 }) => {
  const initMap = (el) => {
    if (!el || el.dataset.inited) return;
    el.dataset.inited = "1";

    // SDK guard
    if (!window.H) {
      console.error("HERE SDK not loaded. Add mapsjs scripts to <head>.");
      el.innerHTML = '<div style="padding:8px;color:#b91c1c;">HERE SDK not loaded.</div>';
      return;
    }

    // Resolve API key: prop → global → meta
    const resolvedKey =
      apiKey ||
      window.HERE_API_KEY ||
      (document.querySelector('meta[name="here-apikey"]')?.content || "");

    if (!resolvedKey) {
      console.error("HERE API key missing. Provide via prop, window.HERE_API_KEY, or <meta name='here-apikey'>.");
      el.innerHTML = '<div style="padding:8px;color:#b91c1c;">HERE API key missing.</div>';
      return;
    }

    const H = window.H;

    // --- Boilerplate (aligned to your raw JS), but using RASTER base map ---
    const platform = new H.service.Platform({ apikey: resolvedKey });
    const defaultLayers = platform.createDefaultLayers();

    const map = new H.Map(
      el,
      defaultLayers.raster.normal.map,   // ✅ raster (not vector) avoids Tangram errors
      {
        center: new H.geo.Point(52.515, 13.405),
        zoom: 14,
        pixelRatio: window.devicePixelRatio || 1,
      }
    );

    // Resize like raw snippet
    el._onResize = () => map.getViewPort().resize();
    window.addEventListener("resize", el._onResize);

    // Interactions + UI
    new H.mapevents.Behavior(new H.mapevents.MapEvents(map));
    H.ui.UI.createDefault(map, defaultLayers);

    // --- Your overlay function (1:1 logic) ---
    function overlayHistoricalBerlin(map) {
      // Create a tile provider from images of historical Berlin
      var tileProvider = new H.map.provider.ImageTileProvider({
        // We have tiles only for zoom levels 12–15,
        // so on all other zoom levels only base map will be visible
        min: 12,
        max: 15,
        opacity: 0.5,
        getURL: function (column, row, zoom) {
          // If Berlin is not displayed, return a blank tile.
          if (
            (zoom == 12 && (row != 1343 || column != 2200)) ||
            (zoom == 13 &&
              (row < 2686 || column < 4400 || row > 2687 || column > 4401)) ||
            (zoom == 14 &&
              (row < 5372 || column < 8800 || row > 5375 || column > 8803)) ||
            (zoom == 15 &&
              (row < 10744 || column < 17601 || row > 10750 || column > 17607))
          ) {
            return "https://heremaps.github.io/maps-api-for-javascript-examples/custom-tile-overlay/tiles/blank.png";
          } else {
            // TMS format: /zoom/x(row)/y(column).png
            return (
              "https://heremaps.github.io/maps-api-for-javascript-examples/custom-tile-overlay/tiles/" +
              zoom +
              "/" +
              row +
              "/" +
              column +
              ".png"
            );
          }
        },
      });

      // Licensing/attribution
      tileProvider.getCopyrights = function () {
        return [
          {
            label:
              "Overlay derived from <a href='http://commons.wikimedia.org/wiki/File%3AMap_de_berlin_1789_%28georeferenced%29.jpg' target='_blank' rel='noopener noreferrer'>WikiMedia Commons</a>",
            alt: "Overlay Based on a WikiMedia Commons Image in the Public Domain",
          },
        ];
      };

      // Layer consuming our provider (semi-transparent)
      var overlayLayer = new H.map.layer.TileLayer(tileProvider, {
        opacity: 0.5,
      });

      // Add old Berlin to the map
      map.addLayer(overlayLayer);
    }

    // Main logic
    overlayHistoricalBerlin(map);

    // Keep references for cleanup
    el._map = map;
  };

  const cleanup = (el) => {
    if (el && el._map) {
      if (el._onResize) window.removeEventListener("resize", el._onResize);
      try { el._map.dispose(); } catch {}
      delete el._map;
      delete el._onResize;
      delete el.dataset.inited;
    }
  };

  return (
    <div className="flex justify-center">
      <div className="rounded-md p-6 m-4 w-full max-w-3xl shadow-md border border-gray-300 dark:bg-gray-800 dark:border-gray-600">
        <p className="text-lg font-bold mb-2">{header}</p>
        <p className="mb-4">
          Interactive HERE Map centered on Berlin with the <strong>1789 historical overlay</strong>.
        </p>
        <div
          ref={(el) => { if (el) initMap(el); else cleanup(el); }}
          style={{ width: "100%", height: `${height}px`, borderRadius: "8px" }}
        />
      </div>
    </div>
  );
};
