import React, { useEffect, useState } from "react";

export const MeterGroup = ({ 
  values = [], 
  max = 100, 
  labelPosition = "below", 
  showCards = false, 
  apiReference = false 
}) => {
  const [animatedValues, setAnimatedValues] = useState([]);
  const [showLabels, setShowLabels] = useState(false);
  const [dynamicValues, setDynamicValues] = useState([]);

  useEffect(() => {
    if (apiReference) {
      // Find all API request types in the sidebar
      const apiMethods = Array.from(document.querySelectorAll(".rm-APIMethod"))
        .map(el => el.innerText.toUpperCase())  // Extract the request method (GET, POST, etc.)
        .filter(method => ["GET", "POST", "PATCH", "DELETE", "PUT"].includes(method));  // Filter valid methods

      // Count occurrences of each method
      const methodCounts = apiMethods.reduce((acc, method) => {
        acc[method] = (acc[method] || 0) + 1;
        return acc;
      }, {});

      // Convert to a format that MeterGroup understands
      const newValues = [
        { label: "GET", value: methodCounts.GET || 0, meterBackgroundColor: "#4285F4", cardBackgroundColor: "#D2E3FC", textColor: "#fff", cardTextColor: "#174EA6", icon: "fas fa-arrow-down" },
        { label: "POST", value: methodCounts.POST || 0, meterBackgroundColor: "#34A853", cardBackgroundColor: "#DFF6DD", textColor: "#fff", cardTextColor: "#0D652D", icon: "fas fa-arrow-up" },
        { label: "PATCH", value: methodCounts.PATCH || 0, meterBackgroundColor: "#FBBC05", cardBackgroundColor: "#FEF7E0", textColor: "#fff", cardTextColor: "#B06000", icon: "fas fa-edit" },
        { label: "DELETE", value: methodCounts.DELETE || 0, meterBackgroundColor: "#EA4335", cardBackgroundColor: "#FCE8E6", textColor: "#fff", cardTextColor: "#A50E0E", icon: "fas fa-trash" },
        { label: "PUT", value: methodCounts.PUT || 0, meterBackgroundColor: "#A142F4", cardBackgroundColor: "#EADCFD", textColor: "#fff", cardTextColor: "#5E35B1", icon: "fas fa-sync-alt" }
      ].filter(item => item.value > 0);  // Remove items with 0 value

      setDynamicValues(newValues);
    }
  }, [apiReference]);

  // Select either the provided values OR the dynamically generated ones
  const meterValues = apiReference ? dynamicValues : values;

  useEffect(() => {
    setAnimatedValues(meterValues.map(() => 0));

    const interval = setInterval(() => {
      setAnimatedValues(prev => prev.map((val, i) => Math.min(val + 2, meterValues[i]?.value || 0)));
    }, 50);

    const labelTimeout = setTimeout(() => {
      setShowLabels(true);
    }, 500);

    return () => {
      clearInterval(interval);
      clearTimeout(labelTimeout);
    };
  }, [meterValues]);

  return (
    <div className="w-full flex flex-col items-start">
      {showCards && (
        <div className="flex gap-4 mb-3 w-full">
          {meterValues.map((item, index) => (
            <div 
              key={index} 
              className="shadow-md rounded-lg p-4 flex flex-col items-center w-1/4 border"
              style={{ backgroundColor: item.cardBackgroundColor || item.meterBackgroundColor }}
            >
              <span className="text-sm" style={{ color: item.cardTextColor || item.textColor }}>{item.cardLabel || item.label}</span>
              <span className="text-lg font-bold" style={{ color: item.cardTextColor || item.textColor }}>{item.value}%</span>
              {item.icon && <i className={`${item.icon} text-2xl mt-2`} style={{ color: item.cardTextColor || item.textColor }}></i>}
            </div>
          ))}
        </div>
      )}
      <div className="relative w-full bg-gray-200 rounded-md overflow-hidden h-6 flex items-center">
        {meterValues.map((item, index) => (
          <div 
            key={index} 
            className="h-full transition-all duration-500 flex items-center justify-center relative" 
            style={{ width: `${(animatedValues[index] / max) * 100}%`, backgroundColor: item.meterBackgroundColor }}
          >
            {labelPosition === "inside" && showLabels && (
              <span className="text-xs font-semibold absolute left-2" style={{ color: item.textColor }}>{item.label}</span>
            )}
            {labelPosition === "overlay" && showLabels && (
              <span className="text-xs font-semibold absolute inset-0 flex items-center justify-center" style={{ color: item.textColor }}>
                {item.label} ({item.value}%)
              </span>
            )}
          </div>
        ))}
      </div>
    </div>
  );
};

<MeterGroup apiReference={true} showCards={true} max={100} labelPosition="overlay" />
