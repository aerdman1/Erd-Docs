import React, { useEffect, useState } from "react";
import ReactDOM from "react-dom";

export const AlertBannerPortal = ({
  message = "YOUR CONTENT GOES HERE.",
  bannerColor = "#ffcc00",          // Default bright yellow background
  bannerTextColor = "#050910",        // Dark text and border color
  targetSelector = ".Header-bottom2eLKOFXMEmh5.rm-Header-bottom" // Change this to your desired target element
}) => {
  const [container, setContainer] = useState(null);

  useEffect(() => {
    // Find the target element in the DOM
    const targetEl = document.querySelector(targetSelector);
    if (targetEl) {
      // Create a container div to hold the banner
      const div = document.createElement("div");
      // Insert the container right after the target element
      targetEl.parentNode.insertBefore(div, targetEl.nextSibling);
      setContainer(div);
      return () => {
        // Clean up when unmounting
        if (div.parentNode) {
          div.parentNode.removeChild(div);
        }
      };
    }
  }, [targetSelector]);

  if (!container) return null; // Wait until the container is ready

  return ReactDOM.createPortal(
    <div>
      <div
        className="alert-banner"
        style={{
          background: bannerColor,
          color: bannerTextColor,
          padding: "10px",
          textAlign: "center",
          fontSize: "14px",
          fontWeight: "bold",
          border: "2px solid " + bannerTextColor,
          borderRadius: "5px",
          marginBottom: "20px"
        }}
      >
        {message}
      </div>
      <style>{`
        @keyframes bounceIn {
          0% { transform: scale(0.5); opacity: 0; }
          60% { transform: scale(1.1); opacity: 1; }
          100% { transform: scale(1); }
        }
        .alert-banner {
          animation: bounceIn 0.6s ease-out;
        }
      `}</style>
    </div>,
    container
  );
};
