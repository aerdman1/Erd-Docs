import React, { useState } from 'react';

export const LiveCodeEditor = () => {
  // The code state controls both the textarea and the preview.
  const [code, setCode] = useState(`// Edit this code:
function greet() {
  console.log("Hello, world!");
}

greet();`);

  return (
    <div>
      <h3>Live Code Editor</h3>
      <textarea
        value={code}
        onChange={(e) => setCode(e.target.value)}
        style={{
          width: "100%",
          height: "150px",
          fontFamily: "monospace",
          fontSize: "14px",
          padding: "10px",
          border: "1px solid #ccc",
          borderRadius: "4px"
        }}
      />
      <h4>Preview:</h4>
      <pre
        style={{
          background: "#f5f5f5",
          padding: "10px",
          borderRadius: "4px",
          fontFamily: "monospace",
          whiteSpace: "pre-wrap"
        }}
      >
        {code}
      </pre>
    </div>
  );
};

<LiveCodeEditor />
