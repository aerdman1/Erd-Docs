import React, { useState } from "react";

export const PageModal_Simple = ({ 
  title = "Notice", 
  message = "This is your message. Customize it as needed." 
}) => {
  const [isOpen, setIsOpen] = useState(true);

  // Function to close modal
  const closeModal = () => {
    setIsOpen(false);
  };

  // If modal is closed, don't render anything
  if (!isOpen) return null;

  return (
    <div className="fixed z-50 inset-0 flex items-center justify-center bg-black bg-opacity-50 p-4">
      <div className="bg-white rounded-lg shadow-xl w-full max-w-md transform transition-all">
        
        {/* Modal Header (Title + Close Button) */}
        <div className="flex items-center justify-between p-4 border-b">
          <h3 className="text-lg font-semibold text-gray-900">{title}</h3>
          <button 
            onClick={closeModal} 
            className="text-gray-500 hover:text-gray-700 transition"
            aria-label="Close"
          >
            ✖
          </button>
        </div>

        {/* Modal Body (Message) */}
        <div className="p-6 text-gray-700 text-center">
          {message}
        </div>
      </div>
    </div>
  );
};

<PageModal_Simple 
  title="System Update" 
  message="We will be performing scheduled maintenance at 12 AM UTC." 
/>
