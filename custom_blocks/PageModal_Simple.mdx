name: PageModal_Simple

import React, { useState, useEffect, useRef } from "react";

export const PageModal_Simple = ({
  isActive = true,
  title = "Notice",
  message = "This is your message. Customize it as needed.",
  onClose = () => {},
}) => {
  const [isOpen, setIsOpen] = useState(isActive);
  const modalRef = useRef(null);

  // Sync internal state with isActive prop
  useEffect(() => {
    setIsOpen(isActive);
  }, [isActive]);

  const closeModal = () => {
    setIsOpen(false);
    onClose();
  };

  // Set focus to the modal when it opens for accessibility.
  useEffect(() => {
    if (isOpen && modalRef.current) {
      modalRef.current.focus();
    }
  }, [isOpen]);

  // Close the modal when Escape key is pressed.
  useEffect(() => {
    const handleKeyDown = (e) => {
      if (e.key === "Escape") {
        closeModal();
      }
    };
    if (isOpen) {
      document.addEventListener("keydown", handleKeyDown);
    }
    return () => document.removeEventListener("keydown", handleKeyDown);
  }, [isOpen]);

  if (!isOpen) return null;

  // Inline style for the close button.
  const closeButtonStyle = {
    background: "transparent",
    border: "none",
    fontSize: "1.5rem",
    color: "#4a5568",
    cursor: "pointer",
    transition: "color 0.2s ease-in-out",
  };

  return (
    <>
      <div
        className="fixed inset-0 flex items-center justify-center p-4 z-50 backdrop-blur-sm"
        role="dialog"
        aria-modal="true"
        aria-labelledby="modal-title"
        aria-describedby="modal-desc"
      >
        <div
          className="bg-white shadow-lg w-full max-w-md rounded-md transition-all duration-300 focus:outline-none"
          ref={modalRef}
          tabIndex="-1"
        >
          {/* Modal Header */}
          <div className="flex items-center justify-between p-5">
            <h3 id="modal-title" className="text-xl font-bold text-gray-800">
              {title}
            </h3>
            <button
              type="button"
              onClick={closeModal}
              style={closeButtonStyle}
              onMouseOver={(e) => (e.currentTarget.style.color = "#2d3748")}
              onMouseOut={(e) => (e.currentTarget.style.color = "#4a5568")}
              aria-label="Close modal"
              className="close-button focus:outline-none"
            >
              &times;
            </button>
          </div>

          {/* Modal Body */}
          <div id="modal-desc" className="p-6 text-gray-600 text-center">
            {message}
          </div>
        </div>
      </div>
    </>
  );
};

<PageModal_Simple 
  title="System Update" 
  message="We will be performing scheduled maintenance at 12 AM UTC." 
/>
