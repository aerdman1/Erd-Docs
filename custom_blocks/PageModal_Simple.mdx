name: PageModal_Simple
---
import React, { useState, useEffect, useRef } from "react";

export const PageModal_Simple = ({
  isActive = true,
  title = "Notice",
  message = "This is your message. Customize it as needed.",
  onClose = () => {},
}) => {
  const [isOpen, setIsOpen] = useState(isActive);
  const modalRef = useRef(null);

  useEffect(() => {
    setIsOpen(isActive);
  }, [isActive]);

  const closeModal = () => {
    setIsOpen(false);
    onClose();
  };

  // Accessibility: Focus on the modal when it opens
  useEffect(() => {
    if (isOpen && modalRef.current) {
      modalRef.current.focus();
    }
  }, [isOpen]);

  // Close modal on Escape key press
  useEffect(() => {
    const handleKeyDown = (e) => {
      if (e.key === "Escape") {
        closeModal();
      }
    };
    if (isOpen) {
      document.addEventListener("keydown", handleKeyDown);
    }
    return () => document.removeEventListener("keydown", handleKeyDown);
  }, [isOpen]);

  if (!isOpen) return null;

  return (
    <div className="readme-tailwind">
      {/* Modal Background Overlay */}
      <div className="fixed inset-0 flex items-center justify-center p-4 z-50 bg-black/40 backdrop-blur-sm">
        {/* Modal Content */}
        <div
          className="bg-white shadow-xl w-full max-w-md rounded-lg transition-all duration-300 focus:outline-none"
          ref={modalRef}
          tabIndex="-1"
          role="dialog"
          aria-modal="true"
          aria-labelledby="modal-title"
          aria-describedby="modal-desc"
        >
          {/* Modal Header */}
          <div className="flex items-center justify-between p-5 border-b border-gray-200">
            <h3 id="modal-title" className="text-xl font-semibold text-gray-800">
              {title}
            </h3>
            <button
              type="button"
              onClick={closeModal}
              aria-label="Close modal"
              className="text-gray-500 hover:text-gray-700 transition-all text-2xl font-bold focus:outline-none"
            >
              &times;
            </button>
          </div>

          {/* Modal Body */}
          <div id="modal-desc" className="p-6 text-gray-600 text-center">
            {message}
          </div>

          {/* Modal Footer */}
          <div className="flex justify-center p-4 border-t border-gray-200">
            <button
              onClick={closeModal}
              className="px-4 py-2 bg-blue-600 text-white font-semibold rounded-md shadow-md transition-all hover:bg-blue-700 active:scale-95"
            >
              Close
            </button>
          </div>
        </div>
      </div>
    </div>
  );
};

// Example Usage
<PageModal_Simple 
  title="System Update" 
  message="We will be performing scheduled maintenance at 12 AM UTC." 
/>
