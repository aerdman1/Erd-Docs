import React, { useState } from "react";

export const PageModal_Simple = ({ 
  message = "Are you sure you want to deactivate your account? This action cannot be undone." 
}) => {
  const [isOpen, setIsOpen] = useState(false);

  return (
    <>
      {/* Open Modal Button */}
      <button 
        onClick={() => setIsOpen(true)} 
        className="px-4 py-2 bg-blue-600 text-white rounded-md hover:bg-blue-700 transition"
      >
        Open Modal
      </button>

      {/* Modal Backdrop & Container */}
      {isOpen && (
        <div 
          className="fixed inset-0 flex items-center justify-center bg-black bg-opacity-50 p-4"
          onClick={() => setIsOpen(false)} // Clicking outside closes modal
        >
          {/* Modal Content */}
          <div 
            className="relative bg-white rounded-lg shadow-lg p-6 max-w-sm w-full"
            onClick={(e) => e.stopPropagation()} // Prevents closing when clicking inside modal
          >
            {/* Close Button */}
            <button 
              onClick={() => setIsOpen(false)} 
              className="absolute top-3 right-3 text-gray-500 hover:text-gray-700 transition"
              aria-label="Close"
            >
              <svg xmlns="http://www.w3.org/2000/svg" className="w-6 h-6" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth="2">
                <path strokeLinecap="round" strokeLinejoin="round" d="M6 18L18 6M6 6l12 12" />
              </svg>
            </button>

            {/* Modal Title & Message */}
            <h3 className="text-lg font-semibold text-gray-900">Deactivate Account</h3>
            <p className="mt-2 text-sm text-gray-600">{message}</p>

            {/* Buttons */}
            <div className="mt-4 flex justify-end gap-2">
              <button 
                onClick={() => setIsOpen(false)} 
                className="px-4 py-2 bg-gray-300 text-gray-700 rounded-md hover:bg-gray-400 transition"
              >
                Cancel
              </button>
              <button 
                onClick={() => setIsOpen(false)} 
                className="px-4 py-2 bg-red-600 text-white rounded-md hover:bg-red-700 transition"
              >
                Deactivate
              </button>
            </div>
          </div>
        </div>
      )}
    </>
  );
};

<PageModal_Simple 
  message="This modal is displayed inline."  
  modalColor="#00bcff" 
  modalTextColor="#ffffff" 
  fontSize="14px" 
  fontWeight="bold" 
/>
