import React, { useEffect, useState } from "react";

export const ClaudeArtifact = ({ url }) => {
  const [html, setHtml] = useState("");
  const [status, setStatus] = useState("Loading…");

  useEffect(() => {
    if (!url) return;

    let cancelled = false;

    async function load() {
      try {
        const res = await fetch(url, {
          method: "GET",
          credentials: "omit",
          headers: {
            Accept: "text/html"
          }
        });

        if (!res.ok) {
          throw new Error(`HTTP ${res.status}`);
        }

        const text = await res.text();

        if (!cancelled) {
          setHtml(text);
          setStatus("");
        }
      } catch (err) {
        if (!cancelled) {
          setStatus(`FAILED: ${err.message}`);
        }
      }
    }

    load();

    return () => {
      cancelled = true;
    };
  }, [url]);

  if (!url) return null;

  if (!html) {
    return (
      <pre style={{ whiteSpace: "pre-wrap" }}>
        {status}
      </pre>
    );
  }

  return (
    <iframe
      title="Claude Artifact"
      srcDoc={html}
      style={{
        width: "100%",
        minHeight: "800px",
        border: 0
      }}
    />
  );
};
