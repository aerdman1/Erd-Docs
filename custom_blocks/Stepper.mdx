import React, { useState, useEffect } from "react";

export function StepperComponent({ steps, verticalTrue = false }) {
  const [currentStep, setCurrentStep] = useState(0);
  const [isVisible, setIsVisible] = useState(false); // Controls fade-in

  useEffect(() => {
    // Hide stepper completely on page load and fade in smoothly
    const timeout = setTimeout(() => {
      setIsVisible(true);
    }, 300); // Delay fade-in

    return () => clearTimeout(timeout);
  }, []);

  useEffect(() => {
    // Inject CSS dynamically into the <head>
    const styleOverride = document.createElement("style");
    styleOverride.textContent = `
      .stepper-container {
        position: relative;
        text-align: center;
        width: 100%;
        max-width: 600px;
        margin: 20px auto;
        padding: 24px;
        border-radius: 12px;
        border: 1px solid rgba(255, 255, 255, 0.2);
				background: radial-gradient(circle at top, #835db9 20%, #4a2a8f 80%);
        box-shadow: 0 4px 10px rgba(0, 0, 0, 0.2);
      }
      /* HORIZONTAL LAYOUT (DEFAULT) */
      .progress-container {
        display: flex;
        justify-content: space-between;
        align-items: center;
        position: relative;
        width: 100%;
        max-width: 500px;
        margin: 20px auto;
      }
      .progress-container::before {
        background-color: rgba(255, 255, 255, 0.2);
        content: "";
        height: 3px;
        position: absolute;
        top: 50%;
        left: 10%; /* Start after first circle */
        right: 10%; /* End before last circle */
        transform: translateY(-50%);
        z-index: 2;
        border-radius: 2px;
      }
      .progress {
        background-color: rgba(255, 255, 255, 0.3); /* Dim when inactive */
        height: 3px;
        position: absolute;
        top: 50%;
        left: 10%;
        transition: width 0.3s ease-in-out, background-color 0.3s ease-in-out;
        width: 0%;
        z-index: 1; /* Ensure it renders above the default line */
        border-radius: 2px;
      }
      
      .progress.active {
        background-color: oklch(0.811 0.111 293.571) /* Brighten when active */
      }
      .circle-container {
        position: relative;
        display: flex;
        flex-direction: column;
        align-items: center;
        flex: 1; /* Dynamically spaces circles based on number of steps */
        text-align: center;
      }
      /* Circle Step Style */
      .circle {
        align-items: center;
        background: linear-gradient(to top, #b78cff, #8c5eff);
        border: 2px solid rgba(236, 233, 254, 0.6);
        border-radius: 50%;
        color: #fff;
        display: flex;
        font-size: 14px;
        font-weight: bold;
        height: 34px;
        justify-content: center;
        transition: 0.3s ease;
        width: 34px;
        z-index: 99999;
        position:relative;
      }
      .circle.active {
        background: linear-gradient(to top, rgb(193 131 246), rgb(183, 140, 255));
        color: #fff;
        border: 2px solid #fff;
        z-index: 99999;
        transition: .3s ease;
      }
      /* Square Step Style */
      .square {
        align-items: center;
        background: linear-gradient(to top, #ff8c6b, #ff5e42);
        border: 2px solid rgba(255, 255, 255, 0.6);
        border-radius: 6px; /* Slightly rounded edges */
        color: #fff;
        display: flex;
        font-size: 14px;
        font-weight: bold;
        height: 34px;
        justify-content: center;
        transition: 0.3s ease;
        width: 34px;
        z-index: 99999;
        position:relative;
      }
      .square.active {
        background: linear-gradient(to top, #ffb09c, #ff8c6b);
        color: #fff;
        border: 2px solid #fff;
        z-index: 99999;
      }
      .step-title {
        position: absolute;
        top: 45px;
        font-size: 14px;
        font-weight: bold;
        color: rgba(255, 255, 255, 0.6);
        transition: color 0.3s ease;
        width: 100px;
        text-align: center;
      }
      .step-title.active {
        color: rgba(255, 255, 255, 1);
      }
      .step-content {
        margin-top: 50px;
        font-size: 16px;
        padding: 10px;
        color: white;
        font-weight: 400;
      }
      .stepper-buttons {
        margin-top: 20px;
        display: flex;
        justify-content: space-between;
      }
      .btn {
        background: transparent;
        border: none;
        color: rgba(255, 255, 255, 0.6);
        font-size: 14px;
        font-weight: bold;
        cursor: pointer;
        padding: 6px 12px;
        transition: color 0.2s ease;
      }
      .btn:hover {
        color: rgba(255, 255, 255, 0.9);
      }
      .btn:disabled {
        color: rgba(255, 255, 255, 0.3);
        cursor: not-allowed;
      }
      .content-body .rm-Markdown {
        opacity: 1 !important;
      }

      /* VERTICAL LAYOUT */
      .vertical-layout {
        display: flex;
        flex-direction: row;
        align-items: flex-start;
      }
      /* The timeline is given a fixed width to properly center circles. */
      .vertical-timeline {
        position: relative;
        width: 60px; /* Enough space for circles to center on line */
        margin-right: 20px;
      }
      .vertical-timeline::before {
        content: "";
        position: absolute;
        left: 30px; /* Center of .vertical-timeline */
        top: 34px; /* Start AFTER the first circle */
        bottom: 34px; /* Stop BEFORE the last circle */
        width: 3px;
        background-color: rgba(255, 255, 255, 0.2);
        border-radius: 2px;
        z-index: 0;
      }           
      .vertical-progress {
        position: absolute;
        left: 30px;
        top: 34px;
        width: 3px;
        background-color: rgba(255, 255, 255, 0.3); /* Dim when inactive */
        transition: height 0.3s ease-in-out, background-color 0.3s ease-in-out;
        height: 0%; /* Start empty */
        z-index: 1;
        border-radius: 2px;
      }
      
      .vertical-progress.active {
        background-color: rgba(255, 255, 255, 0.9); /* Brighten when active */
      }      
      .vertical-circle-container {
        position: relative;
        margin: 20px 0;
      }
      /* Center the shape on the line (which is at x=30px) */
      .vertical-circle-container .circle,
      .vertical-circle-container .square {
        margin-left: 13px; /* 30 - (34/2) = 13 */
      }

      /* Title to the RIGHT of each shape (vertically centered) */
      .vertical-circle-container .step-title {
        position: absolute;
        top: 50%;
        left: 60px; /* Enough space to clear the shape */
        transform: translateY(-50%);
        width: auto;
        min-width: 140px; /* Ensures we have space for the text */
        text-align: left;
        font-size: 14px;
        font-weight: bold;
        color: rgba(255, 255, 255, 0.6);
        transition: color 0.3s ease;
      }
      .vertical-circle-container .step-title.active {
        color: rgba(255, 255, 255, 1);
      }

      /* Push vertical step content further right to avoid overlap */
      .vertical-layout .step-content {
        margin-left: 100px; /* Extra spacing from timeline+titles */
      }
    `;
    document.head.appendChild(styleOverride);

    return () => {
      document.head.removeChild(styleOverride); // Cleanup on unmount
    };
  }, []);

  // If "vertical" is true, render vertical layout
  if (verticalTrue) {
    return (
      <div
        className={`stepper-container transition-opacity duration-500 ${
          isVisible ? "opacity-100 visible" : "opacity-0 invisible"
        }`}
      >
        <div className="vertical-layout">
          {/* Timeline on the left */}
          <div className="vertical-timeline">
  <div
    className={`vertical-progress ${currentStep > 0 ? "active" : ""}`}
    style={{
      height: `${(currentStep / (steps.length - 1)) * 100}%`,
    }}
  ></div>

            {steps.map((step, index) => {
              const shapeClass = step.stepType === "square" ? "square" : "circle";
              return (
                <div key={index} className="vertical-circle-container">
                  <div className={`${shapeClass} ${index <= currentStep ? "active" : ""}`}>
                    {index + 1}
                  </div>
                  <div className={`step-title ${index === currentStep ? "active" : ""}`}>
                    {step.header}
                  </div>
                </div>
              );
            })}
          </div>

          {/* Step Content on the right */}
          <div className="step-content">
            {steps[currentStep].content}

            <div className="stepper-buttons">
              <button
                className="btn"
                onClick={() => setCurrentStep(currentStep - 1)}
                disabled={currentStep === 0}
              >
                BACK
              </button>
              <button
                className="btn"
                onClick={() => setCurrentStep(currentStep + 1)}
                disabled={currentStep === steps.length - 1}
              >
                NEXT
              </button>
            </div>
          </div>
        </div>
      </div>
    );
  }

  // Default Horizontal Layout
  return (
    <div
      className={`stepper-container transition-opacity duration-500 ${
        isVisible ? "opacity-100 visible" : "opacity-0 invisible"
      }`}
    >
      {/* Progress Bar */}
      <div className="progress-container">
  <div
    className={`progress ${currentStep > 0 ? "active" : ""}`}
    style={{
      width: `${(currentStep / (steps.length - 1)) * 80}%`,
    }}
  ></div>


        {/* Step Circles or Squares with Titles Below */}
        {steps.map((step, index) => {
          const shapeClass = step.stepType === "square" ? "square" : "circle";
          return (
            <div key={index} className="circle-container">
              <div className={`${shapeClass} ${index <= currentStep ? "active" : ""}`}>
                {index + 1}
              </div>
              <div className={`step-title ${index === currentStep ? "active" : ""}`}>
                {step.header}
              </div>
            </div>
          );
        })}
      </div>

      {/* Step Content */}
      <div className="step-content">
        {steps[currentStep].content}
      </div>

      {/* Navigation Buttons */}
      <div className="stepper-buttons">
        <button
          className="btn"
          onClick={() => setCurrentStep(currentStep - 1)}
          disabled={currentStep === 0}
        >
          BACK
        </button>
        <button
          className="btn"
          onClick={() => setCurrentStep(currentStep + 1)}
          disabled={currentStep === steps.length - 1}
        >
          NEXT
        </button>
      </div>
    </div>
  );
}

     </p>, stepType: "circle" },

  ]}
