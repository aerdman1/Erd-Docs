import React, { useState } from 'react';

export const Stepper1 = ({ steps, position = "top" }) => {
  const [currentStep, setCurrentStep] = useState(0);

  const handleNext = () => {
    if (currentStep < steps.length - 1) {
      setCurrentStep(currentStep + 1);
    }
  };

  const handlePrev = () => {
    if (currentStep > 0) {
      setCurrentStep(currentStep - 1);
    }
  };

  return (
    <div className="bg-opacity-80 border-1 border-white/10 rounded-lg shadow-md text-white p-6 w-150 bg-radial from-purple-200 from-30% to-violet-950 to-80%">
      <div className={`flex flex-col ${position === 'bottom' ? 'flex-col-reverse' : ''} w-full`}>
        <div
          className={`flex ${position === 'top' || position === 'bottom' ? 'flex-row' : 'flex-col'} items-center justify-center space-x-6`}
        >
          {/* Step Circles with Text */}
          {steps.map((step, index) => (
            <div className="flex flex-col items-center w-full" key={index}>
              <div
                className={`stepper-step w-8 h-8 rounded-full flex items-center justify-center border-2 ${
                  index <= currentStep
                    ? 'bg-linear-to-t from-violet-300 to-violet-400 text-white border-violet-100'
                    : 'bg-violet-200 text-white border-violet-100/10 opacity-40'
                }`}
              >
                {index + 1}
              </div>
              <div className="font-semibold text-center text-sm mt-2">{step.header}</div>
            </div>
          ))}
        </div>

        {/* Connecting Lines Between Steps */}
        <div className="absolute flex gap-8 w-135 ml-25 mt-4">
          {steps.map((step, index) => (
            <div key={index} className="relative flex-grow">
              {index < steps.length - 1 && (
                <div
                  className={`absolute top-1/2 left-0 w-full h-1 ${index < currentStep ? 'bg-violet-300' : 'bg-gray-300'}`}
                  style={{ transition: 'width 0.3s ease' }}
                />
              )}
            </div>
          ))}
        </div>

        {/* Step Content and Navigation Buttons */}
        <div className="mt-4 flex flex-col items-center ml-5 mr-5">
          {/* Step Content */}
          <div className="w-full">
            {steps[currentStep].content}
          </div>

          {/* Back and Next buttons */}
          <div className="flex pt-4 justify-between w-full">
            <button
              className="border-transparent bg-transparent text-gray-400 hover:text-gray-200 font-medium font-bold uppercase text-sm"
              onClick={handlePrev}
              disabled={currentStep === 0}
            >
              Back
            </button>
            <button
              className="border-transparent bg-transparent font-medium font-bold text-sm text-purple-200 uppercase"
              onClick={handleNext}
              disabled={currentStep === steps.length - 1}
            >
              Next
            </button>
          </div>
        </div>
      </div>
     </div>
  );
};
