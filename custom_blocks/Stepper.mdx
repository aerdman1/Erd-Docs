import React, { useState } from "react";

export function StepperComponent({ steps, position = "top" }) {
  const [currentStep, setCurrentStep] = useState(0);

  // Move forward one step
  const handleNext = () => {
    if (currentStep < steps.length - 1) {
      setCurrentStep(currentStep + 1);
    }
  };

  // Move backward one step
  const handlePrev = () => {
    if (currentStep > 0) {
      setCurrentStep(currentStep - 1);
    }
  };

  if (!steps || !steps.length) {
    return <p>No steps provided.</p>;
  }

  /**
   * Build an array of alternating "circle" and "line" elements.
   * For N steps, you'll end up with (2N - 1) elements total:
   *    Circle -> Line -> Circle -> Line -> ... -> Circle
   */
  const circleLineElements = [];
  for (let i = 0; i < steps.length; i++) {
    const isActive = i <= currentStep;

    // Circle + label
    circleLineElements.push(
      <div className="flex flex-col items-center" key={`step-${i}`}>
        <div
          className={`stepper-step w-8 h-8 rounded-full flex items-center justify-center border-2 transition-colors duration-300 ${
            isActive
              ? "bg-gradient-to-t from-violet-300 to-violet-400 text-white border-violet-100"
              : "bg-violet-200 text-white border-violet-100/10 opacity-40"
          }`}
        >
          {i + 1}
        </div>
        <div className="font-semibold text-center text-sm mt-2">
          {steps[i].header}
        </div>
      </div>
    );

    // Connector line (except after the last step)
    if (i < steps.length - 1) {
      circleLineElements.push(
        <div
          className={`flex-1 h-1 ${
            i < currentStep ? "bg-violet-300" : "bg-gray-300"
          } mx-2`}
          style={{ transition: "width 0.3s ease" }}
          key={`line-${i}`}
        />
      );
    }
  }

  return (
    <div className="bg-opacity-80 border border-white/10 rounded-lg shadow-md text-white p-6 w-full bg-radial from-purple-200 from-30% to-violet-950 to-80%">
      <div
        className={`flex flex-col ${
          position === "bottom" ? "flex-col-reverse" : ""
        } w-full`}
      >
        {/* Step Circles + Connector Lines */}
        <div className="flex items-center justify-center w-full">
          {circleLineElements}
        </div>

        {/* Current Step Content + Navigation */}
        <div className="mt-4 flex flex-col items-center ml-5 mr-5">
          <div className="w-full">{steps[currentStep].content}</div>

          <div className="flex pt-4 justify-between w-full">
            <button
              className="border-transparent bg-transparent text-gray-400 hover:text-gray-200 font-bold uppercase text-sm"
              onClick={handlePrev}
              disabled={currentStep === 0}
            >
              Back
            </button>
            <button
              className="border-transparent bg-transparent font-bold text-sm text-purple-200 uppercase"
              onClick={handleNext}
              disabled={currentStep === steps.length - 1}
            >
              Next
            </button>
          </div>
        </div>
      </div>
    </div>
  );
}


<StepperComponent
  position="top"
  steps={[
    {
      header: "Introduction",
      content:  
      <div>
      <p>This is the first step of the process. You will get introduced to the system. Make sure to review the provided information carefully.</p>
      </div>,
    },
    {
      header: "Details",
      content:         
      <div>
      <p>This step asks for your personal details. Fill out the required fields. Ensure that all your details are correct before moving on.</p>
     </div>,
    },
    {
      header: "Confirmation",
      content:        
      <div>
      <p>Review the details you have entered. If everything looks good, submit the form. If you need to make any changes, you can go back to the previous step.</p>
      </div>,
    },
  ]}
/>
