import React, { useState, useEffect } from "react";

export function StepperComponent({ steps, position = "top" }) {
  const [currentStep, setCurrentStep] = useState(0);
  const visibleSteps = steps.slice(0, 4); // Show only the first 4 steps

  useEffect(() => {
    // Inject CSS dynamically into the <head>
    const styleOverride = document.createElement("style");
    styleOverride.textContent = `
      .stepper-container {
        position: relative;
        text-align: center;
        width: 100%;
        max-width: 500px;
        margin: 20px auto;
      }
      .progress-container {
        display: flex;
        justify-content: space-between;
        align-items: center;
        position: relative;
        width: 100%;
        max-width: 400px;
        margin: 20px auto;
      }
      .progress-container::before {
        background-color: #e0e0e0;
        content: "";
        height: 4px;
        left: 0;
        position: absolute;
        top: 50%;
        transform: translateY(-50%);
        width: 100%;
        z-index: -1;
      }
      .progress {
        background-color: #7c3aed;
        height: 4px;
        position: absolute;
        top: 50%;
        transform: translateY(-50%);
        transition: width 0.4s ease;
        width: 0%;
        z-index: -1;
      }
      .circle {
        align-items: center;
        background-color: #fff;
        border: 3px solid #e0e0e0;
        border-radius: 50%;
        color: #999;
        display: flex;
        font-size: 14px;
        font-weight: bold;
        height: 40px;
        justify-content: center;
        transition: 0.4s ease;
        width: 40px;
        z-index: 1;
      }
      .circle.active {
        border-color: #7c3aed;
        color: #7c3aed;
      }
      .step-content {
        margin-top: 20px;
        font-size: 16px;
        padding: 10px;
      }
      .stepper-buttons {
        margin-top: 20px;
      }
      .btn {
        background-color: #7c3aed;
        border: 0;
        border-radius: 6px;
        color: #fff;
        cursor: pointer;
        font-size: 14px;
        margin: 5px;
        padding: 8px 30px;
        transition: 0.2s;
      }
      .btn:disabled {
        background-color: #e0e0e0;
        cursor: not-allowed;
      }
    `;
    document.head.appendChild(styleOverride);

    return () => {
      document.head.removeChild(styleOverride); // Cleanup on unmount
    };
  }, []);

  return (
    <div className="stepper-container">
      {/* Progress Bar */}
      <div className="progress-container">
        <div
          className="progress"
          style={{
            width: `${(currentStep / (visibleSteps.length - 1)) * 100}%`,
          }}
        ></div>

        {/* Circles */}
        {visibleSteps.map((step, index) => (
          <div
            key={index}
            className={`circle ${index <= currentStep ? "active" : ""}`}
          >
            {index + 1}
          </div>
        ))}
      </div>

      {/* Step Content */}
      <div className="step-content">{visibleSteps[currentStep].content}</div>

      {/* Navigation Buttons */}
      <div className="stepper-buttons">
        <button className="btn" onClick={() => setCurrentStep(currentStep - 1)} disabled={currentStep === 0}>
          Back
        </button>
        <button className="btn" onClick={() => setCurrentStep(currentStep + 1)} disabled={currentStep === visibleSteps.length - 1}>
          Next
        </button>
      </div>
    </div>
  );
}


// Example usage with 5 steps but only showing 4
<StepperComponent
  position="top"
  steps={[
    {
      header: "Introduction",
      content:  
      <div>
      <p>This is the first step of the process. You will get introduced to the system. Make sure to review the provided information carefully.</p>
      </div>,
    },
    {
      header: "Details",
      content:         
      <div>
      <p>This step asks for your personal details. Fill out the required fields. Ensure that all your details are correct before moving on.</p>
     </div>,
    },
    {
      header: "Confirmation",
      content:        
      <div>
      <p>Review the details you have entered. If everything looks good, submit the form. If you need to make any changes, you can go back to the previous step.</p>
      </div>,
    },
    {
      header: "Completion",
      content:        
      <div>
      <p>Congratulations! You have completed the steps successfully.</p>
      </div>,
    },
    {
      header: "Extra Step (Hidden)",
      content:        
      <div>
      <p>This step is not displayed by default but exists in the code.</p>
      </div>,
    },
  ]}
/>
