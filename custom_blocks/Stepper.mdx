import React, { useState, useEffect } from "react";

export function StepperComponent({ steps }) {
  const [currentStep, setCurrentStep] = useState(0);
  const visibleSteps = steps.slice(0, 4); // Show only the first 4 steps

  useEffect(() => {
    // Inject CSS dynamically into the <head>
    const styleOverride = document.createElement("style");
    styleOverride.textContent = `
      .stepper-container {
        position: relative;
        text-align: center;
        width: 100%;
        max-width: 600px;
        margin: 20px auto;
        padding: 24px;
        border-radius: 12px;
        border: 1px solid rgba(255, 255, 255, 0.2);
        background: radial-gradient(circle at center, #c9a4ff 20%, #4a2a8f 80%);
        box-shadow: 0 4px 10px rgba(0, 0, 0, 0.2);
      }
      .progress-container {
        display: flex;
        justify-content: space-between;
        align-items: center;
        position: relative;
        width: 100%;
        max-width: 500px;
        margin: 20px auto;
      }
      .progress-container::before {
        background-color: rgba(255, 255, 255, 0.2);
        content: "";
        height: 3px;
        left: 0;
        position: absolute;
        top: 50%;
        transform: translateY(-50%);
        width: 100%;
        z-index: -1;
        border-radius: 2px;
      }
      .progress {
        background-color: rgba(255, 255, 255, 0.8);
        height: 3px;
        position: absolute;
        top: 50%;
        transform: translateY(-50%);
        transition: width 0.4s ease;
        width: 0%;
        z-index: -1;
        border-radius: 2px;
      }
      .circle {
        align-items: center;
        background: linear-gradient(to top, #b78cff, #8c5eff);
        border: 3px solid rgba(255, 255, 255, 0.6);
        border-radius: 50%;
        color: #fff;
        display: flex;
        font-size: 14px;
        font-weight: bold;
        height: 34px;
        justify-content: center;
        transition: 0.4s ease;
        width: 34px;
        z-index: 1;
      }
      .circle.active {
        background: linear-gradient(to top, #e2c0ff, #b78cff);
        color: #fff;
        border: 3px solid white;
      }
      .step-content {
        margin-top: 20px;
        font-size: 16px;
        padding: 10px;
        color: white;
        font-weight: 400;
      }
      .step-title {
        font-weight: bold;
        font-size: 18px;
        color: white;
        margin-bottom: 4px;
      }
      .step-subtitle {
        font-weight: bold;
        font-size: 16px;
        color: white;
        margin-bottom: 8px;
      }
      .stepper-buttons {
        margin-top: 20px;
        display: flex;
        justify-content: space-between;
      }
      .btn {
        background: transparent;
        border: none;
        color: rgba(255, 255, 255, 0.6);
        font-size: 14px;
        font-weight: bold;
        cursor: pointer;
        padding: 6px 12px;
        transition: color 0.2s ease;
      }
      .btn:hover {
        color: rgba(255, 255, 255, 0.9);
      }
      .btn:disabled {
        color: rgba(255, 255, 255, 0.3);
        cursor: not-allowed;
      }
    `;
    document.head.appendChild(styleOverride);

    return () => {
      document.head.removeChild(styleOverride); // Cleanup on unmount
    };
  }, []);

  return (
    <div className="stepper-container">
      {/* Progress Bar */}
      <div className="progress-container">
        <div
          className="progress"
          style={{
            width: `${(currentStep / (visibleSteps.length - 1)) * 100}%`,
          }}
        ></div>

        {/* Circles */}
        {visibleSteps.map((step, index) => (
          <div key={index} className={`circle ${index <= currentStep ? "active" : ""}`}>
            {index + 1}
          </div>
        ))}
      </div>

      {/* Step Content */}
      <div className="step-content">
        <div className="step-title">{visibleSteps[currentStep].header}</div>
        <div className="step-subtitle">Getting Started</div>
        {visibleSteps[currentStep].content}
      </div>

      {/* Navigation Buttons */}
      <div className="stepper-buttons">
        <button className="btn" onClick={() => setCurrentStep(currentStep - 1)} disabled={currentStep === 0}>
          BACK
        </button>
        <button className="btn" onClick={() => setCurrentStep(currentStep + 1)} disabled={currentStep === visibleSteps.length - 1}>
          NEXT
        </button>
      </div>
    </div>
  );
}


// Example usage with 5 steps but only showing 4
<StepperComponent
  position="top"
  steps={[
    {
      header: "Introduction",
      content:  
      <div>
      <p>This is the first step of the process. You will get introduced to the system. Make sure to review the provided information carefully.</p>
      </div>,
    },
    {
      header: "Details",
      content:         
      <div>
      <p>This step asks for your personal details. Fill out the required fields. Ensure that all your details are correct before moving on.</p>
     </div>,

    {
      header: "Confirmation",
      content:        
      <div>
      <p>Review the details you have entered. If everything looks good, submit the form. If you need to make any changes, you can go back to the previous step.</p>
      </div>,
    },
    {
      header: "Completion",
      content:        
      <div>
      <p>Congratulations! You have completed the steps successfully.</p>
      </div>,
    },
    {
      header: "Extra Step (Hidden)",
      content:        
      <div>
      <p>This step is not displayed by default but exists in the code.</p>
      </div>,
    },
  ]}
/>
