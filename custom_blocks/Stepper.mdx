import React, { useState, useEffect, useMemo } from "react";

export function StepperComponent({ 
  steps, 
  verticalTrue = false, 
  backgroundTheme = "purple", 
  verticalSpacing = "20px", 
  horizontalSpacing = "0px" 
}) {
  const themes = {
    purple: {
      background: "radial-gradient(circle at top, #835db9 20%, #4a2a8f 80%)",
      circle: "linear-gradient(to top, #b78cff, #8c5eff)",
      circleActive: "linear-gradient(to top, rgb(193, 131, 246), rgb(183, 140, 255))",
      square: "linear-gradient(to top, #ff8c6b, #ff5e42)",
      squareActive: "linear-gradient(to top, #ffb09c, #ff8c6b)"
    },
    blue: {
      background: "radial-gradient(circle at top, #4da6ff 20%, #0073e6 80%)",
      circle: "linear-gradient(to top, #80ccff, #4da6ff)",
      circleActive: "linear-gradient(to top, #1a8cff, #0066cc)",
      square: "linear-gradient(to top, #66ccff, #3399ff)",
      squareActive: "linear-gradient(to top, #99ddff, #66ccff)"
    }
  };

  const currentTheme = themes[backgroundTheme] || themes["purple"];
  const uniqueId = useMemo(() => Math.random().toString(36).substr(2, 9), []);
  const [currentStep, setCurrentStep] = useState(0);
  const [isVisible, setIsVisible] = useState(false);

  useEffect(() => {
    const timeout = setTimeout(() => {
      setIsVisible(true);
    }, 300);
    return () => clearTimeout(timeout);
  }, []);

  useEffect(() => {
    const styleOverride = document.createElement("style");
    styleOverride.textContent = `
      .stepper-container-${uniqueId} {
        position: relative;
        text-align: center;
        width: 100%;
        max-width: 90%;
        margin: 20px auto;
        padding: 24px 36px;
        border-radius: 12px;
        border: 1px solid rgba(255, 255, 255, 0.2);
        background: ${currentTheme.background};
        box-shadow: 0 4px 10px rgba(0, 0, 0, 0.2);
      }
      .stepper-container-${uniqueId} .progress-container {
        display: flex;
        justify-content: space-between;
        align-items: center;
        gap: ${horizontalSpacing};
        position: relative;
        width: 100%;
        max-width: 90%;
        margin: 20px auto;
      }
      .stepper-container-${uniqueId} .progress-container::before {
        background-color: rgba(255, 255, 255, 0.2);
        content: "";
        height: 3px;
        position: absolute;
        top: 50%;
        left: 10%;
        right: 10%;
        transform: translateY(-50%);
        z-index: 2;
        border-radius: 2px;
      }
      .stepper-container-${uniqueId} .progress {
        background-color: rgba(255, 255, 255, 0.3);
        height: 3px;
        position: absolute;
        top: 50%;
        left: 10%;
        transition: width 0.3s ease-in-out, background-color 0.3s ease-in-out;
        width: 0%;
        z-index: 1;
        border-radius: 2px;
      }
      .stepper-container-${uniqueId} .progress.active {
        background-color: oklch(0.811 0.111 293.571);
      }
      .stepper-container-${uniqueId} .circle-container {
        position: relative;
        display: flex;
        flex-direction: column;
        align-items: center;
        flex: 1;
        text-align: center;
      }
      .stepper-container-${uniqueId} .circle {
        align-items: center;
        background: ${currentTheme.circle};
        border: 2px solid rgba(236, 233, 254, 0.6);
        border-radius: 50%;
        color: #fff;
        display: flex;
        font-size: 14px;
        font-weight: bold;
        height: 34px;
        justify-content: center;
        transition: 0.3s ease;
        width: 34px;
        z-index: 99999;
        position: relative;
      }
      .stepper-container-${uniqueId} .circle.active {
        background: ${currentTheme.circleActive};
        color: #fff;
        border: 2px solid #fff;
        z-index: 99999;
        transition: 0.3s ease;
      }
      .stepper-container-${uniqueId} .square {
        align-items: center;
        background: ${currentTheme.square};
        border: 2px solid rgba(255, 255, 255, 0.6);
        border-radius: 6px;
        color: #fff;
        display: flex;
        font-size: 14px;
        font-weight: bold;
        height: 34px;
        justify-content: center;
        transition: 0.3s ease;
        width: 34px;
        z-index: 99999;
        position: relative;
      }
      .stepper-container-${uniqueId} .square.active {
        background: ${currentTheme.squareActive};
        color: #fff;
        border: 2px solid #fff;
        z-index: 99999;
      }
      .stepper-container-${uniqueId} .step-title {
        position: absolute;
        top: 45px;
        font-size: 14px;
        font-weight: bold;
        color: rgba(255, 255, 255, 0.6);
        transition: color 0.3s ease;
        width: auto;
        text-align: center;
      }
      .stepper-container-${uniqueId} .step-title.active {
        color: rgba(255, 255, 255, 1);
      }
      .stepper-container-${uniqueId} .step-content {
        margin: 80px auto 0 auto;
        max-width: 90%;
        font-size: 16px;
        padding: 10px;
        color: white;
        font-weight: 400;
      }
      .stepper-container-${uniqueId} .step-header {
        text-align: center;
        font-size: 24px;
        font-weight: bold;
        margin-bottom: 20px;
        color: white;
      }
      .stepper-container-${uniqueId} .stepper-buttons {
        margin-top: 20px;
        display: flex;
        justify-content: space-between;
      }
      .stepper-container-${uniqueId} .btn {
        background: transparent;
        border: none;
        color: rgba(255, 255, 255, 0.6);
        font-size: 14px;
        font-weight: bold;
        cursor: pointer;
        padding: 6px 12px;
        transition: color 0.2s ease;
      }
      .stepper-container-${uniqueId} .btn:hover {
        color: rgba(255, 255, 255, 0.9);
      }
      .stepper-container-${uniqueId} .btn:disabled {
        color: rgba(255, 255, 255, 0.3);
        cursor: not-allowed;
      }
      .stepper-container-${uniqueId} .content-body .rm-Markdown {
        opacity: 1 !important;
      }
      .stepper-container-${uniqueId} .vertical-layout {
        display: flex;
        flex-direction: row;
        align-items: flex-start;
      }
      .stepper-container-${uniqueId} .vertical-timeline {
        position: relative;
        width: 60px;
        margin-right: 20px;
      }
      .stepper-container-${uniqueId} .vertical-timeline::before {
        content: "";
        position: absolute;
        left: 30px;
        top: 34px;
        bottom: 34px;
        width: 3px;
        background-color: rgba(255, 255, 255, 0.2);
        border-radius: 2px;
        z-index: 0;
      }
      .stepper-container-${uniqueId} .vertical-progress {
        position: absolute;
        left: 30px;
        top: 34px;
        width: 3px;
        background-color: rgba(255, 255, 255, 0.3);
        transition: height 0.3s ease-in-out, background-color 0.3s ease-in-out;
        height: 0%;
        z-index: 1;
        border-radius: 2px;
      }
      .stepper-container-${uniqueId} .vertical-progress.active {
        background-color: rgba(255, 255, 255, 0.9);
      }
      .stepper-container-${uniqueId} .vertical-circle-container {
        position: relative;
        margin: ${verticalSpacing} 0;
      }
      .stepper-container-${uniqueId} .vertical-circle-container .circle,
      .stepper-container-${uniqueId} .vertical-circle-container .square {
        margin-left: 13px;
      }
      .stepper-container-${uniqueId} .vertical-circle-container .step-title {
        position: absolute;
        top: 50%;
        left: 60px;
        transform: translateY(-50%);
        width: auto;
        min-width: 140px;
        text-align: left;
        font-size: 14px;
        font-weight: bold;
        color: rgba(255, 255, 255, 0.6);
        transition: color 0.3s ease;
      }
      .stepper-container-${uniqueId} .vertical-circle-container .step-title.active {
        color: rgba(255, 255, 255, 1);
      }
      .stepper-container-${uniqueId} .vertical-layout .step-content {
        margin: 0 0 0 190px;
        align-self: center;
      }
    `;
    document.head.appendChild(styleOverride);
    return () => {
      document.head.removeChild(styleOverride);
    };
  }, [backgroundTheme, verticalSpacing, horizontalSpacing, uniqueId]);

  if (verticalTrue) {
    return (
      <div className={`stepper-container-${uniqueId} transition-opacity duration-500 ${isVisible ? "opacity-100 visible" : "opacity-0 invisible"}`}>
        <div className="vertical-layout">
          <div className="vertical-timeline">
            <div
              className={`vertical-progress ${currentStep > 0 ? "active" : ""}`}
              style={{
                height: currentStep === 0 ? "0%" : `${(currentStep / (steps.length - 1)) * 80}%`
              }}
            ></div>
            {steps.map((step, index) => {
              const shapeClass = step.stepType === "square" ? "square" : "circle";
              return (
                <div key={index} className="vertical-circle-container">
                  <div className={`${shapeClass} ${index <= currentStep ? "active" : ""}`}>
                    {index + 1}
                  </div>
                  <div className={`step-title ${index === currentStep ? "active" : ""}`}>
                    {step.header}
                  </div>
                </div>
              );
            })}
          </div>
          <div className="step-content">
            <h2 className="step-header">{steps[currentStep].header}</h2>
            {steps[currentStep].content}
            <div className="stepper-buttons">
              <button className="btn" onClick={() => setCurrentStep(currentStep - 1)} disabled={currentStep === 0}>
                <i className="fa-solid fa-chevron-left"></i> BACK
              </button>
              <button className="btn" onClick={() => setCurrentStep(currentStep + 1)} disabled={currentStep === steps.length - 1}>
                NEXT <i className="fa-solid fa-chevron-right"></i>
              </button>
            </div>
          </div>
        </div>
      </div>
    );
  }

  return (
    <div className={`stepper-container-${uniqueId} transition-opacity duration-500 ${isVisible ? "opacity-100 visible" : "opacity-0 invisible"}`}>
      <div className="progress-container">
        <div
          className={`progress ${currentStep > 0 ? "active" : ""}`}
          style={{ width: `${(currentStep / (steps.length - 1)) * 80}%` }}
        ></div>
        {steps.map((step, index) => {
          const shapeClass = step.stepType === "square" ? "square" : "circle";
          return (
            <div key={index} className="circle-container">
              <div className={`${shapeClass} ${index <= currentStep ? "active" : ""}`}>
                {index + 1}
              </div>
              <div className={`step-title ${index === currentStep ? "active" : ""}`}>
                {step.header}
              </div>
            </div>
          );
        })}
      </div>
      <div className="step-content">
        <h2 className="step-header">{steps[currentStep].header}</h2>
        {steps[currentStep].content}
      </div>
      <div className="stepper-buttons">
        <button className="btn" onClick={() => setCurrentStep(currentStep - 1)} disabled={currentStep === 0}>
          <i className="fa-solid fa-chevron-left"></i> BACK
        </button>
        <button className="btn" onClick={() => setCurrentStep(currentStep + 1)} disabled={currentStep === steps.length - 1}>
          NEXT <i className="fa-solid fa-chevron-right"></i>
        </button>
      </div>
    </div>
  );
}

  return (
    <StepperComponent
      verticalFalse

      verticalSpacing="20px"
      horizontalSpacing="0px"
      steps={[
        { header: "Plan Your Documentation", content: <p>Define the purpose of your documentation and identify your target audience. Outline key topics and gather relevant resources before writing.</p>, stepType: "circle" },
        { header: "Write Clear Content", content: <p>Use simple language, structured explanations, and break complex concepts into digestible sections.</p>, stepType: "circle" },
        { header: "Format Content Effectively", content: <p>Utilize headings, bullet points, and tables to improve readability. Add images, code snippets, and examples to enhance clarity.</p>, stepType: "circle" },
        { header: "Use Interactive Elements", content: <p>Incorporate API explorers, live code examples, and try-it-yourself features for better engagement and hands-on learning.</p>, stepType: "circle" },
