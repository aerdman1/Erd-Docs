import React, { useEffect, useState } from "react";

export function LinkScannerV2({
  initialApiKey = "",
  initialVersion = "",
}) {
  const [apiKey, setApiKey] = useState(initialApiKey);
  const [version, setVersion] = useState(initialVersion);
  const [issues, setIssues] = useState([]);
  const [loading, setLoading] = useState(false);

  // Only start scanning once we have a key
  useEffect(() => {
    if (!apiKey) return;

    (async () => {
      setLoading(true);
      try {
        const origin = "https://api.readme.com";
        const headers = new Headers({ Accept: "application/json" });
        headers.set("Authorization", `Bearer ${apiKey}`);

        // 1) get guides list
        const listRes = await fetch(
          `${origin}/v2/versions/${encodeURIComponent(version)}/guides`,
          { headers }
        );
        if (!listRes.ok) throw new Error("Fetch guides failed");
        const { data: guides } = await listRes.json();

        // 2) fetch each guide’s body and scan
        const hrefRegex = /<a\s+[^>]*href="([^"]+)"/g;
        const found = [];

        for (const g of guides) {
          const slug = g.slug;
          const pageRes = await fetch(
            `${origin}/v2/versions/${encodeURIComponent(version)}/guides/${slug}`,
            { headers }
          );
          if (!pageRes.ok) continue;
          const { data } = await pageRes.json();
          let m;
          while ((m = hrefRegex.exec(data.content.body))) {
            let href = m[1];
            // absolute URL
            if (!/^https?:\/\//.test(href)) {
              href = window.location.origin + (href.startsWith("/") ? "" : "/") + href;
            }
            try {
              const head = await fetch(href, { method: "HEAD", redirect: "manual" });
              if (!(head.status >= 200 && head.status < 300)) {
                found.push({ slug, href, status: head.status });
              }
            } catch {
              found.push({ slug, href, status: "ERR" });
            }
          }
        }

        setIssues(found);
      } catch (err) {
        console.error(err);
      } finally {
        setLoading(false);
      }
    })();
  }, [apiKey, version]);

  // If no key yet, prompt for it (and optionally version)
  if (!apiKey) {
    return (
      <div className="mb-4 w-full">
        <label className="block text-sm font-medium mb-1">ReadMe API Key</label>
        <div className="flex gap-2 mb-3">
          <input
            type="password"
            className="flex-1 border rounded p-2"
            placeholder="Paste your API key…"
            value={initialApiKey}
            onChange={(e) => setApiKey(e.target.value.trim())}
          />
        </div>
        <label className="block text-sm font-medium mb-1">Project Version (branch)</label>
        <div className="flex gap-2 mb-3">
          <input
            type="text"
            className="flex-1 border rounded p-2"
            placeholder="e.g. stable or v2.0.0"
            value={version}
            onChange={(e) => setVersion(e.target.value.trim())}
          />
        </div>
        <button
          onClick={() => apiKey && setApiKey(apiKey)}
          className="px-4 py-2 bg-blue-600 text-white rounded"
        >
          Load
        </button>
      </div>
    );
  }

  if (loading) {
    return <p>Scanning for broken links…</p>;
  }

  if (issues.length === 0) {
    return <p>No broken links found!</p>;
  }

  return (
    <div className="mt-4">
      <h4 className="font-medium mb-2">Broken Links:</h4>
      <ul className="list-disc ml-6">
        {issues.map((b, i) => (
          <li key={i}>
            <code>{b.slug}</code>:{" "}
            <a href={b.href} target="_blank" rel="noopener noreferrer">
              {b.href}
            </a>{" "}
            [{b.status}]
          </li>
        ))}
      </ul>
    </div>
  );
}

<LinkScannerV2 />
