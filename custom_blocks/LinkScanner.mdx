import React, { useState } from "react";

export function LinkScanner({
  basePath      = "/erdfactored/docs", // adjust to your sidebar prefix
  initialBranch = "1.0",               // default branch/version
  initialApiKey = "",                  // user-entered rdme_xxxxx
}) {
  const [branch, setBranch]   = useState(initialBranch);
  const [apiKey, setApiKey]   = useState(initialApiKey);
  const [records, setRecords] = useState([]);
  const [loading, setLoading] = useState(false);

  const scanLinks = async () => {
    setLoading(true);
    setRecords([]);

    // 1) scrape sidebar slugs
    const els = document.querySelectorAll(`a[href^="${basePath}/"]`);
    const slugs = Array.from(els)
      .map((a) => a.getAttribute("href").split("/").pop())
      .filter((s, i, all) => all.indexOf(s) === i);

    console.log("🔍 Found slugs:", slugs);
    if (!slugs.length) {
      console.warn("No slugs found—check your basePath!");
      setLoading(false);
      return;
    }

    // 2) fetch each guide JSON and extract links from the raw Markdown body
    const hdrs = {
      Accept:        "application/json",
      Authorization: `Bearer ${apiKey}`,
    };
    const out = [];

    for (const slug of slugs) {
      const guideUrl = `/erdfactored/api-next/v2/versions/${encodeURIComponent(branch)}/guides/${encodeURIComponent(slug)}`;
      console.log(`➡️ Fetching guide for slug "${slug}":`, guideUrl);

      let data;
      try {
        const res = await fetch(guideUrl, { credentials: "include", headers: hdrs });
        console.log(`   ← status for "${slug}":`, res.status);
        if (!res.ok) continue;
        const json = await res.json();
        data = json.data;
      } catch (e) {
        console.error("   ✖️ Error fetching guide JSON for", slug, e);
        continue;
      }

      // 3) pull links directly from the Markdown body
      const md = data.content?.body || "";
      const mdLinkRe = /\[([^\]]+)\]\((https?:\/\/[^\s)]+|\/[^\s)]+)\)/g;
      const anchors = [];
      let m, mdCount = 0;
      while ((m = mdLinkRe.exec(md))) {
        mdCount++;
        anchors.push({
          text: m[1],
          href: m[2],
          _fromMarkdown: true,
        });
      }
      console.log(`   🔄 Parsed ${mdCount} links from Markdown for "${slug}"`);

      // 4) HEAD-check each anchor, but only if it's on your root domain
      for (const a of anchors) {
        const linkText = a._fromMarkdown ? a.text : a.textContent.trim();
        const fullUrl  = a._fromMarkdown ? a.href : a.href;

        // **NEW:** accept any URL on your root domain
        const urlObj = new URL(fullUrl, location.href);
        const isInternal = urlObj.hostname === "erdman-refactored-group.readme.io";
        if (!isInternal) {
          console.log(`   ⏭️ Skipping external link: ${fullUrl}`);
          continue;
        }

        let status = "ERR", valid = false;
        try {
          const headRes = await fetch(fullUrl, {
            method:   "HEAD",
            redirect: "manual",
            mode:     "cors",
            headers:  { Authorization: `Bearer ${apiKey}` },
          });
          console.log(`   ↪️ HEAD ${fullUrl} → status: ${headRes.status}`);
          status = headRes.status;
          valid  = status >= 200 && status < 300;
        } catch (e) {
          console.warn(`   ✖️ HEAD failed for ${fullUrl}`, e);
        }

        out.push({ page: slug, text: linkText, url: fullUrl, status, valid });
      }
    }

    setRecords(out);
    setLoading(false);
  };

  const total  = records.length;
  const valid  = records.filter(r => r.valid).length;
  const broken = total - valid;

  // If records is empty (table isn't rendering), also log a Markdown version:
  if (!loading && records.length) {
    console.log("## Link Scanner Results (Markdown)\n");
    console.log("| Page | Text | URL | Status | Valid |");
    console.log("| ---- | ---- | --- | ------ | ----- |");
    records.forEach(r => {
      console.log(`| ${r.page} | ${r.text} | ${r.url} | ${r.status} | ${r.valid ? "✔️" : "❌"} |`);
    });
  }

  return (
    <div style={{ fontFamily: "sans-serif", maxWidth: 800, margin: "auto" }}>
      <h3>Link Scanner</h3>

      <div style={{ marginBottom: 16 }}>
        <label>
          Branch/Version:<br/>
          <input
            type="text"
            value={branch}
            onChange={e => setBranch(e.target.value.trim())}
            placeholder="e.g. 1.0"
            style={{ width: "100%", padding: 6, marginTop: 4 }}
          />
        </label>

        <label style={{ display: "block", marginTop: 12 }}>
          API Key:<br/>
          <input
            type="password"
            value={apiKey}
            onChange={e => setApiKey(e.target.value.trim())}
            placeholder="rdme_xxxxx"
            style={{ width: "100%", padding: 6, marginTop: 4 }}
          />
        </label>

        <button
          onClick={scanLinks}
          disabled={!branch || !apiKey || loading}
          style={{
            marginTop: 16,
            padding: "8px 16px",
            backgroundColor: "#007bff",
            color: "#fff",
            border: "none",
            cursor: "pointer",
            opacity: (!branch || !apiKey || loading) ? 0.5 : 1,
          }}
        >
          {loading ? "Scanning…" : "Scan Links"}
        </button>
      </div>

      {records.length > 0 && !loading && (
        <>
          <p>
            <strong>Results:</strong> Total: {total} | Valid: {valid} | Broken: {broken}
          </p>
          <div style={{ maxHeight: 300, overflow: "auto" }}>
            <table style={{ width: "100%", borderCollapse: "collapse" }}>
              <thead>
                <tr>
                  {["Page","Text","URL","Status","Valid"].map(h => (
                    <th
                      key={h}
                      style={{
                        border: "1px solid #ccc",
                        padding: 6,
                        background: "#f9f9f9",
                        textAlign: "left",
                      }}
                    >
                      {h}
                    </th>
                  ))}
                </tr>
              </thead>
              <tbody>
                {records.map((r, i) => (
                  <tr key={i}>
                    <td style={{ border: "1px solid #ccc", padding: 6 }}>{r.page}</td>
                    <td style={{ border: "1px solid #ccc", padding: 6 }}>{r.text}</td>
                    <td style={{ border: "1px solid #ccc", padding: 6 }}>
                      <a href={r.url} target="_blank" rel="noopener noreferrer">
                        {r.url}
                      </a>
                    </td>
                    <td style={{ border: "1px solid #ccc", padding: 6 }}>{r.status}</td>
                    <td style={{ border: "1px solid #ccc", padding: 6 }}>
                      {r.valid ? "✔️" : "❌"}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </>
      )}
    </div>
  );
}

<LinkScanner />
