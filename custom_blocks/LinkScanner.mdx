import React, { useState } from "react";

export function LinkScannerV2({
  // your project docs live under /docs/, so leave basePath for sidebar scraping only
  basePath       = "/docs",
  initialVersion = "1.0",
  initialApiKey  = "",
}) {
  const [version,    setVersion]    = useState(initialVersion);
  const [apiKey,     setApiKey]     = useState(initialApiKey);
  const [foundLinks, setFoundLinks] = useState(null);
  const [issues,     setIssues]     = useState(null);
  const [loading,    setLoading]    = useState(false);

  // Step 1: scrape sidebar <a> slugs → fetch each guide JSON → extract Markdown links
  const findLinks = async () => {
    setLoading(true);
    setFoundLinks(null);
    setIssues(null);

    // 1a) get unique slugs from your sidebar nav
    const navAs = document.querySelectorAll(`a[href^="${basePath}/"]`);
    const pages = Array.from(navAs)
      .map((a) => a.getAttribute("href").split("/").pop())
      .filter((slug, i, arr) => arr.indexOf(slug) === i);

    // 1b) auth header
    const hdrs = new Headers({ Accept: "application/json" });
    hdrs.set("Authorization", `Basic ${btoa(apiKey + ":")}`);

    // 1c) regex for Markdown links
    const mdLinkRe = /\[([^\]]+)\]\((https?:\/\/[^\s)]+)\)/g;
    const links = [];

    // 1d) fetch each page’s Markdown via the v2 Refactored branch endpoint
    await Promise.all(
      pages.map(async (slug) => {
        try {
          const url = `${window.location.origin}/api-next/v2/branches/${encodeURIComponent(
            version
          )}/guides/${slug}`;
          const res = await fetch(url, { headers: hdrs });
          if (!res.ok) {
            console.warn(`404 for ${slug}, skipping`);
            return;
          }
          const { data } = await res.json();
          const md = data.content.body || "";
          let m;
          while ((m = mdLinkRe.exec(md))) {
            links.push({ slug, href: m[2] });
          }
        } catch (e) {
          console.error(`Error fetching ${slug}:`, e);
        }
      })
    );

    setFoundLinks(links);
    setLoading(false);
  };

  // Step 2: HEAD-check each extracted link
  const checkBroken = async () => {
    setLoading(true);
    const bad = [];
    const hdrs = new Headers();
    hdrs.set("Authorization", `Basic ${btoa(apiKey + ":")}`);

    for (const { slug, href } of foundLinks) {
      try {
        const head = await fetch(href, {
          method: "HEAD",
          headers: hdrs,
          redirect: "manual",
        });
        if (!(head.status >= 200 && head.status < 300)) {
          bad.push({ slug, href, status: head.status });
        }
      } catch {
        bad.push({ slug, href, status: "ERR" });
      }
    }

    setIssues(bad);
    setLoading(false);
  };

  // — RENDER FLOW —

  // 0) Inputs + Find Links
  if (foundLinks === null && !loading) {
    return (
      <div className="mb-4 w-full">
        <label className="block text-sm font-medium mb-1">Project Branch</label>
        <input
          type="text"
          className="w-full border rounded p-2 mb-3"
          placeholder="e.g. 1.0"
          value={version}
          onChange={(e) => setVersion(e.target.value.trim())}
        />

        <label className="block text-sm font-medium mb-1">
          API Key (Basic auth, required)
        </label>
        <input
          type="password"
          className="w-full border rounded p-2 mb-3"
          placeholder="rdme_xxxxx"
          value={apiKey}
          onChange={(e) => setApiKey(e.target.value.trim())}
        />

        <button
          onClick={findLinks}
          disabled={!version || !apiKey}
          className="px-4 py-2 bg-blue-600 text-white rounded disabled:opacity-50"
        >
          Find Links in Content
        </button>
      </div>
    );
  }

  // 1) Loading
  if (loading) {
    return <p>Working…</p>;
  }

  // 2) Found links → show list + “Check Broken”
  if (foundLinks !== null && issues === null) {
    return (
      <div className="mt-4">
        <h4 className="font-medium mb-2">
          Found {foundLinks.length} Content Links
        </h4>
        <ul className="list-disc ml-6 max-h-48 overflow-auto mb-3">
          {foundLinks.map((l, i) => (
            <li key={i}>
              <code>{l.slug}</code>:{" "}
              <a href={l.href} target="_blank" rel="noopener noreferrer">
                {l.href}
              </a>
            </li>
          ))}
        </ul>
        <button
          onClick={checkBroken}
          className="px-4 py-2 bg-red-600 text-white rounded"
        >
          Check Broken
        </button>
      </div>
    );
  }

  // 3) Show broken results
  if (issues !== null) {
    if (issues.length === 0) {
      return <p>No broken links found!</p>;
    }
    return (
      <div className="mt-4">
        <h4 className="font-medium mb-2">Broken Links Detected:</h4>
        <ul className="list-disc ml-6">
          {issues.map((b, i) => (
            <li key={i}>
              <code>{b.slug}</code>:{" "}
              <a href={b.href} target="_blank" rel="noopener noreferrer">
                {b.href}
              </a>{" "}
              [{b.status}]
            </li>
          ))}
        </ul>
      </div>
    );
  }

  return null;
}

<LinkScannerV2 basePath="/erdfactored/docs" initialVersion="1.0" />
