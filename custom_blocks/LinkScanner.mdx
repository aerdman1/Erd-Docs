import React, { useState } from "react";

export function LinkScannerV2({
  initialApiKey = "",
  initialVersion = "",
}) {
  const [apiKey, setApiKey] = useState(initialApiKey);
  const [version, setVersion] = useState(initialVersion);
  const [issues, setIssues] = useState(null);
  const [loading, setLoading] = useState(false);

  const scanLinks = async () => {
    setLoading(true);
    setIssues(null);
    try {
      const headers = new Headers({ Accept: "application/json" });
      headers.set("Authorization", `Bearer ${apiKey}`);

      // 1) fetch all guides via proxy
      const listRes = await fetch(
        `/api/v2/versions/${encodeURIComponent(version)}/guides`,
        { headers, credentials: "include" }
      );
      if (!listRes.ok) throw new Error("Could not fetch guides");
      const { data: guides } = await listRes.json();

      // 2) scan each guide
      const hrefRegex = /<a\s+[^>]*href="([^"]+)"/g;
      const found = [];

      for (const g of guides) {
        const pageRes = await fetch(
          `/api/v2/versions/${encodeURIComponent(version)}/guides/${g.slug}`,
          { headers, credentials: "include" }
        );
        if (!pageRes.ok) continue;
        const { data } = await pageRes.json();
        let m;
        while ((m = hrefRegex.exec(data.content.body))) {
          let href = m[1];
          // make absolute
          if (!/^https?:\/\//.test(href)) {
            href = window.location.origin + (href.startsWith("/") ? "" : "/") + href;
          }
          try {
            const head = await fetch(href, { method: "HEAD", redirect: "manual" });
            if (!(head.status >= 200 && head.status < 300)) {
              found.push({ slug: g.slug, href, status: head.status });
            }
          } catch {
            found.push({ slug: g.slug, href, status: "ERR" });
          }
        }
      }

      setIssues(found);
    } catch (err) {
      console.error(err);
      setIssues([]); // treat as “no results” to avoid endless loading
    } finally {
      setLoading(false);
    }
  };

  // Prompt until user clicks Scan
  if (issues === null && !loading) {
    return (
      <div className="mb-4 w-full">
        <label className="block text-sm font-medium mb-1">ReadMe API Key</label>
        <input
          type="password"
          className="w-full border rounded p-2 mb-3"
          placeholder="Paste your API key…"
          value={apiKey}
          onChange={(e) => setApiKey(e.target.value.trim())}
        />
        <label className="block text-sm font-medium mb-1">Project Version (branch)</label>
        <input
          type="text"
          className="w-full border rounded p-2 mb-3"
          placeholder="e.g. stable or v2.0.0"
          value={version}
          onChange={(e) => setVersion(e.target.value.trim())}
        />
        <button
          onClick={scanLinks}
          disabled={!apiKey || !version}
          className="px-4 py-2 bg-blue-600 text-white rounded disabled:opacity-50"
        >
          Scan
        </button>
      </div>
    );
  }

  if (loading) {
    return <p>Scanning for broken links…</p>;
  }

  // After scan completes
  if (issues && issues.length === 0) {
    return <p>No broken links found!</p>;
  }

  return (
    <div className="mt-4">
      <h4 className="font-medium mb-2">Broken Links Detected:</h4>
      <ul className="list-disc ml-6">
        {issues.map((b, i) => (
          <li key={i}>
            <code>{b.slug}</code>:{" "}
            <a href={b.href} target="_blank" rel="noopener noreferrer">
              {b.href}
            </a>{" "}
            [{b.status}]
          </li>
        ))}
      </ul>
    </div>
  );
}

<LinkScannerV2 />
