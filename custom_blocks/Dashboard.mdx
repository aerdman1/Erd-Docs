import React, { useEffect, useState } from 'react';

export const DocumentsAPI = () => {
  const [endpoints, setEndpoints] = useState([]);
  const [loading, setLoading] = useState(true);
  const [err, setErr] = useState(null);

  useEffect(() => {
    const fetchData = async () => {
      try {
        if (typeof document === 'undefined') return;
        const sidebar = document.querySelector('.rm-Sidebar');

        if (!sidebar) {
          setErr("Sidebar not found.");
          setLoading(false);
          return;
        }

        // Find all sections with a specific heading (e.g., "Document Management API")
        const targetSection = Array.from(sidebar.querySelectorAll('.rm-Sidebar-section'))
          .find(section => section.querySelector('.rm-Sidebar-heading')?.textContent.includes("Document Management API"));

        if (!targetSection) {
          setErr("Target API category not found.");
          setLoading(false);
          return;
        }

        // Select only the links within this section and exclude the Overview page
        const allLinks = Array.from(targetSection.querySelectorAll('a.rm-Sidebar-link'))
          .filter(link => {
            const href = link.getAttribute('href');
            return href && href.startsWith('/reference/') && href !== '/reference/overview';
          });

        // Extract HTTP method and title from each link
        const endpointsData = allLinks.map(link => {
          const href = link.getAttribute('href'); // This is the ReadMe reference page URL
          const methodEl = link.querySelector('.rm-APIMethod');
          const method = methodEl ? methodEl.textContent.trim().toUpperCase() : "";
          const titleEl = link.querySelector('.Sidebar-link-text_label1gCT_uPnx7Gu');
          const title = titleEl ? titleEl.textContent.trim() : link.textContent.trim();
          return { href, method, title };
        });

        // Fetch endpoint details
        const detailsPromises = endpointsData.map(async ep => {
          const fullUrl = window.location.origin + ep.href;
          try {
            const response = await fetch(fullUrl);
            if (!response.ok) throw new Error("Network error");
            const html = await response.text();
            const parser = new DOMParser();
            const doc = parser.parseFromString(html, "text/html");
            const urlEl = doc.querySelector('[data-testid="serverurl"]') ||
                          doc.querySelector('.headline-container-article-info-url2nV_XrjpFuVQ');
            const endpointUrl = urlEl ? urlEl.textContent.trim() : "Not found";
            return { ...ep, endpointUrl };
          } catch (error) {
            console.error("Error fetching", ep.href, error);
            return { ...ep, endpointUrl: "Error" };
          }
        });

        const results = await Promise.all(detailsPromises);
        setEndpoints(results);
      } catch (e) {
        setErr(e.message);
      } finally {
        setLoading(false);
      }
    };

    fetchData();
  }, []);

  if (loading) return <p>Loading endpoints...</p>;
  if (err) return <p>Error: {err}</p>;

  return (
    <div className="grid">
      {endpoints.map((ep, index) => (
        <a key={index} href={ep.href} className="card" target="_self">
          <div className="card-content">
            {ep.method && (
              <span className={`method ${ep.method.toLowerCase()}`}>
                {ep.method}
              </span>
            )}
            <h3>{ep.title}</h3>
            <div className="path">{ep.endpointUrl}</div>
          </div>
          <span className="arrow">›</span>
        </a>
      ))}
      <style dangerouslySetInnerHTML={{ __html: `
        .grid {
          display: grid;
          grid-template-columns: 1fr;
          gap: 8px;
        }
        .card {
          display: flex;
          align-items: center;
          justify-content: space-between;
          background: #ffffff;
          border: 1px solid #d1d5db;
          border-radius: 8px;
          padding: 16px;
          text-decoration: none !important;
          color: inherit;
          transition: background 0.2s;
        }
        .card:hover {
          background: #f9fafb;
        }
        .card-content {
          flex-grow: 1;
        }
        .card h3 {
          margin: 0;
          font-size: 1rem;
          font-weight: bold;
          color: #111827;
        }
        .card .method {
          font-weight: bold;
          text-transform: uppercase;
          font-size: 0.85rem;
          margin-bottom: 4px;
          display: block;
        }
        .card .method.get { color: #2563eb; }
        .card .method.post { color: #16a34a; }
        .card .method.put { color: #ca8a04; }
        .card .method.delete { color: #dc2626; }
        .card .path {
          font-family: monospace;
          font-size: 0.9rem;
          color: #6b7280;
        }
        .arrow {
          font-size: 1.2rem;
          color: #6b7280;
        }
      ` }} />
    </div>
  );
};

<DocumentsAPI />
