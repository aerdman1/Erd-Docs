import React, { useEffect, useState } from 'react';

export const DashboardComponent = ({ header = "My Dashboard" }) => {
  // Existing states
  const [pageCount, setPageCount] = useState(0);
  const [totalCategories, setTotalCategories] = useState(0);
  const [categories, setCategories] = useState([]);
  const [hubData, setHubData] = useState(null);
  const [expandedCategories, setExpandedCategories] = useState({});

  // New states for endpoint/guide details
  const [pageDetails, setPageDetails] = useState([]);
  const [fetchingPages, setFetchingPages] = useState(false);
  const [pageError, setPageError] = useState(null);

  // Determine current page type (reference vs guides)
  const currentPath = typeof window !== "undefined" ? window.location.pathname : "";
  const isReference = currentPath.includes("/reference/");

  // On mount, grab sidebar links and user info
  useEffect(() => {
    if (typeof document !== "undefined") {
      const navElement = document.querySelector('.rm-Sidebar');
      if (navElement) {
        // For reference pages, only count links starting with "/reference/"
        // For guides pages, count all links (or those starting with "/docs/" if needed)
        let pageLinks;
        if (isReference) {
          pageLinks = navElement.querySelectorAll('a[href^="/reference/"]');
        } else {
          pageLinks = navElement.querySelectorAll('a');
        }
        setPageCount(pageLinks.length);

        // Extract categories from sidebar sections
        const sectionElements = navElement.querySelectorAll('section.rm-Sidebar-section');
        const catArray = [];
        sectionElements.forEach((section) => {
          const heading = section.querySelector('h2');
          if (heading) {
            const title = heading.textContent.trim();
            // Grab all links inside this category.
            const links = section.querySelectorAll('a');
            const linkArray = Array.from(links).map(link => ({
              title: link.textContent.trim(),
              href: link.getAttribute('href')
            }));
            catArray.push({ title, links: linkArray });
          }
        });
        setCategories(catArray);
        setTotalCategories(catArray.length);
      }

      // Parse Hub‑Me user data from the embedded script.
      const hubMeElement = document.getElementById("hub-me");
      if (hubMeElement) {
        let dataJson = hubMeElement.getAttribute("data-json");
        dataJson = dataJson.replace(/&quot;/g, '"');
        try {
          const parsedHubData = JSON.parse(dataJson);
          setHubData(parsedHubData);
        } catch (error) {
          console.error("Error parsing Hub‑Me data:", error);
        }
      }
    }
  }, [isReference]);

  const toggleCategory = (index) => {
    setExpandedCategories(prev => ({
      ...prev,
      [index]: !prev[index]
    }));
  };

  // Helper: Extract key page info from an HTML string
  const extractPageInfo = (htmlString) => {
    try {
      const parser = new DOMParser();
      const doc = parser.parseFromString(htmlString, "text/html");

      // Title: try header with id "content-head" then fallback to document title
      const titleEl = doc.querySelector("header#content-head h1");
      const pageTitle = titleEl ? titleEl.innerText.trim() : doc.title || "No Title Found";

      // Description: from element with role "doc-subtitle"
      const descEl = doc.querySelector("div[role='doc-subtitle']");
      const description = descEl ? descEl.innerText.trim() : "No Description Found";

      // HTTP Method (likely for endpoints): look for any element with class "rm-APIMethod"
      const methodEl = doc.querySelector("span.rm-APIMethod");
      const method = methodEl ? methodEl.innerText.trim() : "No Method Found";

      // Endpoint URL (for reference pages): from a span with a unique class (adjust as needed)
      const urlEl = doc.querySelector("span.headline-container-article-info-url2nV_XrjpFuVQ");
      const endpointUrl = urlEl ? urlEl.innerText.trim() : "No Endpoint URL Found";

      // Last Updated: try several methods to locate the updated info.
      let updatedEl = doc.querySelector("div.UpdatedAt p.DateLine");
      if (!updatedEl) {
        // Look for any DateLine within the main article that contains "Updated"
        const articleEl = doc.querySelector(".rm-Article");
        if (articleEl) {
          const candidates = articleEl.querySelectorAll("p.DateLine");
          updatedEl = Array.from(candidates).find(p => p.innerText.includes("Updated"));
        }
      }
      const updated = updatedEl ? updatedEl.innerText.trim() : "No Update Info Found";

      return { pageTitle, description, method, endpointUrl, updated };
    } catch (e) {
      console.error("Error extracting page info:", e);
      return null;
    }
  };

  // Function to fetch details for all reference or guide links
  const fetchAllPageDetails = async () => {
    setFetchingPages(true);
    setPageError(null);
    // Flatten links from categories
    const flattenedLinks = categories.flatMap(cat => cat.links);
    let targetLinks;
    if (isReference) {
      targetLinks = flattenedLinks.filter(link => link.href && link.href.startsWith("/reference/"));
    } else {
      // For guides pages, adjust the filter as needed (here we assume guides use "/docs/")
      targetLinks = flattenedLinks.filter(link => link.href && link.href.startsWith("/docs/"));
    }
    // Build full URL for each link (assuming same origin)
    const fullUrls = targetLinks.map(link => ({
      ...link,
      fullUrl: window.location.origin + link.href
    }));

    try {
      const detailsPromises = fullUrls.map(async (link) => {
        try {
          const response = await fetch(link.fullUrl);
          if (!response.ok) throw new Error("Network response was not ok");
          const html = await response.text();
          const info = extractPageInfo(html);
          return { ...link, ...info };
        } catch (err) {
          console.error("Error fetching", link.href, err);
          return { ...link, error: err.message };
        }
      });
      const results = await Promise.all(detailsPromises);
      setPageDetails(results);
    } catch (err) {
      setPageError(err.message);
    } finally {
      setFetchingPages(false);
    }
  };

  // Compute a tally of pages by HTTP method (useful mostly for reference pages)
  const methodTally = pageDetails.reduce((acc, page) => {
    if (page.method) {
      acc[page.method] = (acc[page.method] || 0) + 1;
    }
    return acc;
  }, {});

  return (
    <>
      {/* Global Custom CSS */}
      <style>{`
        section.content-toc.grid-25 { display: none; }
        .rm-ReferenceMain .rm-Article { max-width: 100%; }
        li { list-style: none; }
        .readme-tailwind .text-blue-500 { text-decoration: none !important; }
        .field-description a[href], .field-description a:not([href=""]),
        .markdown-body a[href], .markdown-body a:not([href=""]) {
          text-decoration: none;
          padding: 10px;
          display: inline-block;
          border-radius: 10px;
          min-height:50px;
        }
        .p-4.bg-white.rounded-xl.border.hover\\:shadow-2xl.transition.transform.hover\\:scale-105 {
          border: 1px solid #e3e3e3;
        }
        .field-description li+li, .markdown-body li+li { margin-top: 0; }
      `}</style>

      {/* Main Container */}
      <div className="max-w-screen-xl mx-auto p-4">
        <div className="grid grid-cols-2 gap-4">
          {/* Left Column: Dashboard Info */}
          <div className="rounded-xl p-6 shadow-xl border border-gray-300 bg-white">
            <p className="text-2xl font-bold mb-4 text-center bg-clip-text text-transparent bg-gradient-to-r from-blue-500 to-green-500">
              {header}
            </p>
            <div className="space-y-6">
              {/* User Information */}
              <div className="p-4 rounded-xl shadow-md bg-gradient-to-br from-purple-400 to-pink-300 text-white">
                <p className="text-xl font-semibold mb-2">User Information</p>
                {hubData ? (
                  <div className="space-y-1">
                    <p><strong>Name:</strong> {hubData.user?.name}</p>
                    <p><strong>Email:</strong> {hubData.user?.email}</p>
                    <p><strong>Logged In:</strong> {hubData.loggedIn ? "Yes" : "No"}</p>
                    {hubData.isAdmin && <p><strong>Admin:</strong> Yes</p>}
                    {hubData.metaData && hubData.metaData.length > 0 && (
                      <p><strong>Project:</strong> {hubData.metaData[0].name}</p>
                    )}
                  </div>
                ) : (
                  <p>No user data</p>
                )}
              </div>
              {/* Stats */}
              <div className="flex flex-row gap-4">
                <div className="flex-1 p-4 rounded-xl shadow-md bg-gradient-to-br from-blue-400 to-blue-600 text-white text-center transform transition hover:scale-105">
                  <p className="text-xl font-semibold">Total Pages</p>
                  <p className="text-3xl">{pageCount}</p>
                </div>
                <div className="flex-1 p-4 rounded-xl shadow-md bg-gradient-to-br from-green-400 to-green-600 text-white text-center transform transition hover:scale-105">
                  <p className="text-xl font-semibold">Total Categories</p>
                  <p className="text-3xl">{totalCategories}</p>
                </div>
              </div>
            </div>
          </div>

          {/* Right Column: Categories Launchpad */}
          <div className="rounded-xl p-6 shadow-xl border border-gray-300">
            <p className="text-2xl font-bold mb-4 text-center">Categories Launchpad</p>
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
              {categories.map((cat, index) => (
                <div key={index} className="p-4 bg-white rounded-xl border hover:shadow-2xl transition transform hover:scale-105">
                  <div className="flex justify-between items-center mb-2">
                    <p className="text-xl font-semibold">{cat.title}</p>
                    <button
                      onClick={() => toggleCategory(index)}
                      className="text-sm text-blue-500 underline"
                    >
                      {expandedCategories[index] ? 'Hide' : 'Show'} ({cat.links.length})
                    </button>
                  </div>
                  {expandedCategories[index] && (
                    <ul className="mt-2 grid grid-cols-2 gap-2">
                      {cat.links.map((link, idx) => (
                        <li key={idx}>
                          <a 
                            href={link.href} 
                            className="block w-full text-gray-800 no-underline hover:text-gray-900 bg-gray-100 p-2 rounded transition"
                          >
                            {link.title}
                          </a>
                        </li>
                      ))}
                    </ul>
                  )}
                </div>
              ))}
            </div>
          </div>
        </div>

        {/* New Section: Page Details Overview */}
        <div className="mt-10 rounded-xl p-6 shadow-xl border border-gray-300 bg-white">
          <p className="text-2xl font-bold mb-4 text-center">
            {isReference ? "Endpoint Details Overview" : "Guide Page Details Overview"}
          </p>
          <div className="mb-4 flex justify-center">
            <button 
              onClick={fetchAllPageDetails} 
              className="px-4 py-2 bg-blue-500 text-white rounded hover:bg-blue-600"
              disabled={fetchingPages}
            >
              {fetchingPages
                ? "Fetching Data..."
                : isReference 
                  ? "Fetch Endpoint Data" 
                  : "Fetch Guide Page Data"}
            </button>
          </div>
          {pageError && <p className="text-red-500">{pageError}</p>}
          {pageDetails.length > 0 && (
            <>
              {/* Tally Section (only really relevant for endpoints) */}
              {isReference && (
                <div className="mb-6">
                  <h3 className="text-xl font-semibold">Endpoint Method Tally</h3>
                  <ul className="list-disc list-inside">
                    {Object.entries(methodTally).map(([method, count]) => (
                      <li key={method}>
                        <strong>{method}:</strong> {count}
                      </li>
                    ))}
                  </ul>
                </div>
              )}
              {/* Page Cards Grid */}
              <div style={{
                display: 'grid',
                gridTemplateColumns: 'repeat(auto-fit, minmax(250px, 1fr))',
                gap: '1rem'
              }}>
                {pageDetails.map((page, idx) => (
                  <div key={idx} className="p-4 border rounded shadow-md">
                    <p><strong>Page Title:</strong> {page.pageTitle || "N/A"}</p>
                    <p><strong>Description:</strong> {page.description || "N/A"}</p>
                    {isReference && (
                      <>
                        <p><strong>Method:</strong> {page.method || "N/A"}</p>
                        <p><strong>Endpoint URL:</strong> {page.endpointUrl || "N/A"}</p>
                      </>
                    )}
                    <p><strong>Last Updated:</strong> {page.updated || "N/A"}</p>
                    {page.error && <p className="text-red-500">Error: {page.error}</p>}
                    <a href={page.href} target="_blank" rel="noopener noreferrer" className="text-blue-500 underline">
                      View Page
                    </a>
                  </div>
                ))}
              </div>
            </>
          )}
        </div>
      </div>
    </>
  );
};

<DashboardComponent header="Interactive Dashboard" />
