import React, { useEffect, useState } from 'react';

export default function GuidesIframe({ endpoint }) {
  const [codeText, setCodeText] = useState('');
  const [error, setError] = useState('');
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    if (!endpoint) {
      setError('IFRAME TESTING');
      setLoading(false);
      return;
    }

    let docsUrl = mapEndpointToDocsUrl(endpoint);
    console.log('Loading endpoint via iframe:', docsUrl);

    const iframe = document.createElement('iframe');
    iframe.style.position = 'absolute';
    iframe.style.top = '-9999px';
    iframe.style.width = '1px'; 
    iframe.style.height = '1px';
    iframe.src = docsUrl;
    document.body.appendChild(iframe);

    const cleanup = () => {
      if (iframe.parentNode) {
        iframe.parentNode.removeChild(iframe);
      }
    };

    iframe.onload = () => {
      try {
        const iDoc = iframe.contentDocument || iframe.contentWindow.document;
        console.log('Iframe loaded for:', endpoint);

        setTimeout(() => {
          const exampleButton = Array.from(iDoc.querySelectorAll('button')).find(btn =>
            btn.innerText.trim().toUpperCase().includes('EXAMPLE')
          );

          if (!exampleButton) {
            setError('EXAMPLE button not found in iframe.');
            setLoading(false);
            cleanup();
            return;
          }

          console.log('Clicking EXAMPLE button.');
          exampleButton.click();

          setTimeout(() => {
            const popup = iDoc.querySelector('[data-tippy-root]');
            if (!popup) {
              setError('Popup (data-tippy-root) not found.');
              setLoading(false);
              cleanup();
              return;
            }

            console.log('Popup found, looking for "Request Example".');
            const reqExampleOption = Array.from(popup.querySelectorAll('.Menu-Item')).find(el =>
              el.innerText.trim().toUpperCase().includes('REQUEST EXAMPLE')
            );

            if (!reqExampleOption) {
              setError('"Request Example" option not found.');
              setLoading(false);
              cleanup();
              return;
            }

            console.log('Clicking Request Example option.');
            reqExampleOption.click();

            setTimeout(() => {
              const codeContainer = iDoc.querySelector('.react-codemirror2.CodeEditor');

              if (!codeContainer) {
                setError('CodeMirror container not found.');
                setLoading(false);
                cleanup();
                return;
              }

              const codeLines = codeContainer.querySelectorAll('.CodeMirror-lines pre.CodeMirror-line');

              if (codeLines.length === 0) {
                setError('No content found in CodeMirror.');
                setLoading(false);
                cleanup();
                return;
              }

              const extractedCode = Array.from(codeLines).map(line => line.textContent).join('\n');
              console.log('Extracted code:', extractedCode);

              setCodeText(extractedCode);
              setLoading(false);
              cleanup();
            }, 3000);
          }, 2000);
        }, 2000);
      } catch (err) {
        console.error('Error processing iframe:', err);
        setError(`Error accessing iframe: ${err.message}`);
        setLoading(false);
        cleanup();
      }
    };

    return cleanup;
  }, [endpoint]);

  const mapEndpointToDocsUrl = (path) => {
    switch (path) {
      case '/pet':
        return 'https://erdtest123.readme.io/reference/updatepet';
      case '/pet/{petId}':
        return 'https://erdtest123.readme.io/reference/getpetbyid';
      case '/pet/findByStatus':
        return 'https://erdtest123.readme.io/reference/findpetbystatus';
      default:
        return `https://erdtest123.readme.io/reference${path}`;
    }
  };

  if (loading) {
    return <div>Loading example response (iframe mode)...</div>;
  }

  if (error) {
    return <div style={{ color: 'red', whiteSpace: 'pre-wrap' }}>{error}</div>;
  }

  if (!codeText) {
    return <div>No example response found.</div>;
  }

  return (
    <div style={{ border: '1px solid #ddd', padding: '16px', borderRadius: '8px', marginTop: '16px' }}>
      <h3>Example Response (from iframe) for {endpoint}</h3>
      <pre style={{
        whiteSpace: 'pre-wrap',
        background: '#f5f5f5',
        padding: '8px',
        borderRadius: '4px',
        marginTop: '8px',
      }}>
        {codeText}
      </pre>
    </div>
  );
}

<GuidesIframe/>
