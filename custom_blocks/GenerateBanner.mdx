name: GenerateBanner
---
import React, { useState } from 'react';

export const BannerGenerator = () => {
  const [message, setMessage] = useState('This is my banner message!');
  const [bannerColor, setBannerColor] = useState('#00bcff');
  const [bannerTextColor, setBannerTextColor] = useState('#ffffff');
  const [fontSize, setFontSize] = useState('16px');
  const [fontWeight, setFontWeight] = useState('normal');
  const [preset, setPreset] = useState('custom'); // 'critical', 'warning', 'standard', 'custom'

  const handlePresetChange = (e) => {
    const selectedPreset = e.target.value;
    setPreset(selectedPreset);

    if (selectedPreset === 'critical') {
      setBannerColor('#dc2626');
      setBannerTextColor('#ffffff');
    } else if (selectedPreset === 'warning') {
      setBannerColor('#facc15');
      setBannerTextColor('#000000');
    } else if (selectedPreset === 'standard') {
      setBannerColor('#2563eb');
      setBannerTextColor('#ffffff');
    } else if (selectedPreset === 'custom') {
      setBannerColor('#00bcff');
      setBannerTextColor('#ffffff');
    }
  };

  const generateBannerProps = () => {
    let props = `message="${message}"`;
    if (preset !== 'custom') {
      props += ` ${preset}={true}`;
    } else {
      props += ` bannerColor="${bannerColor}" bannerTextColor="${bannerTextColor}"`;
    }
    props += ` fontSize="${fontSize}" fontWeight="${fontWeight}"`;
    return props;
  };

  const generatedSnippet = `<Banner ${generateBannerProps()} />`;

  const actualBannerColor =
    preset === 'custom' ? bannerColor :
    preset === 'critical' ? '#dc2626' :
    preset === 'warning' ? '#facc15' :
    preset === 'standard' ? '#2563eb' : bannerColor;

  const actualBannerTextColor =
    preset === 'custom' ? bannerTextColor :
    preset === 'critical' ? '#ffffff' :
    preset === 'warning' ? '#000000' :
    preset === 'standard' ? '#ffffff' : bannerTextColor;

  // Centralized style object for a modern look
  const styles = {
    container: {
      maxWidth: '100%',
      margin: '2rem auto',
      padding: '2rem',
      background: '#fff',
      borderRadius: '10px',
      boxShadow: '0 4px 12px rgba(0, 0, 0, 0.1)',
      fontFamily: '"Segoe UI", Tahoma, Geneva, Verdana, sans-serif',
    },
    heading: {
      textAlign: 'center',
      marginBottom: '1.5rem',
      color: '#333',
    },
    label: {
      display: 'block',
      marginBottom: '1rem',
      fontWeight: '500',
      color: '#555',
    },
    input: {
      width: '100%',
      padding: '0.5rem',
      marginTop: '0.5rem',
      borderRadius: '5px',
      border: '1px solid #ccc',
    },
    select: {
      width: '100%',
      padding: '0.5rem',
      marginTop: '0.5rem',
      borderRadius: '5px',
      border: '1px solid #ccc',
    },
    preview: {
      backgroundColor: actualBannerColor,
      color: actualBannerTextColor,
      fontSize,
      fontWeight,
      padding: '1rem',
      textAlign: 'center',
      borderRadius: '5px',
      marginBottom: '1.5rem',
    },
    code: {
      background: '#f0f0f0',
      padding: '1rem',
      borderRadius: '5px',
      overflowX: 'auto',
      fontFamily: 'monospace',
      marginBottom: '1.5rem',
    },
    button: {
      padding: '0.75rem 1.5rem',
      cursor: 'pointer',
      background: '#4CAF50',
      color: 'white',
      border: 'none',
      borderRadius: '5px',
      display: 'block',
      margin: '0 auto',
      fontSize: '1rem',
    },
    sectionHeading: {
      fontSize: '1.25rem',
      marginBottom: '1rem',
      textAlign: 'center',
      color: '#333',
    },
  };

  return (
    <div style={styles.container}>
      <h3 style={styles.heading}>Banner Generator</h3>

      <label style={styles.label}>
        Message:
        <input
          type="text"
          value={message}
          onChange={(e) => setMessage(e.target.value)}
          style={styles.input}
        />
      </label>

      <label style={styles.label}>
        Preset:
        <select value={preset} onChange={handlePresetChange} style={styles.select}>
          <option value="custom">Custom (you choose colors)</option>
          <option value="critical">Critical (Red)</option>
          <option value="warning">Warning (Yellow)</option>
          <option value="standard">Standard (Blue)</option>
        </select>
      </label>

      {preset === 'custom' && (
        <>
          <label style={styles.label}>
            Banner Background Color:
            <input
              type="color"
              value={bannerColor}
              onChange={(e) => setBannerColor(e.target.value)}
              style={{ ...styles.input, padding: '0.2rem' }}
            />
          </label>
          <label style={styles.label}>
            Banner Text Color:
            <input
              type="color"
              value={bannerTextColor}
              onChange={(e) => setBannerTextColor(e.target.value)}
              style={{ ...styles.input, padding: '0.2rem' }}
            />
          </label>
        </>
      )}

      <label style={styles.label}>
        Font Size:
        <input
          type="text"
          value={fontSize}
          onChange={(e) => setFontSize(e.target.value)}
          style={styles.input}
        />
      </label>

      <label style={styles.label}>
        Font Weight:
        <input
          type="text"
          value={fontWeight}
          onChange={(e) => setFontWeight(e.target.value)}
          style={styles.input}
        />
      </label>

      {/* Banner Preview */}
      <h4 style={styles.sectionHeading}>Banner Preview:</h4>
      <div style={styles.preview}>{message}</div>

      {/* Generated Code */}
      <h4 style={styles.sectionHeading}>Generated Code:</h4>
      <pre style={styles.code}>{generatedSnippet}</pre>

      <button
        onClick={() => navigator.clipboard.writeText(generatedSnippet)}
        style={styles.button}
      >
        Copy to Clipboard
      </button>
    </div>
  );
};

<BannerGenerator />
