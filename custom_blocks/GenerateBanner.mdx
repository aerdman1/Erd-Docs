---
name: GenerateBanner
---
import React, { useState, useEffect } from 'react';

export const BannerGenerator = ({
  message: propMessage = 'This is my banner message!',
  bannerColor: propBannerColor = '#00bcff',
  bannerTextColor: propBannerTextColor = '#ffffff',
  fontSize: propFontSize = '16px',
  fontWeight: propFontWeight = 'normal',
}) => {
  // Check if the user is an admin based on the hub-me script data
  let isAdmin = false;
  if (typeof document !== 'undefined') {
    const hubElem = document.getElementById('hub-me');
    if (hubElem) {
      try {
        const hubData = JSON.parse(hubElem.getAttribute('data-json'));
        isAdmin = hubData.isAdmin;
      } catch (e) {
        console.error('Error parsing hub-me data', e);
      }
    }
  }

  // Use the incoming props to initialize state
  const [message, setMessage] = useState(propMessage);
  const [bannerColor, setBannerColor] = useState(propBannerColor);
  const [bannerTextColor, setBannerTextColor] = useState(propBannerTextColor);
  const [fontSize, setFontSize] = useState(propFontSize);
  const [fontWeight, setFontWeight] = useState(propFontWeight);
  const [preset, setPreset] = useState('custom'); // 'critical', 'warning', 'standard', 'custom'

  // Sync state with props if they change externally
  useEffect(() => {
    setMessage(propMessage);
    setBannerColor(propBannerColor);
    setBannerTextColor(propBannerTextColor);
    setFontSize(propFontSize);
    setFontWeight(propFontWeight);
  }, [propMessage, propBannerColor, propBannerTextColor, propFontSize, propFontWeight]);

  const handlePresetChange = (e) => {
    const selectedPreset = e.target.value;
    setPreset(selectedPreset);

    if (selectedPreset === 'critical') {
      setBannerColor('#dc2626');
      setBannerTextColor('#ffffff');
    } else if (selectedPreset === 'warning') {
      setBannerColor('#facc15');
      setBannerTextColor('#000000');
    } else if (selectedPreset === 'standard') {
      setBannerColor('#2563eb');
      setBannerTextColor('#ffffff');
    } else if (selectedPreset === 'custom') {
      setBannerColor('#00bcff');
      setBannerTextColor('#ffffff');
    }
  };

  const generateBannerProps = () => {
    let props = `message="${message}"`;
    if (preset !== 'custom') {
      props += ` ${preset}={true}`;
    } else {
      props += ` bannerColor="${bannerColor}" bannerTextColor="${bannerTextColor}"`;
    }
    props += ` fontSize="${fontSize}" fontWeight="${fontWeight}"`;
    return props;
  };

  // Updated snippet now outputs the unified component's name
  const generatedSnippet = `<BannerGenerator ${generateBannerProps()} />`;

  const actualBannerColor =
    preset === 'custom' ? bannerColor :
    preset === 'critical' ? '#dc2626' :
    preset === 'warning' ? '#facc15' :
    preset === 'standard' ? '#2563eb' : bannerColor;

  const actualBannerTextColor =
    preset === 'custom' ? bannerTextColor :
    preset === 'critical' ? '#ffffff' :
    preset === 'warning' ? '#000000' :
    preset === 'standard' ? '#ffffff' : bannerTextColor;

  // Centralized style object for the generator UI
  const styles = {
    container: {
      maxWidth: '600px',
      margin: '2rem auto',
      padding: '2rem',
      background: '#fff',
      borderRadius: '10px',
      boxShadow: '0 4px 12px rgba(0, 0, 0, 0.1)',
      fontFamily: '"Segoe UI", Tahoma, Geneva, Verdana, sans-serif',
      display: isAdmin ? undefined : 'none', // Only admins see the generator UI
    },
    heading: {
      textAlign: 'center',
      marginBottom: '1.5rem',
      color: '#333',
    },
    label: {
      display: 'block',
      marginBottom: '0.5rem',
      fontWeight: '500',
      color: '#555',
      fontSize: '0.9rem',
    },
    input: {
      width: '100%',
      padding: '0.5rem',
      marginTop: '0.25rem',
      borderRadius: '5px',
      border: '1px solid #ccc',
    },
    select: {
      width: '100%',
      padding: '0.5rem',
      marginTop: '0.25rem',
      borderRadius: '5px',
      border: '1px solid #ccc',
    },
    row: {
      display: 'flex',
      gap: '1rem',
      marginBottom: '1rem',
    },
    fourColumnRow: {
      display: 'flex',
      gap: '1rem',
      marginBottom: '1rem',
    },
    colorInput: {
      width: '100%',
      height: '40px',
      padding: '0',
      border: '1px solid #ccc',
      borderRadius: '5px',
      cursor: 'pointer',
      appearance: 'none',
    },
    preview: {
      backgroundColor: actualBannerColor,
      color: actualBannerTextColor,
      fontSize,
      fontWeight,
      padding: '1rem',
      textAlign: 'center',
      borderRadius: '5px',
      marginBottom: '1.5rem',
    },
    code: {
      background: '#f0f0f0',
      padding: '1rem',
      borderRadius: '5px',
      overflowX: 'auto',
      fontFamily: 'monospace',
      marginBottom: '1.5rem',
    },
    button: {
      padding: '0.75rem 1.5rem',
      cursor: 'pointer',
      background: '#4CAF50',
      color: 'white',
      border: 'none',
      borderRadius: '5px',
      display: 'block',
      margin: '0 auto',
      fontSize: '1rem',
    },
    sectionHeading: {
      fontSize: '1.25rem',
      marginBottom: '1rem',
      textAlign: 'center',
      color: '#333',
    },
  };

  // Banner defaults (from your original Banner component)
  const defaultPresets = {
    critical: {
      bannerColor: "#dc2626", // red-600
      bannerTextColor: "#ffffff", // white
      fontSize: "16px",
      fontWeight: "normal",
      message: "Critical Alert: Please check immediately!"
    },
    warning: {
      bannerColor: "#facc15", // yellow-400
      bannerTextColor: "#000000", // black
      fontSize: "16px",
      fontWeight: "normal",
      message: "Warning Alert: There might be an issue."
    },
    standard: {
      bannerColor: "#2563eb", // blue-600
      bannerTextColor: "#ffffff", // white
      fontSize: "16px",
      fontWeight: "normal",
      message: "Standard Alert: Here is some information."
    },
    custom: {
      bannerColor: "#00BCFF",    
      bannerTextColor: "#ffffff",
      fontSize: "14px",
      fontWeight: "bold",
      message: "Here is my banner message - hope you enjoy!"
    }
  };

  const defaults = defaultPresets[preset];

  const bannerStyles = {
    background: bannerColor || defaults.bannerColor,
    color: bannerTextColor || defaults.bannerTextColor,
    fontSize: fontSize || defaults.fontSize,
    fontWeight: fontWeight || defaults.fontWeight,
    width: "100%",
    boxSizing: "border-box"
  };

  const containerStyles = {
    maxWidth: "var(--rm-container-width, 1200px)",
    margin: "0 auto",
    padding: "0.75rem 1rem",
    textAlign: "center",
    boxSizing: "border-box"
  };

  // Update (or create) the banner container without removing it if it exists.
  useEffect(() => {
    const moveBannerToHeader = () => {
      const headerElement = document.querySelector('header.rm-Header');
      if (!headerElement) return;
      let bannerWrapper = document.getElementById('custom-banner-container');
      const bannerContent = document.getElementById('custom-banner-content');
      if (bannerWrapper) {
        // Update the existing container's styles
        Object.assign(bannerWrapper.style, bannerStyles);
        if (bannerContent && bannerContent.parentNode !== bannerWrapper) {
          bannerWrapper.appendChild(bannerContent);
          bannerContent.style.display = '';
        }
      } else {
        bannerWrapper = document.createElement('div');
        bannerWrapper.id = 'custom-banner-container';
        Object.assign(bannerWrapper.style, bannerStyles);
        headerElement.appendChild(bannerWrapper);
        if (bannerContent) {
          bannerWrapper.appendChild(bannerContent);
          bannerContent.style.display = '';
        }
      }
    };

    moveBannerToHeader();
  }, [bannerColor, bannerTextColor, fontSize, fontWeight, message, preset]);

  return (
    <>
      {/* Generator UI (visible only to admins) */}
      <div style={styles.container}>
        <h3 style={styles.heading}>Banner Generator</h3>

        <label style={styles.label}>
          Message:
          <input
            type="text"
            value={message}
            onChange={(e) => setMessage(e.target.value)}
            style={styles.input}
          />
        </label>

        <label style={styles.label}>
          Preset:
          <select value={preset} onChange={handlePresetChange} style={styles.select}>
            <option value="custom">Custom (you choose colors)</option>
            <option value="critical">Critical (Red)</option>
            <option value="warning">Warning (Yellow)</option>
            <option value="standard">Standard (Blue)</option>
          </select>
        </label>

        {preset === 'custom' ? (
          <div style={styles.fourColumnRow}>
            <div style={{ flex: 1 }}>
              <label style={styles.label}>BG Color 🎨</label>
              <input
                type="color"
                value={bannerColor}
                onChange={(e) => setBannerColor(e.target.value)}
                style={styles.colorInput}
              />
            </div>
            <div style={{ flex: 1 }}>
              <label style={styles.label}>Text Color 🎨</label>
              <input
                type="color"
                value={bannerTextColor}
                onChange={(e) => setBannerTextColor(e.target.value)}
                style={styles.colorInput}
              />
            </div>
            <div style={{ flex: 1 }}>
              <label style={styles.label}>Font Size</label>
              <input
                type="text"
                value={fontSize}
                onChange={(e) => setFontSize(e.target.value)}
                style={styles.input}
              />
            </div>
            <div style={{ flex: 1 }}>
              <label style={styles.label}>Font Weight</label>
              <input
                type="text"
                value={fontWeight}
                onChange={(e) => setFontWeight(e.target.value)}
                style={styles.input}
              />
            </div>
          </div>
        ) : (
          <div style={styles.row}>
            <div style={{ flex: 1 }}>
              <label style={styles.label}>Font Size</label>
              <input
                type="text"
                value={fontSize}
                onChange={(e) => setFontSize(e.target.value)}
                style={styles.input}
              />
            </div>
            <div style={{ flex: 1 }}>
              <label style={styles.label}>Font Weight</label>
              <input
                type="text"
                value={fontWeight}
                onChange={(e) => setFontWeight(e.target.value)}
                style={styles.input}
              />
            </div>
          </div>
        )}

        {/* Live Preview within the generator */}
        <h4 style={styles.sectionHeading}>Banner Preview:</h4>
        <div style={styles.preview}>{message}</div>

        {/* Generated Code Output */}
        <h4 style={styles.sectionHeading}>Generated Code:</h4>
        <pre style={styles.code}>{generatedSnippet}</pre>

        <button
          onClick={() => navigator.clipboard.writeText(generatedSnippet)}
          style={styles.button}
        >
          Copy to Clipboard
        </button>
      </div>

      {/* Banner Injection Content (always used to update the header) */}
      <div id="custom-banner-content" style={{ display: 'none', width: '100%' }}>
        <div style={containerStyles}>
          {message || defaults.message}
        </div>
      </div>
    </>
  );
};

<BannerGenerator />
