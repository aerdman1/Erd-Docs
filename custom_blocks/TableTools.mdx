import React, { useEffect, useRef } from "react";

export const TableTools = ({ children }) => {
  const containerRef = useRef(null);

  // Helper to load a script
  const loadScript = (src) =>
    new Promise((resolve, reject) => {
      if (document.querySelector(`script[src="${src}"]`)) return resolve();
      const script = Object.assign(document.createElement("script"), {
        src,
        async: true,
        onload: resolve,
        onerror: () => reject(new Error(`Failed to load: ${src}`)),
      });
      document.body.appendChild(script);
    });

  // Helper to load a CSS file
  const loadCSS = (href) => {
    if (!document.querySelector(`link[href="${href}"]`)) {
      document.head.appendChild(
        Object.assign(document.createElement("link"), { rel: "stylesheet", href })
      );
    }
  };

  // Load all external dependencies
  const loadDependencies = async () => {
    await loadScript("https://code.jquery.com/jquery-3.6.0.min.js");
    await loadScript("https://cdn.datatables.net/1.13.6/js/jquery.dataTables.min.js");
    loadCSS("https://cdn.datatables.net/1.13.6/css/jquery.dataTables.min.css");
    const deps = [
      "https://cdn.datatables.net/buttons/2.4.1/js/dataTables.buttons.min.js",
      "https://cdn.datatables.net/buttons/2.4.1/js/buttons.html5.min.js",
      "https://cdn.datatables.net/buttons/2.4.1/js/buttons.print.min.js",
      "https://cdn.datatables.net/buttons/2.4.1/js/buttons.colVis.min.js",
      "https://cdnjs.cloudflare.com/ajax/libs/jszip/3.10.1/jszip.min.js",
      "https://cdnjs.cloudflare.com/ajax/libs/pdfmake/0.2.8/pdfmake.min.js",
      "https://cdnjs.cloudflare.com/ajax/libs/pdfmake/0.2.8/vfs_fonts.js",
    ];
    deps.forEach(loadScript);
    loadCSS("https://cdn.datatables.net/buttons/2.4.1/css/buttons.dataTables.min.css");
  };

  // Function to initialize a table (attach trigger button and DataTable functionality)
  const initializeTable = (table) => {
    // Avoid re-processing tables that have been processed
    if (table.getAttribute("data-dt-processed")) return;
    table.setAttribute("data-dt-processed", "true");
    table.setAttribute("data-original-html", table.innerHTML);

    // Use the closest wrapper if available, otherwise the table's parent element
    const wrapper = table.closest(".rdmd-table") || table.parentElement;
    if (!wrapper) return;
    wrapper.style.position = "relative";

    // Create the trigger button
    const trigger = document.createElement("button");
    trigger.innerHTML =
      '<i class="fa fa-ellipsis-v" aria-hidden="true" style="font-size: 12px;"></i>';
    Object.assign(trigger, {
      type: "button",
      title: "Enable Filter",
      "aria-label": "Enable filtering options",
      style:
        "position:absolute;top:0;right:0;cursor:pointer;z-index:9999;" +
        "border-radius:0 0 0 2px;color:#fff;background:linear-gradient(rgb(86,184,255),rgba(1,142,245,0.79));" +
        "filter:saturate(1.15);display:block;border:0;",
    });
    wrapper.insertBefore(trigger, wrapper.firstChild);

    // On click, initialize DataTables for this table
    trigger.onclick = () => {
      if (table.getAttribute("data-filter-enabled")) return;
      const dt = window.jQuery(table).DataTable({
        // Using Tailwind CSS classes for a flex layout
        dom:
          "<'dt-nav flex items-center justify-between'<'w-1/2'f><'w-1/2 text-right'B>>rtip",
        language: { search: "", searchPlaceholder: "Search Table:" },
        buttons: [
          {
            extend: "copy",
            text: '<i class="fa fa-copy" style="font-size: 12px;"></i>',
            className: "dt-btn",
          },
          {
            extend: "csv",
            text: '<i class="fa fa-file-csv" style="font-size: 12px;"></i>',
            className: "dt-btn",
          },
          {
            extend: "excel",
            text: '<i class="fa fa-file-excel" style="font-size: 12px;"></i>',
            className: "dt-btn",
          },
          {
            extend: "pdf",
            text: '<i class="fa fa-file-pdf" style="font-size: 12px;"></i>',
            className: "dt-btn",
          },
          {
            extend: "print",
            text: '<i class="fa fa-print" style="font-size: 12px;"></i>',
            className: "dt-btn",
            action: function () {
              setTimeout(() => {
                window.print();
              }, 100);
            },
          },
          {
            extend: "colvis",
            text: '<i class="fa fa-columns" style="font-size: 12px;"></i>',
            className: "dt-btn",
          },
          {
            text: '<i class="fa fa-ban" style="font-size: 12px;"></i>',
            className: "dt-btn",
            action: function () {
              dt.destroy();
              table.innerHTML = table.getAttribute("data-original-html");
              table.removeAttribute("data-original-html");
              table.removeAttribute("data-filter-enabled");
              trigger.style.display = "block";
              initializeTable(table);
            },
          },
        ],
        initComplete: function () {
          // Fallback inline styling for the buttons
          document.querySelectorAll(".dt-btn").forEach((btn) => {
            btn.style.border = "none";
            btn.style.padding = "4px 6px";
            btn.style.fontSize = "0.75rem";
            btn.style.background = "#f8f9fa";
            btn.style.margin = "2px";
            btn.style.borderRadius = "3px";
            btn.style.minWidth = "30px";
          });
          // Adjust the search input margin
          const searchInput = document.querySelector(".dataTables_filter input");
          if (searchInput) {
            searchInput.style.marginRight = "10px";
          }
          const nav = document.querySelector(".dt-nav");
          if (nav) {
            nav.style.marginBottom = "12px";
          }
        },
      });
      table.setAttribute("data-filter-enabled", "true");
      trigger.style.display = "none";
    };
  };

  // Inline styles using Tailwind selectors.
  // In this example, we reverse the flex order on the container that wraps the table controls.
  const inlineStyles = `
    .dt-btn {
      border: none !important;
      padding: 4px 6px !important;
      font-size: 0.75rem !important;
      background: #f8f9fa !important;
      margin: 2px !important;
      border-radius: 3px !important;
      min-width: 30px !important;
    }

  `;

  // Use an async function with a MutationObserver to process any new tables
  useEffect(() => {
    async function initTables() {
      await loadDependencies();
      const processTables = (container) => {
        const tables = container.querySelectorAll("table:not([data-dt-processed])");
        tables.forEach(initializeTable);
      };
      if (containerRef.current) {
        processTables(containerRef.current);
        const observer = new MutationObserver((mutationsList) => {
          for (const mutation of mutationsList) {
            mutation.addedNodes.forEach((node) => {
              if (node.nodeType === 1) {
                if (node.matches("table:not([data-dt-processed])")) {
                  initializeTable(node);
                }
                // Also check for any nested tables
                node.querySelectorAll &&
                  node.querySelectorAll("table:not([data-dt-processed])").forEach(initializeTable);
              }
            });
          }
        });
        observer.observe(containerRef.current, { childList: true, subtree: true });
        // Clean up observer on unmount
        return () => observer.disconnect();
      }
    }
    initTables();
  }, []);

  return (
    <div ref={containerRef}>
      <style>{inlineStyles}</style>
      {children}
    </div>
  );
};

export default TableTools;
