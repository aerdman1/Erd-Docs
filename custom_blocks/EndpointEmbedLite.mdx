import React, { useMemo, useState } from "react";

export const EndpointEmbedLite = ({
  endpoint,
  method = "GET",
  title,
  description,
  linkText,
  displayMode = "popup",      // "popup" | "inline"
  websiteBaseUrl,             // optional; auto-detects if omitted
  docSlug,                    // optional; if provided, link to /reference/{docSlug}
  apiBaseUrl,                 // optional; if provided, show "API: METHOD {apiBaseUrl}{endpoint}"
  labels = true
}) => {
  const baseUrl = useMemo(() => {
    if (websiteBaseUrl) return websiteBaseUrl.replace(/\/$/, "");
    if (typeof window !== "undefined") return `${window.location.protocol}//${window.location.host}`;
    return "";
  }, [websiteBaseUrl]);

  const derivedSlug = useMemo(() => {
    if (docSlug) return docSlug;
    const s = String(endpoint || "")
      .replace(/[{}]/g, "")
      .replace(/\//g, "-")
      .replace(/^-+|-+$/g, "");
    return s || "endpoint";
  }, [endpoint, docSlug]);

  const readmeDocUrl = baseUrl ? `${baseUrl}/reference/${derivedSlug}` : "#";
  const displayTitle = title || endpoint || "Endpoint";
  const displayLinkText = linkText || displayTitle;

  const [hover, setHover] = useState(false);

  // Always solid card styling
  const cardStyle = {
    position: "relative",
    background: "var(--color-bg-page, #fff)",
    color: "var(--color-text-default, #000)",
    padding: "14px 16px",
    borderRadius: 8,
    border: "1px solid var(--color-border-default, rgba(0,0,0,.12))",
    boxShadow: "0 6px 16px rgba(0,0,0,.06)",
    minWidth: 240,
    maxWidth: "100%"
  };

  const dividerStyle = {
    borderBottom: "1px solid var(--color-border-default, rgba(0,0,0,.08))",
    paddingBottom: 6
  };

  const mono = "ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, monospace";

  // Method pill
  const METHOD_RGB = {
    GET: [37, 99, 235],
    POST: [22, 163, 74],
    PUT: [202, 138, 4],
    PATCH: [217, 119, 6],
    HEAD: [14, 116, 144],
    OPTIONS: [107, 114, 128]
  };

  const MethodTag = ({ m }) => {
    const M = String(m).toUpperCase();
    const [r, g, b] = METHOD_RGB[M] || [99, 102, 241]; // fallback indigo
    return (
      <span
        title={M}
        style={{
          display: "inline-block",
          fontWeight: 800,
          textTransform: "uppercase",
          fontSize: ".72rem",
          lineHeight: "1.4",
          padding: "3px 8px",
          borderRadius: 999,
          color: `rgb(${r}, ${g}, ${b})`,
          background: `rgba(${r}, ${g}, ${b}, 0.12)`,
          border: `1px solid rgba(${r}, ${g}, ${b}, 0.35)`,
          letterSpacing: ".02em"
        }}
      >
        {M}
      </span>
    );
  };

  const Label = ({ children }) => (labels ? <strong>{children}: </strong> : null);

  const CardContent = (
    <a
      href={readmeDocUrl}
      style={{
        textDecoration: "none",
        color: "var(--color-text-default, #000)",
        display: "block"
      }}
    >
      <div style={cardStyle}>
        <div style={{ display: "flex", alignItems: "center", gap: 8, justifyContent: "flex-end", ...dividerStyle }}>
          <MethodTag m={String(method).toUpperCase()} />
        </div>
        <div style={{ paddingTop: 8 }}>
          <h3 style={{ margin: "6px 0 4px" }}>{displayTitle}</h3>
          {endpoint && (
            <div style={{ fontFamily: mono, fontSize: ".9rem", opacity: .8 }}>
              {endpoint}
            </div>
          )}
          {description && (
            <p style={{ margin: "8px 0 0", fontSize: ".95rem", opacity: .9 }}>{description}</p>
          )}
          {apiBaseUrl && endpoint && (
            <div style={{ marginTop: 8, fontFamily: mono, fontSize: ".9rem", opacity: .85 }}>
              <Label>API</Label>{String(method).toUpperCase()} {apiBaseUrl}{endpoint}
            </div>
          )}
        </div>
      </div>
    </a>
  );

  if (displayMode === "popup") {
    return (
      <span
        style={{ display: "inline-block", position: "relative" }}
        onMouseEnter={() => setHover(true)}
        onMouseLeave={() => setHover(false)}
      >
        <a
          href={readmeDocUrl}
          style={{
            textDecoration: "none",
            fontWeight: 700,
            borderRadius: 4,
            padding: "2px 6px",
            background: "var(--color-bg-muted, rgba(0,0,0,.06))",
            color: "var(--color-text-default, #000)"
          }}
        >
          {displayLinkText}
        </a>
        {hover && (
          <div style={{ position: "absolute", left: 0, top: "calc(100% + 6px)", zIndex: 1000 }}>
            {CardContent}
          </div>
        )}
      </span>
    );
  }

  return CardContent;
};
