import { useState } from 'react';

export default function LinkChecker() {
  const [baseUrl, setBaseUrl] = useState('https://');
  const [projects, setProjects] = useState([]);
  const [loading, setLoading] = useState(false);

  // Fetch projects using your backend endpoint
  async function fetchProjects() {
    setLoading(true);
    try {
      const response = await fetch('/api/fetch-metadata', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ baseUrl }),
      });
      const data = await response.json();
      if (data.projects) {
        setProjects(data.projects);
      } else {
        alert('No projects found.');
      }
    } catch (error) {
      console.error('Error fetching projects:', error);
      alert('Error fetching projects.');
    } finally {
      setLoading(false);
    }
  }

  // Trigger link validation for a given project
  async function validateProject(projectSlug) {
    const apiKeyInput = document.getElementById(`api-key-${projectSlug}`);
    const apiKey = apiKeyInput && apiKeyInput.value.trim();
    if (!apiKey) {
      alert('API Key is required to validate.');
      return;
    }

    setLoading(true);
    try {
      const response = await fetch('/api/validate', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ apiKey, baseUrl, projectSlug }),
      });
      if (!response.ok) throw new Error('Validation failed');
      // Create a link to download the CSV file
      const blob = await response.blob();
      const downloadUrl = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = downloadUrl;
      a.download = `${projectSlug}_broken_links.csv`;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
    } catch (error) {
      console.error('Validation error:', error);
      alert('Error during validation.');
    } finally {
      setLoading(false);
    }
  }

  return (
    <div style={{ padding: '20px' }}>
      <h1>ReadMe Page Validator</h1>
      <form onSubmit={(e) => { e.preventDefault(); fetchProjects(); }}>
        <label>
          Enter Your ReadMe Website Domain:
          <input
            type="text"
            value={baseUrl}
            onChange={(e) => setBaseUrl(e.target.value)}
            placeholder="e.g., https://docs.myCompany.com"
            required
            style={{ width: '100%', padding: '8px', marginTop: '8px' }}
          />
        </label>
        <button type="button" onClick={fetchProjects} style={{ marginTop: '10px', padding: '10px 20px' }}>
          Fetch Projects
        </button>
      </form>

      {loading && <p>Loading...</p>}

      {projects.length > 0 && (
        <div style={{ marginTop: '20px' }}>
          <h2>Projects</h2>
          {projects.map(project => (
            <div key={project.subdomain} style={{ border: '1px solid #ccc', padding: '10px', marginBottom: '10px' }}>
              <p><strong>Project:</strong> {project.name}</p>
              <label>
                API Key for {project.subdomain}:
                <input type="password" id={`api-key-${project.subdomain}`} placeholder={`API Key for ${project.subdomain}`} style={{ width: '100%', padding: '8px', marginTop: '5px' }}/>
              </label>
              <button onClick={() => validateProject(project.subdomain)} style={{ marginTop: '10px', padding: '8px 16px' }}>
                Validate
              </button>
            </div>
          ))}
        </div>
      )}
    </div>
  );
}
