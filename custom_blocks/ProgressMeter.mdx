import React, { useEffect, useState, useMemo } from "react";

export const ProgressMeter = ({
  values = null,
  websiteBaseUrl = "https://yourproject.readme.io",
  specFile,
  apiGroup,

  // users can still pass an initial key, but nothing is ever
  // read from `window` unless it exists:
  metricsApiKey: initialMetricsKey = null,

  max = 100,
  labelPosition = "below",
  showCards = false,
}) => {
  const STORAGE_KEY = "metricsApiKey";

  // 🔑 lazy‑initialize from localStorage only in the browser
  const [metricsApiKey, setMetricsApiKey] = useState(() => {
    if (typeof window === "undefined") return initialMetricsKey || "";
    return (
      initialMetricsKey ||
      window.localStorage.getItem(STORAGE_KEY) ||
      ""
    );
  });

  // persist any pasted key
  useEffect(() => {
    if (metricsApiKey && typeof window !== "undefined") {
      window.localStorage.setItem(STORAGE_KEY, metricsApiKey);
    }
  }, [metricsApiKey]);

  const [specData, setSpecData] = useState(null);
  const [metricsValues, setMetricsValues] = useState(null);

  useEffect(() => {
    if (values && values.length) return;
    if (!specFile) return;
    fetch(`${websiteBaseUrl}/openapi/${specFile}?nocache=${Date.now()}`)
      .then((res) => (res.ok ? res.json() : Promise.reject(res.status)))
      .then(setSpecData)
      .catch(console.error);
  }, [values, websiteBaseUrl, specFile]);

  useEffect(() => {
    if ((values && values.length) || specFile) return;
    if (!metricsApiKey) return;
    const url =
      "https://metrics.readme.io/v2/search/top-search-terms?rangeLength=20&resolution=day";
    fetch(url, {
      headers: { Authorization: `Basic ${btoa(metricsApiKey + ":")}` },
    })
      .then((res) => (res.ok ? res.json() : Promise.reject(res.status)))
      .then((data) => {
        const top4 = (data.search || []).slice(0, 4);
        const total = top4.reduce((sum, e) => sum + e.count, 0) || 1;
        setMetricsValues(
          top4.map((e, i) => ({
            label: e.searchTerm,
            cardLabel: `Top #${i + 1}`,
            value: Math.round((e.count / total) * 100),
            meterBackgroundColor: "#6C63FF",
            cardBackgroundColor: "#E6E6FA",
            textColor: "#fff",
            cardTextColor: "#1A1A7F",
            icon: "fas fa-search",
          }))
        );
      })
      .catch(console.error);
  }, [values, specFile, metricsApiKey]);

  const dynamicValues = useMemo(() => {
    if (values && values.length) return values;
    if (specData) {
      const counts = { get: 0, post: 0, patch: 0, delete: 0 };
      Object.values(specData.paths || {}).forEach((pi) =>
        Object.entries(pi).forEach(([m, op]) => {
          const key = m.toLowerCase();
          if (
            counts[key] !== undefined &&
            (!apiGroup || (op.tags || []).includes(apiGroup))
          ) {
            counts[key]++;
          }
        })
      );
      const total = Object.values(counts).reduce((s, v) => s + v, 0) || 1;
      const cmap = {
        get: {
          meter: "#4285F4",
          card: "#D2E3FC",
          text: "#174EA6",
          icon: "fas fa-arrow-down",
        },
        post: {
          meter: "#34A853",
          card: "#DFF6DD",
          text: "#0D652D",
          icon: "fas fa-arrow-up",
        },
        patch: {
          meter: "#FBBC05",
          card: "#FEF7E0",
          text: "#B06000",
          icon: "fas fa-edit",
        },
        delete: {
          meter: "#EA4335",
          card: "#FCE8E6",
          text: "#A50E0E",
          icon: "fas fa-trash",
        },
      };
      return Object.entries(counts)
        .filter(([, c]) => c > 0)
        .map(([m, c]) => ({
          label: m.toUpperCase(),
          cardLabel: `${m.toUpperCase()} Requests`,
          value: Math.round((c / total) * 100),
          meterBackgroundColor: cmap[m].meter,
          cardBackgroundColor: cmap[m].card,
          textColor: "#fff",
          cardTextColor: cmap[m].text,
          icon: cmap[m].icon,
        }));
    }
    if (metricsValues) return metricsValues;
    return [];
  }, [values, specData, metricsValues, apiGroup]);

  const [animatedValues, setAnimatedValues] = useState([]);
  const [showLabels, setShowLabels] = useState(false);
  useEffect(() => {
    setAnimatedValues(dynamicValues.map(() => 0));
    if (!dynamicValues.length) return;
    const iv = setInterval(() => {
      setAnimatedValues((prev) =>
        prev.map((v, i) => Math.min(v + 2, dynamicValues[i].value))
      );
    }, 50);
    const lt = setTimeout(() => setShowLabels(true), 500);
    return () => {
      clearInterval(iv);
      clearTimeout(lt);
    };
  }, [dynamicValues]);

  // If we have nothing to show *and* no key, prompt the user:
  const needKey =
    !values?.length && !specFile && dynamicValues.length === 0 && !metricsApiKey;
  if (needKey) {
    return (
      <div className="mb-4 w-full">
        <label className="block text-sm font-medium mb-1">
          Metrics API Key
        </label>
        <input
          type="password"
          className="w-full border rounded p-2"
          placeholder="Paste your key here…"
          value={metricsApiKey}
          onChange={(e) => setMetricsApiKey(e.target.value)}
        />
      </div>
    );
  }

  // …and render exactly like ProgressMeter_Simple from here on out…
  return (
    <div className="w-full flex flex-col items-start">
      {showCards && (
        <div className="flex gap-4 mb-3 w-full">
          {dynamicValues.map((item, i) => (
            <div
              key={i}
              className="shadow-md rounded-lg p-4 flex flex-col items-center w-1/4 border"
              style={{
                backgroundColor:
                  item.cardBackgroundColor || item.meterBackgroundColor,
              }}
            >
              <span
                className="text-sm"
                style={{ color: item.cardTextColor || item.textColor }}
              >
                {item.cardLabel || item.label}
              </span>
              <span
                className="text-lg font-bold"
                style={{ color: item.cardTextColor || item.textColor }}
              >
                {item.value}%
              </span>
              {item.icon && (
                <i
                  className={`${item.icon} text-2xl mt-2`}
                  style={{ color: item.cardTextColor || item.textColor }}
                />
              )}
            </div>
          ))}
        </div>
      )}
      {labelPosition === "above" && (
        <div
          className="flex justify-between w-full mb-1 opacity-0 transition-opacity duration-500"
          style={{ opacity: showLabels ? 1 : 0 }}
        >
          {dynamicValues.map((item, i) => (
            <span
              key={i}
              className="text-sm font-medium"
              style={{ color: item.textColor }}
            >
              {item.label}
            </span>
          ))}
        </div>
      )}
      <div className="relative w-full bg-gray-200 rounded-md overflow-hidden h-6 flex items-center">
        {dynamicValues.map((item, i) => (
          <div
            key={i}
            className="h-full transition-all duration-500 flex items-center justify-center relative"
            style={{
              width: `${(animatedValues[i] / max) * 100}%`,
              backgroundColor: item.meterBackgroundColor,
            }}
          >
            {labelPosition === "inside" && showLabels && (
              <span
                className="text-xs font-semibold absolute left-2"
                style={{ color: item.textColor }}
              >
                {item.label}
              </span>
            )}
            {labelPosition === "overlay" && showLabels && (
              <span
                className="text-xs font-semibold absolute inset-0 flex items-center justify-center"
                style={{ color: item.textColor }}
              >
                {item.label} ({item.value}%)
              </span>
            )}
          </div>
        ))}
      </div>
      {labelPosition === "below" && (
        <div
          className="flex justify-between w-full mt-1 opacity-0 transition-opacity duration-500"
          style={{ opacity: showLabels ? 1 : 0 }}
        >
          {dynamicValues.map((item, i) => (
            <span
              key={i}
              className="text-sm font-medium"
              style={{ color: item.textColor }}
            >
              {item.label} ({item.value}%)
            </span>
          ))}
        </div>
      )}
      {labelPosition === "side" && (
        <div
          className="flex flex-col items-start mt-2 opacity-0 transition-opacity duration-500"
          style={{ opacity: showLabels ? 1 : 0 }}
        >
          {dynamicValues.map((item, i) => (
            <span
              key={i}
              className="text-sm font-medium"
              style={{ color: item.textColor }}
            >
              {item.label}
            </span>
          ))}
        </div>
      )}
    </div>
  );
};
