import React, { useEffect, useState } from "react";

export const MeterGroup = ({ values = [], max = 100, labelPosition = "below", showCards = false }) => {
  const [animatedValues, setAnimatedValues] = useState(values.map(() => 0));
  const [showLabels, setShowLabels] = useState(false);

  useEffect(() => {
    const interval = setInterval(() => {
      setAnimatedValues((prev) => prev.map((val, i) => Math.min(val + 2, values[i].value)));
    }, 50);
    
    const labelTimeout = setTimeout(() => {
      setShowLabels(true);
    }, 500);
    
    return () => {
      clearInterval(interval);
      clearTimeout(labelTimeout);
    };
  }, [values]);

  return (
    <div className="w-full flex flex-col items-start">
      {showCards && (
        <div className="flex gap-4 mb-3 w-full">
          {values.map((item, index) => (
            <div 
              key={index} 
              className="shadow-md rounded-lg p-4 flex flex-col items-center w-1/4 border"
              style={{ backgroundColor: item.cardBackgroundColor || item.meterBackgroundColor }}
            >
              <span className="text-sm" style={{ color: item.cardTextColor || item.textColor }}>{item.cardLabel || item.label}</span>
              <span className="text-lg font-bold" style={{ color: item.cardTextColor || item.textColor }}>{item.value}%</span>
              {item.icon && <i className={`${item.icon} text-2xl mt-2`} style={{ color: item.cardTextColor || item.textColor }}></i>}
            </div>
          ))}
        </div>
      )}
      {labelPosition === "above" && (
        <div className="flex justify-between w-full mb-1 opacity-0 transition-opacity duration-500" style={{ opacity: showLabels ? 1 : 0 }}>
          {values.map((item, index) => (
            <span key={index} className="text-sm font-medium" style={{ color: item.textColor }}>{item.label}</span>
          ))}
        </div>
      )}
      <div className="relative w-full bg-gray-200 rounded-md overflow-hidden h-6 flex items-center">
        {values.map((item, index) => (
          <div 
            key={index} 
            className="h-full transition-all duration-500 flex items-center justify-center relative" 
            style={{ width: `${(animatedValues[index] / max) * 100}%`, backgroundColor: item.meterBackgroundColor }}
          >
            {labelPosition === "inside" && showLabels && (
              <span className="text-xs font-semibold absolute left-2" style={{ color: item.textColor }}>{item.label}</span>
            )}
            {labelPosition === "overlay" && showLabels && (
              <span className="text-xs font-semibold absolute inset-0 flex items-center justify-center" style={{ color: item.textColor }}>
                {item.label} ({item.value}%)
              </span>
            )}
          </div>
        ))}
      </div>
      {labelPosition === "below" && (
        <div className="flex justify-between w-full mt-1 opacity-0 transition-opacity duration-500" style={{ opacity: showLabels ? 1 : 0 }}>
          {values.map((item, index) => (
            <span key={index} className="text-sm font-medium" style={{ color: item.textColor }}>{item.label} ({item.value}%)</span>
          ))}
        </div>
      )}
      {labelPosition === "side" && (
        <div className="flex flex-col items-start mt-2 opacity-0 transition-opacity duration-500" style={{ opacity: showLabels ? 1 : 0 }}>
          {values.map((item, index) => (
            <span key={index} className="text-sm font-medium" style={{ color: item.textColor }}>{item.label}</span>
          ))}
        </div>
      )}
    </div>
  );
};

<MeterGroup  
  values={[
    { label: "GET", cardLabel: "GET Requests", value: 40, meterBackgroundColor: "#3B82F6", cardBackgroundColor: "#E0F2FE", textColor: "#fff", cardTextColor: "#1E40AF", icon: "fas fa-arrow-down" },
    { label: "POST", cardLabel: "POST Requests", value: 27, meterBackgroundColor: "#10B981", cardBackgroundColor: "#D1FAE5", textColor: "#fff", cardTextColor: "#047857", icon: "fas fa-arrow-up" },
    { label: "PATCH", cardLabel: "PATCH Updates", value: 21, meterBackgroundColor: "#ff0087", cardBackgroundColor: "#ffbfe1", textColor: "#fff", cardTextColor: "#92400E", icon: "fas fa-edit" },
    { label: "DELETE", cardLabel: "DELETE Actions", value: 13, meterBackgroundColor: "#EF4444", cardBackgroundColor: "#FEE2E2", textColor: "#fff", cardTextColor: "#991B1B", icon: "fas fa-trash" }
  ]} 
  max={100} 
  labelPosition="overlay"
  showCards={true}
/>
