import React, { useState, useEffect, useMemo } from "react";
//One Theme, Horizontal Only
export function StepperOneTheme({ steps }) {
  // PURPLE theme only
  const currentTheme = {
    background: "radial-gradient(circle at top, #835db9 20%, #4a2a8f 80%)",
    circle: "linear-gradient(to top, #b78cff, #8c5eff)",
    circleActive: "linear-gradient(to top, rgb(193, 131, 246), rgb(183, 140, 255))",
    textColor: "white",
    line: "rgba(255,255,255,0.2)",
    lineActive: "rgba(255,255,255,0.3)",
  };

  const textColor = currentTheme.textColor;
  const inactiveTextColor = textColor === "black" ? "rgba(0,0,0,0.6)" : "rgba(255,255,255,0.6)";
  const activeTextColor = textColor === "black" ? "rgba(0,0,0,1)" : "rgba(255,255,255,1)";
  const activeBoxShadow =
    textColor === "black"
      ? "0 0 8px rgba(0,0,0,0.8)"
      : "0 0 8px rgba(255,255,255,0.8)";

  const [currentStep, setCurrentStep] = useState(0);
  const [isVisible, setIsVisible] = useState(false);
  const uniqueId = useMemo(() => "stepper-" + Math.random().toString(36).substr(2, 9), []);

  useEffect(() => {
    const timeout = setTimeout(() => {
      setIsVisible(true);
    }, 300);
    return () => clearTimeout(timeout);
  }, []);

  const styleContent = `
    /* Container styling scoped to this instance */
    #${uniqueId} {
      position: relative;
      text-align: center;
      width: 100%;
      max-width: 90%;
      margin: 20px auto;
      padding: 24px 36px;
      border-radius: 12px;
      border: 1px solid rgba(255, 255, 255, 0.2);
      background: ${currentTheme.background};
      box-shadow: 0 4px 10px rgba(0, 0, 0, 0.2);
    }
    /* HORIZONTAL LAYOUT */
    #${uniqueId} .progress-container {
      display: flex;
      justify-content: space-between;
      align-items: center;
      position: relative;
      width: 100%;
      max-width: 100%;
      margin: 20px auto;
    }
    #${uniqueId} .progress-container::before {
      background-color: ${currentTheme.line};
      content: "";
      height: 3px;
      position: absolute;
      top: 17px; /* Align with center of 34px circles */
      left: 10%;
      right: 10%;
      z-index: 2;
      border-radius: 2px;
    }
    #${uniqueId} .progress {
      background-color: ${currentTheme.line};
      height: 3px;
      position: absolute;
      top: 17px; /* Align with center of circles */
      left: 10%;
      transition: width 0.3s ease-in-out, background-color 0.3s ease-in-out;
      width: 0%;
      z-index: 1;
      border-radius: 2px;
    }
    #${uniqueId} .progress.active {
      background-color: ${currentTheme.lineActive};
    }
    #${uniqueId} .circle-container {
      position: relative;
      display: flex;
      flex-direction: column;
      align-items: center;
      flex: 1;
      text-align: center;
    }
    /* Circle Step Style */
    #${uniqueId} .circle {
      align-items: center;
      background: ${currentTheme.circle};
      border: 2px solid rgba(236, 233, 254, 0.6);
      border-radius: 50%;
      color: ${textColor};
      display: flex;
      font-size: 14px;
      font-weight: bold;
      height: 34px;
      justify-content: center;
      transition: 0.3s ease;
      width: 34px;
      z-index: 99999;
      position: relative;
    }
    #${uniqueId} .circle.active {
      background: ${currentTheme.circleActive};
      color: ${textColor};
      border: 2px solid ${textColor};
      z-index: 99999;
      transition: 0.3s ease;
      box-shadow: ${activeBoxShadow};
    }
    #${uniqueId} .step-title {
      margin-top: 8px;
      font-size: 14px;
      font-weight: bold;
      color: ${inactiveTextColor};
      transition: color 0.3s ease;
      width: auto;
      text-align: center;
    }
    #${uniqueId} .step-title.active {
      color: ${activeTextColor};
    }
    /* Horizontal Layout Content */
    #${uniqueId} .step-content {
      margin: 10px auto 0 auto;
      max-width: 90%;
      font-size: 16px;
      padding: 10px;
      color: ${textColor};
      font-weight: 400;
    }
    #${uniqueId} .step-header {
      text-align: center;
      font-size: 24px;
      font-weight: bold;
      margin-bottom: 20px;
      color: ${textColor};
    }
    #${uniqueId} .stepper-buttons {
      margin-top: 20px;
      display: flex;
      justify-content: space-between;
    }
    #${uniqueId} .btn {
      background: transparent;
      border: none;
      color: ${inactiveTextColor};
      font-size: 14px;
      font-weight: bold;
      cursor: pointer;
      padding: 6px 12px;
      transition: color 0.2s ease;
    }
    #${uniqueId} .btn:hover {
      color: ${activeTextColor};
    }
    #${uniqueId} .btn:disabled {
      color: ${inactiveTextColor};
      cursor: not-allowed;
    }
    .content-body .rm-Markdown {
      opacity: 1 !important;
    }
  `;

  return (
    <>
      <style>{styleContent}</style>
      <div id={uniqueId} className={`stepper-container ${isVisible ? "opacity-100 visible" : "opacity-0 invisible"}`}>
        <div className="progress-container">
          <div
            className={`progress ${currentStep > 0 ? "active" : ""}`}
            style={{ width: `${(currentStep / (steps.length - 1)) * 80}%` }}
          ></div>
          {steps.map((step, index) => (
            <div key={index} className="circle-container">
              <div className={`circle ${index <= currentStep ? "active" : ""}`}>
                {index + 1}
              </div>
              <div className={`step-title ${index === currentStep ? "active" : ""}`}>
                {step.header}
              </div>
            </div>
          ))}
        </div>
        <div className="step-content">
          <h2 className="step-header">{steps[currentStep].header}</h2>
          {steps[currentStep].content}
        </div>
        <div className="stepper-buttons">
          <button type="button" className="btn" onClick={() => setCurrentStep(currentStep - 1)} disabled={currentStep === 0}>
            <i className="fa fa-arrow-left"></i> BACK
          </button>
          <button type="button" className="btn" onClick={() => setCurrentStep(currentStep + 1)} disabled={currentStep === steps.length - 1}>
            NEXT <i className="fa fa-arrow-right"></i>
          </button>
        </div>
      </div>
    </>
  );
}

  steps={[
    { header: "Plan Your Documentation", content: <p>Define the purpose of your documentation and identify your target audience. Outline key topics and gather relevant resources before writing.</p>, stepType: "circle" },

    { header: "Format Content Effectively", content: <p>Utilize headings, bullet points, and tables to improve readability. Add images, code snippets, and examples to enhance clarity.</p>, stepType: "circle" },
    { header: "Use Interactive Elements", content: <p>Incorporate API explorers, live code examples, and try-it-yourself features for better engagement and hands-on learning.</p>, stepType: "circle" },
    { header: "Publish and Maintain Content", content: <p>Test documentation for accuracy and clarity before publishing. Keep it updated to reflect new features, changes, or improvements.</p>, stepType: "circle" },
  ]}
/>
