import React, { useEffect, useState } from 'react';

const EndpointExampleResponseIframe = () => {
  const [codeText, setCodeText] = useState("");
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // Create an off-screen iframe (positioned off-screen so it's not visible)
    const iframe = document.createElement('iframe');
    iframe.style.position = 'absolute';
    iframe.style.top = '-9999px';
    iframe.src = 'https://erdtest123.readme.io/reference/updatepet';
    document.body.appendChild(iframe);

    iframe.onload = function() {
      try {
        const iDoc = iframe.contentDocument || iframe.contentWindow.document;
        console.log('Iframe loaded.');

        // Wait for the page's JavaScript to fully initialize.
        setTimeout(() => {
          // 1. Find and click the "EXAMPLE" button.
          const exampleButton = Array.from(iDoc.querySelectorAll('button')).find(btn =>
            btn.innerText.trim().toUpperCase().includes("EXAMPLE")
          );
          if (!exampleButton) {
            setError('EXAMPLE button not found.');
            setLoading(false);
            document.body.removeChild(iframe);
            return;
          }
          console.log('Clicking the "EXAMPLE" button:', exampleButton);
          exampleButton.click();

          // 2. Wait for the Tippy popup to appear.
          setTimeout(() => {
            const popup = iDoc.querySelector('[data-tippy-root]');
            if (!popup) {
              setError('Popup (data-tippy-root) not found.');
              setLoading(false);
              document.body.removeChild(iframe);
              return;
            }
            console.log('Popup found:', popup);

            // 3. In the popup, find and click the "Request Example" option.
            const reqExampleOption = Array.from(popup.querySelectorAll('.Menu-Item')).find(el =>
              el.innerText.trim().toUpperCase().includes("REQUEST EXAMPLE")
            );
            if (!reqExampleOption) {
              setError('"Request Example" option not found in popup.');
              setLoading(false);
              document.body.removeChild(iframe);
              return;
            }
            console.log('Clicking the "Request Example" option:', reqExampleOption);
            reqExampleOption.click();

            // 4. Wait for CodeMirror to update with the example code.
            setTimeout(() => {
              const codeContainer = iDoc.querySelector('.react-codemirror2.CodeEditor');
              if (!codeContainer) {
                setError('CodeMirror container not found.');
                setLoading(false);
                document.body.removeChild(iframe);
                return;
              }
              const codeLines = codeContainer.querySelectorAll('.CodeMirror-lines pre.CodeMirror-line');
              if (!codeLines || codeLines.length === 0) {
                setError('No CodeMirror lines found.');
                setLoading(false);
                document.body.removeChild(iframe);
                return;
              }
              let extractedCode = "";
              codeLines.forEach(line => {
                extractedCode += line.textContent + "\n";
              });
              console.log("Extracted CodeMirror text:");
              console.log(extractedCode);
              setCodeText(extractedCode);
              setLoading(false);
              document.body.removeChild(iframe);
            }, 3000); // Adjust delay as needed for CodeMirror content to load.
          }, 2000); // Delay for popup to appear.
        }, 2000); // Initial delay for page JS to run.
      } catch (err) {
        console.error("Error accessing iframe content:", err);
        setError(err.message);
        setLoading(false);
        document.body.removeChild(iframe);
      }
    };

    // Cleanup: Remove iframe if component unmounts.
    return () => {
      if (iframe && iframe.parentNode) {
        iframe.parentNode.removeChild(iframe);
      }
    };
  }, []);

  if (loading) return <div>Loading endpoint example (iframe mode)...</div>;
  if (error) return <div style={{ color: 'red' }}>Error: {error}</div>;
  return (
    <div style={{ border: '1px solid #ddd', padding: '16px', borderRadius: '8px', marginTop: '16px' }}>
      <h3>Response from Iframe Mode</h3>
      <pre style={{
        whiteSpace: 'pre-wrap',
        background: '#f5f5f5',
        padding: '8px',
        borderRadius: '4px',
        marginTop: '8px'
      }}>
        {codeText}
      </pre>
    </div>
  );
};

export default EndpointExampleResponseIframe;
