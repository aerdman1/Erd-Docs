import React, { useEffect } from "react";

export const DownloadableTables = () => {
  useEffect(() => {
    // Load a script dynamically.
    const loadScript = (src) => {
      return new Promise((resolve, reject) => {
        const script = document.createElement("script");
        script.src = src;
        script.async = true;
        script.onload = resolve;
        script.onerror = () => reject(new Error(`Failed to load script: ${src}`));
        document.body.appendChild(script);
      });
    };

    // Load a CSS file dynamically.
    const loadCSS = (href) => {
      const link = document.createElement("link");
      link.rel = "stylesheet";
      link.href = href;
      document.head.appendChild(link);
    };

    const loadDependencies = async () => {
      if (!window.jQuery) {
        await loadScript("https://code.jquery.com/jquery-3.6.0.min.js");
      }
      if (!window.jQuery.fn.DataTable) {
        await loadScript("https://cdn.datatables.net/1.10.24/js/jquery.dataTables.min.js");
        loadCSS("https://cdn.datatables.net/1.10.24/css/jquery.dataTables.min.css");
      }
      await loadScript("https://cdn.datatables.net/buttons/2.3.6/js/dataTables.buttons.min.js");
      loadCSS("https://cdn.datatables.net/buttons/2.3.6/css/buttons.dataTables.min.css");
      await loadScript("https://cdn.datatables.net/buttons/2.3.6/js/buttons.html5.min.js");
      await loadScript("https://cdn.datatables.net/buttons/2.3.6/js/buttons.print.min.js");
      await loadScript("https://cdn.datatables.net/buttons/2.3.6/js/buttons.colVis.min.js");
      await loadScript("https://cdnjs.cloudflare.com/ajax/libs/jszip/3.10.1/jszip.min.js");
      await loadScript("https://cdnjs.cloudflare.com/ajax/libs/pdfmake/0.2.7/pdfmake.min.js");
      await loadScript("https://cdnjs.cloudflare.com/ajax/libs/pdfmake/0.2.7/vfs_fonts.js");
    };

    loadDependencies()
      .then(() => {
        // Optional style override for the nav row
        const styleOverride = document.createElement("style");
        styleOverride.textContent = `
          table.dataTable, table.dataTable th, table.dataTable td {
            box-sizing: inherit;
          }
          .dt-nav {
            margin-bottom: 10px;
          }
          .dt-nav .dt-button {
            margin-right: 5px;
          }
        `;
        document.head.appendChild(styleOverride);

        const tables = document.querySelectorAll("table");
        tables.forEach((table) => {
          if (table.getAttribute("data-dt-processed")) return;
          table.setAttribute("data-dt-processed", "true");

          // Store the original table HTML to allow reverting later.
          table.setAttribute("data-original-html", table.innerHTML);

          // Find the table wrapper (.rdmd-table) and ensure it’s positioned relatively.
          const tableWrapper = table.closest(".rdmd-table");
          if (!tableWrapper) return;
          tableWrapper.style.position = "relative";

          // Create the initial trigger button (3-dot icon).
          const trigger = document.createElement("div");
          trigger.innerHTML = '<i class="fa fa-ellipsis-v" aria-hidden="true"></i>';
          trigger.title = "Enable Filter";
          trigger.style.position = "absolute";
          trigger.style.top = "0px";
          trigger.style.right = "0px";
          trigger.style.padding = "4px 6px";
          trigger.style.cursor = "pointer";
          trigger.style.fontSize = "0.75rem";
          trigger.style.zIndex = "10";
          trigger.style.borderRadius = "0px 0px 0px 3px";
          trigger.style.color = "rgb(255, 255, 255)";
          trigger.style.background = "linear-gradient(rgb(86, 184, 255), rgba(1, 142, 245, 0.79))";
          trigger.style.filter = "saturate(1.15)";
          trigger.setAttribute("data-dt-trigger", "true");
          tableWrapper.insertBefore(trigger, tableWrapper.firstChild);

          // On clicking the trigger, initialize DataTables with a custom nav bar.
          trigger.onclick = () => {
            const $table = window.jQuery(table);
            if (table.getAttribute("data-filter-enabled") === "true") return;
            const dt = $table.DataTable({
              // Custom DOM: A single nav row containing the buttons (B) and filter input (f)
              dom: "<'dt-nav row'<'col-sm-12 d-flex align-items-center justify-content-between'Bf>>rtip",
              buttons: [
                {
                  extend: "copy",
                  text: '<i class="fa fa-copy"></i> Copy'
                },
                {
                  extend: "csv",
                  text: '<i class="fa fa-file-csv"></i> CSV'
                },
                {
                  extend: "excel",
                  text: '<i class="fa fa-file-excel"></i> Excel'
                },
                {
                  extend: "pdf",
                  text: '<i class="fa fa-file-pdf"></i> PDF'
                },
                {
                  extend: "print",
                  text: '<i class="fa fa-print"></i> Print'
                },
                {
                  extend: "colvis",
                  text: '<i class="fa fa-columns"></i> Column Visibility'
                },
                {
                  text: '<i class="fa fa-arrows-alt"></i> Popup',
                  action: function (e, dt, node, config) {
                    // Popup modal code:
                    let wrapper = table.closest(".rdmd-table") || table.closest(".rdmd-table-inner");
                    let clone;
                    if (wrapper) {
                      clone = wrapper.cloneNode(true);
                      const injected = clone.querySelectorAll('[data-dt-trigger], [data-dt-buttons]');
                      injected.forEach((el) => el.remove());
                      clone.style.margin = "0";
                    } else {
                      clone = table.cloneNode(true);
                    }
                    const markdownWrapper = document.createElement("div");
                    markdownWrapper.className = "markdown-body";
                    markdownWrapper.appendChild(clone);
                    const modalOverlay = document.createElement("div");
                    modalOverlay.style.position = "fixed";
                    modalOverlay.style.top = "0";
                    modalOverlay.style.left = "0";
                    modalOverlay.style.width = "100%";
                    modalOverlay.style.height = "100%";
                    modalOverlay.style.backgroundColor = "rgba(0, 0, 0, 0.5)";
                    modalOverlay.style.display = "flex";
                    modalOverlay.style.justifyContent = "center";
                    modalOverlay.style.alignItems = "center";
                    modalOverlay.style.zIndex = "9999";
                    modalOverlay.style.overflow = "auto";
                    const modalContent = document.createElement("div");
                    modalContent.style.backgroundColor = "#fff";
                    modalContent.style.padding = "20px";
                    modalContent.style.paddingTop = "40px";
                    modalContent.style.borderRadius = "4px";
                    modalContent.style.maxHeight = "90%";
                    modalContent.style.overflow = "auto";
                    modalContent.style.position = "relative";
                    modalContent.style.minWidth = "800px";
                    modalContent.style.margin = "20px";
                    modalContent.appendChild(markdownWrapper);
                    modalOverlay.appendChild(modalContent);
                    document.body.appendChild(modalOverlay);
                    const closeButton = document.createElement("button");
                    closeButton.innerHTML = '<i class="fa fa-times" aria-hidden="true"></i>';
                    closeButton.title = "Close";
                    closeButton.style.position = "absolute";
                    closeButton.style.top = "10px";
                    closeButton.style.right = "10px";
                    closeButton.style.background = "transparent";
                    closeButton.style.border = "none";
                    closeButton.style.fontSize = "1.5rem";
                    closeButton.style.cursor = "pointer";
                    closeButton.onclick = () => {
                      document.body.removeChild(modalOverlay);
                    };
                    modalContent.appendChild(closeButton);
                    modalOverlay.onclick = (e) => {
                      if (e.target === modalOverlay) {
                        document.body.removeChild(modalOverlay);
                      }
                    };
                  }
                },
                {
                  text: '<i class="fa fa-ban"></i> Close Filter',
                  action: function (e, dt, node, config) {
                    dt.destroy();
                    const original = table.getAttribute("data-original-html");
                    if (original) {
                      table.innerHTML = original;
                      table.removeAttribute("data-original-html");
                    }
                    table.setAttribute("data-filter-enabled", "false");
                    // Show trigger button again.
                    trigger.style.display = "block";
                  }
                }
              ]
            });
            table.setAttribute("data-filter-enabled", "true");
            // Hide the trigger after enabling the filter.
            trigger.style.display = "none";
          };
        });
      })
      .catch((err) => {
        console.error("Error loading dependencies:", err);
      });
  }, []);

  return null;
};

<DownloadableTables />
