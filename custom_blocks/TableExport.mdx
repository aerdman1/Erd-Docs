import React, { useEffect } from "react";

export const DataTableEnhancer = () => {
  useEffect(() => {
    // Function to load a script by URL
    const loadScript = (src) => {
      return new Promise((resolve, reject) => {
        const script = document.createElement("script");
        script.src = src;
        script.onload = resolve;
        script.onerror = reject;
        document.head.appendChild(script);
      });
    };

    // Function to load a stylesheet by URL
    const loadStyle = (href) => {
      return new Promise((resolve, reject) => {
        const link = document.createElement("link");
        link.href = href;
        link.rel = "stylesheet";
        link.onload = resolve;
        link.onerror = reject;
        document.head.appendChild(link);
      });
    };

    // Load jQuery if not already loaded.
    const ensurejQuery = () => {
      return new Promise((resolve) => {
        if (window.jQuery) {
          resolve();
        } else {
          loadScript("https://code.jquery.com/jquery-3.6.0.min.js").then(resolve);
        }
      });
    };

    // Load DataTables CSS and JS.
    Promise.all([
      ensurejQuery(),
      loadScript("https://cdn.datatables.net/1.13.1/js/jquery.dataTables.min.js"),
      loadStyle("https://cdn.datatables.net/1.13.1/css/jquery.dataTables.min.css")
    ])
      .then(() => {
        // Initialize DataTables on all tables.
        window.jQuery(document).ready(function ($) {
          // You might want to target only specific tables.
          $("table").DataTable();
        });
      })
      .catch((err) => {
        console.error("Failed to load DataTables:", err);
      });
  }, []);

  return null;
};

< DataTableEnhancer />
