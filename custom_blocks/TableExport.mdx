import React, { useEffect } from "react";

export const DownloadableTables = () => {
  useEffect(() => {
    const tables = document.querySelectorAll("table");

    tables.forEach((table, index) => {
      // Skip if already processed
      if (table.getAttribute("data-csv-processed")) return;
      table.setAttribute("data-csv-processed", "true");

      // Create a container for the buttons (inserted above the table)
      const container = document.createElement("div");
      container.style.marginBottom = "8px";
      container.style.display = "flex";
      container.style.gap = "8px";
      container.style.justifyContent = "flex-end";
      container.style.width = "100%";

      // Helper: Build CSV string from table rows
      const buildCSV = () => {
        const rows = Array.from(table.querySelectorAll("tr"));
        let csvData = "";
        rows.forEach((row) => {
          const cells = Array.from(row.querySelectorAll("th, td"));
          let rowData = [];
          cells.forEach((cell) => {
            let text = cell.innerText.trim();
            text = text.split('"').join('""');
            if (text.indexOf(",") !== -1 || text.indexOf('"') !== -1) {
              text = '"' + text + '"';
            }
            rowData.push(text);
          });
          csvData += rowData.join(",") + "\n";
        });
        return csvData;
      };

      // Button: Download (icon only)
      const csvButton = document.createElement("button");
      csvButton.innerHTML = '<i class="fa fa-download" aria-hidden="true"></i>';
      csvButton.title = "Download as CSV";
      csvButton.style.backgroundColor = "#3490dc";
      csvButton.style.border = "none";
      csvButton.style.color = "#fff";
      csvButton.style.fontWeight = "bold";
      csvButton.style.padding = "6px 12px";
      csvButton.style.borderRadius = "4px";
      csvButton.style.cursor = "pointer";
      csvButton.style.fontSize = "0.625rem";
      csvButton.style.display = "inline-flex";
      csvButton.style.alignItems = "center";
      csvButton.onclick = () => {
        const csvData = buildCSV();
        const blob = new Blob([csvData], { type: "text/csv;charset=utf-8;" });
        const url = URL.createObjectURL(blob);
        const link = document.createElement("a");
        link.href = url;
        link.download = "table-" + (index + 1) + ".csv";
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
      };

      // Button: Print (icon only)
      const printButton = document.createElement("button");
      printButton.innerHTML = '<i class="fa fa-print" aria-hidden="true"></i>';
      printButton.title = "Print Table";
      printButton.style.backgroundColor = "#38c172";
      printButton.style.border = "none";
      printButton.style.color = "#fff";
      printButton.style.fontWeight = "bold";
      printButton.style.padding = "6px 12px";
      printButton.style.borderRadius = "4px";
      printButton.style.cursor = "pointer";
      printButton.style.fontSize = "0.625rem";
      printButton.style.display = "inline-flex";
      printButton.style.alignItems = "center";
      printButton.onclick = () => {
        const printWindow = window.open("", "", "width=800,height=600");
        if (printWindow) {
          printWindow.document.write("<html><head><title>Print Table</title>");
          printWindow.document.write(
            '<link rel="stylesheet" href="https://cdnjs.cloudflare.com/ajax/libs/font-awesome/4.7.0/css/font-awesome.min.css">'
          );
          printWindow.document.write("</head><body>");
          // Use the table's outerHTML (or its wrapper if available)
          printWindow.document.write(table.outerHTML);
          printWindow.document.write("</body></html>");
          printWindow.document.close();
          printWindow.focus();
          setTimeout(() => {
            printWindow.print();
            printWindow.close();
          }, 500);
        }
      };

      // Button: Copy (icon only)
      const copyButton = document.createElement("button");
      copyButton.innerHTML = '<i class="fa fa-clipboard" aria-hidden="true"></i>';
      copyButton.title = "Copy Table Data";
      copyButton.style.backgroundColor = "#9561e2";
      copyButton.style.border = "none";
      copyButton.style.color = "#fff";
      copyButton.style.fontWeight = "bold";
      copyButton.style.padding = "6px 12px";
      copyButton.style.borderRadius = "4px";
      copyButton.style.cursor = "pointer";
      copyButton.style.fontSize = "0.625rem";
      copyButton.style.display = "inline-flex";
      copyButton.style.alignItems = "center";
      copyButton.onclick = async () => {
        const csvData = buildCSV();
        try {
          await navigator.clipboard.writeText(csvData);
          alert("Table data copied!");
        } catch (err) {
          alert("Copy failed: " + err);
        }
      };

      // Button: Popup (icon only) – Opens a modal displaying the full table with all classes intact
      const popupButton = document.createElement("button");
      popupButton.innerHTML = '<i class="fa fa-arrows-alt" aria-hidden="true"></i>';
      popupButton.title = "View Table in Popup";
      popupButton.style.backgroundColor = "#f6993f";
      popupButton.style.border = "none";
      popupButton.style.color = "#fff";
      popupButton.style.fontWeight = "bold";
      popupButton.style.padding = "6px 12px";
      popupButton.style.borderRadius = "4px";
      popupButton.style.cursor = "pointer";
      popupButton.style.fontSize = "0.625rem";
      popupButton.style.display = "inline-flex";
      popupButton.style.alignItems = "center";
      popupButton.onclick = () => {
        // Check if the table is wrapped in an element with class "rdmd-table-inner"
        const wrapper = table.closest(".rdmd-table-inner");
        let clone;
        if (wrapper) {
          // Clone the entire wrapper so we keep all classes and styles
          clone = wrapper.cloneNode(true);
          // Remove the first child if it is our injected button container (assumes it's a div)
          if (
            clone.firstElementChild &&
            clone.firstElementChild.tagName.toLowerCase() === "div"
          ) {
            clone.removeChild(clone.firstElementChild);
          }
        } else {
          clone = table.cloneNode(true);
        }

        // Create modal overlay
        const modalOverlay = document.createElement("div");
        modalOverlay.style.position = "fixed";
        modalOverlay.style.top = "0";
        modalOverlay.style.left = "0";
        modalOverlay.style.width = "100%";
        modalOverlay.style.height = "100%";
        modalOverlay.style.backgroundColor = "rgba(0, 0, 0, 0.5)";
        modalOverlay.style.display = "flex";
        modalOverlay.style.justifyContent = "center";
        modalOverlay.style.alignItems = "center";
        modalOverlay.style.zIndex = "9999";

        // Create modal content container with extra padding so the close button doesn't overlap the table
        const modalContent = document.createElement("div");
        modalContent.style.backgroundColor = "#fff";
        modalContent.style.padding = "20px";
        modalContent.style.paddingTop = "40px"; // extra top padding for close button
        modalContent.style.borderRadius = "4px";
        modalContent.style.maxHeight = "90%";
        modalContent.style.overflow = "auto";
        modalContent.style.position = "relative";

        // Append the cloned table (or wrapper) to modal content
        modalContent.appendChild(clone);

        // Create close button (no icons inside the modal)
        const closeButton = document.createElement("button");
        closeButton.textContent = "X";
        closeButton.style.position = "absolute";
        closeButton.style.top = "10px";
        closeButton.style.right = "10px";
        closeButton.style.background = "transparent";
        closeButton.style.border = "none";
        closeButton.style.fontSize = "1.5rem";
        closeButton.style.cursor = "pointer";
        closeButton.onclick = () => {
          document.body.removeChild(modalOverlay);
        };

        modalContent.appendChild(closeButton);
        modalOverlay.appendChild(modalContent);
        document.body.appendChild(modalOverlay);

        // Close modal when clicking outside the modal content
        modalOverlay.onclick = (e) => {
          if (e.target === modalOverlay) {
            document.body.removeChild(modalOverlay);
          }
        };
      };

      // Append all buttons to the container
      container.appendChild(csvButton);
      container.appendChild(printButton);
      container.appendChild(copyButton);
      container.appendChild(popupButton);

      // Insert the container above the table
      table.parentNode.insertBefore(container, table);
    });
  }, []);

  return null;
}; 

<DownloadableTables />
