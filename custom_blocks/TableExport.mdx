import React, { useEffect } from "react";

export const DownloadableTables = () => {
  useEffect(() => {
    // Helper to load a script dynamically.
    const loadScript = (src) => {
      return new Promise((resolve, reject) => {
        const script = document.createElement("script");
        script.src = src;
        script.async = true;
        script.onload = () => resolve();
        script.onerror = () =>
          reject(new Error(`Failed to load script: ${src}`));
        document.body.appendChild(script);
      });
    };

    // Helper to load a CSS stylesheet dynamically.
    const loadCSS = (href) => {
      const link = document.createElement("link");
      link.rel = "stylesheet";
      link.href = href;
      document.head.appendChild(link);
    };

    // Load required dependencies using newer CDN links.
    const loadDependencies = async () => {
      if (!window.jQuery) {
        await loadScript("https://code.jquery.com/jquery-3.6.0.min.js");
      }
      if (!window.jQuery.fn.DataTable) {
        await loadScript("https://cdn.datatables.net/1.13.6/js/jquery.dataTables.min.js");
        loadCSS("https://cdn.datatables.net/1.13.6/css/jquery.dataTables.min.css");
      }
      await loadScript("https://cdn.datatables.net/buttons/2.4.1/js/dataTables.buttons.min.js");
      loadCSS("https://cdn.datatables.net/buttons/2.4.1/css/buttons.dataTables.min.css");
      await loadScript("https://cdn.datatables.net/buttons/2.4.1/js/buttons.html5.min.js");
      await loadScript("https://cdn.datatables.net/buttons/2.4.1/js/buttons.print.min.js");
      await loadScript("https://cdn.datatables.net/buttons/2.4.1/js/buttons.colVis.min.js");
      await loadScript("https://cdnjs.cloudflare.com/ajax/libs/jszip/3.10.1/jszip.min.js");
      await loadScript("https://cdnjs.cloudflare.com/ajax/libs/pdfmake/0.2.8/pdfmake.min.js");
      await loadScript("https://cdnjs.cloudflare.com/ajax/libs/pdfmake/0.2.8/vfs_fonts.js");
    };

    loadDependencies()
      .then(() => {
        // Append style overrides (preserving your original styles)
        const styleOverride = document.createElement("style");
        styleOverride.textContent = `
table.dataTable, table.dataTable th, table.dataTable td {
  box-sizing: inherit;
}
.dataTables_wrapper {
  font-family: "Helvetica Neue", Arial, sans-serif;
  font-size: 0.875rem;
  color: #333;
  background: #fff;
  margin: 0 auto;
  border: none;
}
.dataTables_wrapper .dataTables_filter {
  margin-bottom: 0.5rem;
  font-size: 12px;
  order: 0 !important;
}
.dataTables_wrapper .dataTables_filter input {
  border: 1px solid #ccc;
  margin-left: 25px;
  border-radius: 6px;
  padding: 4px 10px !important;
  outline: none;
  transition: border-color 0.2s ease, box-shadow 0.2s ease;
}
.dataTables_wrapper .dataTables_filter input:focus {
  border-color: #007bff;
  box-shadow: 0 0 4px rgba(0, 123, 255, 0.4);
}
.dataTables_wrapper .dataTables_info {
  font-size: 12px;
  color: #666;
}
.dataTables_wrapper .dataTables_paginate .paginate_button {
  color: #fff !important;
  border: none !important;
  border-radius: 6px;
  margin: 2px;
  padding: 6px 12px;
  cursor: pointer;
  font-size: 12px !important;
  font-weight: 500;
  line-height: 1;
  height: auto;
  box-sizing: border-box;
  box-shadow: none;
  transform: none !important;
  outline: none !important;
}
.dataTables_wrapper .dataTables_paginate .paginate_button:hover {
  background: #0056b3;
  color: #fff !important;
  border: none !important;
  opacity: 0.9;
}
.dataTables_wrapper .dataTables_paginate .paginate_button.current {
  background: #003f7f !important;
  color: #fff !important;
  font-weight: bold;
  cursor: default;
  box-shadow: none;
  transform: none !important;
  outline: none !important;
  border: none;
}
.dataTables_wrapper .dataTables_paginate .paginate_button.current:hover {
  background: #003f7f !important;
  color: #fff !important;
  border: none !important;
  opacity: 1;
}
.dataTables_wrapper .dataTables_paginate .paginate_button.disabled {
  background: transparent !important;
  cursor: not-allowed;
}
.dt-button {
  background-color: #007bff !important;
  background: linear-gradient(rgb(86, 184, 255), rgba(1, 142, 245, 0.79)) !important;
  color: #fff !important;
  border: none !important;
  border-radius: 20px !important;
  padding: 1px 14px !important;
  font-size: 0.65rem !important;
  font-weight: 500 !important;
  cursor: pointer !important;
  box-shadow: none !important;
  margin: 2px;
  height: auto;
  line-height: 1.4;
  box-sizing: border-box;
  transform: none !important;
  outline: none !important;
}
.dt-button:hover {
  background: #0056b3 !important;
  color: #fff !important;
}
.dataTables_wrapper .dataTables_paginate .paginate_button,
.dt-button {
  transform: none !important;
  background-color: #007bff !important;
  background: linear-gradient(rgb(86, 184, 255), rgba(1, 142, 245, 0.79)) !important;
}
/* Force search to appear first and buttons second */
.dt-nav .dataTables_filter {
  order: 0 !important;
}
.dt-nav .dt-buttons {
  order: 1 !important;
}
/* Custom classes for Popup and Close buttons */
.custom-popup-button {
  background-color: #FFA500 !important; /* Clean modern orange */
  border: none !important;
  color: #fff !important;
}
.custom-close-button {
  background-color: #E74C3C !important; /* Not too harsh red */
  border: none !important;
  color: #fff !important;
}
        `;
        document.head.appendChild(styleOverride);

        // Process each table.
        const tables = document.querySelectorAll("table");
        tables.forEach((table) => {
          if (table.getAttribute("data-dt-processed")) return;
          table.setAttribute("data-dt-processed", "true");
          // Store the original table HTML for later restoration.
          table.setAttribute("data-original-html", table.innerHTML);
          const tableWrapper = table.closest(".rdmd-table");
          if (!tableWrapper) return;
          tableWrapper.style.position = "relative";

          // Create the trigger button (3-dot icon) with your original styling.
          const trigger = document.createElement("div");
          trigger.innerHTML = '<i class="fa fa-ellipsis-v" aria-hidden="true"></i>';
          trigger.title = "Enable Filter";
          trigger.style.position = "absolute";
          trigger.style.top = "0px";
          trigger.style.right = "0px";
          trigger.style.padding = "4px 6px";
          trigger.style.cursor = "pointer";
          trigger.style.fontSize = "0.75rem";
          trigger.style.zIndex = "9999";  // Increased z-index for visibility
          trigger.style.borderRadius = "0px 0px 0px 3px";
          trigger.style.color = "rgb(255, 255, 255)";
          trigger.style.background = "linear-gradient(rgb(86, 184, 255), rgba(1, 142, 245, 0.79))";
          trigger.style.filter = "saturate(1.15)";
          trigger.style.display = "block";
          trigger.setAttribute("data-dt-trigger", "true");
          tableWrapper.insertBefore(trigger, tableWrapper.firstChild);

          // When the trigger is clicked, initialize DataTables with a unified nav bar.
          trigger.onclick = () => {
            const $table = window.jQuery(table);
            if (table.getAttribute("data-filter-enabled") === "true") return;
            const dt = $table.DataTable({
              // Custom DOM: search field (f) first then buttons (B) in one nav row above the table.
              dom: "<'dt-nav row'<'col-sm-12 d-flex align-items-center justify-content-between'fB>>rtip",
              // Update language: hide "Search" label and use a placeholder.
              language: {
                search: "",
                searchPlaceholder: "Search Table:"
              },
              buttons: [
                {
                  extend: "copy",
                  text: '<i class="fa fa-copy"></i> Copy'
                },
                {
                  extend: "csv",
                  text: '<i class="fa fa-file-csv"></i> CSV'
                },
                {
                  extend: "excel",
                  text: '<i class="fa fa-file-excel"></i> Excel'
                },
                {
                  extend: "pdf",
                  text: '<i class="fa fa-file-pdf"></i> PDF'
                },
                {
                  extend: "print",
                  text: '<i class="fa fa-print"></i> Print'
                },
                {
                  extend: "colvis",
                  text: '<i class="fa fa-columns"></i> Columns'
                },
                {
                  text: '<i class="fa fa-arrows-alt"></i> Popup',
                  className: "custom-popup-button",
                  action: function (e, dt, node, config) {
                    // Build the modal popup, removing buttons and search.
                    let wrapper =
                      table.closest(".rdmd-table") ||
                      table.closest(".rdmd-table-inner");
                    let clone;
                    if (wrapper) {
                      clone = wrapper.cloneNode(true);
                      const injected = clone.querySelectorAll('[data-dt-trigger], [data-dt-buttons]');
                      injected.forEach((el) => el.remove());
                      const searchDiv = clone.querySelector(".dataTables_filter");
                      if (searchDiv) searchDiv.remove();
                      clone.style.margin = "0";
                    } else {
                      clone = table.cloneNode(true);
                    }
                    const markdownWrapper = document.createElement("div");
                    markdownWrapper.className = "markdown-body";
                    markdownWrapper.appendChild(clone);
                    const modalOverlay = document.createElement("div");
                    modalOverlay.style.position = "fixed";
                    modalOverlay.style.top = "0";
                    modalOverlay.style.left = "0";
                    modalOverlay.style.width = "100%";
                    modalOverlay.style.height = "100%";
                    modalOverlay.style.backgroundColor = "rgba(0, 0, 0, 0.5)";
                    modalOverlay.style.display = "flex";
                    modalOverlay.style.justifyContent = "center";
                    modalOverlay.style.alignItems = "center";
                    modalOverlay.style.zIndex = "9999";
                    modalOverlay.style.overflow = "auto";
                    const modalContent = document.createElement("div");
                    modalContent.style.backgroundColor = "#fff";
                    modalContent.style.padding = "20px";
                    modalContent.style.paddingTop = "40px";
                    modalContent.style.borderRadius = "4px";
                    modalContent.style.maxHeight = "90%";
                    modalContent.style.overflow = "auto";
                    modalContent.style.position = "relative";
                    modalContent.style.minWidth = "800px";
                    modalContent.style.margin = "20px";
                    modalContent.appendChild(markdownWrapper);
                    modalOverlay.appendChild(modalContent);
                    document.body.appendChild(modalOverlay);
                    const closeButton = document.createElement("button");
                    closeButton.innerHTML = '<i class="fa fa-times" aria-hidden="true"></i>';
                    closeButton.title = "Close";
                    closeButton.className = "custom-close-button";
                    // Style the close button with a soft red background.
                    closeButton.style.position = "absolute";
                    closeButton.style.top = "10px";
                    closeButton.style.right = "10px";
                    closeButton.style.background = "#E74C3C";
                    closeButton.style.border = "none";
                    closeButton.style.fontSize = "1.5rem";
                    closeButton.style.cursor = "pointer";
                    closeButton.style.color = "#fff";
                    closeButton.onclick = () => {
                      document.body.removeChild(modalOverlay);
                    };
                    modalContent.appendChild(closeButton);
                    modalOverlay.onclick = (e) => {
                      if (e.target === modalOverlay) {
                        document.body.removeChild(modalOverlay);
                      }
                    };
                  }
                },
                {
                  text: '<i class="fa fa-ban"></i> Close',
                  action: function (e, dt, node, config) {
                    dt.destroy();
                    const original = table.getAttribute("data-original-html");
                    if (original) {
                      table.innerHTML = original;
                      table.removeAttribute("data-original-html");
                    }
                    table.setAttribute("data-filter-enabled", "false");
                    trigger.style.display = "block";
                  }
                }
              ]
            });
            table.setAttribute("data-filter-enabled", "true");
            // Hide the trigger button once filter mode is enabled.
            trigger.style.display = "none";
          };
        });
      })
      .catch((err) => {
        console.error("Error loading dependencies:", err);
      });
  }, []);

  return null;
};

<DownloadableTables />
