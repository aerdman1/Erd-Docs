import React, { useEffect } from "react";

export const DownloadableTables = () => {
  useEffect(() => {
    const tables = document.querySelectorAll("table");

    tables.forEach((table, index) => {
      // Skip if already processed
      if (table.getAttribute("data-csv-processed")) return;
      table.setAttribute("data-csv-processed", "true");

      // Create a container for the buttons
      const container = document.createElement("div");
      container.style.marginBottom = "8px";
      container.style.display = "flex";
      container.style.gap = "8px";

      // Button: Download to CSV
      const csvButton = document.createElement("button");
      csvButton.textContent = "Download to CSV";
      csvButton.style.backgroundColor = "#3490dc"; // blue
      csvButton.style.border = "none";
      csvButton.style.color = "#fff";
      csvButton.style.fontWeight = "bold";
      csvButton.style.padding = "8px 16px";
      csvButton.style.borderRadius = "4px";
      csvButton.style.cursor = "pointer";
      csvButton.onclick = () => {
        const rows = Array.from(table.querySelectorAll("tr"));
        let csvData = "";
        rows.forEach(row => {
          const cells = Array.from(row.querySelectorAll("th, td"));
          let rowData = [];
          cells.forEach(cell => {
            let text = cell.innerText.trim();
            // Escape double quotes by replacing them with two double quotes
            text = text.split('"').join('""');
            // Wrap in quotes if text contains a comma or a double quote
            if (text.indexOf(",") !== -1 || text.indexOf('"') !== -1) {
              text = '"' + text + '"';
            }
            rowData.push(text);
          });
          csvData += rowData.join(",") + "\n";
        });
        const blob = new Blob([csvData], { type: "text/csv;charset=utf-8;" });
        const url = URL.createObjectURL(blob);
        const link = document.createElement("a");
        link.href = url;
        link.download = "table-" + (index + 1) + ".csv";
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
      };

      // Button: Print Table (opens a new window with the table's HTML)
      const printButton = document.createElement("button");
      printButton.textContent = "Print Table";
      printButton.style.backgroundColor = "#38c172"; // green
      printButton.style.border = "none";
      printButton.style.color = "#fff";
      printButton.style.fontWeight = "bold";
      printButton.style.padding = "8px 16px";
      printButton.style.borderRadius = "4px";
      printButton.style.cursor = "pointer";
      printButton.onclick = () => {
        const printWindow = window.open("", "", "width=800,height=600");
        if (printWindow) {
          printWindow.document.write("<html><head><title>Print Table</title></head><body>");
          printWindow.document.write(table.outerHTML);
          printWindow.document.write("</body></html>");
          printWindow.document.close();
          printWindow.focus();
          setTimeout(() => {
            printWindow.print();
            printWindow.close();
          }, 500);
        }
      };

      // Button: Copy Table HTML to clipboard
      const copyButton = document.createElement("button");
      copyButton.textContent = "Copy Table HTML";
      copyButton.style.backgroundColor = "#9561e2"; // purple
      copyButton.style.border = "none";
      copyButton.style.color = "#fff";
      copyButton.style.fontWeight = "bold";
      copyButton.style.padding = "8px 16px";
      copyButton.style.borderRadius = "4px";
      copyButton.style.cursor = "pointer";
      copyButton.onclick = async () => {
        try {
          await navigator.clipboard.writeText(table.outerHTML);
          alert("Table HTML copied to clipboard!");
        } catch (err) {
          alert("Failed to copy: " + err);
        }
      };

      // Append the buttons to the container and then insert it before the table
      container.appendChild(csvButton);
      container.appendChild(printButton);
      container.appendChild(copyButton);
      table.parentNode.insertBefore(container, table);
    });
  }, []);

  return null;
};

<DownloadableTables />
