import React, { useEffect, useRef } from 'react';

export const KeywordTesting123 = ({ keywords = [] }) => {
  const markRef = useRef(null);
  const contentRef = useRef(null);

  // Loads Mark.js and resolves when it's available.
  const loadMarkJs = () =>
    new Promise((resolve) => {
      if (window.Mark) return resolve(window.Mark);
      const existing = document.querySelector('script[src*="mark.min.js"]');
      if (existing) {
        const checkInterval = setInterval(() => {
          if (window.Mark) {
            clearInterval(checkInterval);
            resolve(window.Mark);
          }
        }, 50);
        return;
      }
      const script = document.createElement('script');
      script.src = 'https://unpkg.com/mark.js/dist/mark.min.js';
      script.onload = () => resolve(window.Mark);
      document.head.appendChild(script);
    });

  useEffect(() => {
    let retries = 10; // Limit the number of retry attempts
    let styleInjected = false;

    const attemptHighlight = async () => {
      const Mark = await loadMarkJs();
      // Try to target published docs (.content-body) or MDX preview ([data-testid="RDMD"])
      const target =
        document.querySelector('.content-body') ||
        document.querySelector('.test');
      // If either Mark.js or the target element is not ready yet, retry.
      if (!Mark || !target || !target.innerText.trim()) {
        if (retries-- > 0) {
          setTimeout(attemptHighlight, 300);
        } else {
          console.warn('❌ Mark.js or content not ready after multiple attempts.'); 
        }
        return;
      }

      // Save target reference.
      contentRef.current = target;

      // Inject highlight style only once.
      if (!styleInjected) {
        const style = document.createElement('style');
        style.textContent = `
          mark[data-keyword-highlight] {
            background-color: #ffeeba !important;
            font-weight: bold !important;
            border-radius: 4px !important;
            padding: 2px 4px !important;
          }
        `;
        style.setAttribute('data-keyword-style', 'true');
        document.head.appendChild(style);
        styleInjected = true;
      }

      // Instantiate Mark.js on the target element.
      const markInstance = new Mark(target);
      markRef.current = markInstance;

      // Perform the highlighting.
      markInstance.mark(keywords, {
        separateWordSearch: false,
        element: 'mark',
        className: '',
        attributes: { 'data-keyword-highlight': 'true' },
        done: (count) => console.log(`✨ Highlighted ${count} matches.`),
      });
    };

    attemptHighlight();

    return () => {
      // On cleanup, unmark any changes if needed.
      markRef.current?.unmark();
    };
  }, [keywords]);

  return null;
};
