import React, { useEffect, useRef, useState } from 'react';

export default function KeywordHighlighterInline({ keywords = [] }) {
  const markRef = useRef(null);
  const contentRef = useRef(null);

  const searchSelector = `
    .content-body p, .content-body li, .content-body ul, .content-body ol,
    .content-body details, .content-body code, .content-body table,
    .content-body th, .content-body td, .content-body pre,
    .content-body blockquote, .content-body h1, .content-body h2,
    .content-body h3, .content-body h4, .content-body h5, .content-body h6
  `;

  const loadMarkJs = () =>
    new Promise((resolve) => {
      if (window.Mark) return resolve(window.Mark);
      const script = document.createElement('script');
      script.src = 'https://unpkg.com/mark.js/dist/mark.min.js';
      script.onload = () => resolve(window.Mark);
      document.head.appendChild(script);
    });

  useEffect(() => {
    contentRef.current = document.querySelector('.content-body');

    if (!contentRef.current) {
      console.warn('No .content-body found');
      return;
    }

    const style = document.createElement('style');
    style.textContent = `
      .highlighted-keyword {
        background-color: #ffeeba;
        font-weight: bold;
        border-radius: 4px;
        padding: 2px 4px;
      }
    `;
    document.head.appendChild(style);

    let markInstance;

    loadMarkJs().then((Mark) => {
      if (!contentRef.current) return;
      markInstance = new Mark(contentRef.current);
      markRef.current = markInstance;

      markInstance.mark(keywords, {
        separateWordSearch: false,
        className: 'highlighted-keyword',
      });
    });

    return () => {
      markInstance?.unmark();
      document.head.contains(style) && document.head.removeChild(style);
    };
  }, [keywords]);

  return null; // No UI needed — just highlight
}
