import React, { useEffect, useRef } from 'react';

export const KeywordTesting123 = ({ keywords = [] }) => {
  const markRef = useRef(null);
  const contentRef = useRef(null);

  const loadMarkJs = () =>
    new Promise((resolve) => {
      if (window.Mark) return resolve(window.Mark);
      const script = document.createElement('script');
      script.src = 'https://unpkg.com/mark.js/dist/mark.min.js';
      script.onload = () => resolve(window.Mark);
      document.head.appendChild(script);
    });

  useEffect(() => {
    let markInstance;
    let interval;
    let style;

    const applyHighlighting = () => {
      // Prefer .content-body (published), fallback to .rm-Markdown (preview)
      const target =
        document.querySelector('.content-body') ||
        document.querySelector('.rm-Markdown');

      if (!target || target.innerText.trim().length === 0) return;

      clearInterval(interval);
      contentRef.current = target;

      // Add custom highlight style
      style = document.createElement('style');
      style.textContent = `
        .highlighted-keyword {
          background-color: #ffeeba;
          font-weight: bold;
          border-radius: 4px;
          padding: 2px 4px;
        }
      `;
      document.head.appendChild(style);

      loadMarkJs().then((Mark) => {
        if (!contentRef.current) return;
        markInstance = new Mark(contentRef.current);
        markRef.current = markInstance;

        markInstance.mark(keywords, {
          separateWordSearch: false,
          className: 'highlighted-keyword',
        });
      });
    };

    // Poll every 100ms until content is available
    interval = setInterval(applyHighlighting, 100);

    return () => {
      clearInterval(interval);
      markInstance?.unmark();
      if (style && document.head.contains(style)) {
        document.head.removeChild(style);
      }
    };
  }, [keywords]);

  return null;
};

<KeywordTesting123 keywords={['MDX', 'Tailwind', 'Marketplace']} /> 
<p>This page supports MDX and uses Tailwind. See the Marketplace.</p>
