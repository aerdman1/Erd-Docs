import React, { useState, useEffect } from 'react';

export const OpenApiSpecViewer = () => {
  const [spec, setSpec] = useState(null);
  const [error, setError] = useState(null);

  useEffect(() => {
    async function fetchSpec() {
      try {
        const response = await fetch('https://erdtest123.readme.io/openapi/ErdSpec1.json');
        if (!response.ok) {
          throw new Error(`HTTP error: ${response.status}`);
        }
        const data = await response.json();
        setSpec(data);
      } catch (err) {
        setError(err.message);
      }
    }
    fetchSpec();
  }, []);

  if (error) return <div>Error: {error}</div>;
  if (!spec) return <div>Loading specification...</div>;

  // Example: Render the raw JSON; you can modify this to render just specific parts.
  return (
    <div>
      <h1>OpenAPI Specification</h1>
      <pre style={{ whiteSpace: 'pre-wrap', wordWrap: 'break-word' }}>
        {JSON.stringify(spec, null, 2)}
      </pre>
    </div>
  );
};

export default OpenApiSpecViewer;
