import { useState, useEffect } from 'react';

export function SimplePopUpAlert({ text, onClose, onCheckboxChange }) {
  const [remindNextTime, setRemindNextTime] = useState(true);

  const handleCheckboxChange = (e) => {
    const isChecked = e.target.checked;
    setRemindNextTime(isChecked);
    onCheckboxChange(isChecked);
  };

  return (
    <div className="fixed inset-0 bg-white/60 backdrop-blur-sm flex items-center justify-center z-50">
      <div className="bg-white p-6 rounded-lg shadow-xl max-w-md text-center relative">
        <p className="text-lg text-gray-800">{text}</p>
        <div className="mt-4 flex items-center justify-center">
          <input
            type="checkbox"
            id="remindNextTime"
            checked={remindNextTime}
            onChange={handleCheckboxChange}
            className="mr-2"
          />
          <label htmlFor="remindNextTime" className="text-gray-700">
            Remind me next time
          </label>
        </div>
        <button
          onClick={onClose}
          className="mt-4 px-4 py-2 bg-blue-600 text-white rounded hover:bg-blue-700"
        >
          Close
        </button>
      </div>
    </div>
  );
}

export default function AutoPopUpAlert() {
  const [showAlert, setShowAlert] = useState(false);

  useEffect(() => {
    const getPageKey = () => {
      if (typeof window === 'undefined') return null; // Skip on server-side
      return `popup_reminder_${window.location.pathname}`;
    };

    const pageKey = getPageKey();
    if (pageKey && localStorage.getItem(pageKey) !== 'false') {
      setShowAlert(true);
    }
  }, []);

  const handleClose = (remindNextTime) => {
    const pageKey = `popup_reminder_${window.location.pathname}`;
    localStorage.setItem(pageKey, remindNextTime ? 'true' : 'false');
    setShowAlert(false);
  };

  return (
    <>
      {showAlert && (
        <SimplePopUpAlert
          text="This is an important alert message that you can control! Place any text you want here."
          onClose={() => handleClose(true)} // Defaults to true if user closes without interaction
          onCheckboxChange={(checked) => handleClose(checked)}
        />
      )}
    </>
  );
}
