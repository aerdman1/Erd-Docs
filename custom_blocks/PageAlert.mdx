import { useState } from 'react'

export function PopUpAlert({ title, message, type = 'info' }) {
  const [isOpen, setIsOpen] = useState(true)

  const typeStyles = {
    info: 'bg-blue-100 border-blue-500 text-blue-700',
    success: 'bg-green-100 border-green-500 text-green-700',
    warning: 'bg-yellow-100 border-yellow-500 text-yellow-700',
    error: 'bg-red-100 border-red-500 text-red-700',
  }

  if (!isOpen) return null

  return (
    <div className={`fixed inset-0 flex items-center justify-center z-50`}>
      <div className={`p-4 border-l-4 rounded shadow-lg max-w-lg ${typeStyles[type]} relative`}>
        <strong className="font-bold">{title}</strong>
        <p className="text-sm mt-2">{message}</p>
        <button
          className="absolute top-2 right-2 text-xl leading-none text-gray-500 hover:text-gray-700"
          onClick={() => setIsOpen(false)}
        >
          &times;
        </button>
      </div>
    </div>
  )
}
