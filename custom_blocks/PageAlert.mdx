import React, { useState, useEffect } from "react";

export const PopUpAlert = ({ children }) => {
  const [showAlert, setShowAlert] = useState(false);
  const [remindNextTime, setRemindNextTime] = useState(true);

  useEffect(() => {
    const pageKey = `popup_reminder_${window.location.pathname}`;
    if (localStorage.getItem(pageKey) !== "false") {
      setShowAlert(true);
    }
  }, []);

  const handleClose = () => {
    const pageKey = `popup_reminder_${window.location.pathname}`;
    localStorage.setItem(pageKey, remindNextTime ? "true" : "false");
    setShowAlert(false);
  };

  return (
    <>
      {showAlert && (
        <div className="fixed inset-0 bg-white/60 backdrop-blur-sm flex items-center justify-center z-50">
          <div className="bg-white p-6 rounded-lg shadow-lg max-w-md text-center">
            <div>{children}</div>
            <div className="mt-4 flex items-center justify-center">
              <input
                type="checkbox"
                id="remindNextTime"
                checked={remindNextTime}
                onChange={(e) => setRemindNextTime(e.target.checked)}
                className="mr-2"
              />
              <label htmlFor="remindNextTime" className="text-gray-600">
                Remind me next time
              </label>
            </div>
            <button
              onClick={handleClose}
              className="mt-4 px-4 py-2 bg-purple-700 text-white rounded-md hover:bg-purple-800"
            >
              Close
            </button>
          </div>
        </div>
      )}
    </>
  );
};

export const PopUpContent = ({ title, children }) => (
  <div>
    <h2 className="text-gray-800 text-lg mb-2 font-semibold">{title}</h2>
    <p className="text-gray-600 text-sm">{children}</p>
  </div>
);
