import React, { useState, useEffect } from "react";

export const PopUpAlert = ({ title, children }) => {
  const [showAlert, setShowAlert] = useState(false);
  const [remindNextTime, setRemindNextTime] = useState(true);

  useEffect(() => {
    const pageKey = `popup_reminder_${window.location.pathname}`;
    if (localStorage.getItem(pageKey) !== "false") {
      setShowAlert(true);
    }
  }, []);

  const handleClose = () => {
    const pageKey = `popup_reminder_${window.location.pathname}`;
    localStorage.setItem(pageKey, remindNextTime ? "true" : "false");
    setShowAlert(false);
  };

  return (
    showAlert && (
      <div className="fixed inset-0 flex items-center justify-center bg-gray-900 bg-opacity-75 z-50">
        <div className="bg-white rounded-lg shadow-xl transform transition-all sm:max-w-lg sm:w-full">
          <div className="px-6 py-4 border-b">
            <h2 className="text-lg font-semibold text-gray-800">{title}</h2>
          </div>
          <div className="px-6 py-4 text-gray-700">{children}</div>
          <div className="bg-gray-100 px-6 py-4 text-right">
            <button
              onClick={handleClose}
              className="py-2 px-4 bg-blue-500 text-white rounded font-medium hover:bg-blue-700 transition duration-500"
            >
              Close
            </button>
          </div>
        </div>
      </div>
    )
  );
};

export const PopUpContent = ({ title, children }) => (
  <div>
    <h2 className="text-gray-800 text-lg mb-2 font-semibold">{title}</h2>
    <p className="text-gray-600 text-sm">{children}</p>
  </div>
);

</PopUpAlert>
