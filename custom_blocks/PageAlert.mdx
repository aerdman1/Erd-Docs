import React, { useState, useEffect } from "react";

export const PopUpAlert = ({ children }) => {
  const [showAlert, setShowAlert] = useState(false);
  const [remindNextTime, setRemindNextTime] = useState(true);

  useEffect(() => {
    const getPageKey = () => `popup_reminder_${window.location.pathname}`;
    const pageKey = getPageKey();

    if (localStorage.getItem(pageKey) !== "false") {
      setShowAlert(true);
    }
  }, []);

  const handleClose = () => {
    const pageKey = `popup_reminder_${window.location.pathname}`;
    localStorage.setItem(pageKey, remindNextTime ? "true" : "false");
    setShowAlert(false);
  };

  return (
    <>
      {showAlert && (
        <div
          style={{
            position: "fixed",
            inset: 0,
            backgroundColor: "rgba(255, 255, 255, 0.6)",
            backdropFilter: "blur(5px)",
            display: "flex",
            alignItems: "center",
            justifyContent: "center",
            zIndex: 50,
          }}
        >
          <div
            style={{
              backgroundColor: "white",
              padding: "20px",
              borderRadius: "8px",
              boxShadow: "0 4px 10px rgba(0, 0, 0, 0.2)",
              textAlign: "center",
              maxWidth: "400px",
            }}
          >
            <div>{children}</div>
            <div style={{ marginTop: "15px", display: "flex", alignItems: "center", justifyContent: "center" }}>
              <input
                type="checkbox"
                id="remindNextTime"
                checked={remindNextTime}
                onChange={(e) => setRemindNextTime(e.target.checked)}
                style={{ marginRight: "8px" }}
              />
              <label htmlFor="remindNextTime" style={{ color: "#555" }}>
                Remind me next time
              </label>
            </div>
            <button
              onClick={handleClose}
              style={{
                marginTop: "10px",
                padding: "8px 16px",
                backgroundColor: "#5A3E8B",
                color: "white",
                borderRadius: "5px",
                border: "none",
                cursor: "pointer",
              }}
            >
              Close
            </button>
          </div>
        </div>
      )}
    </>
  );
};

export const PopUpContent = ({ title, children }) => (
  <div>
    <h2 style={{ color: "#333", fontSize: "18px", marginBottom: "10px" }}>{title}</h2>
    <p style={{ color: "#555", fontSize: "14px" }}>{children}</p>
  </div>
);

<PopUpAlert> 
  <PopUpContent title="Important Notice">
    This is a custom alert message. Adjust content as needed!
  </PopUpContent>
</PopUpAlert>
