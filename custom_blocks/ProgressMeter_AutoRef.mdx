import React, { useEffect, useState } from "react";

export const ProgressMeter_Automatic = ({ apiCategory, apiGroup, max = 100, labelPosition = "below", showCards = false }) => {
  const [animatedValues, setAnimatedValues] = useState([]);
  const [showLabels, setShowLabels] = useState(false);
  const [dynamicValues, setDynamicValues] = useState([]);

  useEffect(() => {
    setTimeout(() => {
      const sidebarSections = document.querySelectorAll(".Sidebar-listWrapper6Q9_yUrG906C");

      let selectedCategory = null;
      sidebarSections.forEach(section => {
        const heading = section.querySelector(".Sidebar-headingTRQyOa2pk0gh");
        if (heading && heading.textContent.trim() === apiCategory) {
          selectedCategory = section;
        }
      });

      if (selectedCategory) {
        const groupElements = Array.from(selectedCategory.querySelectorAll(".Sidebar-link-text_label1gCT_uPnx7Gu"));
        const targetGroup = groupElements.find(el => el.textContent.trim() === apiGroup);

        if (targetGroup) {
          const parentItem = targetGroup.closest(".Sidebar-item23D-2Kd61_k3");
          if (parentItem) {
            const endpointItems = parentItem.querySelectorAll(".Sidebar-method-container2yBYD-KB_IfC");

            const colorMap = {
              get: { meter: "#4285F4", card: "#D2E3FC", text: "#174EA6", icon: "fas fa-arrow-down" },
              post: { meter: "#34A853", card: "#DFF6DD", text: "#0D652D", icon: "fas fa-arrow-up" },
              patch: { meter: "#FBBC05", card: "#FEF7E0", text: "#B06000", icon: "fas fa-edit" },
              delete: { meter: "#EA4335", card: "#FCE8E6", text: "#A50E0E", icon: "fas fa-trash" }
            };

            const parsedEndpoints = Array.from(endpointItems)
              .map(endpoint => {
                const methodElement = endpoint.querySelector(".rm-APIMethod");
                const labelElement = endpoint.closest(".Sidebar-item23D-2Kd61_k3").querySelector(".Sidebar-link-text_label1gCT_uPnx7Gu");

                const method = methodElement ? methodElement.textContent.trim().toLowerCase() : "unknown";
                const label = labelElement ? labelElement.textContent.trim() : "Unnamed Endpoint";

                if (!colorMap[method]) return null;

                return {
                  label: method.toUpperCase(),
                  cardLabel: label,
                  value: Math.floor(Math.random() * 30) + 10,
                  meterBackgroundColor: colorMap[method].meter,
                  cardBackgroundColor: colorMap[method].card,
                  textColor: "#fff",
                  cardTextColor: colorMap[method].text,
                  icon: colorMap[method].icon
                };
              })
              .filter(Boolean);

            setDynamicValues(parsedEndpoints);
          }
        }
      }
    }, 500); // Give ReadMe.io time to fully render the sidebar before querying
  }, [apiCategory, apiGroup]);

  useEffect(() => {
    setAnimatedValues(dynamicValues.map(() => 0));

    const interval = setInterval(() => {
      setAnimatedValues(prev => prev.map((val, i) => Math.min(val + 2, dynamicValues[i].value)));
    }, 50);

    const labelTimeout = setTimeout(() => {
      setShowLabels(true);
    }, 500);

    return () => {
      clearInterval(interval);
      clearTimeout(labelTimeout);
    };
  }, [dynamicValues]);

  return (
    <div className="w-full flex flex-col items-start">
      {/* Cards Section */}
      {showCards && (
        <div className="flex gap-4 mb-3 w-full">
          {dynamicValues.map((item, index) => (
            <div 
              key={index} 
              className="shadow-md rounded-lg p-4 flex flex-col items-center w-1/4 border bg-opacity-90"
              style={{ backgroundColor: item.cardBackgroundColor }}
            >
              <span className="text-sm font-medium" style={{ color: item.cardTextColor }}>
                {item.cardLabel || item.label}
              </span>
              <span className="text-lg font-bold" style={{ color: item.cardTextColor }}>
                {item.value}%
              </span>
              {item.icon && <i className={`${item.icon} text-2xl mt-2`} style={{ color: item.cardTextColor }}></i>}
            </div>
          ))}
        </div>
      )}

      {/* Labels Above */}
      {labelPosition === "above" && (
        <div className="flex justify-between w-full mb-1 transition-opacity duration-500" style={{ opacity: showLabels ? 1 : 0 }}>
          {dynamicValues.map((item, index) => (
            <span key={index} className="text-sm font-medium" style={{ color: item.textColor }}>{item.label}</span>
          ))}
        </div>
      )}

      {/* Progress Bar */}
      <div className="relative w-full bg-gray-200 rounded-md overflow-hidden h-6 flex items-center">
        {dynamicValues.map((item, index) => (
          <div 
            key={index} 
            className="h-full transition-all duration-500 flex items-center justify-center relative bg-opacity-90" 
            style={{ width: `${(animatedValues[index] / max) * 100}%`, backgroundColor: item.meterBackgroundColor }}
          >
            {labelPosition === "inside" && showLabels && (
              <span className="text-xs font-semibold absolute left-2" style={{ color: item.textColor }}>{item.label}</span>
            )}
            {labelPosition === "overlay" && showLabels && (
              <span className="text-xs font-semibold absolute inset-0 flex items-center justify-center" style={{ color: item.textColor }}>
                {item.label} ({item.value}%)
              </span>
            )}
          </div>
        ))}
      </div>

      {/* Labels Below */}
      {labelPosition === "below" && (
        <div className="flex justify-between w-full mt-1 transition-opacity duration-500" style={{ opacity: showLabels ? 1 : 0 }}>
          {dynamicValues.map((item, index) => (
            <span key={index} className="text-sm font-medium" style={{ color: item.textColor }}>{item.label} ({item.value}%)</span>
          ))}
        </div>
      )}
    </div>
  );
};
 
<ProgressMeter_Automatic 
  apiCategory="Erdman API Test 1" 
  apiGroup="pet" 
  max={100} 
  labelPosition="overlay"
  showCards={true}
/>
