import React, { useEffect, useRef } from "react";

export const AlertBannerInjected = ({
  message = "YOUR CONTENT GOES HERE.",
  bannerColor = "#ffcc00",          // Default bright yellow background
  bannerTextColor = "#050910",        // Dark text and border color
  targetSelector = ".Header-bottom2eLKOFXMEmh5.rm-Header-bottom" // Update this selector if needed
}) => {
  const bannerRef = useRef(null);

  useEffect(() => {
    // Delay to allow the target element to appear in the DOM.
    const timer = setTimeout(() => {
      const target = document.querySelector(targetSelector);
      console.log("Target found?", target);
      if (target && bannerRef.current) {
        // Move the banner node to just after the target element.
        target.parentNode.insertBefore(bannerRef.current, target.nextSibling);
      } else {
        console.warn("Target element not found for selector:", targetSelector);
      }
    }, 1000); // 1 second delay; adjust if necessary

    return () => clearTimeout(timer);
  }, [targetSelector]);

  return (
    <div
      ref={bannerRef}
      className="alert-banner"
      style={{
        background: bannerColor,
        color: bannerTextColor,
        padding: "10px",
        textAlign: "center",
        fontSize: "14px",
        fontWeight: "bold",
        border: "2px solid " + bannerTextColor,
        borderRadius: "5px",
        marginBottom: "20px"
      }}
    >
      {message}
      <style>{`
        @keyframes bounceIn {
          0% { transform: scale(0.5); opacity: 0; }
          60% { transform: scale(1.1); opacity: 1; }
          100% { transform: scale(1); }
        }
        .alert-banner {
          animation: bounceIn 0.6s ease-out;
        }
      `}</style>
    </div>
  );
};
