import React from "react";

export const AlertBanner = ({
  message = "YOUR CONTENT GOES HERE.",
  preset = "custom",  // Options: "critical", "warning", "standard", or "custom"
  bannerColor,
  bannerTextColor,
  fontSize,
  fontWeight
}) => {
  // Define default settings for each preset
  const presets = {
    critical: {
      bannerColor: "#DC3545",    // Red background
      bannerTextColor: "#FFFFFF", // White text
      fontSize: "16px",
      fontWeight: "normal"
    },
    warning: {
      bannerColor: "#FFC107",    // Yellow background
      bannerTextColor: "#000000", // Black text
      fontSize: "16px",
      fontWeight: "normal"
    },
    standard: {
      bannerColor: "#007BFF",    // Blue background
      bannerTextColor: "#FFFFFF", // White text
      fontSize: "16px",
      fontWeight: "normal"
    },
    custom: {
      bannerColor: "#ffcc00",    // Bright yellow (default custom)
      bannerTextColor: "#050910", // Dark text and border
      fontSize: "14px",
      fontWeight: "bold"
    }
  };

  // Determine the default settings from the chosen preset,
  // and allow props to override these values.
  const defaults = presets[preset] || presets.custom;
  const finalBannerColor = bannerColor || defaults.bannerColor;
  const finalBannerTextColor = bannerTextColor || defaults.bannerTextColor;
  const finalFontSize = fontSize || defaults.fontSize;
  const finalFontWeight = fontWeight || defaults.fontWeight;

  return (
    <div>
      <div
        className="alert-banner"
        style={{
          background: finalBannerColor,
          color: finalBannerTextColor,
          padding: "10px",
          textAlign: "center",
          fontSize: finalFontSize,
          fontWeight: finalFontWeight,
          border: "2px solid " + finalBannerTextColor,
          borderRadius: "5px",
          marginBottom: "20px"
        }}
      >
        {message}
      </div>
      <style>{`
        @keyframes bounceIn {
          0% { transform: scale(0.5); opacity: 0; }
          60% { transform: scale(1.1); opacity: 1; }
          100% { transform: scale(1); }
        }
        .alert-banner {
          animation: bounceIn 0.6s ease-in 0.5s;
          animation-fill-mode: forwards;
        }
      `}</style>
    </div>
  );
};
