import React from "react";

export const Banner = ({
  // Select one to be true - defaults to custom if all are false
  critical = false,
  warning = false,
  standard = false,
  custom = false,
  message,
  bannerColor,
  bannerTextColor,
  fontSize,
  fontWeight
}) => {
  // Determine the preset based on the boolean props.
  let selectedPreset;
  if (critical) {
    selectedPreset = "critical";
  } else if (warning) {
    selectedPreset = "warning";
  } else if (standard) {
    selectedPreset = "standard";
  } else if (custom) {
    selectedPreset = "custom";
  } else {
    // Default to custom if no preset is explicitly set.
    selectedPreset = "custom";
  }

  // Define default settings for each preset.
  const defaultPresets = {
    critical: {
      bannerColor: "bg-red-600",    // Tailwind: red background
      bannerTextColor: "text-white", // Tailwind: white text
      fontSize: "text-base",         // Tailwind: base font size (~16px)
      fontWeight: "font-normal",
      message: "Critical Alert: Please check immediately!"
    },
    warning: {
      bannerColor: "bg-yellow-400",    // Tailwind: yellow background
      bannerTextColor: "text-black",   // Tailwind: black text
      fontSize: "text-base",
      fontWeight: "font-normal",
      message: "Warning Alert: There might be an issue."
    },
    standard: {
      bannerColor: "bg-blue-600",    // Tailwind: blue background
      bannerTextColor: "text-white", // Tailwind: white text
      fontSize: "text-base",
      fontWeight: "font-normal",
      message: "Standard Alert: Here is some information."
    },
    custom: {
      // These values are for inline styles since they're custom hex values.
      bannerColor: "#00BCFF",    
      bannerTextColor: "#fff",
      fontSize: "14px",
      fontWeight: "bold",
      message: "Here is my banner message - hope you enjoy!"
    }
  };

  // Get the default settings for the selected preset.
  const defaults = defaultPresets[selectedPreset];
  const finalMessage = message || defaults.message;

  // Base Tailwind classes for layout, padding, border, and margin.
  let containerClasses = "p-2 text-center rounded-md mb-5";
  let inlineStyles = {};

  // Use inline styles if it's the custom preset or if any override is provided.
  if (selectedPreset === "custom" || bannerColor || bannerTextColor || fontSize || fontWeight) {
    inlineStyles.background = bannerColor || (selectedPreset === "custom" ? defaults.bannerColor : undefined);
    inlineStyles.color = bannerTextColor || (selectedPreset === "custom" ? defaults.bannerTextColor : undefined);
    inlineStyles.borderColor = bannerTextColor || (selectedPreset === "custom" ? defaults.bannerTextColor : undefined);
    inlineStyles.fontSize = fontSize || (selectedPreset === "custom" ? defaults.fontSize : undefined);
    inlineStyles.fontWeight = fontWeight || (selectedPreset === "custom" ? defaults.fontWeight : undefined);
  } else {
    // For presets that use Tailwind classes, add those classes.
    containerClasses += ` ${defaults.bannerColor} ${defaults.bannerTextColor} ${defaults.fontSize} ${defaults.fontWeight}`;
  }

  return (
    <div>
      <div className={containerClasses} style={inlineStyles}>
        {finalMessage}
      </div>
    </div>
  );
};

<Banner />
