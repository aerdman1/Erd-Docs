import React from "react";

export const AlertBanner = ({
  preset = "a", // Options: "critical", "warning", "standard", or "custom"
  message,
  bannerColor,
  bannerTextColor,
  fontSize,
  fontWeight
}) => {
  // Define default settings for each preset type.
  const defaultPresets = {
    critical: {
      bannerColor: "#DC3545",    // Red
      bannerTextColor: "#FFFFFF", // White text
      fontSize: "16px",
      fontWeight: "normal",
      message: "Critical Alert: Please check immediately!"
    },
    warning: {
      bannerColor: "#FFC107",    // Yellow
      bannerTextColor: "#000000", // Black text
      fontSize: "16px",
      fontWeight: "normal",
      message: "Warning Alert: There might be an issue."
    },
    standard: {
      bannerColor: "#007BFF",    // Blue
      bannerTextColor: "#FFFFFF", // White text
      fontSize: "16px",
      fontWeight: "normal",
      message: "Standard Alert: Here is some information."
    },
    custom: {
      bannerColor: "#ffcc00",    // Bright yellow (default for custom)
      bannerTextColor: "#050910", // Dark text and border
      fontSize: "14px",
      fontWeight: "bold",
      message: "YOUR CONTENT GOES HERE."
    }
  };

  // Get the default settings for the chosen preset.
  const defaults = defaultPresets[preset] || defaultPresets.custom;

  // Allow props to override preset defaults.
  const finalBannerColor = bannerColor || defaults.bannerColor;
  const finalBannerTextColor = bannerTextColor || defaults.bannerTextColor;
  const finalFontSize = fontSize || defaults.fontSize;
  const finalFontWeight = fontWeight || defaults.fontWeight;
  const finalMessage = message || defaults.message;

  return (
    <div>
      <div
        className="alert-banner"
        style={{
          background: finalBannerColor,
          color: finalBannerTextColor,
          padding: "10px",
          textAlign: "center",
          fontSize: finalFontSize,
          fontWeight: finalFontWeight,
          border: "2px solid " + finalBannerTextColor,
          borderRadius: "5px",
          marginBottom: "20px"
        }}
      >
        {finalMessage}
      </div>
    </div>
  );
};

<AlertBanner preset="critical" />
