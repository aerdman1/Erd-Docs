import React, { useEffect, useState } from "react";

export const PageBanner_Simple = ({
  isHeader = false,
  message = "Hello! 👋 This is your page-specific announcement banner — customize it to share updates, reminders, or anything important.",
  bannerColor = "#006fff",
  bannerTextColor = "#ffffff",
  fontSize = "16px",
  fontWeight = "400"
}) => {
  const [bannerContainer, setBannerContainer] = useState(null);

  // Function to remove the banner on navigation
  const removeBanner = () => {
    const container = document.getElementById("simple-banner-container");
    if (container) {
      container.remove();
    }
  };

  // When isHeader is true, find the header element and create/reuse the banner container.
  useEffect(() => {
    if (isHeader) {
      const header = document.querySelector("header.rm-Header");
      if (header) {
        let container = document.getElementById("simple-banner-container");
        if (!container) {
          container = document.createElement("div");
          container.id = "simple-banner-container";
          // Always append the container to the bottom of the header
          header.appendChild(container);
        }
        setBannerContainer(container);
      }
    } else {
      setBannerContainer(null);
    }
  }, [isHeader]);

  // Update container styles and content whenever props change.
  useEffect(() => {
    if (bannerContainer) {
      bannerContainer.style.background = bannerColor;
      bannerContainer.style.color = bannerTextColor;
      bannerContainer.style.fontSize = fontSize;
      bannerContainer.style.fontWeight = fontWeight;
      bannerContainer.style.padding = "8px";
      bannerContainer.style.textAlign = "center";
      bannerContainer.style.borderRadius = "6px";
      bannerContainer.textContent = message;
    }
  }, [bannerContainer, message, bannerColor, bannerTextColor, fontSize, fontWeight]);

  // Remove the banner when navigating to another page
  useEffect(() => {
    // Override history methods to detect navigation
    const originalPushState = history.pushState;
    const originalReplaceState = history.replaceState;

    history.pushState = function () {
      originalPushState.apply(history, arguments);
      removeBanner();
    };

    history.replaceState = function () {
      originalReplaceState.apply(history, arguments);
      removeBanner();
    };

    window.addEventListener("popstate", removeBanner);

    return () => {
      history.pushState = originalPushState;
      history.replaceState = originalReplaceState;
      window.removeEventListener("popstate", removeBanner);
    };
  }, []);

  // When isHeader is false, render the banner inline.
  if (!isHeader) {
    return (
      <div
        style={{
          background: bannerColor,
          color: bannerTextColor,
          fontSize: fontSize,
          fontWeight: fontWeight,
          padding: "8px",
          textAlign: "center",
          borderRadius: "6px",
          marginBottom: "20px"
        }}
      >
        {message}
      </div>
    );
  }

  // If isHeader is true, the banner is injected into the header, so nothing is rendered inline.
  return null;
};

<PageBanner_Simple 
  isHeader={false}
  message="Hello! 👋 This is your page-specific banner — customize it to share updates, reminders, or anything important." 
  bannerColor="#006fff" 
  bannerTextColor="#ffffff" 
  fontSize="16px" 
  fontWeight="400" 
/>
