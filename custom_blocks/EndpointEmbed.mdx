import React, { useEffect, useMemo, useState } from "react";

export const EndpointEmbed = ({
  websiteBaseUrl = "https://erdtest123.readme.io",
  displayMode = "popup",
  theme = "endpoint-block",
  apiTargetUrl,
  endpoint,
  method,
  linkText,
  specFile,
  labels = true,
  show = {
    summary: true,
    description: true,
    parameters: true,
    tags: true,
    requestBody: false,
    responses: false,
    security: false,
    example: false,
    readmeUrl: false,
    apiUrl: false,
  },
}) => {
  const [endpointData, setEndpointData] = useState(null);
  const [loading, setLoading] = useState(false);
  const [isHovered, setIsHovered] = useState(false);
  const [fadeIn, setFadeIn] = useState(false);

  const displayLinkText = linkText || endpoint;
  // Append a query parameter to bypass cache if needed for debugging
  const fullSpecUrl = useMemo(() => {
    return websiteBaseUrl + "/openapi/" + specFile + "?nocache=" + Date.now();
  }, [websiteBaseUrl, specFile]);

  /* ------------------------------------------------------------------
   *  Inject shared CSS once
   * ----------------------------------------------------------------*/
  useEffect(() => {
    if (document.getElementById("embed-styles")) return;
    const style = document.createElement("style");
    style.id = "embed-styles";
    style.textContent = `
      /* Shared color palette */
      :root {
        --gray-500: #6b7280;
        --gray-800: #111827;
        --gray-light: #f9f9f9;
        --blue-hover: #f0f8ff;
        --white: #fff;
        --shadow-default: 0 8px 16px rgba(0, 0, 0, 0.1);
        --shadow-hover: 0 4px 20px rgba(0, 0, 0, 0.08);
      }

      /* Base block styles */
      .endpoint-block,
      .endpoint-simple {
        display: flex;
        flex-direction: column;
        text-decoration: none;
        color: inherit;
        margin-bottom: 8px;
        width: auto;
        max-width: 100%;
        position: relative;
      }

      .endpoint-block {
        background: var(--gray-light);
        border: 1px solid #d1d5db;
        border-radius: 8px;
        padding: 16px;
      }

      .endpoint-block:hover {
        background: var(--blue-hover);
        border-color: var(--gray-500);
      }

      .endpoint-simple {
        background: transparent;
        border: none;
        padding: 16px;
      }

      /* Shared arrow indicator */
      .endpoint-block::after,
      .endpoint-simple::after,
      .popup-card::after {
        content: '↗';
        font-size: 0.75rem;
        position: absolute;
        top: 8px;
        right: 8px;
        opacity: 0;
        color: var(--gray-500);
        transition: opacity 0.2s ease;
      }

      .endpoint-block:hover::after,
      .endpoint-simple:hover::after,
      .popup-card:hover::after {
        opacity: 1;
      }

      /* Header section */
      .header-row {
        display: flex;
        align-items: center;
        justify-content: flex-end;
        flex-direction: row-reverse;
        padding-bottom: 3px;
        border-bottom: 0.5px solid #6b728057;
      }

      /* Text and metadata */
      .endpoint-block h3 {
        margin: 0 0 0.5em;
        font-size: 1rem;
        font-weight: bold;
        color: var(--gray-800);
      }

      .method {
        font-weight: bold;
        text-transform: uppercase;
        font-size: 0.85rem;
        display: inline-block;
      }

      .method.get { color: #2563eb; }
      .method.post { color: #16a34a; }
      .method.put { color: #ca8a04; }
      .method.delete { color: #dc2626; }

      .path {
        font-family: monospace;
        font-size: 0.9rem;
        color: var(--gray-500);
        word-break: break-word;
      }

      /* Popup-specific styling */
      .popup-card {
        position: relative;
        background: var(--white);
        border: none;
        border-radius: 8px;
        padding: 12px 16px;
        width: fit-content;
        min-width: 300px;
        max-width: 100%;
        font-size: 0.85rem;
        box-shadow: var(--shadow-default);
        transition: box-shadow 0.2s ease, background 0.2s ease;
      }

      .popup-card:hover {
        background: var(--blue-hover);
        box-shadow: var(--shadow-hover);
      }

      /* When using the "simple" theme, the popup-card gets minimal styling */
      .popup-card.simple {
        border: none;
        border-radius: 0;
        padding: 0;
      }

      /* Neutralize container colors */
      .CardsGrid .Card,
      .Accordion .Accordion-title,
      .CardsGrid .Card-icon,
      .Accordion {
        background: inherit;
        color: inherit;
      }

      /* Skeleton animation & styling */
      @keyframes pulse {
        0% { opacity: 1; }
        50% { opacity: 0.4; }
        100% { opacity: 1; }
      }
      
      .skeleton-card {
        animation: pulse 1.5s infinite ease-in-out;
      }
    `;
    document.head.appendChild(style);
  }, []);

  /* ------------------------------------------------------------------
   *  Fetch & parse spec with logging and cache-bypass query param
   * ----------------------------------------------------------------*/
  useEffect(() => {
    if (!endpoint) return;
    const run = async () => {
      setLoading(true);
      try {
        if (!window.__endpointSpecCache) window.__endpointSpecCache = {};
        let spec = window.__endpointSpecCache[fullSpecUrl];
        if (spec) {
          console.log(`[CACHE HIT] Spec from cache for ${fullSpecUrl}:`, spec);
        } else {
          const res = await fetch(fullSpecUrl);
          if (!res.ok) throw new Error("fetch failed");
          spec = await res.json();
          console.log(`[FETCH] New spec retrieved from ${fullSpecUrl}:`, spec);
          window.__endpointSpecCache[fullSpecUrl] = spec;
        }
        if (!spec.paths || !spec.paths[endpoint]) {
          console.error(`Spec does not contain endpoint "${endpoint}":`, spec);
          throw new Error("missing path");
        }
        const ops = spec.paths[endpoint];
        const m = method ? method.toLowerCase() : undefined;
        const opMethod = (m && ops[m]) ? m : Object.keys(ops)[0];
        const op = ops[opMethod];
        const example = extractExample(op, spec.components);

        // Extract global file-level last updated timestamp.
        const fileLastUpdated =
          spec["x-readme"] && spec["x-readme"]["x-last-updated"]
            ? spec["x-readme"]["x-last-updated"]
            : null;
        // Extract endpoint-specific last updated timestamp if available.
        const endpointLastUpdated = op["x-last-updated"] ? op["x-last-updated"] : null;

        setEndpointData({
          path: endpoint,
          method: opMethod.toUpperCase(),
          summary: op.summary || "",
          description: op.description || "",
          example: example,
          operationId: op.operationId || "",
          parameters: op.parameters || [],
          requestBody: op.requestBody || null,
          responses: op.responses || {},
          security: op.security || [],
          tags: op.tags || [],
          endpointLastUpdated,
          fileLastUpdated,
        });
      } catch (e) {
        console.error("Error fetching or processing spec:", e);
        setEndpointData(null);
      } finally {
        setLoading(false);
      }
    };
    run();
  }, [endpoint, fullSpecUrl, method]);

  // When endpointData loads, trigger the fade-in effect.
  useEffect(() => {
    if (endpointData) {
      setTimeout(() => setFadeIn(true), 100);
    }
  }, [endpointData]);

  /* Helper functions – no optional chaining or Object.fromEntries */
  function resolveSchema(schema, comps) {
    if (!schema) return null;
    if (schema.example !== undefined) return schema.example;
    if (schema.$ref) {
      const refKey = schema.$ref.split("/").pop();
      return resolveSchema(comps.schemas[refKey], comps);
    }
    if (schema.type === "object" && schema.properties) {
      const result = {};
      Object.keys(schema.properties).forEach(k => {
        result[k] = resolveSchema(schema.properties[k], comps);
      });
      return result;
    }
    if (schema.type === "array") return [resolveSchema(schema.items, comps)];
    return null;
  }

  function extractExample(op, comps) {
    const responses = op.responses || {};
    const resp = responses["200"] || Object.values(responses)[0];
    const media = resp && resp.content && resp.content["application/json"];
    if (!media) return null;
    if (media.example) return media.example;
    if (media.examples) {
      const vals = Object.values(media.examples);
      if (vals.length > 0 && vals[0].value) return vals[0].value;
    }
    return resolveSchema(media.schema, comps);
  }

  const readmeDocUrl =
    (endpointData && endpointData.operationId)
      ? websiteBaseUrl + "/reference/" + endpointData.operationId
      : websiteBaseUrl + "/reference/" + endpoint.replace(/[\/{}]/g, "");
  const apiFullUrl = apiTargetUrl ? (apiTargetUrl + endpoint) : null;
  const label = txt => (labels ? <strong>{txt}: </strong> : null);

  // ConnectionIndicator renders as a small inline-block element.
  const ConnectionIndicator = ({ connected, inline }) => {
    const style = inline
      ? {
          display: "inline-block",
          width: 8,
          height: 8,
          borderRadius: "50%",
          marginRight: 8,
          alignSelf: "center",
        }
      : {
          position: "absolute",
          top: 2,
          right: 4,
          width: 10,
          height: 10,
          borderRadius: "50%",
        };
    return (
      <div
        style={{ ...style, backgroundColor: connected ? "green" : "red" }}
        title={connected ? "Synced" : "Not Connected"}
      />
    );
  };

  /* UI Pieces */
  const Card = ({ children, extra = "" }) => {
    return (
      <div
        className={(theme === "simple" ? "endpoint-simple" : "endpoint-block") + " " + extra}
        style={{ position: "relative" }}
      >
        {children}
      </div>
    );
  };

  const MethodTag = ({ method }) => {
    return <span className={"method " + method.toLowerCase()}>{method}</span>;
  };

  // Skeleton placeholder while loading.
  const SkeletonCard = () => (
    <div
      className="skeleton-card"
      style={{
        background: "#e2e2e2",
        borderRadius: 8,
        padding: 16,
        width: "fit-content",
        minWidth: 250,
      }}
    >
      <div style={{ height: 20, width: "60%", background: "#ccc", marginBottom: 8 }} />
      <div style={{ height: 16, width: "80%", background: "#ccc", marginBottom: 4 }} />
      <div style={{ height: 16, width: "90%", background: "#ccc", marginBottom: 4 }} />
    </div>
  );

  // Fallback card when no data is available.
  const fallbackCard = (
    <a href={readmeDocUrl} style={{ textDecoration: "none", color: "inherit" }}>
      <div
        className={("popup-card " + (theme === "simple" ? "simple" : "")).trim()}
        style={{ position: "relative" }}
      >
        {show.summary && method && (
          <div className="header-row" style={{ width: "100%" }}>
            <MethodTag method={method} />
            <ConnectionIndicator connected={!!endpointData && !loading} inline={true} />
          </div>
        )}
        <div style={{ flexGrow: 1 }}>
          <h3 style={{ margin: "0.5em 0 0.5em" }}>
            {linkText || displayLinkText}
          </h3>
          <div
            style={{
              fontFamily: "monospace",
              fontSize: "0.9rem",
              color: "#6b7280",
            }}
          >
            {endpoint}
          </div>
          {show.apiUrl && apiTargetUrl && (
            <div
              style={{
                fontFamily: "monospace",
                fontSize: "0.9rem",
                color: "#4b5563",
              }}
            >
              API: {method ? method.toUpperCase() : ""} {apiTargetUrl + endpoint}
            </div>
          )}
          {show.readmeUrl && <div className="path">{readmeDocUrl}</div>}
        </div>
      </div>
    </a>
  );

  // Full rendered content (all extra fields) used for inline mode.
  const renderedContent =
    endpointData && (
      <div
        style={{
          opacity: fadeIn ? 1 : 0,
          transition: "opacity 0.5s ease-in-out"
        }}
      >
        <a href={readmeDocUrl} style={{ textDecoration: "none", color: "inherit" }}>
          <Card>
            {show.summary && endpointData.method && (
              <div className="header-row" style={{ width: "100%" }}>
                <MethodTag method={endpointData.method} />
                <ConnectionIndicator connected={true} inline={true} />
              </div>
            )}
            <div style={{ flexGrow: 1 }}>
              <h3 style={{ margin: "0.5em 0 0.5em" }}>
                {linkText || endpointData.summary || displayLinkText}
              </h3>
              {show.description && endpointData.description && (
                <p style={{ margin: "0 0 0.5em", fontSize: "0.9rem", color: "#374151" }}>
                  {endpointData.description}
                </p>
              )}
              {show.parameters &&
                endpointData.parameters.length > 0 && (
                  <div>
                    {label("Parameters")}
                    {endpointData.parameters.map(p => p.name).join(", ")}
                  </div>
                )}
              {show.requestBody && endpointData.requestBody && (
                <div>
                  {label("Request Body")}Available
                </div>
              )}
              {show.responses && endpointData.responses && (
                <div>
                  {label("Responses")}
                  {Object.keys(endpointData.responses).join(", ")}
                </div>
              )}
              {show.security &&
                endpointData.security.length > 0 && (
                  <div>
                    {label("Security")}
                    {JSON.stringify(endpointData.security)}
                  </div>
                )}
              {show.tags &&
                endpointData.tags.length > 0 && (
                  <div>
                    {label("Tags")}
                    {endpointData.tags.join(", ")}
                  </div>
                )}
              {show.example && endpointData.example && (
                <pre
                  style={{
                    background: "#f3f4f6",
                    padding: 8,
                    margin: "5px 0px",
                    borderRadius: 6,
                    fontSize: "0.85rem",
                    overflowX: "auto",
                  }}
                >
                  {JSON.stringify(endpointData.example, null, 2)}
                </pre>
              )}
              {show.apiUrl && apiFullUrl && (
                <div
                  style={{
                    fontFamily: "monospace",
                    fontSize: "0.9rem",
                    color: "#4b5563",
                  }}
                >
                  API: {endpointData.method} {apiFullUrl}
                </div>
              )}
              {show.readmeUrl && <div className="path">{readmeDocUrl}</div>}
              {/* Render the updated bubble.
                  If the endpoint has a specific timestamp, show that; otherwise, if available, show the file-level timestamp. */}
              {endpointData.endpointLastUpdated ? (
                <div
                  style={{
                    fontSize: "0.75rem",
                    color: "#6b7280",
                    marginTop: 8,
                    textAlign: "right",
                    background: "#f9f9f9",
                    padding: "2px 4px",
                    borderRadius: "4px",
                  }}
                >
                  Endpoint Last Updated:{" "}
                  {new Date(endpointData.endpointLastUpdated).toLocaleString()}
                </div>
              ) : (
                endpointData.fileLastUpdated && (
                  <div
                    style={{
                      fontSize: "0.75rem",
                      color: "#6b7280",
                      marginTop: 8,
                      textAlign: "right",
                      background: "#f9f9f9",
                      padding: "2px 4px",
                      borderRadius: "4px",
                    }}
                  >
                    File Last Updated:{" "}
                    {new Date(endpointData.fileLastUpdated).toLocaleString()}
                  </div>
                )
              )}
            </div>
          </Card>
        </a>
      </div>
    );

  /* ------------------------------------------------------------------
   *  Render based on displayMode
   * ----------------------------------------------------------------*/
  if (displayMode === "popup") {
    const linkStyle = {
      textDecoration: "none",
      cursor: "pointer",
      color: "#1a73e8",
      background: "#e8f0fe",
      borderRadius: "4px",
      padding: "2px 4px",
      fontWeight: 600,
    };
    return (
      <span
        style={{ display: "inline-block", position: "relative" }}
        onMouseEnter={() => setIsHovered(true)}
        onMouseLeave={() => setIsHovered(false)}
      >
        <a href={readmeDocUrl} style={linkStyle}>
          {displayLinkText}
        </a>
        {isHovered && (
          <div style={{ position: "absolute", left: 0, zIndex: 1000, width: 220 }}>
            {loading ? (
              <SkeletonCard />
            ) : (
              endpointData ? (
                <div
                  style={{
                    opacity: fadeIn ? 1 : 0,
                    transition: "opacity 0.5s ease-in-out",
                  }}
                >
                  <a href={readmeDocUrl} style={{ textDecoration: "none", color: "inherit" }}>
                    <div className={"popup-card " + (theme === "simple" ? "simple" : "")}>
                      {renderedContent}
                    </div>
                  </a>
                </div>
              ) : (
                fallbackCard
              )
            )}
          </div>
        )}
      </span>
    );
  }

  // Inline mode: show fallback if no data.
  if (loading) return <div style={{ position: "relative" }}><SkeletonCard /></div>;
  if (!endpointData) return <div style={{ position: "relative" }}>{fallbackCard}</div>;
  return renderedContent;
};
