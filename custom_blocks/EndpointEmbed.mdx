import React, { useEffect, useState } from 'react';

export const EndpointEmbed = ({
  baseUrl = 'https://erdtest123.readme.io',  // New prop for the base URL
  endpoint,
  method, // Optional prop to specify HTTP method, e.g., 'GET', 'POST'
  displayMode = 'popup',
  linkText,
  specFile = "ErdSpec1.json",
  show = {
    summary: true,
    description: true,
    parameters: true,
    requestBody: true,
    responses: true,
    security: true,
    tags: true,
    example: true,
  }
}) => {
  const [endpointData, setEndpointData] = useState(null);
  const [error, setError] = useState(null);
  const [loading, setLoading] = useState(false);
  const [isHovered, setIsHovered] = useState(false);

  // Default link text to the endpoint value if none is provided.
  const displayLinkText = linkText || endpoint;
  // Build the full spec URL using the baseUrl and specFile prop.
  const fullSpecUrl = `${baseUrl}/openapi/${specFile}`;

  useEffect(() => {
    if (!endpoint) {
      setError('Endpoint not provided.');
      return;
    }
    fetchEndpointExample();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [endpoint, fullSpecUrl, method]);

  const fetchEndpointExample = async () => {
    setLoading(true);
    setError(null);
    try {
      const response = await fetch(fullSpecUrl);
      if (!response.ok) {
        throw new Error("This endpoint may be private or restricted.");
      }
      const spec = await response.json();
      const operations = spec.paths[endpoint];
      if (!operations) {
        throw new Error(`No endpoint found in spec for path ${endpoint}`);
      }

      let chosenOperation = null;
      let chosenMethod = null;
      
      // If a specific method is provided and exists in the spec, use it.
      if (typeof method === 'string' && operations[method.toLowerCase()]) {
        chosenOperation = operations[method.toLowerCase()];
        chosenMethod = method.toUpperCase();
      } else {
        // Otherwise, iterate over operations and pick one with an example.
        for (const [m, op] of Object.entries(operations)) {
          const example = extractResponseExample(op, spec.components);
          if (example) {
            chosenOperation = op;
            chosenMethod = m.toUpperCase();
            break;
          }
        }
        // Fallback to the first available operation if no operation had an example.
        if (!chosenOperation) {
          const [m, op] = Object.entries(operations)[0];
          chosenOperation = op;
          chosenMethod = m.toUpperCase();
        }
      }

      const summary = chosenOperation.summary || '';
      const description = chosenOperation.description || '';
      const example = extractResponseExample(chosenOperation, spec.components);
      const operationId = chosenOperation.operationId || '';

      // Additional metadata from the operation.
      const parameters = chosenOperation.parameters || [];
      const requestBody = chosenOperation.requestBody || null;
      const responses = chosenOperation.responses || {};
      const security = chosenOperation.security || [];
      const tags = chosenOperation.tags || [];
      const deprecated = chosenOperation.deprecated || false;

      setEndpointData({
        path: endpoint,
        method: chosenMethod,
        summary,
        description,
        example,
        operationId,
        parameters,
        requestBody,
        responses,
        security,
        tags,
        deprecated,
      });
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };

  // Helper: Recursively resolve an example from a schema.
  const resolveExampleFromSchema = (schema, components) => {
    if (!schema) return null;
    if (schema.example !== undefined) {
      return schema.example;
    }
    if (schema.$ref) {
      const refParts = schema.$ref.split('/');
      const schemaName = refParts[refParts.length - 1];
      const resolved = components.schemas[schemaName];
      return resolveExampleFromSchema(resolved, components);
    }
    if (schema.type === 'object' && schema.properties) {
      const obj = {};
      Object.entries(schema.properties).forEach(([prop, propSchema]) => {
        obj[prop] = resolveExampleFromSchema(propSchema, components);
      });
      return obj;
    }
    if (schema.type === 'array' && schema.items) {
      return [resolveExampleFromSchema(schema.items, components)];
    }
    return null;
  };

  // Helper: Extract an example from an operation’s responses.
  const extractResponseExample = (operation, components) => {
    if (!operation.responses) return null;
    let response = operation.responses['200'] || operation.responses['default'];
    if (!response) {
      response = Object.values(operation.responses)[0];
    }
    if (response && response.content) {
      const media = response.content['application/json'];
      if (media) {
        if (media.example) {
          return media.example;
        }
        if (media.examples) {
          const firstExample = Object.values(media.examples)[0];
          if (firstExample && firstExample.value) return firstExample.value;
        }
        if (media.schema) {
          return resolveExampleFromSchema(media.schema, components);
        }
      }
    }
    return null;
  };

  // Base card styling.
  const baseCardStyle = {
    border: '1px solid #ddd',
    background: '#fff',
    padding: '16px',
    borderRadius: '8px',
    boxShadow: '0px 0px 10px rgba(0,0,0,0.1)',
  };

  // Use full width for inline mode; fixed 400px width for popup mode.
  const cardWidthStyle = displayMode === 'inline'
    ? { width: '100%' }
    : { width: '400px' };

  // For popup mode, the card is absolutely positioned so that it overlays content without moving it.
  const popupPositionStyle = displayMode === 'popup'
    ? { position: 'absolute', top: '100%', left: '0', zIndex: 1000, marginTop: '1px' }
    : {};

  const cardStyle = displayMode === 'inline'
    ? { padding: 0, background: 'transparent', border: 'none', boxShadow: 'none' }
    : { ...baseCardStyle, ...cardWidthStyle, ...popupPositionStyle };

  // Custom link styling so it stands out.
  const customLinkStyle = {
    textDecoration: 'none',
    cursor: 'pointer',
    color: '#1a73e8',
    backgroundColor: '#e8f0fe',
    borderRadius: '4px',
    padding: '2px 4px',
    fontWeight: '600',
  };

  // Determine the reference URL.
  // Use operationId if available; otherwise, fall back to a sanitized endpoint string.
  const referenceUrl = endpointData && endpointData.operationId
    ? `${baseUrl}/reference/${endpointData.operationId}`
    : `${baseUrl}/reference/${endpoint ? endpoint.replace(/[\/{}]/g, '') : ''}`;

  // Render additional metadata sections using the "show" prop to toggle visibility.
  const renderParameters = () => {
    if (!show.parameters || !endpointData.parameters.length) return null;
    return (
      <div>
        <strong>Parameters:</strong>
        <ul>
          {endpointData.parameters.map((param, idx) => (
            <li key={idx}>
              <strong>{param.name}</strong> (<em>{param.in}</em>) {param.required ? '[Required]' : ''}
              {param.description && ` - ${param.description}`}
              {param.schema && param.schema.type && ` (type: ${param.schema.type})`}
              {param.schema && param.schema.default !== undefined && ` [default: ${param.schema.default}]`}
            </li>
          ))}
        </ul>
      </div>
    );
  };

  const renderRequestBody = () => {
    if (!show.requestBody || !endpointData.requestBody) return null;
    const contentTypes = endpointData.requestBody.content ? Object.keys(endpointData.requestBody.content) : [];
    return (
      <div>
        <strong>Request Body:</strong>
        {endpointData.requestBody.description && <p>{endpointData.requestBody.description}</p>}
        {contentTypes.length > 0 && (
          <p>Supported Content Types: {contentTypes.join(', ')}</p>
        )}
      </div>
    );
  };

  const renderResponses = () => {
    if (!show.responses || !endpointData.responses || Object.keys(endpointData.responses).length === 0) return null;
    return (
      <div>
        <strong>Responses:</strong>
        <ul>
          {Object.entries(endpointData.responses).map(([code, response]) => (
            <li key={code}>
              <strong>{code}:</strong> {response.description}
            </li>
          ))}
        </ul>
      </div>
    );
  };

  const renderSecurity = () => {
    if (!show.security || !endpointData.security.length) return null;
    return (
      <div>
        <strong>Security:</strong>
        <pre style={{ whiteSpace: 'pre-wrap', fontSize: '0.9em' }}>
          {JSON.stringify(endpointData.security, null, 2)}
        </pre>
      </div>
    );
  };

  const renderTags = () => {
    if (!show.tags || !endpointData.tags.length) return null;
    return (
      <div>
        <strong>Tags:</strong> {endpointData.tags.join(', ')}
      </div>
    );
  };

  const renderDeprecated = () => {
    if (!endpointData.deprecated) return null;
    return (
      <div style={{ color: 'red', fontWeight: 'bold' }}>
        Warning: This endpoint is deprecated.
      </div>
    );
  };

  const renderSummary = () => {
    if (!show.summary || !endpointData.summary) return null;
    return <p><strong>Summary:</strong> {endpointData.summary}</p>;
  };

  const renderDescription = () => {
    if (!show.description || !endpointData.description) return null;
    return <p><strong>Description:</strong> {endpointData.description}</p>;
  };

  const renderExample = () => {
    if (!show.example || !endpointData.example) return null;
    return (
      <div>
        <strong>Example Response:</strong>
        <pre
          style={{
            whiteSpace: 'pre-wrap',
            padding: '8px',
            borderRadius: '4px',
            marginTop: '8px'
          }}
        >
          {JSON.stringify(endpointData.example, null, 2)}
        </pre>
      </div>
    );
  };

  // Render the main content card.
  const renderedContent = endpointData && (
    <div style={cardStyle}>
      <h3 style={{ margin: '5px 0' }}>
        <a href={referenceUrl} style={customLinkStyle}>
          {displayLinkText}
        </a>
      </h3>
      {renderDeprecated()}
      {renderSummary()}
      {renderDescription()}
      {renderParameters()}
      {renderRequestBody()}
      {renderResponses()}
      {renderSecurity()}
      {renderTags()}
      {renderExample()}
    </div>
  );

  // Skeleton Loader component to use during loading.
  const SkeletonLoader = () => (
    <div className="skeleton">
      <div>
        <div className="skeleton-method"></div>
        <div className="skeleton-title"></div>
        <div className="skeleton-path"></div>
      </div>
    </div>
  );

  if (displayMode === 'popup') {
    return (
      <span
        style={{ display: 'inline-block', position: 'relative' }}
        onMouseEnter={() => setIsHovered(true)}
        onMouseLeave={() => setIsHovered(false)}
      >
        <a href={referenceUrl} style={customLinkStyle}>
          {displayLinkText}
        </a>
        {isHovered && (
          <>
            {loading && (
              <div
                style={{
                  ...cardStyle,
                  position: 'absolute',
                  top: '100%',
                  left: '0',
                  zIndex: 1000,
                  marginTop: '1px'
                }}
              >
                <SkeletonLoader />
              </div>
            )}
            {error && (
              <div
                style={{
                  ...cardStyle,
                  color: 'red',
                  position: 'absolute',
                  top: '100%',
                  left: '0',
                  zIndex: 1000,
                  marginTop: '1px'
                }}
              >
                {error}
              </div>
            )}
            {!loading && !error && renderedContent}
          </>
        )}
      </span>
    );
  }

  return (
    <>
      <div style={{ marginBottom: '16px', width: '100%' }}>
        {loading && <SkeletonLoader />}
        {error && <div style={{ color: 'red' }}>{error}</div>}
        {!loading && !error && renderedContent}
      </div>
    </>
  );
};

### Basic Usage 

These examples show the default behavior with minimal details (only summary and description):

<EndpointEmbed 
  endpoint="/pet" 
  method="PUT" 
  specFile="ErdSpec1.json" 
  linkText="Pet Endpoint" 
  displayMode="popup" 
  show={{ summary: true, description: true, parameters: false, requestBody: false, responses: false, security: false, tags: false, example: false }}
/>

<EndpointEmbed 
  endpoint="/pet" 
  method="PUT" 
  specFile="ErdSpec1.json" 
  linkText="Pet Endpoint" 
  displayMode="inline" 
  show={{ summary: true, description: true, parameters: false, requestBody: false, responses: false, security: false, tags: false, example: false }}
/>

# Advanced Usage

Below are more advanced examples that demonstrate the component in different display modes and container types.

## 🔗 Popup Mode

When you want to keep your page uncluttered, use the "popup" display mode. The endpoint renders as a hyperlink—like this  
<EndpointEmbed endpoint="/pet" method="PUT" specFile="ErdSpec1.json" linkText="Pet Endpoint" displayMode="popup" />.  
When users hover over it, they’ll see a card with full details.

Another one:  
<EndpointEmbed endpoint="/pet/findByStatus" method="GET" specFile="ErdSpec1.json" linkText="Find Pets by Status" displayMode="popup" />  
which lets you preview how the search works without cluttering your layout.

***

## 📦 Inline Mode

If you want all endpoint info to be immediately visible, `"inline"` mode will render the card directly in the page flow.

<EndpointEmbed endpoint="/pet" method="PUT" specFile="ErdSpec1.json" displayMode="inline" linkText="Inline Pet Endpoint (PUT)" />

<EndpointEmbed endpoint="/pet" method="POST" specFile="ErdSpec1.json" displayMode="inline" linkText="Inline Pet Endpoint (POST)" />

This is great when you're walking users through specific request/response flows.

***

## 🗂️ Works with Any Spec File

You can point to other OpenAPI files by changing the `specFile` prop. Useful if you’re documenting multiple services.

<EndpointEmbed endpoint="/v1/users" method="GET" specFile="NovaTest.json" displayMode="inline" linkText="List Users (NovaTest)" />

<EndpointEmbed endpoint="/v1/users" method="POST" specFile="NovaTest.json" displayMode="inline" linkText="Create User (NovaTest)" />

***

## 🧱 Inside Cards

This component plays nicely inside ReadMe's `<Cards>` layout. Great for overview sections or dashboards.

<Cards columns={2}>
  <Card title="User Management" icon="fa-user-cog">
    <EndpointEmbed endpoint="/v1/users" method="GET" specFile="NovaTest.json" displayMode="inline" linkText="List Users" />
    <EndpointEmbed endpoint="/v1/users" method="POST" specFile="NovaTest.json" displayMode="inline" linkText="Create User" />
  </Card>

  <Card title="Posts API" icon="fa-file-alt">
    <EndpointEmbed endpoint="/v1/posts" method="POST" specFile="NovaTest.json" displayMode="inline" linkText="Create Post" />
    <EndpointEmbed endpoint="/v1/posts/{postId}" method="GET" specFile="NovaTest.json" displayMode="inline" linkText="Get Post by ID" />
  </Card>
</Cards>

***

## 🔽 Inside Accordions

Perfect for FAQs or progressive disclosure.

<Accordion title="Upload a Pet Image" icon="fa-image">
  <EndpointEmbed endpoint="/pet/{petId}/uploadImage" method="POST" specFile="ErdSpec1.json" displayMode="inline" linkText="Upload Pet Image" />
</Accordion>

<Accordion title="Delete Pet by ID" icon="fa-trash">
  <EndpointEmbed endpoint="/pet/{petId}" method="DELETE" specFile="ErdSpec1.json" displayMode="inline" linkText="Delete Pet" />
</Accordion>

***

## ➗ Inside Columns

Use `<Columns>` to organize multiple endpoints side by side.

<Columns layout="auto">
  <Column>
    <EndpointEmbed endpoint="/pet/findByStatus" method="GET" specFile="ErdSpec1.json" displayMode="inline" linkText="Find by Status" />
  </Column>

  <Column>
    <EndpointEmbed endpoint="/pet/findByTags" method="GET" specFile="ErdSpec1.json" displayMode="inline" linkText="Find by Tags" />
  </Column>

  <Column>
    <EndpointEmbed endpoint="/pet/{petId}/uploadImage" method="POST" specFile="ErdSpec1.json" displayMode="inline" linkText="Upload Image" />
  </Column>
</Columns>
