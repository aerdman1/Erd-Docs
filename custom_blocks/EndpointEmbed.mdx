import React, { useEffect, useMemo, useState } from "react";

export const EndpointEmbed = ({
  websiteBaseUrl = "https://erdtest123.readme.io",
  displayMode = "popup",            // "popup" or "inline"
  theme = "card-block",             // "card-block" or "card-simple"
  cardBg,                           // optional override for background
  cardPadding = "15px",        // optional override for padding 
  apiTargetUrl,
  endpoint,
  method,
  linkText,
  specFile, 
  labels = true,
  show = {
    summary: true,
    description: true,
    parameters: true,
    tags: true,
    requestBody: false,
    responses: false,
    security: false,
    example: false,
    readmeUrl: false,
    apiUrl: false,
  },
}) => {
  const DEFAULT_SHOW = {
    summary:    true,
    description:true,
    parameters: true,
    tags:       true,
    requestBody:false, 
    responses:  true,
    security:   false,
    example:    false,
    readmeUrl:  false,
    apiUrl:     false,
  };
  const mergedShow = { ...DEFAULT_SHOW, ...show };

  const [endpointData, setEndpointData] = useState(null);
  const [loading, setLoading] = useState(false);
  const [isHovered, setIsHovered] = useState(false);
  const [fadeIn, setFadeIn] = useState(false);

  const displayLinkText = linkText || endpoint;
  const fullSpecUrl = useMemo(
    () => `${websiteBaseUrl}/openapi/${specFile}?nocache=${Date.now()}`,
    [websiteBaseUrl, specFile]
  );

  // Inject shared CSS once
  useEffect(() => {
    if (document.getElementById("embed-styles")) return;
    const style = document.createElement("style");
    style.id = "embed-styles";
    style.textContent = `
      /* Shared color palette */
      :root {
        --gray-500: #6b7280;
        --gray-800: #111827;
        --gray-light: #f9f9f9;
        --blue-hover: #f0f8ff;
        --white: #fff;
        --shadow-default: 0 8px 16px rgba(0,0,0,0.1);
        --shadow-hover:   0 4px 20px rgba(0,0,0,0.08);
      }
      .card-block,
      .card-simple {
        display: flex;
        flex-direction: column;
        text-decoration: none;
        color: inherit;
        margin-bottom: 8px;
        max-width: 100%;
        position: relative;
        min-width:250px !important;
      }
      .method {
        font-weight: bold;
        text-transform: uppercase;
        font-size: 0.85rem;
        display: inline-block;
      }
      .method.get    { color: #2563eb; }
      .method.post   { color: #16a34a; }
      .method.put    { color: #ca8a04; }
      .method.delete { color: #dc2626; }
      .card-block::after,
      .card-simple::after,
      .card-popup::after {
        content: '↗';
        font-size: 0.75rem;
        position: absolute;
        top: 8px;
        right: 8px;
        opacity: 0;
        color: var(--gray-500);
        transition: opacity 0.2s ease;
      }
      .card-block:hover::after,
      .card-simple:hover::after,
      .card-popup:hover::after {
        opacity: 1;
      }
      .header-row {
        display: flex;
        align-items: center;
        justify-content: flex-end;
        flex-direction: row-reverse;
        padding-bottom: 3px;
        border-bottom: 0.5px solid rgba(107,115,128,0.35);
      }
      .card-popup {
        position: relative;
        border: none;
        border-radius: 8px;
        font-size: 0.85rem;
        box-shadow: var(--shadow-default);
        transition: box-shadow 0.2s ease, background 0.2s ease;
        min-width:250px !important;
      }
      .card-popup.simple {
        border-radius: 0;
      }
      .CardsGrid .Card,
      .Accordion,
      .Accordion .Accordion-title,
      .CardsGrid .Card-icon {
        background: inherit !important;
        color:     inherit !important;
      }
      @keyframes pulse {
        0%   { opacity: 1; }
        50%  { opacity: 0.4; }
        100% { opacity: 1; }
      }
      .skeleton-card {
        animation: pulse 1.5s infinite ease-in-out;
      }
    `;
    document.head.appendChild(style);
  }, []);

  // Fetch & parse spec
  useEffect(() => {
    if (!endpoint) return;
    setLoading(true);
    (async () => {
      try {
        if (!window.__endpointSpecCache) window.__endpointSpecCache = {};
        let spec = window.__endpointSpecCache[fullSpecUrl];
        if (!spec) {
          const res = await fetch(fullSpecUrl);
          if (!res.ok) throw new Error("fetch failed");
          spec = await res.json();
          window.__endpointSpecCache[fullSpecUrl] = spec;
        }
        const ops = spec.paths?.[endpoint];
        if (!ops) throw new Error(`missing path "${endpoint}"`);
        const m = method?.toLowerCase();
        const opKey = m && ops[m] ? m : Object.keys(ops)[0];
        const op = ops[opKey];

        const example = extractExample(op, spec.components);
        const fileLastUpdated     = spec["x-readme"]?.["x-last-updated"] || null;
        const endpointLastUpdated = op["x-last-updated"] || null;

        setEndpointData({
          path: endpoint,
          method: opKey.toUpperCase(),
          summary:     op.summary     || "",
          description: op.description || "",
          example,
          operationId: op.operationId || "",
          parameters:  op.parameters  || [],
          requestBody: op.requestBody || null,
          responses:   op.responses   || {},
          security:    op.security    || [],
          tags:        op.tags        || [],
          endpointLastUpdated,
          fileLastUpdated,
        });
      } catch (e) {
        console.error(e);
        setEndpointData(null);
      } finally {
        setLoading(false);
      }
    })();
  }, [endpoint, fullSpecUrl, method]);

  useEffect(() => {
    if (endpointData) setTimeout(() => setFadeIn(true), 100);
  }, [endpointData]);

  // Helpers
  function resolveSchema(schema, comps) {
    if (!schema) return null;
    if (schema.example !== undefined) return schema.example;
    if (schema.$ref) {
      const key = schema.$ref.split("/").pop();
      return resolveSchema(comps.schemas[key], comps);
    }
    if (schema.type === "object") {
      const obj = {};
      Object.entries(schema.properties || {}).forEach(([k,v]) => {
        obj[k] = resolveSchema(v, comps);
      });
      return obj;
    }
    if (schema.type === "array") {
      return [resolveSchema(schema.items, comps)];
    }
    return null;
  }
  function extractExample(op, comps) {
    const resp = op.responses?.["200"] || Object.values(op.responses || {})[0];
    const media = resp?.content?.["application/json"];
    if (!media) return null;
    if (media.example) return media.example;
    if (media.examples) {
      const ex = Object.values(media.examples)[0];
      return ex?.value ?? null;
    }
    return resolveSchema(media.schema, comps);
  }
  function parseCustomDate(s) {
    const [y,m,d] = s.split("-");
    return new Date(`${y}-${m.padStart(2,"0")}-${d.padStart(2,"0")}`);
  }
  function formatLastUpdated(s) {
    return parseCustomDate(s).toLocaleDateString();
  }

  const readmeDocUrl = endpointData?.operationId
    ? `${websiteBaseUrl}/reference/${endpointData.operationId}`
    : `${websiteBaseUrl}/reference/${endpoint.replace(/[\/{}]/g,"")}`;

  const label = txt => labels ? <strong>{txt}: </strong> : null;

  // Card wrapper
  const Card = ({ children, variant }) => {
    const defaults = {
      popup:    { bg: theme==="card-block" ? "var(--gray-light)" : "var(--white)",    pad: "12px 16px" },
      inline:   { bg: theme==="card-block" ? "var(--gray-light)" : "transparent",     pad: "16px"     },
      fallback: { bg: theme==="card-block" ? "var(--gray-light)" : "transparent",     pad: "16px"     },
    }[variant];

    return (
      <div
        className={variant==="popup" ? "card-popup" : theme}
        style={{
          position:    "relative",
          background:  cardBg     ?? defaults.bg,
          padding:     cardPadding?? defaults.pad,
          border:      theme==="card-block" ? "1px solid #d1d5db" : "none",
          borderRadius:8,
          boxShadow:   variant==="popup" ? "var(--shadow-default)" : "none",
        }}
      >
        {children}
      </div>
    );
  };

  const MethodTag    = ({ method }) => <span className={`method ${method.toLowerCase()}`}>{method}</span>;
  const ConnectionIndicator = ({ connected, inline }) => {
    const s = inline
      ? { display:"inline-block", width:8, height:8, borderRadius:"50%", marginRight:8 }
      : { position:"absolute", top:2, right:4, width:10, height:10, borderRadius:"50%" };
    return <div style={{ ...s, backgroundColor: connected?"green":"red" }} />;
  };
  const SkeletonCard = () => (
    <div className="skeleton-card" style={{ background:"#e2e2e2", borderRadius:8, padding:16, width:250 }}>
      <div style={{ height:20, width:"60%", background:"#ccc", marginBottom:8 }} />
      <div style={{ height:16, width:"80%", background:"#ccc", marginBottom:4 }} />
      <div style={{ height:16, width:"90%", background:"#ccc", marginBottom:4 }} />
    </div>
  );

  // Fallback
  const fallbackCard = (
    <a href={readmeDocUrl} style={{ textDecoration:"none", color:"inherit" }}>
      <Card variant="fallback">
        {mergedShow.summary && method && (
          <div className="header-row">
            <MethodTag method={method} />
            <ConnectionIndicator connected={!!endpointData && !loading} inline />
          </div>
        )}
        <div style={{ flexGrow:1 }}>
          <h3 style={{ margin:"0.5em 0" }}>{displayLinkText}</h3>
          <div style={{ fontFamily:"monospace", fontSize:"0.9rem", color:"#6b7280" }}>
            {endpoint}
          </div>
          {mergedShow.apiUrl && apiTargetUrl && (
            <div style={{ fontFamily:"monospace", fontSize:"0.9rem", color:"#4b5563" }}>
              API: {method?.toUpperCase()} {apiTargetUrl + endpoint}
            </div>
          )}
          {mergedShow.readmeUrl && <div className="path">{readmeDocUrl}</div>}
        </div>
      </Card>
    </a>
  );

  // Full content
  const renderedContent = endpointData && (
    <a href={readmeDocUrl} style={{ textDecoration:"none", color:"inherit" }}>
      <Card variant={displayMode}>
        {mergedShow.summary && endpointData.method && (
          <div className="header-row">
            <MethodTag method={endpointData.method} />
            <ConnectionIndicator connected inline />
          </div>
        )}
        <div style={{ flexGrow:1, opacity: fadeIn?1:0, transition:"opacity 0.5s" }}>
          <h3 style={{ margin:"0.5em 0" }}>{endpointData.summary || displayLinkText}</h3>
          {mergedShow.description && endpointData.description && (
            <p style={{ margin:"0 0 0.5em", fontSize:"0.9rem", color:"#374151" }}>
              {endpointData.description}
            </p>
          )}
          {mergedShow.parameters && endpointData.parameters.length>0 && (
            <div>{label("Parameters")}{endpointData.parameters.map(p=>p.name).join(", ")}</div>
          )}
          {mergedShow.requestBody && endpointData.requestBody && (
            <div>{label("Request Body")}Available</div>
          )}
          {mergedShow.responses && (
            <div>{label("Responses")}{Object.keys(endpointData.responses).join(", ")}</div>
          )}
          {mergedShow.security && endpointData.security.length>0 && (
            <div>{label("Security")}{JSON.stringify(endpointData.security)}</div>
          )}
          {mergedShow.tags && endpointData.tags.length>0 && (
            <div>{label("Tags")}{endpointData.tags.join(", ")}</div>
          )}
          {mergedShow.example && endpointData.example && (
            <pre style={{ background:"#f3f4f6", padding:8, borderRadius:6, overflowX:"auto" }}>
              {JSON.stringify(endpointData.example,null,2)}
            </pre>
          )}
          {mergedShow.apiUrl && apiTargetUrl && (
            <div style={{ fontFamily:"monospace", fontSize:"0.9rem", color:"#4b5563" }}>
              API: {endpointData.method} {apiTargetUrl+endpoint}
            </div>
          )}
          {mergedShow.readmeUrl && <div className="path">{readmeDocUrl}</div>}
          {(endpointData.endpointLastUpdated||endpointData.fileLastUpdated)&&(
            <div style={{ fontSize:"0.75rem", color:"#6b7280", marginTop:8, textAlign:"right" }}>
              {endpointData.endpointLastUpdated
                ? `Endpoint Modified: ${formatLastUpdated(endpointData.endpointLastUpdated)}`
                : `File Last Updated: ${formatLastUpdated(endpointData.fileLastUpdated)}`}
            </div>
          )}
        </div>
      </Card>
    </a>
  );

  // Final render
  if (displayMode === "popup") {
    return (
      <span
        style={{ display:"inline-block", position:"relative", cursor:"pointer" }}
        onMouseEnter={()=>setIsHovered(true)}
        onMouseLeave={()=>setIsHovered(false)}
      >
        <a
          href={readmeDocUrl}
          style={{
            textDecoration:"none",
            color:"#1a73e8",
            background:"#e8f0fe",
            borderRadius:4,
            padding:"2px 4px",
            fontWeight:600
          }}
        >
          {displayLinkText}
        </a>
        {isHovered && (
          <div style={{ position:"absolute", left:0, top:"100%", zIndex:1000 }}>
            {loading ? <SkeletonCard/> : (endpointData ? renderedContent : fallbackCard)}
          </div>
        )}
      </span>
    );
  }

  // inline
  if (loading) return <SkeletonCard />;
  if (!endpointData) return fallbackCard;
  return renderedContent;
};

<Accordion title="All props/default values">
| **Prop**           | **Type**      | **Default**                          | **Description**                                                                 |
|--------------------|---------------|--------------------------------------|---------------------------------------------------------------------------------|
| `websiteBaseUrl`   | `string`      | `'https://yourdocsite.com'`          | The ReadMe project base URL (used for linking to reference pages).             |
| `apiTargetUrl`     | `string`      | _none_                               | Optional override for the real API server base URL (for displaying API URL).   |
| `endpoint`         | `string`      | _required_                           | The API path you want to display (e.g. `/pet/{petId}`).                         |
| `method`           | `string`      | _optional_                           | The HTTP method (e.g. `GET`, `POST`). If omitted, auto-selects best match.     |
| `specFile`         | `string`      | _required_                           | The OpenAPI JSON file name to load (e.g. `ErdSpec1.json`).                     |
| `displayMode`      | `string`      | `'popup'`                            | Determines how the component renders: `'popup'` or `'inline'`.                 |
| `linkText`         | `string`      | _fallbacks to `endpoint`_            | Custom text to display instead of the raw endpoint path.                        |
| `theme`            | `string`      | `'endpoint-block'`                   | `'endpoint-block'` renders with background chrome; `'simple'` removes padding and borders.|
| `labels`           | `boolean`     | `true`                               | If `false`, hides labels like `Summary:`, `Parameters:`, etc.                  |
| `show.summary`     | `boolean`     | `true`                               | Toggles display of the summary/method line.                                    |
| `show.description` | `boolean`     | `true`                               | Toggles display of the endpoint description.                                   |
| `show.parameters`  | `boolean`     | `true`                               | Toggles display of parameter names.                                            |
| `show.tags`        | `boolean`     | `true`                               | Toggles display of OpenAPI tags.                                               |
| `show.requestBody` | `boolean`     | `false`                              | Toggles request body presence indicator.                                       |
| `show.responses`   | `boolean`     | `false`                              | Toggles list of response codes.                                                |
| `show.security`    | `boolean`     | `false`                              | Toggles display of security scheme info.                                       |
| `show.example`     | `boolean`     | `false`                              | Toggles display of a sample JSON response.                                     |
| `show.readmeUrl`   | `boolean`     | `false`                              | Toggles display of the ReadMe reference doc URL.                               |
| `show.apiUrl`      | `boolean`     | `false`                              | Toggles display of the full API call URL (with method and `apiTargetUrl`).     |
</Accordion>
