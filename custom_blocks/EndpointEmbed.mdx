import React, { useEffect, useMemo, useState } from 'react';

export const EndpointEmbed = ({
  websiteBaseUrl = 'https://erdtest123.readme.io', // The website URL for fetching the OpenAPI spec
  apiTargetUrl, // Optional: The actual API URL for the endpoint (if different)
  endpoint,
  method,
  displayMode = 'popup',
  linkText,
  specFile,
  show = {
    summary: true,
    description: true,
    parameters: true,
    tags: true,
    requestBody: false,
    responses: false,
    security: false,
    example: false,
    readmeUrl: false,
    apiUrl: false, // New toggle to display the actual API URL
  }
}) => {
  const [endpointData, setEndpointData] = useState(null);
  const [loading, setLoading] = useState(false);
  const [isHovered, setIsHovered] = useState(false);
  const arrow = '›';

  // Default link text if not provided.
  const displayLinkText = linkText || endpoint;
  const fullSpecUrl = useMemo(() => `${websiteBaseUrl}/openapi/${specFile}`, [websiteBaseUrl, specFile]);

  // Inject shared CSS once.
  useEffect(() => {
    if (!document.getElementById('endpoint-embed-styles')) {
      const style = document.createElement('style');
      style.id = 'endpoint-embed-styles';
      style.textContent = `
        .card {
          display: flex;
          align-items: center;
          justify-content: space-between;
          background: #f9f9f9 !important;
          border: 1px solid #d1d5db;
          border-radius: 8px;
          padding: 16px; 
          text-decoration: none !important;
          color: inherit;
          transition: background 0.2s;
          margin-bottom: 8px;
        }
        .card:hover {
          background: #7dbeff1c;
          border: 1px solid #018ef5;
        }
        .card-content { flex-grow: 1; }
        .card h3 {
          margin: 0;
          font-size: 1rem;
          font-weight: bold;
          color: #111827 !important;
        }
        .method {
          font-weight: bold;
          text-transform: uppercase;
          font-size: 0.85rem;
          margin-bottom: 4px;
          display: block;
        }
        .method.get { color: #2563eb; }
        .method.post { color: #16a34a; }
        .method.put { color: #ca8a04; }
        .method.delete { color: #dc2626; }
        .path {
          font-family: monospace;
          font-size: 0.9rem;
          color: #6b7280;
        }
        .arrow { font-size: 1.2rem; color: #6b7280; }
        .skeleton {
          width: 420px !important;
          background: #f3f4f6;
          border: 1px solid #e5e7eb;
          border-radius: 8px;
          padding: 16px;
          display: flex;
          align-items: center;
          justify-content: space-between;
          opacity: 0.7;
          min-height: 100px;
        }
        .skeleton-method, .skeleton-title, .skeleton-path {
          background: linear-gradient(90deg, #e5e7eb 25%, #f3f4f6 50%, #e5e7eb 75%);
          background-size: 200% 100%;
          animation: shimmer 1.5s infinite linear;
          border-radius: 4px;
        }
        .skeleton-method { width: 50px; height: 12px; margin-bottom: 8px; }
        .skeleton-title { width: 60%; height: 16px; margin-bottom: 6px; }
        .skeleton-path { width: 80%; height: 12px; }
        @keyframes shimmer {
          0% { background-position: -200% 0; }
          100% { background-position: 200% 0; }
        }
        [data-color-mode=dark] .card { background: #000; }
        @media (prefers-color-scheme: dark) {
          [data-color-mode=system] .card { background:#000; }
        }
        [data-color-mode=dark] .card:hover { background: #000; }
        @media (prefers-color-scheme: dark) {
          [data-color-mode=system] .card:hover { background: #000; }
        }
        [data-color-mode=dark] .card h3 { color: #fff; }
        @media (prefers-color-scheme: dark) {
          [data-color-mode=system] .card h3 { color: #fff; }
        }
        /* Popup specific styles */
        .popup-card {
          box-shadow: 0px 4px 10px rgba(0,0,0,0.2);
          background: #fff !important;
          border: 1px solid #ccc;
          padding:10px !important;
          border-radius:20px !important
          width: fit-content;
					max-width: 100%;
        }
        .popup-card:hover {
          border: 1px solid #018ef5;
        }
      `;
      document.head.appendChild(style);
    }
  }, []);

  // Fetch the spec file and extract endpoint info.
  useEffect(() => {
    if (!endpoint) {
      // Instead of setting an error, simply stop loading and show fallback.
      setLoading(false);
      setEndpointData(null);
      return;
    }
    const fetchEndpointExample = async () => {
      setLoading(true);
      // Use a simple cache if available.
      let cache = {};
      if (typeof window !== 'undefined') {
        if (!window.__endpointSpecCache) {
          window.__endpointSpecCache = {};
        }
        cache = window.__endpointSpecCache;
      }
      try {
        let spec;
        if (cache[fullSpecUrl]) {
          spec = cache[fullSpecUrl];
        } else {
          const response = await fetch(fullSpecUrl);
          if (!response.ok) {
            // Instead of throwing an error, fall back.
            setEndpointData(null);
            return;
          }
          spec = await response.json();
          cache[fullSpecUrl] = spec;
        }
        const operations = spec.paths[endpoint];
        if (!operations) {
          setEndpointData(null);
          return;
        }
        let chosenOperation = null;
        let chosenMethod = null;
        const lowerMethod = typeof method === 'string' ? method.toLowerCase() : null;
        if (lowerMethod && operations[lowerMethod]) {
          chosenOperation = operations[lowerMethod];
          chosenMethod = method.toUpperCase();
        } else {
          for (const [m, op] of Object.entries(operations)) {
            const example = extractResponseExample(op, spec.components);
            if (example) {
              chosenOperation = op;
              chosenMethod = m.toUpperCase();
              break;
            }
          }
          if (!chosenOperation) {
            const [m, op] = Object.entries(operations)[0];
            chosenOperation = op;
            chosenMethod = m.toUpperCase();
          }
        }
        const summary = chosenOperation.summary || '';
        const description = chosenOperation.description || '';
        const example = extractResponseExample(chosenOperation, spec.components);
        const operationId = chosenOperation.operationId || '';
        const parameters = chosenOperation.parameters || [];
        const requestBody = chosenOperation.requestBody || null;
        const responses = chosenOperation.responses || {};
        const security = chosenOperation.security || [];
        const tags = chosenOperation.tags || [];
        setEndpointData({
          path: endpoint,
          method: chosenMethod,
          summary,
          description,
          example,
          operationId,
          parameters,
          requestBody,
          responses,
          security,
          tags,
        });
      } catch (err) {
        // On any error, simply clear the endpoint data.
        setEndpointData(null);
      } finally {
        setLoading(false);
      }
    };
    fetchEndpointExample();
  }, [endpoint, fullSpecUrl, method]);

  // Recursively resolve an example from a schema.
  const resolveExampleFromSchema = (schema, components) => {
    if (!schema) return null;
    if (schema.example !== undefined) return schema.example;
    if (schema.$ref) {
      const schemaName = schema.$ref.split('/').pop();
      const resolved = components.schemas[schemaName];
      return resolveExampleFromSchema(resolved, components);
    }
    if (schema.type === 'object' && schema.properties) {
      return Object.fromEntries(
        Object.entries(schema.properties).map(([prop, propSchema]) => [
          prop,
          resolveExampleFromSchema(propSchema, components)
        ])
      );
    }
    if (schema.type === 'array' && schema.items) {
      return [resolveExampleFromSchema(schema.items, components)];
    }
    return null;
  };

  // Extract an example from an operation’s responses.
  const extractResponseExample = (operation, components) => {
    if (!operation.responses) return null;
    let response = operation.responses['200'] || operation.responses['default'];
    if (!response) {
      response = Object.values(operation.responses)[0];
    }
    if (response && response.content) {
      const media = response.content['application/json'];
      if (media) {
        if (media.example) return media.example;
        if (media.examples) {
          const firstExample = Object.values(media.examples)[0];
          if (firstExample && firstExample.value) return firstExample.value;
        }
        if (media.schema) return resolveExampleFromSchema(media.schema, components);
      }
    }
    return null;
  };

  // Card component for inline usage.
  const Card = ({ children, classNameExtra = '' }) => (
    <div className={`card ${classNameExtra}`}>
      <div className="card-content">{children}</div>
      <span className="arrow">{arrow}</span>
    </div>
  );

  // Method tag component.
  const MethodTag = ({ method }) => {
    const lowerMethod = (method || '').toLowerCase();
    return <span className={`method ${lowerMethod}`}>{method}</span>;
  };

  // Construct the ReadMe reference URL.
  const readmeDocUrl =
    endpointData && endpointData.operationId
      ? `${websiteBaseUrl}/reference/${endpointData.operationId}`
      : `${websiteBaseUrl}/reference/${endpoint ? endpoint.replace(/[\/{}]/g, '') : ''}`;

  // Build the API endpoint URL if provided.
  const apiFullUrl = apiTargetUrl ? `${apiTargetUrl}${endpoint}` : null;

  // Reusable fallback card when no valid endpoint data is available.
  const fallbackCard = (
    <a href={readmeDocUrl} style={{ textDecoration: 'none', color: 'inherit' }}>
      <Card>
        <div style={{ flexGrow: 1 }}>
          {show.summary && method && <MethodTag method={method} />}
          <h3 style={{ margin: '0 0 0.5em', fontSize: '1rem', fontWeight: 'bold' }}>
            {linkText || displayLinkText}
          </h3>
          <div style={{ marginBottom: '0.25em', fontFamily: 'monospace', fontSize: '0.9rem', color: '#6b7280' }}>
            {endpoint}
          </div>
          {show.apiUrl && apiTargetUrl && (
            <div style={{ fontFamily: 'monospace', fontSize: '0.9rem', color: '#4b5563' }}>
              API: {method?.toUpperCase()} {apiTargetUrl + endpoint}
            </div>
          )}
          {show.readmeUrl && (
            <div className="path" style={{ fontFamily: 'monospace', fontSize: '0.9rem', color: '#6b7280' }}>
              {readmeDocUrl}
            </div>
          )}
        </div>
      </Card>
    </a>
  );

  // Skeleton card to display during loading.
  const skeletonCard = (
    <div className="card skeleton popup-card">
      <div className="card-content">
        <div className="skeleton-method"></div>
        <div className="skeleton-title"></div>
        <div className="skeleton-path"></div>
      </div>
      <span className="arrow">{arrow}</span>
    </div>
  );

  // Fully rendered content when the spec loads successfully.
  const renderedContent = endpointData && (
    <a href={readmeDocUrl} style={{ textDecoration: 'none', color: 'inherit' }}>
      <Card>
        <div style={{ flexGrow: 1 }}>
          {show.summary && endpointData.method && <MethodTag method={endpointData.method} />}
          <h3 style={{ margin: '0 0 0.5em', fontSize: '1rem', fontWeight: 'bold' }}>
            {linkText ? linkText : (endpointData.summary || displayLinkText)}
          </h3>
          {show.description && endpointData.description && (
            <p style={{ margin: '0 0 0.5em', fontSize: '0.9rem', color: '#374151' }}>
              {endpointData.description}
            </p>
          )}
          <div style={{ marginBottom: '0.5em' }}>
            {show.parameters && endpointData.parameters?.length > 0 && (
              <div style={{ marginBottom: '0.25em' }}>
                <strong>Parameters:</strong> {endpointData.parameters.map(p => p.name).join(', ')}
              </div>
            )}
            {show.requestBody && endpointData.requestBody && (
              <div style={{ marginBottom: '0.25em' }}>
                <strong>Request Body:</strong> Available
              </div>
            )}
            {show.responses && endpointData.responses && (
              <div style={{ marginBottom: '0.25em' }}>
                <strong>Responses:</strong> {Object.keys(endpointData.responses).join(', ')}
              </div>
            )}
            {show.security && endpointData.security?.length > 0 && (
              <div style={{ marginBottom: '0.25em' }}>
                <strong>Security:</strong> {JSON.stringify(endpointData.security)}
              </div>
            )}
            {show.tags && endpointData.tags?.length > 0 && (
              <div style={{ marginBottom: '0.25em' }}>
                <strong>Tags:</strong> {endpointData.tags.join(', ')}
              </div>
            )}
          </div>
          {show.example && endpointData.example && (
            <pre style={{
              background: '#f3f4f6',
              padding: '8px',
              borderRadius: '6px',
              fontSize: '0.85rem',
              overflowX: 'auto',
              marginBottom: '0.5em'
            }}>
              {JSON.stringify(endpointData.example, null, 2)}
            </pre>
          )}
          {show.apiUrl && apiFullUrl && (
            <div style={{
              fontFamily: 'monospace',
              fontSize: '0.9rem',
              color: '#4b5563',
              marginBottom: '0.25em'
            }}>
              API: {endpointData.method} {apiFullUrl}
            </div>
          )}
          {show.readmeUrl !== false && (
            <div className="path" style={{
              fontFamily: 'monospace',
              fontSize: '0.9rem',
              color: '#6b7280'
            }}>
              {readmeDocUrl}
            </div>
          )}
        </div>
      </Card>
    </a>
  );

  if (displayMode === 'popup') {
    const popupLinkStyle = {
      textDecoration: 'none',
      cursor: 'pointer',
      color: '#1a73e8',
      backgroundColor: '#e8f0fe',
      borderRadius: '4px',
      padding: '2px 4px',
      fontWeight: '600',
    };

    return (
      <span
        style={{ display: 'inline-block', position: 'relative' }}
        onMouseEnter={() => setIsHovered(true)}
        onMouseLeave={() => setIsHovered(false)}
      >
        <a href={readmeDocUrl} style={popupLinkStyle}>
          {displayLinkText}
        </a>
        {isHovered && (
          <div style={{ position: 'absolute', left: '0', zIndex: 1000, width: '420px' }}>
            {loading && skeletonCard}
            {!loading &&
              (!endpointData ? fallbackCard : (
                <a href={readmeDocUrl} style={{ textDecoration: 'none', color: 'inherit' }}>
                  <Card classNameExtra="popup-card">
                    <div style={{ flexGrow: 1 }}>
                      {show.summary && endpointData.method && <MethodTag method={endpointData.method} />}
                      <h3 style={{ margin: '0 0 0.5em', fontSize: '1rem', fontWeight: 'bold' }}>
                        {linkText ? linkText : (endpointData.summary || displayLinkText)}
                      </h3>
                      {show.description && endpointData.description && (
                        <p style={{ margin: '0 0 0.5em', fontSize: '0.9rem', color: '#374151' }}>
                          {endpointData.description}
                        </p>
                      )}
                      <div style={{ marginBottom: '0.5em' }}>
                        {show.parameters && endpointData.parameters?.length > 0 && (
                          <div style={{ marginBottom: '0.25em' }}>
                            <strong>Parameters:</strong> {endpointData.parameters.map(p => p.name).join(', ')}
                          </div>
                        )}
                        {show.requestBody && endpointData.requestBody && (
                          <div style={{ marginBottom: '0.25em' }}>
                            <strong>Request Body:</strong> Available
                          </div>
                        )}
                        {show.responses && endpointData.responses && (
                          <div style={{ marginBottom: '0.25em' }}>
                            <strong>Responses:</strong> {Object.keys(endpointData.responses).join(', ')}
                          </div>
                        )}
                        {show.security && endpointData.security?.length > 0 && (
                          <div style={{ marginBottom: '0.25em' }}>
                            <strong>Security:</strong> {JSON.stringify(endpointData.security)}
                          </div>
                        )}
                        {show.tags && endpointData.tags?.length > 0 && (
                          <div style={{ marginBottom: '0.25em' }}>
                            <strong>Tags:</strong> {endpointData.tags.join(', ')}
                          </div>
                        )}
                      </div>
                      {show.example && endpointData.example && (
                        <pre style={{
                          background: '#f3f4f6',
                          padding: '8px',
                          borderRadius: '6px',
                          fontSize: '0.85rem',
                          overflowX: 'auto',
                          marginBottom: '0.5em'
                        }}>
                          {JSON.stringify(endpointData.example, null, 2)}
                        </pre>
                      )}
                      {show.apiUrl && apiFullUrl && (
                        <div style={{
                          fontFamily: 'monospace',
                          fontSize: '0.9rem',
                          color: '#4b5563',
                          marginBottom: '0.25em'
                        }}>
                          API: {endpointData.method} {apiFullUrl}
                        </div>
                      )}
                      {show.readmeUrl !== false && (
                        <div className="path" style={{
                          fontFamily: 'monospace',
                          fontSize: '0.9rem',
                          color: '#6b7280'
                        }}>
                          {readmeDocUrl}
                        </div>
                      )}
                    </div>
                  </Card>
                </a>
              ))
            }
          </div>
        )}
      </span>
    );
  }

  if (loading) return skeletonCard;
  if (!endpointData) return fallbackCard;
  return renderedContent;
};

# Basic Usage

## 💬 Popup Mode

#### Example with minimal props: 
<EndpointEmbed endpoint="/pet" method="PUT" specFile="ErdSpec1.json" linkText="Pet Endpoint" displayMode="popup" /> - This uses the "popup" display mode. When users hover over it, they’ll see specific default info.

#### Example with all props enabled: 

<EndpointEmbed
  endpoint="/pet"
  method="PUT"
  specFile="ErdSpec1.json"
  linkText="Showing all props with a custom label"
  displayMode="popup"
  websiteBaseUrl="https://erdtest123.readme.io"
  apiTargetUrl="https://api.erdman.com"
  show={{
    summary: true,
    description: true,
    parameters: true,
    tags: true,
    requestBody: true,
    responses: true,
    security: true,
    example: false,
    readmeUrl: false,
    apiUrl: false
  }}
/> - This version includes all available options—description, parameters, request body, responses, tags, security, and more. It gives users the most complete view of the endpoint details. 

All props with descriptions and default values where applicable:

| **Prop**           | **Type**      | **Default**                          | **Description**                                                                 |
|--------------------|---------------|--------------------------------------|---------------------------------------------------------------------------------|
| `websiteBaseUrl`   | `string`      | `'https://yourdocsite.com'`     | The ReadMe project base URL (used for linking to reference pages).             |
| `apiTargetUrl`     | `string`      | _none_                               | Optional override for the real API server base URL (for displaying API URL).   |
| `endpoint`         | `string`      | _required_                           | The API path you want to display (e.g. `/pet/{petId}`).                         |
| `method`           | `string`      | _optional_                           | The HTTP method (e.g. `GET`, `POST`). If omitted, auto-selects best match.     |
| `displayMode`      | `string`      | `'popup'`                            | Determines how the component renders: `'popup'` or `'inline'`.                 |
| `linkText`         | `string`      | _fallbacks to `endpoint`_            | Custom text to display instead of the raw endpoint path.                       |
| `specFile`         | `string`      | _required_                           | The OpenAPI JSON file name to load (e.g. `ErdSpec1.json`).                     |
| `show.summary`     | `boolean`     | `true`                               | Toggles display of the summary/method line.                                    |
| `show.description` | `boolean`     | `true`                               | Toggles display of the endpoint description.                                   |
| `show.parameters`  | `boolean`     | `true`                               | Toggles display of parameter names.                                            |
| `show.tags`        | `boolean`     | `true`                               | Toggles display of OpenAPI tags.                                               |
| `show.requestBody` | `boolean`     | `false`                              | Toggles request body presence indicator.                                       |
| `show.responses`   | `boolean`     | `false`                              | Toggles list of response codes.                                                |
| `show.security`    | `boolean`     | `false`                              | Toggles display of security scheme info.                                       |
| `show.example`     | `boolean`     | `false`                              | Toggles display of a sample JSON response.                                     |
| `show.readmeUrl`   | `boolean`     | `false`                              | Toggles display of the ReadMe reference doc URL.                               |
| `show.apiUrl`      | `boolean`     | `false`                              | Toggles display of the full API call URL (with method and `apiTargetUrl`).     |
