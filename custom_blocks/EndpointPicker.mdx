import React, { useEffect, useMemo, useState } from "react";

export const EndpointPicker = props => {
  const { websiteBaseUrl } = props;

  const base = useMemo(() => {
    if (websiteBaseUrl) return websiteBaseUrl.replace(/\/$/, "");
    if (typeof window !== "undefined") return window.location.protocol + "//" + window.location.host;
    return "";
  }, [websiteBaseUrl]);

  const denySlugs = ["health"];

  const [specs, setSpecs] = useState([]);
  const [specUrl, setSpecUrl] = useState("");
  const [specData, setSpecData] = useState(null);
  const [loadingIndex, setLoadingIndex] = useState(true);
  const [endpoints, setEndpoints] = useState([]);
  const [loadingSpec, setLoadingSpec] = useState(false);
  const [selectedPath, setSelectedPath] = useState("");
  const [selectedMethod, setSelectedMethod] = useState("");
  const [search, setSearch] = useState("");
  const [showFullSpec, setShowFullSpec] = useState(false);
  const [fullSpecSearch, setFullSpecSearch] = useState("");
  const [fullSpecTag, setFullSpecTag] = useState("All");
  const [fullSpecSelection, setFullSpecSelection] = useState(null);
  const [copied, setCopied] = useState("");

  const mono = "ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, monospace";

  const slugForReference = (path, opId) => opId || String(path || "")
    .replace(/[{}]/g, "")
    .replace(/\//g, "-")
    .replace(/^-+|-+$/g, "")
    || "endpoint";

  const methodOrder = ["GET", "POST", "PUT", "PATCH", "DELETE", "HEAD", "OPTIONS", "TRACE"];

  const methodClass = method => `method-${String(method || "").toLowerCase()}`;

  const resolveRef = (value, rootSpec) => {
    if (!value || typeof value !== "object" || !value.$ref) return value;
    if (!value.$ref.startsWith("#/")) return value;

    return value.$ref
      .slice(2)
      .split("/")
      .reduce((current, key) => current?.[key.replace(/~1/g, "/").replace(/~0/g, "~")], rootSpec);
  };

  const schemaLabel = (schema, rootSpec) => {
    const resolved = resolveRef(schema, rootSpec);
    if (!resolved) return "any";

    if (resolved.type === "array") {
      return `array<${schemaLabel(resolved.items, rootSpec)}>`;
    }

    if (Array.isArray(resolved.type)) {
      return resolved.type.join(" | ");
    }

    if (resolved.enum?.length) {
      return resolved.type ? `${resolved.type} enum` : "enum";
    }

    if (resolved.type) return resolved.type;
    if (resolved.oneOf) return "oneOf";
    if (resolved.anyOf) return "anyOf";
    if (resolved.allOf) return "allOf";
    if (resolved.properties) return "object";

    return "any";
  };

  const stringifyExample = value => {
    if (value === undefined || value === null) return "";
    if (typeof value === "string") return value;

    try {
      return JSON.stringify(value, null, 2);
    } catch {
      return String(value);
    }
  };

  const getSchemaExample = (schema, rootSpec, depth = 0) => {
    if (!schema || depth > 5) return undefined;

    const resolved = resolveRef(schema, rootSpec);
    if (!resolved) return undefined;

    if (resolved.example !== undefined) return resolved.example;
    if (resolved.default !== undefined) return resolved.default;
    if (resolved.enum?.length) return resolved.enum[0];

    if (resolved.oneOf?.length) return getSchemaExample(resolved.oneOf[0], rootSpec, depth + 1);
    if (resolved.anyOf?.length) return getSchemaExample(resolved.anyOf[0], rootSpec, depth + 1);

    if (resolved.allOf?.length) {
      return resolved.allOf.reduce((combined, part) => {
        const example = getSchemaExample(part, rootSpec, depth + 1);
        return example && typeof example === "object" && !Array.isArray(example)
          ? { ...combined, ...example }
          : combined;
      }, {});
    }

    if (resolved.type === "array") {
      const item = getSchemaExample(resolved.items, rootSpec, depth + 1);
      return item === undefined ? [] : [item];
    }

    if (resolved.type === "object" || resolved.properties) {
      return Object.entries(resolved.properties || {}).reduce((output, [key, propertySchema]) => {
        const example = getSchemaExample(propertySchema, rootSpec, depth + 1);
        output[key] = example === undefined ? null : example;
        return output;
      }, {});
    }

    if (resolved.type === "string") {
      if (resolved.format === "date-time") return "2026-01-01T00:00:00Z";
      if (resolved.format === "date") return "2026-01-01";
      if (resolved.format === "email") return "user@example.com";
      if (resolved.format === "uuid") return "00000000-0000-0000-0000-000000000000";
      return "string";
    }

    if (resolved.type === "integer" || resolved.type === "number") return 0;
    if (resolved.type === "boolean") return true;

    return undefined;
  };

  async function fetchSpecIndex(indexUrl) {
    const res = await fetch(indexUrl, { headers: { Accept: "text/html,*/*" } });
    if (!res.ok) throw new Error(`Unable to load OpenAPI index (${res.status})`);

    const html = await res.text();
    const doc = new DOMParser().parseFromString(html, "text/html");

    return Array.from(doc.querySelectorAll('.container .item a[href^="/openapi/"]'))
      .map(a => ({
        name: (a.textContent || "").trim(),
        href: a.getAttribute("href")
      }));
  }

  async function fetchSpecJson(url) {
    const res = await fetch(url, { headers: { Accept: "application/json" } });
    if (!res.ok) throw new Error(`Unable to load OpenAPI file (${res.status})`);
    return res.json();
  }

  const listPathsAndMethods = spec => {
    const out = [];

    Object.entries(spec?.paths || {}).forEach(([path, pathItem]) => {
      const resolvedPathItem = resolveRef(pathItem, spec) || {};
      const pathParameters = resolvedPathItem.parameters || [];

      Object.entries(resolvedPathItem).forEach(([method, operation]) => {
        if (!["get", "post", "put", "delete", "patch", "head", "options", "trace"].includes(method)) return;

        const resolvedOperation = resolveRef(operation, spec) || {};
        const slug = slugForReference(path, resolvedOperation.operationId || "");
        if (denySlugs.includes(slug.toLowerCase())) return;

        const tags = resolvedOperation.tags?.length ? resolvedOperation.tags : ["General"];

        out.push({
          path,
          method: method.toUpperCase(),
          summary: resolvedOperation.summary || "",
          description: resolvedOperation.description || "",
          slug,
          operationId: resolvedOperation.operationId || "",
          tags,
          operation: resolvedOperation,
          pathParameters
        });
      });
    });

    return out.sort((a, b) =>
      a.path.localeCompare(b.path) ||
      methodOrder.indexOf(a.method) - methodOrder.indexOf(b.method)
    );
  };

  const copyText = async (text, key) => {
    if (!text) return;

    try {
      await navigator.clipboard.writeText(text);
      setCopied(key);
      window.setTimeout(() => setCopied(""), 1600);
    } catch {
      const textarea = document.createElement("textarea");
      textarea.value = text;
      textarea.style.position = "fixed";
      textarea.style.opacity = "0";
      document.body.appendChild(textarea);
      textarea.select();
      document.execCommand("copy");
      textarea.remove();
      setCopied(key);
      window.setTimeout(() => setCopied(""), 1600);
    }

    let cancelled = false;

    (async () => {
      setLoadingIndex(true);

      try {
        const list = await fetchSpecIndex(base + "/openapi");
        if (cancelled) return;

        setSpecs(list);
        if (list?.[0]) setSpecUrl(base + list[0].href);
      } catch {
        if (!cancelled) setSpecs([]);
      } finally {
        if (!cancelled) setLoadingIndex(false);
      }
    })();

    return () => {
      cancelled = true;
    };

    if (!specUrl) return;

    (async () => {
      setLoadingSpec(true);

      try {
        const spec = await fetchSpecJson(specUrl);
        const list = listPathsAndMethods(spec);
        if (cancelled) return;

        setSpecData(spec);
        setEndpoints(list);

        if (list[0]) {
          setSelectedPath(list[0].path);
          setSelectedMethod(list[0].method);
          setFullSpecSelection(list[0]);
        } else {
          setSelectedPath("");
          setSelectedMethod("");
          setFullSpecSelection(null);
        }
      } catch {
        if (cancelled) return;
        setSpecData(null);
        setEndpoints([]);
        setSelectedPath("");
        setSelectedMethod("");
        setFullSpecSelection(null);
      } finally {
        if (!cancelled) setLoadingSpec(false);
      }
    })();

    return () => {
      cancelled = true;
    };

    if (!showFullSpec || typeof document === "undefined") return;

    const originalOverflow = document.body.style.overflow;
    document.body.style.overflow = "hidden";

    const onKeyDown = event => {
      if (event.key === "Escape") setShowFullSpec(false);
    };

    window.addEventListener("keydown", onKeyDown);

    return () => {
      document.body.style.overflow = originalOverflow;
      window.removeEventListener("keydown", onKeyDown);
    };

    const q = search.trim().toLowerCase();

    return endpoints.filter(endpoint =>
      endpoint.path.toLowerCase().includes(q) ||
      endpoint.method.toLowerCase().includes(q) ||
      endpoint.summary.toLowerCase().includes(q)
    );
  }, [endpoints, search]);

  useEffect(() => {
    if (!filtered.length) return;

    const selectedStillExists = filtered.some(endpoint =>
      endpoint.path === selectedPath && endpoint.method === selectedMethod
    );

    if (!selectedStillExists) {
      setSelectedPath(filtered[0].path);
      setSelectedMethod(filtered[0].method);
    }
  }, [filtered, selectedPath, selectedMethod]);

  useEffect(() => {
    const methodsForPath = filtered.filter(endpoint => endpoint.path === selectedPath);
    if (!methodsForPath.length) return;

    if (!methodsForPath.some(endpoint => endpoint.method === selectedMethod)) {
      setSelectedMethod(methodsForPath[0].method);
    }
  }, [selectedPath, selectedMethod, filtered]);

  const current = filtered.find(endpoint =>
    endpoint.path === selectedPath && endpoint.method === selectedMethod
  ) || filtered[0];

  const referenceUrl = current ? `${base}/reference/${current.slug}` : "#";

  const tags = useMemo(() => {
    return ["All", ...Array.from(new Set(endpoints.flatMap(endpoint => endpoint.tags))).sort()];
  }, [endpoints]);

    const q = fullSpecSearch.trim().toLowerCase();

    return endpoints.filter(endpoint => {
      const matchesTag = fullSpecTag === "All" || endpoint.tags.includes(fullSpecTag);
      const matchesSearch =
        !q ||
        endpoint.path.toLowerCase().includes(q) ||
        endpoint.method.toLowerCase().includes(q) ||
        endpoint.summary.toLowerCase().includes(q) ||
        endpoint.tags.some(tag => tag.toLowerCase().includes(q));

      return matchesTag && matchesSearch;
    });
  }, [endpoints, fullSpecSearch, fullSpecTag]);

    return fullSpecFiltered.reduce((groups, endpoint) => {
      const groupName = fullSpecTag === "All"
        ? endpoint.tags[0] || "General"
        : fullSpecTag;

      if (!groups[groupName]) groups[groupName] = [];
      groups[groupName].push(endpoint);
      return groups;
    }, {});
  }, [fullSpecFiltered, fullSpecTag]);

  useEffect(() => {
    if (!showFullSpec || !fullSpecFiltered.length) return;

    const currentStillVisible = fullSpecSelection && fullSpecFiltered.some(endpoint =>
      endpoint.path === fullSpecSelection.path &&
      endpoint.method === fullSpecSelection.method
    );

    if (!currentStillVisible) {
      setFullSpecSelection(fullSpecFiltered[0]);
    }
  }, [showFullSpec, fullSpecFiltered, fullSpecSelection]);

  const selectedFullSpecEndpoint = fullSpecSelection || fullSpecFiltered[0] || endpoints[0];

  const selectedOperation = selectedFullSpecEndpoint?.operation || {};
  const selectedReferenceUrl = selectedFullSpecEndpoint

    : "#";

    if (!selectedFullSpecEndpoint || !specData) return [];

    const combined = [
      ...(selectedFullSpecEndpoint.pathParameters || []),
      ...(selectedOperation.parameters || [])
    ];

    const unique = new Map();

    combined.forEach(parameter => {
      const resolved = resolveRef(parameter, specData);
      if (!resolved) return;
      unique.set(`${resolved.in}:${resolved.name}`, resolved);
    });

    return Array.from(unique.values());
  }, [selectedFullSpecEndpoint, selectedOperation, specData]);

    if (!selectedOperation.requestBody || !specData) return null;
    return resolveRef(selectedOperation.requestBody, specData);
  }, [selectedOperation, specData]);

    if (!specData) return [];

    return Object.entries(selectedOperation.responses || {}).map(([status, response]) => ({
      status,
      response: resolveRef(response, specData) || {}
    }));
  }, [selectedOperation, specData]);

    specData?.info?.title ||
    specs.find(spec => base + spec.href === specUrl)?.name ||
    "OpenAPI Specification";

  const specVersion = specData?.info?.version || "";
  const pathCount = Object.keys(specData?.paths || {}).length;
  const tagCount = new Set(endpoints.flatMap(endpoint => endpoint.tags)).size;

                          Operation ID: {selectedFullSpecEndpoint.operationId}
                        </span>
                      ) : null}

                      <span className="metadata-pill">
                        {selectedParameters.length} parameter{selectedParameters.length === 1 ? "" : "s"}
                      </span>

                      <span className="metadata-pill">
                        {responseEntries.length} response{responseEntries.length === 1 ? "" : "s"}
                      </span>

                      {selectedFullSpecEndpoint.tags.map(tag => (
                        <span className="metadata-pill" key={tag}>{tag}</span>
                      ))}
                    </div>

                    <section className="detail-section">
                      <h3 className="detail-section-title">Parameters</h3>

                      {selectedParameters.length ? (
                        <table className="detail-table">
                          <thead>
                            <tr>
                              <th style={{ width: "24%" }}>Name</th>
                              <th style={{ width: "18%" }}>Type</th>
                              <th>Description</th>
                            </tr>
                          </thead>
                          <tbody>
                            {selectedParameters.map(parameter => (
                              <tr key={`${parameter.in}:${parameter.name}`}>
                                <td>
                                  <div className="parameter-name">{parameter.name}</div>
                                  <div className="parameter-meta">
                                    {parameter.in}
                                    {parameter.required ? " · required" : ""}
                                  </div>
                                </td>
                                <td>{schemaLabel(parameter.schema, specData)}</td>
                                <td>{parameter.description || "No description provided."}</td>
                              </tr>
                            ))}
                          </tbody>
                        </table>
                      ) : (
                        <div className="empty-detail">This endpoint has no parameters.</div>
                      )}
                    </section>

                    <section className="detail-section">
                      <h3 className="detail-section-title">Request body</h3>

                      {requestBody?.content ? (
                        <div className="request-grid">
                          {Object.entries(requestBody.content).map(([contentType, media]) => {
                            const example =
                              media.example !== undefined
                                ? media.example
                                : getSchemaExample(media.schema, specData);

                            return (
                              <div className="request-card" key={contentType}>
                                <div className="request-card-header">
                                  <span>{contentType}</span>
                                  <span>
                                    {schemaLabel(media.schema, specData)}
                                    {requestBody.required ? " · required" : ""}
                                  </span>
                                </div>

                                <div className="request-card-body">
                                  {example !== undefined ? (
                                    <div className="code">{stringifyExample(example)}</div>
                                  ) : (
                                    requestBody.description || "No request example provided."
                                  )}
                                </div>
                              </div>
                            );
                          })}
                        </div>
                      ) : (
                        <div className="empty-detail">This endpoint has no request body.</div>
                      )}
                    </section>

                    <section className="detail-section">
                      <h3 className="detail-section-title">Responses</h3>

                      {responseEntries.length ? (
                        <div className="response-list">
                          {responseEntries.map(({ status, response }) => {
                            const responseContent = Object.entries(response.content || {});

                            return (
                              <div className="response-card" key={status}>
                                <div className="response-card-header">
                                  <span className="response-status">{status}</span>
                                  <span>{response.description || "Response"}</span>
                                </div>

                                <div className="response-card-body">
                                  {responseContent.length ? (
                                    responseContent.map(([contentType, media]) => {
                                      const example =
                                        media.example !== undefined
                                          ? media.example
                                          : getSchemaExample(media.schema, specData);

                                      return (
                                        <div key={contentType} style={{ marginBottom: 12 }}>
                                          <div style={{ fontWeight: 700, marginBottom: 7 }}>
                                            {contentType} · {schemaLabel(media.schema, specData)}
                                          </div>

                                          {example !== undefined ? (
                                            <div className="code">{stringifyExample(example)}</div>
                                          ) : (
                                            <div>No response example provided.</div>
                                          )}
                                        </div>
                                      );
                                    })
                                  ) : (
                                    <div>No response body documented.</div>
                                  )}
                                </div>
                              </div>
                            );
                          })}
                        </div>
                      ) : (
                        <div className="empty-detail">No responses are documented for this endpoint.</div>
                      )}
                    </section>
                  </div>
                ) : (
                  <div className="empty-detail" style={{ marginTop: 28 }}>
                    Select an endpoint from the sidebar to view its details.
                  </div>
                )}
              </main>
            </div>

            <div className="full-spec-footer">
              <button
                className="button-secondary"
                type="button"
                onClick={() => copyText(specUrl, "footer-spec-url")}
              >
                {copied === "footer-spec-url" ? "Copied!" : "Copy spec URL"}
              </button>

              <button
                className="button-link"
                type="button"
                onClick={() => setShowFullSpec(false)}
              >
                Close
              </button>
            </div>
          </div>
        </div>
      ) : null}
    </div>
  );
};
