import React, { useState, useEffect } from "react";

export default function PageSearchHighlight() {
  const [searchTerm, setSearchTerm] = useState("");
  const [matchCount, setMatchCount] = useState(0);
  const [currentMatchIndex, setCurrentMatchIndex] = useState(0);
  const [keywords, setKeywords] = useState([]);
  const [keywordCounts, setKeywordCounts] = useState({});

  useEffect(() => {
    console.log("🔍 Initializing Search Component...");

    // Extract keywords from meta tag and store in state
    const metaTag = document.querySelector('meta[property="keywords"]');
    if (metaTag) {
      const extracted = metaTag.content.split(",").map((kw) => kw.trim());
      setKeywords(extracted);
      console.log("📌 Extracted Keywords:", extracted);
    }

    // Store original text for paragraphs
    document.querySelectorAll(".content-body p").forEach((p) => {
      if (!p.dataset.originalText) {
        p.dataset.originalText = p.innerHTML;
      }
    });
  }, []);

  useEffect(() => {
    if (!searchTerm.trim()) {
      restoreOriginalText();
      setMatchCount(0);
      return;
    }

    let matchCounter = 0;
    let matches = [];
    const regex = new RegExp(`(${searchTerm})`, "gi");

    document.querySelectorAll(".content-body p").forEach((p) => {
      if (p.dataset.originalText) {
        const modifiedText = p.dataset.originalText.replace(regex, (match) => {
          matches.push({ element: p });
          matchCounter++;
          return `<span style="background-color: #fef3c7; font-weight: 600; border:1px solid white; border-radius: 4px;">${match}</span>`;
        });

        p.innerHTML = modifiedText;
      }
    });

    setMatchCount(matchCounter);
    setCurrentMatchIndex(0);
    console.log(`✅ Total matches found: ${matchCounter}`);

    // Scroll to the first match
    if (matchCounter > 0) {
      setTimeout(() => scrollToMatch(0), 300);
    }
  }, [searchTerm]);

  useEffect(() => {
    if (keywords.length === 0) return;

    const counts = {};
    document.querySelectorAll(".content-body p").forEach((p) => {
      const text = p.innerText;
      keywords.forEach((keyword) => {
        const regex = new RegExp(`\\b${keyword}\\b`, "gi");
        const matches = text.match(regex);
        if (matches) {
          counts[keyword] = (counts[keyword] || 0) + matches.length;
        }
      });
    });

    setKeywordCounts(counts);
    console.log("📊 Keyword Counts:", counts);
  }, [keywords]);

  const restoreOriginalText = () => {
    document.querySelectorAll(".content-body p").forEach((p) => {
      if (p.dataset.originalText) {
        p.innerHTML = p.dataset.originalText;
      }
    });
  };

  const scrollToMatch = (index) => {
    const matches = document.querySelectorAll("span[style*='background-color: #fef3c7']");
    if (matches.length > 0 && matches[index]) {
      matches[index].scrollIntoView({ behavior: "smooth", block: "center" });
      console.log(`📍 Scrolled to match ${index + 1}`);
    }
  };

  const goToNextMatch = () => {
    const matches = document.querySelectorAll("span[style*='background-color: #fef3c7']");
    if (matches.length === 0) return;

    let newIndex = currentMatchIndex + 1;
    if (newIndex >= matches.length) newIndex = 0;
    setCurrentMatchIndex(newIndex);
    scrollToMatch(newIndex);
  };

  const goToPrevMatch = () => {
    const matches = document.querySelectorAll("span[style*='background-color: #fef3c7']");
    if (matches.length === 0) return;

    let newIndex = currentMatchIndex - 1;
    if (newIndex < 0) newIndex = matches.length - 1;
    setCurrentMatchIndex(newIndex);
    scrollToMatch(newIndex);
  };

  const handleKeywordClick = (keyword) => {
    setSearchTerm(keyword);
  };

  return (
    <div style={{ width: "100%", display: "flex", flexDirection: "column", gap: "10px" }}>
      {/* Search Input */}
      <div style={{ display: "flex", alignItems: "center", gap: "10px", width: "100%" }}>
        <div style={{ flex: "1 1 0%", maxWidth: "500px" }}>
          <input
            type="text"
            placeholder="Search page content..."
            style={{
              width: "100%",
              padding: "8px",
              fontSize: "16px",
              border: "1px solid #ccc",
              borderRadius: "4px",
            }}
            value={searchTerm}
            onChange={(e) => setSearchTerm(e.target.value)}
          />
        </div>
        <span style={{ fontSize: "14px", color: "#555", whiteSpace: "nowrap" }}>
          {matchCount} {matchCount === 1 ? "result" : "results"}
        </span>
        {matchCount > 1 && (
          <div style={{ display: "flex", gap: "5px" }}>
            <button
              onClick={goToPrevMatch}
              style={{
                padding: "3px 6px",
                backgroundColor: "#888",
                color: "white",
                borderRadius: "3px",
                cursor: "pointer",
                fontSize: "12px",
                opacity: "0.8",
              }}
            >
              ⬅ Prev
            </button>
            <button
              onClick={goToNextMatch}
              style={{
                padding: "3px 6px",
                backgroundColor: "#888",
                color: "white",
                borderRadius: "3px",
                cursor: "pointer",
                fontSize: "12px",
                opacity: "0.8",
              }}
            >
              Next ➡
            </button>
          </div>
        )}
      </div>

      {/* Keyword Buttons */}
      <div style={{ display: "flex", flexWrap: "wrap", gap: "5px" }}>
        {keywords.map((keyword) => (
          <button
            key={keyword}
            onClick={() => handleKeywordClick(keyword)}
            style={{
              backgroundColor: "#f0f0f0",
              padding: "5px 10px",
              borderRadius: "15px",
              border: "1px solid #ccc",
              cursor: "pointer",
              fontSize: "14px",
            }}
          >
            {keyword} ({keywordCounts[keyword] || 0})
          </button>
        ))}
      </div>
    </div>
  );
}

<PageSearchHighlight />
