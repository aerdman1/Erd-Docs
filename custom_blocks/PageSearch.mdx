import React, { useState, useEffect } from "react";

export default function PageSearchHighlight() {
  const [searchTerm, setSearchTerm] = useState("");

  useEffect(() => {
    const contentElement = document.querySelector(".content-body");
    if (!contentElement) {
      console.warn("❌ Could not find .content-body!");
      return;
    }

    // Store original content once (if not already stored)
    if (!contentElement.dataset.originalText) {
      contentElement.dataset.originalText = contentElement.innerHTML;
    }

    // Function to restore and highlight text
    function applyHighlight() {
      if (!searchTerm || searchTerm.length < 3) {
        contentElement.innerHTML = contentElement.dataset.originalText;
        return;
      }

      const safeTerm = searchTerm.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
      const regex = new RegExp(`(${safeTerm})`, "gi");

      contentElement.innerHTML = contentElement.dataset.originalText.replace(
        regex,
        `<span style="background-color: yellow; font-weight: bold;">$1</span>`
      );
    }

    // Apply highlight on search term change
    applyHighlight();

    // **Detect Page Rehydration** and reapply highlights
    const observer = new MutationObserver(() => {
      console.log("🔄 Page content updated, reapplying highlights...");
      applyHighlight();
    });

    observer.observe(contentElement, { childList: true, subtree: true });

    return () => observer.disconnect(); // Cleanup observer on unmount
  }, [searchTerm]);

  return (
    <div style={{ maxWidth: "600px", margin: "auto", fontFamily: "Arial, sans-serif" }}>
      <input
        type="text"
        placeholder="Search page content..."
        value={searchTerm}
        onChange={(e) => setSearchTerm(e.target.value)}
        style={{
          width: "100%",
          padding: "10px",
          fontSize: "16px",
          border: "1px solid #ccc",
          borderRadius: "5px",
          marginBottom: "10px",
        }}
      />
    </div>
  );
}

<PageSearchHighlight />
