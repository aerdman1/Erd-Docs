import { useState, useEffect } from 'react';

export const PageSearching = () => {
  const [searchTerm, setSearchTerm] = useState('');
  const [highlightedElements, setHighlightedElements] = useState([]);

  // Clears existing highlight <span> wrappers
  const clearHighlights = () => {
    highlightedElements.forEach(el => {
      el.replaceWith(...el.childNodes);
    });
    setHighlightedElements([]);
  };

  useEffect(() => {
    // If below 2 characters, just clear any old highlights and exit
    if (searchTerm.length < 2) {
      clearHighlights();
      return;
    }

    // Clear old highlights first
    clearHighlights();

    const searchAndHighlight = () => {
      const contentArea = document.querySelector('.content-body.grid-75');
      if (!contentArea) {
        console.warn('No .content-body.grid-75 found in DOM.');
        return;
      }

      // Escape special regex chars in searchTerm, then create "global + case-insensitive" regex
      const safeTerm = searchTerm.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
      const regex = new RegExp(`(${safeTerm})`, 'gi');
      const elementsToHighlight = [];

      // Recursively traverse text nodes and highlight matches
      const highlightTextNodes = node => {
        if (node.nodeType === 3) {
          // It's a text node
          if (node.nodeValue.match(regex)) {
            // Create wrapper span
            const wrapper = document.createElement('span');
            // Replace matches with inline styled highlights
            wrapper.innerHTML = node.nodeValue.replace(
              regex,
              `<span style="background-color: yellow; font-weight: bold;">$1</span>`
            );
            // Replace the old text node with our new wrapped node
            const parent = node.parentNode;
            if (parent) {
              parent.replaceChild(wrapper, node);
              // Push each highlighted <span> for future removal
              elementsToHighlight.push(...wrapper.querySelectorAll('span[style]'));
            }
          }
        } else if (
          node.nodeType === 1 &&
          node.childNodes &&
          !['SCRIPT', 'STYLE'].includes(node.nodeName)
        ) {
          // Dive deeper into child nodes
          node.childNodes.forEach(highlightTextNodes);
        }
      };

      highlightTextNodes(contentArea);
      setHighlightedElements(elementsToHighlight);
    };

    // Defer the search slightly to ensure content loads
    const timer = setTimeout(searchAndHighlight, 200);
    return () => clearTimeout(timer);
  }, [searchTerm]);

  const handleClear = () => {
    setSearchTerm(''); // triggers highlight clearing in effect
  };

  return (
    <div style={{ margin: '1rem 0', display: 'flex', alignItems: 'center', gap: '0.5rem' }}>
      <input
        type="text"
        style={{ border: '1px solid #ccc', borderRadius: '4px', padding: '0.5rem', flex: '1' }}
        placeholder="Search (min 2 chars)"
        value={searchTerm}
        onChange={e => setSearchTerm(e.target.value)}
      />
      <button
        style={{ border: '1px solid #ccc', borderRadius: '4px', padding: '0.5rem 1rem' }}
        onClick={handleClear}
      >
        Clear
      </button>
    </div>
  );
};


<PageSearching />
