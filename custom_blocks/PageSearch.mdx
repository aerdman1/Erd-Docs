import React, { useState, useEffect } from "react";

export default function PageSearchHighlight() {
  const [searchTerm, setSearchTerm] = useState("");
  const [keywords, setKeywords] = useState([]);
  const [matchCount, setMatchCount] = useState(0);
  const [currentMatch, setCurrentMatch] = useState(0);

  useEffect(() => {
    const metaKeywords = document.querySelector('meta[property="keywords"]');
    if (metaKeywords) {
      const keywordList = metaKeywords.content.split(",").map((word) => word.trim());
      setKeywords(keywordList);
    }

    const contentElement = document.querySelector(".content-body");
    if (!contentElement) return;

    const paragraphs = contentElement.querySelectorAll("p");

    paragraphs.forEach((p) => {
      if (!p.dataset.originalText) {
        p.dataset.originalText = p.innerHTML;
      }
    });

    function applyHighlight() {
      let matchCounter = 0;
      let highlightedElements = [];

      paragraphs.forEach((p) => {
        if (!searchTerm || searchTerm.length < 3) {
          p.innerHTML = p.dataset.originalText;
          return;
        }

        const safeTerm = searchTerm.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
        const regex = new RegExp(`(${safeTerm})`, "gi");

        p.innerHTML = p.dataset.originalText.replace(regex, (match) => {
          matchCounter++;
          return `<span class="highlighted-match bg-yellow-300 font-bold px-1 rounded">${match}</span>`;
        });

        highlightedElements = [...highlightedElements, ...p.querySelectorAll(".highlighted-match")];
      });

      setMatchCount(matchCounter);
      if (matchCounter > 0) {
        setCurrentMatch(0);
        scrollToMatch(0, highlightedElements);
      }
    }

    applyHighlight();
  }, [searchTerm]);

  function scrollToMatch(index, elements) {
    if (elements.length > 0 && elements[index]) {
      elements[index].scrollIntoView({ behavior: "smooth", block: "center" });
      elements.forEach((el) => el.classList.remove("ring-2", "ring-red-500"));
      elements[index].classList.add("ring-2", "ring-red-500");
    }
  }

  function goToNextMatch() {
    const matches = document.querySelectorAll(".highlighted-match");
    if (matches.length === 0) return;

    let newIndex = currentMatch + 1;
    if (newIndex >= matches.length) newIndex = 0;
    setCurrentMatch(newIndex);
    scrollToMatch(newIndex, matches);
  }

  function goToPrevMatch() {
    const matches = document.querySelectorAll(".highlighted-match");
    if (matches.length === 0) return;

    let newIndex = currentMatch - 1;
    if (newIndex < 0) newIndex = matches.length - 1;
    setCurrentMatch(newIndex);
    scrollToMatch(newIndex, matches);
  }

  return (
    <div className="space-y-3">
      <div className="flex items-center gap-2">
        <input
          type="text"
          placeholder="Search page content..."
          value={searchTerm}
          onChange={(e) => setSearchTerm(e.target.value)}
          className="w-full px-3 py-2 text-base border border-gray-300 rounded-md focus:outline-none focus:ring-2 focus:ring-blue-500"
        />
        {matchCount > 0 && <span className="text-sm text-gray-600">{matchCount} results found</span>}
      </div>

      {matchCount > 0 && (
        <div className="flex gap-2">
          <button onClick={goToPrevMatch} className="px-3 py-2 text-sm font-semibold text-white bg-gray-600 rounded-md hover:bg-gray-700">
            ⬅ Previous
          </button>
          <button onClick={goToNextMatch} className="px-3 py-2 text-sm font-semibold text-white bg-gray-600 rounded-md hover:bg-gray-700">
            Next ➡
          </button>
        </div>
      )}

      {keywords.length > 0 && (
        <div className="flex flex-wrap gap-2">
          {keywords.map((keyword, index) => (
            <button
              key={index}
              onClick={() => setSearchTerm(keyword)}
              className="px-3 py-1 text-sm font-medium bg-gray-100 border border-gray-300 rounded-full hover:bg-gray-200"
            >
              {keyword}
            </button>
          ))}
        </div>
      )}
    </div>
  );
}


<PageSearchHighlight />
