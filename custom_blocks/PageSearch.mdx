import React, { useState, useEffect, useRef } from "react";

export default function PageSearchHighlight() {
  const [searchTerm, setSearchTerm] = useState("");
  const [pageText, setPageText] = useState("");
  const contentRef = useRef(null);

  useEffect(() => {
    if (contentRef.current) {
      const extractedText = contentRef.current.innerText;
      console.log("Extracted Page Text:", extractedText); // 🔍 Debugging
      setPageText(extractedText);
    } else {
      console.log("❌ contentRef is null - No content found!");
    }
  }, []);

  function highlightText(text, search) {
    if (!search || search.length < 3) return text;

    const regex = new RegExp(`(${search})`, "gi");
    const parts = text.split(regex);

    console.log("🔍 Searching for:", search); // Debugging search term
    console.log("📄 Original Text:", text); // Debugging original text

    return parts.map((part, index) =>
      part.toLowerCase() === search.toLowerCase() ? (
        <span key={index} style={{ backgroundColor: "#ffeb3b", fontWeight: "bold", padding: "2px", borderRadius: "4px" }}>
          {part}
        </span>
      ) : (
        part
      )
    );
  }

  return (
    <div style={{ maxWidth: "600px", margin: "auto", fontFamily: "Arial, sans-serif" }}>
      <input
        type="text"
        placeholder="Search page content..."
        value={searchTerm}
        onChange={(e) => {
          console.log("⌨️ User Typed:", e.target.value); // Debugging input
          setSearchTerm(e.target.value);
        }}
        style={{
          width: "100%",
          padding: "10px",
          fontSize: "16px",
          border: "1px solid #ccc",
          borderRadius: "5px",
          marginBottom: "10px",
        }}
      />
      <div ref={contentRef}>
        <p>{highlightText(pageText, searchTerm)}</p>
      </div>
    </div>
  );
}

<PageSearchHighlight />
