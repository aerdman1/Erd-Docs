import React, { useState, useEffect } from "react";

export default function PageSearchHighlight() {
  const [searchTerm, setSearchTerm] = useState("");

  useEffect(() => {
    const contentElement = document.querySelector(".content-body");
    
    if (!contentElement) {
      console.warn("❌ Could not find .content-body!");
      return;
    }
    console.log("✅ Found .content-body");

    // Find all <p> tags inside content-body (avoids affecting the search bar)
    const paragraphs = contentElement.querySelectorAll("p");

    paragraphs.forEach((p) => {
      // Store original paragraph text only once
      if (!p.dataset.originalText) {
        p.dataset.originalText = p.innerHTML;
        console.log("📌 Stored original text for:", p.innerText);
      }
    });

    function applyHighlight() {
      console.log("🔍 Searching for:", searchTerm);

      paragraphs.forEach((p) => {
        if (!searchTerm || searchTerm.length < 3) {
          p.innerHTML = p.dataset.originalText; // Restore original content
          return;
        }

        const safeTerm = searchTerm.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
        const regex = new RegExp(`(${safeTerm})`, "gi");

        p.innerHTML = p.dataset.originalText.replace(
          regex,
          `<span style="background-color: yellow; font-weight: bold;">$1</span>`
        );
      });
    }

    applyHighlight();
  }, [searchTerm]);

  return (
    <input
      type="text"
      placeholder="Search page content..."
      value={searchTerm}
      onChange={(e) => {
        console.log("⌨️ User Typed:", e.target.value);
        setSearchTerm(e.target.value);
      }}
      style={{
        width: "100%",
        padding: "8px",
        fontSize: "16px",
        border: "1px solid #ccc",
        borderRadius: "4px",
        marginBottom: "10px",
      }}
    />
  );
}


<PageSearchHighlight />
