import { useState, useEffect } from 'react';

export const PageSearchComponent = () => {
  const [searchTerm, setSearchTerm] = useState('');
  const [highlightedElements, setHighlightedElements] = useState([]);

  useEffect(() => {
    // Clear previous highlights
    highlightedElements.forEach((el) => {
      el.outerHTML = el.innerText; // Remove spans
    });

    if (searchTerm.trim() === '') return;

    const regex = new RegExp(`(${searchTerm})`, 'gi');
    const elements = [];
    
    // Search through the page body
    const searchInElement = (element) => {
      if (element.nodeType === 3) { // Text node
        const matches = element.nodeValue.match(regex);
        if (matches) {
          const highlightedHTML = element.nodeValue.replace(regex, '<span class="bg-yellow-200 font-bold">$1</span>');
          const wrapper = document.createElement('span');
          wrapper.innerHTML = highlightedHTML;

          const parent = element.parentNode;
          if (parent) {
            parent.replaceChild(wrapper, element);
            elements.push(...wrapper.querySelectorAll('span'));
          }
        }
      } else {
        element.childNodes.forEach(searchInElement);
      }
    };

    document.querySelectorAll('main').forEach(searchInElement);
    setHighlightedElements(elements);
  }, [searchTerm]);

  return (
    <div className="w-full flex justify-start my-4">
      <input
        type="text"
        className="border border-gray-300 rounded-md p-2 w-full max-w-md"
        placeholder="Search this page..."
        value={searchTerm}
        onChange={(e) => setSearchTerm(e.target.value)}
      />
    </div>
  );
};


<PageSearchComponent />
