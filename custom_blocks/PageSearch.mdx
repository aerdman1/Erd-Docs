import React, { useState, useEffect } from "react";

export default function PageSearchHighlight() {
  const [searchTerm, setSearchTerm] = useState("");

  useEffect(() => {
    const contentElement = document.querySelector(".content-body");
    
    if (!contentElement) {
      console.warn("❌ Could not find .content-body!");
      return;
    }
    console.log("✅ Found .content-body");

    // Store the original content only once
    if (!contentElement.dataset.originalText) {
      contentElement.dataset.originalText = contentElement.innerHTML;
      console.log("📌 Stored original text:", contentElement.dataset.originalText);
    }

    function applyHighlight() {
      console.log("🔍 Searching for:", searchTerm);

      if (!searchTerm || searchTerm.length < 3) {
        console.log("ℹ️ Search term too short, restoring original text.");
        contentElement.innerHTML = contentElement.dataset.originalText;
        return;
      }

      const safeTerm = searchTerm.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
      const regex = new RegExp(`(${safeTerm})`, "gi");

      const modifiedText = contentElement.dataset.originalText.replace(
        regex,
        `<span style="background-color: yellow; font-weight: bold;">$1</span>`
      );

      console.log("🖍 Applying modified text:", modifiedText);
      contentElement.innerHTML = modifiedText;
    }

    applyHighlight();
  }, [searchTerm]);

  return (
    <input
      type="text"
      placeholder="Search page content..."
      value={searchTerm}
      onChange={(e) => {
        console.log("⌨️ User Typed:", e.target.value);
        setSearchTerm(e.target.value);
      }}
      style={{
        width: "100%",
        padding: "8px",
        fontSize: "16px",
        border: "1px solid #ccc",
        borderRadius: "4px",
        marginBottom: "10px",
      }}
    />
  );
}

<PageSearchHighlight />
