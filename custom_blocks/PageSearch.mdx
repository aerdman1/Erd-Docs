import { useState, useEffect } from 'react';

export const PageSearching = () => {
  const [searchTerm, setSearchTerm] = useState('');
  const [highlightedElements, setHighlightedElements] = useState([]);

  useEffect(() => {
    // Function to clear existing highlights
    const clearHighlights = () => {
      highlightedElements.forEach((el) => {
        el.replaceWith(...el.childNodes); // Remove highlight <span> but keep text
      });
      setHighlightedElements([]);
    };

    clearHighlights();

    if (!searchTerm.trim()) return;

    const regex = new RegExp(`(${searchTerm.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')})`, 'gi');
    const elementsToHighlight = [];

    // Function to recursively search for text nodes
    const highlightMatches = (node) => {
      if (node.nodeType === 3) { // Text node
        const match = node.nodeValue.match(regex);
        if (match) {
          const wrapper = document.createElement('span');
          wrapper.innerHTML = node.nodeValue.replace(regex, `<span class="bg-yellow-200 font-bold">$1</span>`);

          const parent = node.parentNode;
          parent.replaceChild(wrapper, node);
          elementsToHighlight.push(...wrapper.querySelectorAll('span.bg-yellow-200'));
        }
      } else if (node.nodeType === 1 && node.childNodes && !['SCRIPT', 'STYLE'].includes(node.nodeName)) {
        node.childNodes.forEach(highlightMatches);
      }
    };

    const contentArea = document.querySelector('.content-body.grid-75');
    if (contentArea) {
      highlightMatches(contentArea);
    }

    setHighlightedElements(elementsToHighlight);
  }, [searchTerm]);

  return (
    <div className="w-full flex justify-start my-4">
      <input
        type="text"
        className="border border-gray-300 rounded-md p-2 w-full max-w-md"
        placeholder="Search this page..."
        value={searchTerm}
        onChange={(e) => setSearchTerm(e.target.value)}
      />
    </div>
  );
};


<PageSearching />
