import React, { useState, useEffect } from "react";

export default function PageSearchHighlight() {
  const [searchTerm, setSearchTerm] = useState("");
  const [keywords, setKeywords] = useState([]);
  const [matchCount, setMatchCount] = useState(0);
  const [currentMatch, setCurrentMatch] = useState(0);
  const [keywordCounts, setKeywordCounts] = useState({});

  useEffect(() => {
    console.log("🔍 Initializing Search Component...");

    // Extract keywords from <meta property="keywords">
    const metaKeywords = document.querySelector('meta[property="keywords"]');
    if (metaKeywords) {
      const keywordList = metaKeywords.content.split(",").map((word) => word.trim());
      setKeywords(keywordList);
      countKeywordOccurrences(keywordList);
      console.log("📌 Extracted Keywords:", keywordList);
    }

    const contentElement = document.querySelector(".content-body");
    if (!contentElement) {
      console.warn("❌ Could not find .content-body!");
      return;
    }
    console.log("✅ Found .content-body!");

    // Select only the paragraphs inside .content-body (EXCLUDES search bar & keyword buttons)
    const paragraphs = contentElement.querySelectorAll("p");
    console.log(`📝 Found ${paragraphs.length} paragraphs to scan.`);

    paragraphs.forEach((p, index) => {
      if (!p.dataset.originalText) {
        p.dataset.originalText = p.innerHTML;
        console.log(`📌 Stored original text for paragraph ${index + 1}:`, p.innerText);
      }
    });

    function applyHighlight() {
      console.log("🔍 Searching for:", searchTerm);
      let matchCounter = 0;
      let highlightedElements = [];

      paragraphs.forEach((p, index) => {
        if (!searchTerm || searchTerm.length < 3) {
          p.innerHTML = p.dataset.originalText;
          console.log(`⏪ Restored original text for paragraph ${index + 1}`);
          return;
        }

        const safeTerm = searchTerm.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
        const regex = new RegExp(`(${safeTerm})`, "gi");

        let modifiedText = p.dataset.originalText.replace(regex, (match) => {
          matchCounter++;
          return `<span class="highlighted-match bg-yellow-300 font-bold px-1 rounded">${match}</span>`;
        });

        if (modifiedText !== p.dataset.originalText) {
          console.log(`🎨 Highlighted ${matchCounter} occurrences in paragraph ${index + 1}`);
        }

        p.innerHTML = modifiedText;
        highlightedElements = [...highlightedElements, ...p.querySelectorAll(".highlighted-match")];
      });

      console.log(`✅ Total matches found: ${matchCounter}`);
      setMatchCount(matchCounter);

      if (matchCounter > 0) {
        setCurrentMatch(0);
        scrollToMatch(0, highlightedElements);
      }
    }

    applyHighlight();
  }, [searchTerm]);

  function countKeywordOccurrences(keywordList) {
    const contentElement = document.querySelector(".content-body");
    if (!contentElement) return;

    let counts = {};
    const textContent = contentElement.innerText.toLowerCase();

    keywordList.forEach((keyword) => {
      const regex = new RegExp(`\\b${keyword.toLowerCase()}\\b`, "g");
      const matches = textContent.match(regex);
      counts[keyword] = matches ? matches.length : 0;
    });

    setKeywordCounts(counts);
    console.log("📊 Keyword Counts:", counts);
  }

  function scrollToMatch(index, elements) {
    if (elements.length > 0 && elements[index]) {
      elements[index].scrollIntoView({ behavior: "smooth", block: "center" });
      elements.forEach((el) => el.classList.remove("ring-2", "ring-red-500"));
      elements[index].classList.add("ring-2", "ring-red-500");
      console.log(`📍 Scrolled to match ${index + 1}`);
    }
  }

  function goToNextMatch() {
    const matches = document.querySelectorAll(".highlighted-match");
    if (matches.length === 0) return;

    let newIndex = currentMatch + 1;
    if (newIndex >= matches.length) newIndex = 0;
    setCurrentMatch(newIndex);
    scrollToMatch(newIndex, matches);
  }

  function goToPrevMatch() {
    const matches = document.querySelectorAll(".highlighted-match");
    if (matches.length === 0) return;

    let newIndex = currentMatch - 1;
    if (newIndex < 0) newIndex = matches.length - 1;
    setCurrentMatch(newIndex);
    scrollToMatch(newIndex, matches);
  }

  return (
    <div className="w-full space-y-3">
      {/* Search Input + Buttons */}
      <div className="flex items-center gap-2 w-full">
        <div className="flex-1 max-w-[500px]">
          <input
            type="text"
            placeholder="Search page content..."
            value={searchTerm}
            onChange={(e) => setSearchTerm(e.target.value)}
            className="w-full px-3 py-2 text-base border border-gray-300 rounded-md focus:outline-none focus:ring-2 focus:ring-blue-500"
          />
        </div>
        {matchCount > 0 && (
          <span className="text-sm text-gray-600 whitespace-nowrap">{matchCount} results</span>
        )}
        {matchCount > 0 && (
          <div className="flex gap-2">
            <button
              onClick={goToPrevMatch}
              className="px-3 py-2 text-sm font-semibold text-white bg-gray-600 rounded-md hover:bg-gray-700"
            >
              ⬅ Prev
            </button>
            <button
              onClick={goToNextMatch}
              className="px-3 py-2 text-sm font-semibold text-white bg-gray-600 rounded-md hover:bg-gray-700"
            >
              Next ➡
            </button>
          </div>
        )}
      </div>

      {/* Keyword Bubbles */}
      {keywords.length > 0 && (
        <div className="flex flex-wrap gap-2">
          {keywords.map((keyword, index) => (
            <button
              key={index}
              onClick={() => setSearchTerm(keyword)}
              className="px-3 py-1 text-sm font-medium bg-gray-100 border border-gray-300 rounded-full hover:bg-gray-200"
            >
              {keyword} {keywordCounts[keyword] > 0 ? `(${keywordCounts[keyword]})` : ""}
            </button>
          ))}
        </div>
      )}
    </div>
  );
}


<PageSearchHighlight />
