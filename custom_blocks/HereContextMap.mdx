export const HereContextMap = ({ 
  header = "Context menu demo", 
  apiKey, 
  height = 500 
}) => {
  // Inject HERE SDK CSS/JS only once
  const loadOnce = (tag, attr, url) => {
    return new Promise((resolve, reject) => {
      if ([...document.querySelectorAll(tag)].some(n => n[attr] === url)) return resolve();
      const el = document.createElement(tag);
      el[attr] = url;
      el.onload = resolve;
      el.onerror = reject;
      document.head.appendChild(el);
    });
  };

  const ensureHereLoaded = async () => {
    if (window.H?.service && window.H.Map) return;
    await loadOnce("link", "href", "https://js.api.here.com/v3/3.1/mapsjs-ui.css");
    await loadOnce("script", "src", "https://js.api.here.com/v3/3.1/mapsjs-core.js");
    await loadOnce("script", "src", "https://js.api.here.com/v3/3.1/mapsjs-service.js");
    await loadOnce("script", "src", "https://js.api.here.com/v3/3.1/mapsjs-ui.js");
    await loadOnce("script", "src", "https://js.api.here.com/v3/3.1/mapsjs-mapevents.js");
  };

  const refCb = async (el) => {
    if (!el || el.dataset.inited) return;
    el.dataset.inited = "1";

    try {
      await ensureHereLoaded();
    } catch (e) {
      console.error("HERE SDK load failed", e);
      el.innerHTML = "<div style='padding:8px;color:#b91c1c;'>Failed to load HERE SDK assets.</div>";
      return;
    }

    const key =
      apiKey ||
      window.HERE_API_KEY ||
      document.querySelector("meta[name='here-apikey']")?.content ||
      "";

    if (!key) {
      el.innerHTML = "<div style='padding:8px;color:#b91c1c;'>HERE API key missing.</div>";
      return;
    }

    const H = window.H;
    const platform = new H.service.Platform({ apikey: key });
    const defaultLayers = platform.createDefaultLayers();

    const map = new H.Map(el, defaultLayers.vector.normal.map, {
      center: { lat: 52.55, lng: 13.27 },
      zoom: 9,
      pixelRatio: window.devicePixelRatio || 1,
    });

    // resize + interactions
    const behavior = new H.mapevents.Behavior(new H.mapevents.MapEvents(map));
    H.ui.UI.createDefault(map, defaultLayers);
    const onResize = () => map.getViewPort().resize();
    window.addEventListener("resize", onResize);

    // context menu
    map.addEventListener("contextmenu", (e) => {
      if (e.target instanceof H.map.Circle) {
        e.items.push(
          new H.util.ContextItem({
            label: "Remove circle",
            callback: () => map.removeObject(e.target),
          })
        );
      } else {
        const coord = map.screenToGeo(e.viewportX, e.viewportY);
        e.items.push(
          new H.util.ContextItem({
            label: `${Math.abs(coord.lat.toFixed(4))}${coord.lat > 0 ? "N" : "S"}, ` +
                   `${Math.abs(coord.lng.toFixed(4))}${coord.lng > 0 ? "E" : "W"}`,
          }),
          new H.util.ContextItem({
            label: "Center map here",
            callback: () => map.setCenter(coord, true),
          }),
          H.util.ContextItem.SEPARATOR,
          new H.util.ContextItem({
            label: "Add circle",
            callback: () => map.addObject(new H.map.Circle(coord, 5000)),
          })
        );
      }
    });

    // store cleanup hooks
    el._dispose = () => {
      try { map.dispose(); } catch {}
      window.removeEventListener("resize", onResize);
      delete el.dataset.inited;
    };
  };

  return (
    <div className="flex justify-center">
      <div className="rounded-md p-6 m-4 w-full max-w-3xl shadow-md border border-gray-300 dark:bg-gray-800 dark:border-gray-600">
        <p className="text-lg font-bold mb-2">{header}</p>
        <p className="mb-4">
          Right-click the map to open a context menu. Add/remove circles or re-center.
        </p>
        <div
          ref={refCb}
          style={{ width: "100%", height: `${height}px`, borderRadius: "8px", background: "grey" }}
        />
      </div>
    </div>
  );
};
