import { useState } from "react";

export const Counter = () => {
  const [count, setCount] = useState(0);

  return (
    <div className="flex flex-col items-center gap-4 p-6 bg-gray-900 text-white rounded-lg shadow-lg w-64">
      <h2 className="text-2xl font-bold">Counter: {count}</h2>
      <div className="flex gap-2">
        <button
          onClick={() => setCount(count - 1)}
          className="px-4 py-2 bg-red-600 hover:bg-red-500 text-white font-semibold rounded-md transition"
        >
          -
        </button>
        <button
          onClick={() => setCount(count + 1)}
          className="px-4 py-2 bg-green-600 hover:bg-green-500 text-white font-semibold rounded-md transition"
        >
          +
        </button>
      </div>
    </div>
  );
};

export default () => (
  <Counter />
);
