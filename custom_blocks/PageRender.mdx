import React, { useState } from "react";

export const FlattenedExportButton = ({
  label = "Download flattened content",
  filename = "flattened-doc.txt",
  mode = "text", // "text" or "html"
}) => {
  const [error, setError] = useState(null);

  const handleClick = (e) => {
    e.preventDefault();
    e.stopPropagation();
    setError(null);

    const root = document.querySelector(".rm-Markdown");
    if (!root) {
      setError("Could not find page content.");
      return;
    }

    const title = document.title || "";
    const url = window.location.href || "";

    let content = "";
    let mimeType = "text/plain;charset=utf-8";

    if (mode === "html") {
      const bodyHtml = root.innerHTML || "";
      const header = `<!-- ${title} | ${url} -->\n\n`;
      content = header + bodyHtml;
      mimeType = "text/html;charset=utf-8";
    } else {
      const bodyText = root.innerText || "";
      const header = title ? `${title}\n${url}\n\n` : "";
      content = header + bodyText;
    }

    try {
      const blob = new Blob([content], { type: mimeType });
      const objectUrl = URL.createObjectURL(blob);

      const a = document.createElement("a");
      a.href = objectUrl;
      a.download = filename;
      a.style.display = "none";
      a.target = "_self"; // 👈 ensures same-tab behavior
      document.body.appendChild(a);
      a.click();

      // clean up
      setTimeout(() => {
        URL.revokeObjectURL(objectUrl);
        a.remove();
      }, 100);
    } catch (err) {
      console.error(err);
      setError("Unable to generate download.");
    }
  };

  return (
    <div className="FlattenedExportButton">
      <button type="button" onClick={handleClick}>
        {label}
      </button>
      {error && <div className="error">{error}</div>}
    </div>
  );
};

<FlattenedExportButton
  label="Download flattened page as text"
  filename="flattened-page.txt"
/>

<FlattenedExportButton
  label="Download flattened page as HTML"
  filename="flattened-page.html"
  mode="html"
/>
