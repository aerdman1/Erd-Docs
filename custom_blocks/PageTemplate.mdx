name: PageTemplate
---
import * as React from "react";

export function PageTemplate({ project = "default", children }) {
  // Centralized defaults for each project
  const projectDefaults = {
    projectA: {
      productName: "Product Name A",
      featureName: "Feature Name A",
      image1Url: "https://example.com/projectA-image1.png",
      image2Url: "https://example.com/projectA-image2.png",
    },
    projectB: {
      productName: "Product Name B",
      featureName: "Feature Name B",
      image1Url: "https://example.com/projectB-image1.png",
      image2Url: "https://example.com/projectB-image2.png",
    },
    default: {
      productName: "Default Product Name",
      featureName: "Default Feature Name",
      image1Url: "https://sb-biodatacatalyst.readme.io/path/to/your-image1.png",
      image2Url: "https://sb-biodatacatalyst.readme.io/path/to/your-image2.png",
    },
  };

  const { productName, featureName, image1Url, image2Url } =
    projectDefaults[project] || projectDefaults.default;

  // Map tokens to their replacement values
  const replacements = { productName, featureName, image1Url, image2Url };

  // Recursive function to process children and replace tokens
  function replaceTokens(node) {
    if (typeof node === "string") {
      return node.replace(/\$([a-zA-Z0-9]+)\$/g, (_, token) => {
        return replacements[token] || `$${token}$`;
      });
    }
    if (React.isValidElement(node) && node.props && node.props.children) {
      return React.cloneElement(node, {
        children: React.Children.map(node.props.children, replaceTokens),
      });
    }
    if (Array.isArray(node)) {
      return node.map(replaceTokens);
    }
    return node;
  }

  return <>{replaceTokens(children)}</>;
}

export default function Preview() {
  return (
    <PageTemplate project="projectA">
      <h1>Create a Project in $productName$</h1>
      <p>
        The $featureName$ tool makes it easy to set up new workspaces.
      </p>
      <img src="$image1Url$" alt="Project preview" />
    </PageTemplate>
  );
}
