import * as React from "react";
export function PageTemplate({ project = "default" }) {
  const projectDefaults = {
    projectA: {
      productName: "VARIABLE PRODUCT NAME (A)",
      featureName: "VARIABLE FEATURE NAME (A)",
      image1Url: "https://files.readme.io/89602f1-create-a-project.png",
      image2Url: "https://files.readme.io/287284d-create-a-project-sbg.png",
    },
    projectB: {
      productName: "DIFFERENT PRODUCT NAME(B)",
      featureName: "DYNAMIC FEATURE NAME (B)",
      image1Url: "https://files.readme.io/720b281-create-a-project.png",
      image2Url: "",
    },
    projectC: {
      productName: "VARIABLE PRODUCT NAME (C)",
      featureName: "VARIABLE FEATURE NAME (C)",
      image1Url: "https://files.readme.io/53fc958-create-a-project.png",
      image2Url: "https://files.readme.io/b7ecef3-create-a-project.png",
    },
    projectD: {
      productName: "VARIABLE PRODUCT NAME (D)",
      featureName: "VARIABLE FEATURE NAME (D)",
      image1Url: "https://files.readme.io/a29234d-create-a-project.png",
      image2Url: "https://files.readme.io/94250ff-bdc-create-project.png",
    },
    default: {
      productName: "Default Product Name",
      featureName: "Default Feature Name",
      image1Url: "https://sb-biodatacatalyst.readme.io/path/to/your-image1.png",
      image2Url: "https://sb-biodatacatalyst.readme.io/path/to/your-image2.png",
    },
  };

  // Extract replacement values for the selected project
  const { productName, featureName, image1Url, image2Url } =
    projectDefaults[project] || projectDefaults.default;

  // Map tokens to their replacement values
  const replacements = { productName, featureName, image1Url, image2Url };

  React.useEffect(() => {
    // Recursive function to walk the DOM and replace tokens in text nodes and attributes
    function walk(node) {
      // Process text nodes
      if (node.nodeType === Node.TEXT_NODE) {
        const replacedText = node.nodeValue.replace(/\$([a-zA-Z0-9]+)\$/g, (_, token) => {
          return replacements[token] || `$${token}$`;
        });
        if (replacedText !== node.nodeValue) {
          node.nodeValue = replacedText;
        }
      }
      // Process element nodes (including attributes)
      if (node.nodeType === Node.ELEMENT_NODE) {
        // Check and update attributes if they contain tokens
        if (node.attributes && node.attributes.length > 0) {
          for (let i = 0; i < node.attributes.length; i++) {
            const attr = node.attributes[i];
            const replacedValue = attr.value.replace(/\$([a-zA-Z0-9]+)\$/g, (_, token) => {
              return replacements[token] || `$${token}$`;
            });
            if (replacedValue !== attr.value) {
              attr.value = replacedValue;
            }
          }
        }
        // Recursively process child nodes
        node.childNodes.forEach(child => walk(child));
      }
    }

    // Adjust the selector to match the container that holds your MDX content.
    // In your case, it might be the element with the class "rm-Markdown"
    const container = document.querySelector(".rm-Markdown");
    if (container) {
      walk(container);
    } else {
      console.warn("Markdown container not found. Token replacement was not applied.");
    }
  }, [replacements]);

  // This component only performs side effects, so it doesn't render anything itself.
  return null;
}
