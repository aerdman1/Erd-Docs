import { useState } from "react";

export const Magic8Ball = () => {
  const responses = [
    "Yes!",
    "No.",
    "Ask again later...",
    "Definitely!",
    "Very doubtful.",
    "Without a doubt!",
    "Better not tell you now.",
    "Signs point to yes.",
  ];

  const [answer, setAnswer] = useState("Click to ask!");

  const shakeBall = () => {
    setAnswer(responses[Math.floor(Math.random() * responses.length)]);
  };

  return (
    <div className="flex flex-col items-center p-6 bg-gray-900 text-white rounded-lg shadow-lg w-64">
      <div className="text-lg font-bold">Magic 8-Ball 🎱</div>
      <div className="my-4 p-4 bg-black text-white text-center rounded-full w-32 h-32 flex items-center justify-center text-xl">
        {answer}
      </div>
      <button
        onClick={shakeBall}
        className="px-4 py-2 bg-blue-600 hover:bg-blue-500 text-white font-semibold rounded-md transition"
      >
        Shake!
      </button>
    </div>
  );
};

export default Magic8Ball;
