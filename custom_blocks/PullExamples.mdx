import React, { useEffect, useState } from 'react';

export const EndpointExampleResponse = () => {
  const [endpointData, setEndpointData] = useState(null);
  const [error, setError] = useState(null);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    fetchEndpointExample();
  }, []);

  const fetchEndpointExample = async () => {
    setLoading(true);
    setError(null);

    try {
      // 1. Fetch the OpenAPI spec.
      const response = await fetch('https://erdtest123.readme.io/openapi/ErdSpec1.json');
      if (!response.ok) {
        throw new Error(`HTTP error: ${response.status}`);
      }
      const spec = await response.json();

      // 2. Locate the API endpoint URL from the page.
      // First, try to find an anchor element with a reference URL.
      let fullUrl = "";
      const refLink = document.querySelector('a[href^="https://erdtest123.readme.io/reference/"]');
      if (refLink) {
        fullUrl = refLink.getAttribute('href');
      }
      // Then try to find an actual API endpoint URL in the page text
      // (assuming your API endpoints start with "https://example.com/api/").
      const apiUrlMatch = document.body.innerText.match(/(https:\/\/example\.com\/api\/\S+)/);
      if (apiUrlMatch && apiUrlMatch[1]) {
        fullUrl = apiUrlMatch[1];
      }

      if (!fullUrl) {
        throw new Error("API endpoint URL not found on page.");
      }

      // 3. Extract the relative path from the full URL.
      const urlObj = new URL(fullUrl);
      const relativePath = urlObj.pathname; // e.g. "/api/v3/pet"

      // 4. Find the matching endpoint in the spec.
      const operations = spec.paths[relativePath];
      if (!operations) {
        throw new Error(`No endpoint found in spec for path ${relativePath}`);
      }

      // 5. Choose an operation—prefer one that has an example response.
      let chosenOperation = null;
      let chosenMethod = null;
      for (const [method, operation] of Object.entries(operations)) {
        const example = extractResponseExample(operation, spec.components);
        if (example) {
          chosenOperation = operation;
          chosenMethod = method.toUpperCase();
          break;
        }
      }
      // If none has an example, choose the first available operation.
      if (!chosenOperation) {
        const [method, operation] = Object.entries(operations)[0];
        chosenOperation = operation;
        chosenMethod = method.toUpperCase();
      }

      const summary = chosenOperation.summary || '';
      const description = chosenOperation.description || '';
      const example = extractResponseExample(chosenOperation, spec.components);

      setEndpointData({
        path: relativePath,
        method: chosenMethod,
        summary,
        description,
        example,
      });
      setLoading(false);
    } catch (err) {
      setError(err.message);
      setLoading(false);
    }
  };

  // Helper: Recursively resolve an example from a schema.
  const resolveExampleFromSchema = (schema, components) => {
    if (!schema) return null;
    if (schema.example !== undefined) {
      return schema.example;
    }
    // If schema is a reference, resolve it from components.schemas.
    if (schema.$ref) {
      const refParts = schema.$ref.split('/');
      const schemaName = refParts[refParts.length - 1];
      const resolved = components.schemas[schemaName];
      return resolveExampleFromSchema(resolved, components);
    }
    // For objects, build an example from each property.
    if (schema.type === 'object' && schema.properties) {
      const obj = {};
      Object.entries(schema.properties).forEach(([prop, propSchema]) => {
        obj[prop] = resolveExampleFromSchema(propSchema, components);
      });
      return obj;
    }
    // For arrays, return an array with a single example item.
    if (schema.type === 'array' && schema.items) {
      return [resolveExampleFromSchema(schema.items, components)];
    }
    return null;
  };

  // Helper: Extract an example from an operation’s responses.
  const extractResponseExample = (operation, components) => {
    if (!operation.responses) return null;
    // Prefer a 200 or default response.
    let response = operation.responses['200'] || operation.responses['default'];
    if (!response) {
      response = Object.values(operation.responses)[0];
    }
    if (response && response.content) {
      const media = response.content['application/json'];
      if (media) {
        if (media.example) {
          return media.example;
        }
        if (media.examples) {
          const firstExample = Object.values(media.examples)[0];
          if (firstExample && firstExample.value) return firstExample.value;
        }
        if (media.schema) {
          return resolveExampleFromSchema(media.schema, components);
        }
      }
    }
    return null;
  };

  if (loading) return <div>Loading endpoint example...</div>;
  if (error) return <div style={{ color: 'red' }}>Error: {error}</div>;
  if (!endpointData) return null;
  return (
    <div style={{ border: '1px solid #ddd', padding: '16px', borderRadius: '8px', marginTop: '16px' }}>
      <h3>{endpointData.method} {endpointData.path}</h3>
      {endpointData.summary && <p><strong>Summary:</strong> {endpointData.summary}</p>}
      {endpointData.description && <p><strong>Description:</strong> {endpointData.description}</p>}
      {endpointData.example ? (
        <div>
          <strong>Example Response:</strong>
          <pre style={{
              whiteSpace: 'pre-wrap',
              background: '#f5f5f5',
              padding: '8px',
              borderRadius: '4px',
              marginTop: '8px'
            }}>
            {JSON.stringify(endpointData.example, null, 2)}
          </pre>
        </div>
      ) : (
        <p>No example available.</p>
      )}
    </div>
  );
};

export default EndpointExampleResponse;
