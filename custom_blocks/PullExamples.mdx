import React, { useEffect, useState } from 'react';

export const EndpointExampleResponse = ({ endpoint }) => {
  const [endpointData, setEndpointData] = useState(null);
  const [error, setError] = useState(null);
  const [loading, setLoading] = useState(false);
  // "spec" or "iframe" mode toggle; default is "spec"
  const [mode, setMode] = useState("spec");

  useEffect(() => {
    if (!endpoint) {
      setError("No endpoint provided.");
      return;
    }
    // Clear previous data when switching modes or endpoint changes.
    setEndpointData(null);
    setError(null);
    console.log(`Mode changed: ${mode} | Endpoint: ${endpoint}`);
    if (mode === "spec") {
      fetchSpecMethod();
    } else if (mode === "iframe") {
      fetchIframeMethod();
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [endpoint, mode]);

  // ----------------- SPEC MODE -----------------
  const fetchSpecMethod = async () => {
    setLoading(true);
    setError(null);
    console.log("Spec mode: Fetching OpenAPI spec from https://erdtest123.readme.io/openapi/ErdSpec1.json");
    try {
      const response = await fetch('https://erdtest123.readme.io/openapi/ErdSpec1.json');
      if (!response.ok) throw new Error(`HTTP error: ${response.status}`);
      const spec = await response.json();
      console.log("Spec fetched successfully:", spec);

      // Look up the provided endpoint (must match the spec's key exactly)
      const operations = spec.paths[endpoint];
      if (!operations) {
        throw new Error(`No endpoint found in spec for path ${endpoint}`);
      }
      console.log("Found operations in spec:", operations);

      // Choose an operation that provides an example (if available)
      let chosenOperation = null;
      let chosenMethod = null;
      for (const [method, operation] of Object.entries(operations)) {
        const example = extractResponseExample(operation, spec.components);
        console.log(`Spec mode: Checking method ${method} example:`, example);
        if (example) {
          chosenOperation = operation;
          chosenMethod = method.toUpperCase();
          break;
        }
      }
      // Fallback: if no example is found, use the first available operation.
      if (!chosenOperation) {
        const [method, operation] = Object.entries(operations)[0];
        chosenOperation = operation;
        chosenMethod = method.toUpperCase();
      }
      const summary = chosenOperation.summary || '';
      const description = chosenOperation.description || '';
      const example = extractResponseExample(chosenOperation, spec.components);
      console.log("Spec mode: Final chosen operation:", {
        path: endpoint,
        method: chosenMethod,
        summary,
        description,
        example,
      });
      setEndpointData({
        path: endpoint,
        method: chosenMethod,
        summary,
        description,
        example,
        modeLabel: "Response from Spec Mode",
      });
      setLoading(false);
    } catch (err) {
      console.error("Spec mode error:", err);
      setError(err.message);
      setLoading(false);
    }
  };

  // ----------------- IFRAME MODE -----------------
  const fetchIframeMethod = () => {
    setLoading(true);
    setError(null);
    console.log("Iframe mode: Starting process for endpoint", endpoint);
    // Create an off-screen iframe
    const iframe = document.createElement('iframe');
    iframe.style.position = 'absolute';
    iframe.style.top = '-9999px';
    // Map the provided endpoint to a docs URL.
    // Adjust these mappings as needed.
    let docsUrl = "";
    switch (endpoint) {
      case "/pet":
        docsUrl = "https://erdtest123.readme.io/reference/updatepet";
        break;
      case "/pet/{petId}":
        docsUrl = "https://erdtest123.readme.io/reference/getpetbyid"; // adjust as needed
        break;
      case "/pet/findByStatus":
        docsUrl = "https://erdtest123.readme.io/reference/findpetbystatus";
        break;
      // Add other mappings as necessary.
      default:
        docsUrl = `https://erdtest123.readme.io/reference${endpoint}`;
        break;
    }
    console.log("Iframe mode: Using docs URL:", docsUrl);
    iframe.src = docsUrl;
    document.body.appendChild(iframe);

    iframe.onload = function () {
      try {
        const iDoc = iframe.contentDocument || iframe.contentWindow.document;
        console.log('Iframe mode: Iframe loaded.');

        // Wait for the page's JS to initialize.
        setTimeout(() => {
          // 1. Click the "EXAMPLE" button.
          const exampleButton = Array.from(iDoc.querySelectorAll('button')).find(btn =>
            btn.innerText.trim().toUpperCase().includes("EXAMPLE")
          );
          if (!exampleButton) {
            setError("Iframe mode: EXAMPLE button not found.");
            setLoading(false);
            document.body.removeChild(iframe);
            return;
          }
          console.log('Iframe mode: Clicking EXAMPLE button:', exampleButton);
          exampleButton.click();

          // 2. Wait for the Tippy popup to appear.
          setTimeout(() => {
            const popup = iDoc.querySelector('[data-tippy-root]');
            if (!popup) {
              setError("Iframe mode: Popup not found.");
              setLoading(false);
              document.body.removeChild(iframe);
              return;
            }
            console.log('Iframe mode: Popup found:', popup);

            // 3. In the popup, click the "Request Example" option.
            const reqExampleOption = Array.from(popup.querySelectorAll('.Menu-Item')).find(el =>
              el.innerText.trim().toUpperCase().includes("REQUEST EXAMPLE")
            );
            if (!reqExampleOption) {
              setError('Iframe mode: "Request Example" option not found in popup.');
              setLoading(false);
              document.body.removeChild(iframe);
              return;
            }
            console.log('Iframe mode: Clicking "Request Example" option:', reqExampleOption);
            reqExampleOption.click();

            // 4. Wait for CodeMirror to update.
            setTimeout(() => {
              const codeContainer = iDoc.querySelector('.react-codemirror2.CodeEditor');
              if (!codeContainer) {
                setError("Iframe mode: CodeMirror container not found.");
                setLoading(false);
                document.body.removeChild(iframe);
                return;
              }
              const codeLines = codeContainer.querySelectorAll('.CodeMirror-lines pre.CodeMirror-line');
              if (!codeLines || codeLines.length === 0) {
                setError("Iframe mode: No CodeMirror lines found.");
                setLoading(false);
                document.body.removeChild(iframe);
                return;
              }
              let codeText = "";
              codeLines.forEach(line => {
                codeText += line.textContent + "\n";
              });
              console.log("Iframe mode: Extracted CodeMirror text:", codeText);
              setEndpointData({
                path: endpoint,
                method: "", // Not available via iframe extraction.
                summary: "",
                description: "",
                example: codeText,
                modeLabel: "Response from Iframe Mode",
              });
              setLoading(false);
              document.body.removeChild(iframe);
            }, 3000); // Delay for CodeMirror to load.
          }, 2000); // Delay for popup to appear.
        }, 2000); // Initial delay.
      } catch (err) {
        console.error("Iframe mode error:", err);
        setError(err.message);
        setLoading(false);
        document.body.removeChild(iframe);
      }
    };
  };

  // ----------------- HELPER FUNCTIONS (SPEC MODE) -----------------
  const resolveExampleFromSchema = (schema, components) => {
    if (!schema) return null;
    if (schema.example !== undefined) {
      return schema.example;
    }
    if (schema.$ref) {
      const refParts = schema.$ref.split('/');
      const schemaName = refParts[refParts.length - 1];
      const resolved = components.schemas[schemaName];
      return resolveExampleFromSchema(resolved, components);
    }
    if (schema.type === 'object' && schema.properties) {
      const obj = {};
      Object.entries(schema.properties).forEach(([prop, propSchema]) => {
        obj[prop] = resolveExampleFromSchema(propSchema, components);
      });
      return obj;
    }
    if (schema.type === 'array' && schema.items) {
      return [resolveExampleFromSchema(schema.items, components)];
    }
    return null;
  };

  const extractResponseExample = (operation, components) => {
    if (!operation.responses) return null;
    let response = operation.responses['200'] || operation.responses['default'];
    if (!response) {
      response = Object.values(operation.responses)[0];
    }
    if (response && response.content) {
      const media = response.content['application/json'];
      if (media) {
        if (media.example) {
          return media.example;
        }
        if (media.examples) {
          const firstExample = Object.values(media.examples)[0];
          if (firstExample && firstExample.value) return firstExample.value;
        }
        if (media.schema) {
          return resolveExampleFromSchema(media.schema, components);
        }
      }
    }
    return null;
  };

  // ----------------- RENDER -----------------
  if (loading) return <div>Loading endpoint example...</div>;
  if (error) return <div style={{ color: 'red' }}>Error: {error}</div>;
  if (!endpointData) return null;
  return (
    <div style={{ border: '1px solid #ddd', padding: '16px', borderRadius: '8px', marginTop: '16px' }}>
      <h3>{endpointData.method} {endpointData.path}</h3>
      <p><em>{endpointData.modeLabel}</em></p>
      {endpointData.summary && <p><strong>Summary:</strong> {endpointData.summary}</p>}
      {endpointData.description && <p><strong>Description:</strong> {endpointData.description}</p>}
      {endpointData.example ? (
        <div>
          <strong>Example Response:</strong>
          <pre style={{
            whiteSpace: 'pre-wrap',
            background: '#f5f5f5',
            padding: '8px',
            borderRadius: '4px',
            marginTop: '8px'
          }}>
            {typeof endpointData.example === 'object'
              ? JSON.stringify(endpointData.example, null, 2)
              : endpointData.example}
          </pre>
        </div>
      ) : (
        <p>No example available.</p>
      )}
      <div style={{ marginTop: '16px' }}>
        <button onClick={() => setMode(mode === 'spec' ? 'iframe' : 'spec')}>
          Switch to {mode === 'spec' ? 'Iframe' : 'Spec'} Mode
        </button>
      </div>
    </div>
  );
};

export default EndpointExampleResponse;
