import React, { useState } from "react";

export const SimpleComponent = () => {
  const [url, setUrl] = useState("");
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  const handleFetch = () => {
    if (!url) return;
    setLoading(true);
    setError(null);
    fetch(url)
      .then((res) => {
        if (!res.ok) {
          throw new Error("Network response was not ok");
        }
        return res.text();
      })
      .then((htmlString) => {
        // Parse the HTML string into a document
        const parser = new DOMParser();
        const doc = parser.parseFromString(htmlString, "text/html");

        // Extract the Title from <header id="content-head"><h1>...</h1></header>
        const titleElement = doc.querySelector("header#content-head h1");
        const title = titleElement ? titleElement.innerText.trim() : "No Title Found";

        // Extract the Description from the element with role="doc-subtitle"
        const descElement = doc.querySelector("div[role='doc-subtitle']");
        const description = descElement ? descElement.innerText.trim() : "No Description Found";

        // Extract the API method (e.g. GET, POST) from the first element with class "rm-APIMethod" within the header
        const methodElement = doc.querySelector("header#content-head span.rm-APIMethod");
        const method = methodElement ? methodElement.innerText.trim() : "No Method Found";

        // Extract the endpoint URL from the span with a specific class (adjust the selector as needed)
        const endpointElement = doc.querySelector("span.headline-container-article-info-url2nV_XrjpFuVQ");
        const endpointUrl = endpointElement ? endpointElement.innerText.trim() : "No Endpoint URL Found";

        // Optionally, extract the "Last Updated" info from a paragraph with class "DateLine"
        const updatedElement = doc.querySelector("p.DateLine");
        const updated = updatedElement ? updatedElement.innerText.trim() : "No Updated Info Found";

        // Set the extracted data in state
        setData({
          title,
          description,
          method,
          endpointUrl,
          updated,
        });
      })
      .catch((err) => {
        console.error("Error fetching/parsing content:", err);
        setError(err.message);
      })
      .finally(() => {
        setLoading(false);
      });
  };

  return (
    <div style={{ padding: "1rem", border: "1px solid #ccc" }}>
      <h3>Fetch Endpoint Information</h3>
      <p>Enter the URL of your documentation page (e.g., the Find Pets by Status page):</p>
      <div style={{ marginBottom: "1rem" }}>
        <input
          type="text"
          placeholder="https://yoursite.com/reference/findpetsbystatus"
          value={url}
          onChange={(e) => setUrl(e.target.value)}
          style={{ width: "300px", marginRight: "10px" }}
        />
        <button onClick={handleFetch}>Fetch Info</button>
      </div>
      {loading && <p>Loading...</p>}
      {error && <p style={{ color: "red" }}>Error: {error}</p>}
      {data && (
        <div>
          <h4>Extracted Endpoint Info</h4>
          <p>
            <strong>Title:</strong> {data.title}
          </p>
          <p>
            <strong>Description:</strong> {data.description}
          </p>
          <p>
            <strong>Method:</strong> {data.method}
          </p>
          <p>
            <strong>Endpoint URL:</strong> {data.endpointUrl}
          </p>
          <p>
            <strong>Last Updated:</strong> {data.updated}
          </p>
        </div>
      )}
    </div>
  );
};

<SimpleComponent />
