import React, { useEffect, useMemo, useState } from "react";

export const RSS = props => {
  const url = props.url || props.rssUrl || "/changelog.rss";
  const limit = Number(props.limit || props.maxItems) > 0 ? Number(props.limit || props.maxItems) : 20;
  const compact = props.compact === true || props.compact === "true";
  const showFilters = !(props.filters === false || props.filters === "false" || props.showSearch === false);
  const [items, setItems] = useState(null);
  const [error, setError] = useState("");
  const [type, setType] = useState(props.type || "all");
  const [q, setQ] = useState("");

  useEffect(() => {
    let alive = true;
    const key = "eo-rss:" + url;
    try {
      const hit = JSON.parse(window.sessionStorage.getItem(key) || "null");
      if (hit && Date.now() - hit.at < 5 * 60 * 1000) { setItems(hit.items); return undefined; }
    } catch (e) {}
    fetch(url, { headers: { Accept: "application/rss+xml, application/xml, text/xml" } })
      .then(r => { if (!r.ok) throw new Error("HTTP " + r.status); return r.text(); })
      .then(xml => {
        const doc = new DOMParser().parseFromString(xml, "application/xml");
        const get = (el, tag) => { const n = el.getElementsByTagName(tag)[0]; return n ? (n.textContent || "").trim() : ""; };
        const list = Array.from(doc.getElementsByTagName("item")).map(el => ({
          title: get(el, "title"),
          link: get(el, "link"),
          text: get(el, "description").replace(/<[^>]+>/g, " ").replace(/\s+/g, " ").trim(),
          date: get(el, "pubDate"),
          type: (get(el, "type") || "none").toLowerCase(),
        }));
        if (!alive) return;
        setItems(list);
        try { window.sessionStorage.setItem(key, JSON.stringify({ at: Date.now(), items: list })); } catch (e) {}
      })
      .catch(e => { if (alive) setError(String(e && e.message || e)); });
    return () => { alive = false; };
  }, [url]);

  const META = {
    added: { label: "Added", color: "#12B76A", icon: "+" },
    improved: { label: "Improved", color: "#7B3FF2", icon: "↑" },
    fixed: { label: "Fixed", color: "#0E9384", icon: "✓" },
    deprecated: { label: "Deprecated", color: "#F79009", icon: "!" },
    removed: { label: "Removed", color: "#F04438", icon: "−" },
    none: { label: "Update", color: "#667085", icon: "•" },
  };
  const meta = t => META[t] || META.none;

  const counts = useMemo(() => {
    const c = {};
    (items || []).forEach(it => { c[it.type] = (c[it.type] || 0) + 1; });
    return c;
  }, [items]);

  const shown = useMemo(() => {
    const needle = q.trim().toLowerCase();
    return (items || [])
      .filter(it => type === "all" || it.type === type)
      .filter(it => !needle || (it.title + " " + it.text).toLowerCase().indexOf(needle) !== -1)
      .slice(0, limit);
  }, [items, type, q, limit]);

  const fmt = s => {
    const d = new Date(s);
    if (isNaN(d.getTime())) return "";
    return d.toLocaleDateString(undefined, { year: "numeric", month: "short", day: "numeric" });
  };
  const ago = s => {
    const d = new Date(s).getTime();
    if (isNaN(d)) return "";
    const days = Math.round((Date.now() - d) / 86400000);
    if (days <= 0) return "today";
    if (days === 1) return "yesterday";
    if (days < 30) return days + " days ago";
    return fmt(s);
  };

  const css = `
.eo-feed{--eo-accent:var(--color-link-primary,#2F7BF5);--eo-border:var(--color-border-default,rgba(0,0,0,.12));--eo-muted:var(--color-text-minimum,#6b7280);margin:20px 0}
.eo-feed-top{display:flex;flex-wrap:wrap;align-items:center;gap:8px;margin-bottom:14px}
.eo-feed-pill{border:1px solid var(--eo-border);background:transparent;color:inherit;border-radius:999px;padding:4px 11px;font-size:13px;cursor:pointer;display:inline-flex;align-items:center;gap:6px}
.eo-feed-pill[aria-pressed="true"]{border-color:var(--eo-accent);background:color-mix(in srgb,var(--eo-accent) 12%,transparent);font-weight:600}
.eo-feed-pill i{width:8px;height:8px;border-radius:50%;display:inline-block}
.eo-feed-search{margin-left:auto;border:1px solid var(--eo-border);background:transparent;color:inherit;border-radius:999px;padding:5px 12px;font:inherit;font-size:13px;min-width:160px}
.eo-feed-list{list-style:none;margin:0;padding:0;position:relative}
.eo-feed-item{display:grid;grid-template-columns:110px 1fr;gap:16px;padding:0 0 18px;position:relative}
.eo-feed-date{font-size:13px;color:var(--eo-muted);padding-top:12px;text-align:right}
.eo-feed-card{border:1px solid var(--eo-border);border-radius:12px;padding:12px 16px;position:relative;transition:border-color .15s,transform .15s}
.eo-feed-card:hover{border-color:var(--eo-accent);transform:translateX(2px)}
.eo-feed-badge{display:inline-flex;align-items:center;gap:5px;font-size:11px;font-weight:700;letter-spacing:.05em;text-transform:uppercase;padding:2px 8px;border-radius:999px}
.eo-feed-title{display:block;margin:6px 0 4px;font-weight:700;font-size:16px;color:inherit!important;text-decoration:none!important}
.eo-feed-title:hover{color:var(--eo-accent)!important}
.eo-feed-text{margin:0;font-size:14px;line-height:1.5;color:var(--eo-muted);display:-webkit-box;-webkit-line-clamp:2;-webkit-box-orient:vertical;overflow:hidden}
.eo-feed-compact{list-style:none;margin:0;padding:0;border:1px solid var(--eo-border);border-radius:12px;overflow:hidden}
.eo-feed-compact li{display:flex;align-items:center;gap:10px;padding:10px 14px;border-top:1px solid var(--eo-border)}
.eo-feed-compact li:first-child{border-top:0}
.eo-feed-compact a{flex:1;color:inherit!important;text-decoration:none!important;font-weight:600}
.eo-feed-compact a:hover{color:var(--eo-accent)!important}
.eo-feed-compact span{font-size:13px;color:var(--eo-muted);white-space:nowrap}
.eo-feed-skel{height:64px;border-radius:12px;margin-bottom:10px;background:linear-gradient(90deg,var(--eo-border),transparent,var(--eo-border));background-size:200% 100%;animation:eo-shimmer 1.2s linear infinite}
@keyframes eo-shimmer{to{background-position:-200% 0}}
@media (max-width:600px){.eo-feed-item{grid-template-columns:1fr;gap:4px}.eo-feed-date{text-align:left;padding-top:0}.eo-feed-search{margin-left:0;flex:1}}
`;

  if (error) {
    return <div className="eo-feed"><style>{css}</style><p style={{ color: "var(--color-text-minimum)" }}>Couldn’t load the changelog feed ({error}). <a href="/changelog">Open the changelog →</a></p></div>;
  }
  if (!items) {
    return <div className="eo-feed" aria-busy="true"><style>{css}</style><div className="eo-feed-skel" /><div className="eo-feed-skel" /></div>;
  }

  if (compact) {
    return (
      <div className="eo-feed">
        <style>{css}</style>
        {props.title ? <p style={{ fontWeight: 700, margin: "0 0 8px" }}>{props.title}</p> : null}
        <ul className="eo-feed-compact">
          {shown.map((it, k) => (
            <li key={k}>
              <span className="eo-feed-badge" style={{ color: meta(it.type).color, background: meta(it.type).color + "1f" }}>{meta(it.type).icon} {meta(it.type).label}</span>
              <a href={it.link}>{it.title}</a>
              <span>{ago(it.date)}</span>
            </li>
          ))}
        </ul>
      </div>
    );
  }

  const types = Object.keys(META).filter(t => counts[t]);
  return (
    <div className="eo-feed">
      <style>{css}</style>
      {showFilters ? (
        <div className="eo-feed-top">
          <button type="button" className="eo-feed-pill" aria-pressed={type === "all" ? "true" : "false"} onClick={() => setType("all")}>All <b>{(items || []).length}</b></button>
          {types.map(t => (
            <button key={t} type="button" className="eo-feed-pill" aria-pressed={type === t ? "true" : "false"} onClick={() => setType(type === t ? "all" : t)}>
              <i style={{ background: meta(t).color }} />{meta(t).label} <b>{counts[t]}</b>
            </button>
          ))}
          <input className="eo-feed-search" type="search" placeholder="Search updates…" aria-label="Search updates" value={q} onChange={e => setQ(e.target.value)} />
        </div>
      ) : null}
      {shown.length ? (
        <ol className="eo-feed-list">
          {shown.map((it, k) => (
            <li key={k} className="eo-feed-item">
              <div className="eo-feed-date">{fmt(it.date)}</div>
              <div className="eo-feed-card">
                <span className="eo-feed-badge" style={{ color: meta(it.type).color, background: meta(it.type).color + "1f" }}>{meta(it.type).icon} {meta(it.type).label}</span>
                <a className="eo-feed-title" href={it.link}>{it.title}</a>
                {it.text ? <p className="eo-feed-text">{it.text}</p> : null}
              </div>
            </li>
          ))}
        </ol>
      ) : <p style={{ color: "var(--color-text-minimum)" }}>No updates match.</p>}
      <p style={{ fontSize: "13px", margin: "6px 0 0" }}><a href={url}>Subscribe with RSS</a> · <a href="/changelog">Full changelog →</a></p>
    </div>
  );
};

export const ChangelogRssFeed = RSS;

<RSS limit="5" />
