import React, { useEffect, useMemo, useState } from "react";

export const LiveEndpointFromSpecBlock = props => {
  const specUrl = props.specUrl || "";
  const apiBaseUrlOverride = props.apiBaseUrl || "";
  const endpointPath = props.path || props.endpointPath || "/posts/{id}";
  const endpointMethod = String(props.method || "GET").toUpperCase();
  const defaultAuthMode = props.authMode || "none";
  const defaultApiKeyLocation = props.apiKeyLocation || "header";
  const defaultApiKeyHeaderName = props.apiKeyHeaderName || "x-api-key";
  const defaultApiKeyQueryName = props.apiKeyQueryName || "api_key";
  const defaultHeadersJson = props.defaultHeadersJson || "{}";
  const showAdvancedDefault = props.showAdvanced === true;

  const [spec, setSpec] = useState(null);
  const [endpoint, setEndpoint] = useState(null);
  const [loadingSpec, setLoadingSpec] = useState(false);
  const [error, setError] = useState("");

  const [apiBaseUrlInput, setApiBaseUrlInput] = useState(apiBaseUrlOverride);
  const [pathValues, setPathValues] = useState({});
  const [queryValues, setQueryValues] = useState({});
  const [headerValues, setHeaderValues] = useState({});
  const [requestBody, setRequestBody] = useState("");
  const [headersText, setHeadersText] = useState(defaultHeadersJson);

  const [authMode, setAuthMode] = useState(defaultAuthMode);
  const [apiKeyLocation, setApiKeyLocation] = useState(defaultApiKeyLocation);
  const [apiKeyHeaderName, setApiKeyHeaderName] = useState(defaultApiKeyHeaderName);
  const [apiKeyQueryName, setApiKeyQueryName] = useState(defaultApiKeyQueryName);
  const [bearerToken, setBearerToken] = useState("");
  const [apiKeyValue, setApiKeyValue] = useState("");
  const [basicUsername, setBasicUsername] = useState("");
  const [basicPassword, setBasicPassword] = useState("");

  const [showAdvanced, setShowAdvanced] = useState(showAdvancedDefault);
  const [running, setRunning] = useState(false);
  const [result, setResult] = useState(null);

  const safeString = value => {
    if (value === null || value === undefined) return "";
    return String(value);
  };

  const prettyJson = value => {
    try {
      return JSON.stringify(value, null, 2);
    } catch (e) {
      return "";
    }
  };

  const getFallbackSpec = () => {
    return {
      openapi: "3.0.3",
      info: {
        title: "JSONPlaceholder Sandbox API",
        version: "1.0.0"
      },
      servers: [
        {
          url: "https://jsonplaceholder.typicode.com"
        }
      ],
      paths: {
        "/posts/{id}": {
          get: {
            tags: ["Posts"],
            summary: "Get a post",
            description: "Returns one fake post by ID.",
            parameters: [
              {
                name: "id",
                in: "path",
                required: true,
                description: "Post ID.",
                schema: {
                  type: "integer",
                  example: 1
                }
              }
            ],
            responses: {
              "200": {
                description: "A single post.",
                content: {
                  "application/json": {
                    schema: {
                      "$ref": "#/components/schemas/Post"
                    }
                  }
                }
              }
            }
          }
        },
        "/posts": {
          post: {
            tags: ["Posts"],
            summary: "Create a post",
            description: "Creates a fake post. The response is simulated and not persisted.",
            requestBody: {
              required: true,
              content: {
                "application/json": {
                  schema: {
                    "$ref": "#/components/schemas/NewPost"
                  },
                  example: {
                    title: "Hello from ReadMe MDX",
                    body: "This is a sandbox request body.",
                    userId: 1
                  }
                }
              }
            },
            responses: {
              "201": {
                description: "Fake created post response.",
                content: {
                  "application/json": {
                    schema: {
                      "$ref": "#/components/schemas/Post"
                    }
                  }
                }
              }
            }
          }
        }
      },
      components: {
        schemas: {
          Post: {
            type: "object",
            properties: {
              userId: {
                type: "integer",
                example: 1
              },
              id: {
                type: "integer",
                example: 1
              },
              title: {
                type: "string",
                example: "Hello from ReadMe MDX"
              },
              body: {
                type: "string",
                example: "This is a sandbox post body."
              }
            }
          },
          NewPost: {
            type: "object",
            properties: {
              title: {
                type: "string",
                example: "Hello from ReadMe MDX"
              },
              body: {
                type: "string",
                example: "This is a sandbox request body."
              },
              userId: {
                type: "integer",
                example: 1
              }
            }
          }
        }
      }
    };
  };

  const resolveRef = (obj, ref) => {
    if (!obj || !ref || ref.indexOf("#/") !== 0) return null;

    const parts = ref
      .replace("#/", "")
      .split("/")
      .map(part => part.replace(/~1/g, "/").replace(/~0/g, "~"));

    let current = obj;

    for (let i = 0; i < parts.length; i += 1) {
      if (!current || current[parts[i]] === undefined) return null;
      current = current[parts[i]];
    }

    return current;
  };

  const deref = value => {
    if (value && value.$ref && spec) {
      return resolveRef(spec, value.$ref) || value;
    }

    return value;
  };

  const getExampleFromSchema = schema => {
    const resolved = deref(schema);

    if (!resolved) return null;

    if (resolved.example !== undefined) return resolved.example;
    if (resolved.default !== undefined) return resolved.default;
    if (resolved.enum && resolved.enum.length > 0) return resolved.enum[0];

    if (resolved.type === "string") return "";
    if (resolved.type === "integer" || resolved.type === "number") return 0;
    if (resolved.type === "boolean") return false;

    if (resolved.type === "array") {
      return [getExampleFromSchema(resolved.items)];
    }

    if (resolved.type === "object" || resolved.properties) {
      const obj = {};
      const props = resolved.properties || {};

      Object.entries(props).forEach(entry => {
        obj[entry[0]] = getExampleFromSchema(entry[1]);
      });

      return obj;
    }

    if (resolved.oneOf && resolved.oneOf.length > 0) {
      return getExampleFromSchema(resolved.oneOf[0]);
    }

    if (resolved.anyOf && resolved.anyOf.length > 0) {
      return getExampleFromSchema(resolved.anyOf[0]);
    }

    if (resolved.allOf && resolved.allOf.length > 0) {
      const merged = {};

      resolved.allOf.forEach(item => {
        const example = getExampleFromSchema(item);

        if (example && typeof example === "object" && !Array.isArray(example)) {
          Object.assign(merged, example);
        }
      });

      return merged;
    }

    return null;
  };

  const getRequestBodyExample = operation => {
    const requestBodyObj = deref(operation.requestBody);

    if (!requestBodyObj || !requestBodyObj.content) return "";

    const jsonMedia =
      requestBodyObj.content["application/json"] ||
      requestBodyObj.content["application/*+json"];

    if (!jsonMedia) return "";

    if (jsonMedia.example !== undefined) {
      return prettyJson(jsonMedia.example);
    }

    if (jsonMedia.examples) {
      const firstExample = Object.values(jsonMedia.examples)[0];
      const resolvedExample = deref(firstExample);

      if (resolvedExample && resolvedExample.value !== undefined) {
        return prettyJson(resolvedExample.value);
      }
    }

    if (jsonMedia.schema) {
      const schemaExample = getExampleFromSchema(jsonMedia.schema);

      if (schemaExample !== null) {
        return prettyJson(schemaExample);
      }
    }

    return "";
  };

  const normalizeServerUrl = url => {
    let nextUrl = safeString(url);

    nextUrl = nextUrl.replace("{scheme}", "https");

    return nextUrl.replace(/\/$/, "");
  };

  const getServerUrl = loadedSpec => {
    if (apiBaseUrlOverride) return apiBaseUrlOverride.replace(/\/$/, "");

    if (loadedSpec && loadedSpec.servers && loadedSpec.servers.length > 0) {
      const firstServer = loadedSpec.servers[0];

      if (firstServer && firstServer.url) {
        return normalizeServerUrl(firstServer.url);
      }
    }

    return "";
  };

  const parseSpecResponse = async response => {
    const text = await response.text();

    try {
      return JSON.parse(text);
    } catch (jsonError) {
      if (typeof window !== "undefined" && window.jsyaml && window.jsyaml.load) {
        return window.jsyaml.load(text);
      }

      throw new Error("Could not parse the spec as JSON. If this is YAML, expose a JSON version or load js-yaml on the page.");
    }
  };

  const findEndpoint = loadedSpec => {
    if (!loadedSpec || !loadedSpec.paths) {
      throw new Error("Spec loaded, but no paths object was found.");
    }

    const pathItem = loadedSpec.paths[endpointPath];

    if (!pathItem) {
      throw new Error("Could not find path in spec: " + endpointPath);
    }

    const operation = pathItem[endpointMethod.toLowerCase()];

    if (!operation) {
      throw new Error("Could not find method for " + endpointMethod + " " + endpointPath);
    }

    const operationParams = Array.isArray(operation.parameters) ? operation.parameters : [];
    const pathItemParams = Array.isArray(pathItem.parameters) ? pathItem.parameters : [];
    const allParams = pathItemParams.concat(operationParams).map(param => {
      if (param && param.$ref) return resolveRef(loadedSpec, param.$ref) || param;
      return param;
    });

    return {
      path: endpointPath,
      method: endpointMethod,
      operation: operation,
      summary: operation.summary || "",
      description: operation.description || "",
      params: allParams
    };
  };

  const loadSpec = async () => {
    setLoadingSpec(true);
    setError("");
    setResult(null);

    try {
      let loadedSpec = null;

      if (!specUrl) {
        loadedSpec = getFallbackSpec();
      } else {
        const response = await fetch(specUrl);

        if (!response.ok) {
          throw new Error("Spec fetch failed with HTTP " + response.status);
        }

        loadedSpec = await parseSpecResponse(response);
      }

      const foundEndpoint = findEndpoint(loadedSpec);

      setSpec(loadedSpec);
      setEndpoint(foundEndpoint);
      setApiBaseUrlInput(getServerUrl(loadedSpec));
      setLoadingSpec(false);
    } catch (err) {
      setError(err && err.message ? err.message : "Failed to load endpoint.");
      setLoadingSpec(false);
    }
  };

  useEffect(() => {
    loadSpec();
  }, [specUrl, endpointPath, endpointMethod]);

  const pathParams = useMemo(() => {
    if (!endpoint) return [];
    return endpoint.params.filter(param => param && param.in === "path");
  }, [endpoint]);

  const queryParams = useMemo(() => {
    if (!endpoint) return [];
    return endpoint.params.filter(param => param && param.in === "query");
  }, [endpoint]);

  const headerParams = useMemo(() => {
    if (!endpoint) return [];
    return endpoint.params.filter(param => param && param.in === "header");
  }, [endpoint]);

  useEffect(() => {
    if (!endpoint || !spec) return;

    const nextPathValues = {};
    const nextQueryValues = {};
    const nextHeaderValues = {};

    pathParams.forEach(param => {
      const example =
        param.example !== undefined
          ? param.example
          : getExampleFromSchema(param.schema);

      nextPathValues[param.name] = safeString(example);
    });

    queryParams.forEach(param => {
      const example =
        param.example !== undefined
          ? param.example
          : getExampleFromSchema(param.schema);

      nextQueryValues[param.name] = safeString(example);
    });

    headerParams.forEach(param => {
      const lowerName = safeString(param.name).toLowerCase();

      if (lowerName === "authorization") return;
      if (lowerName === "accept") return;
      if (lowerName === "content-type") return;

      const example =
        param.example !== undefined
          ? param.example
          : getExampleFromSchema(param.schema);

      nextHeaderValues[param.name] = safeString(example);
    });

    setPathValues(nextPathValues);
    setQueryValues(nextQueryValues);
    setHeaderValues(nextHeaderValues);
    setRequestBody(getRequestBodyExample(endpoint.operation));
  }, [endpoint, spec]);

  const buildUrl = () => {
    const cleanBase = safeString(apiBaseUrlInput).replace(/\/$/, "");

    if (!cleanBase) {
      throw new Error("Missing API base URL.");
    }

    let resolvedPath = endpoint.path;

    pathParams.forEach(param => {
      const rawValue = pathValues[param.name] || "";
      resolvedPath = resolvedPath.replace("{" + param.name + "}", encodeURIComponent(rawValue));
    });

    const url = new URL(cleanBase + resolvedPath);

    queryParams.forEach(param => {
      const value = queryValues[param.name];

      if (value !== undefined && value !== null && String(value).trim() !== "") {
        url.searchParams.set(param.name, value);
      }
    });

    if (authMode === "apiKey" && apiKeyLocation === "query" && apiKeyValue.trim()) {
      url.searchParams.set(apiKeyQueryName || "api_key", apiKeyValue.trim());
    }

    return url.toString();
  };

  const getDisplayUrl = url => {
    if (authMode !== "apiKey") return url;
    if (apiKeyLocation !== "query") return url;

    try {
      const parsed = new URL(url);
      parsed.searchParams.set(apiKeyQueryName || "api_key", "••••••••");
      return parsed.toString();
    } catch (e) {
      return url;
    }
  };

  const applyEditableHeaders = headers => {
    if (!headersText || !headersText.trim()) return headers;

    try {
      const parsedHeaders = JSON.parse(headersText);

      Object.entries(parsedHeaders).forEach(entry => {
        if (entry[0] && entry[1] !== undefined && entry[1] !== null) {
          headers[entry[0]] = String(entry[1]);
        }
      });

      return headers;
    } catch (e) {
      throw new Error("Headers JSON is not valid.");
    }
  };

  const applyAuthHeaders = headers => {
    if (authMode === "bearer" && bearerToken.trim()) {
      headers.Authorization = "Bearer " + bearerToken.trim();
    }

    if (authMode === "apiKey" && apiKeyLocation === "header" && apiKeyValue.trim()) {
      headers[apiKeyHeaderName || "x-api-key"] = apiKeyValue.trim();
    }

    if (authMode === "basic" && basicUsername.trim()) {
      headers.Authorization = "Basic " + btoa(basicUsername.trim() + ":" + basicPassword);
    }

    return headers;
  };

  const runRequest = async () => {
    if (!endpoint) return;

    setRunning(true);
    setResult(null);

    try {
      const url = buildUrl();
      const headers = {};

      applyEditableHeaders(headers);

      headerParams.forEach(param => {
        const lowerName = safeString(param.name).toLowerCase();

        if (lowerName === "authorization") return;
        if (lowerName === "accept") return;
        if (lowerName === "content-type") return;

        const value = headerValues[param.name];

        if (value !== undefined && value !== null && String(value).trim() !== "") {
          headers[param.name] = value;
        }
      });

      applyAuthHeaders(headers);

      const options = {
        method: endpoint.method,
        headers: headers
      };

      const methodAllowsBody =
        endpoint.method !== "GET" && endpoint.method !== "HEAD";

      if (methodAllowsBody && requestBody.trim()) {
        headers["Content-Type"] = "application/json";
        options.body = requestBody;
      }

      const startedAt = Date.now();
      const response = await fetch(url, options);
      const duration = Date.now() - startedAt;
      const contentType = response.headers.get("content-type") || "";

      let bodyText = "";

      if (contentType.indexOf("application/json") > -1) {
        const json = await response.json();
        bodyText = prettyJson(json);
      } else {
        bodyText = await response.text();
      }

      setResult({
        ok: response.ok,
        status: response.status,
        statusText: response.statusText,
        duration: duration,
        url: getDisplayUrl(url),
        body: bodyText || "(empty response)"
      });
    } catch (err) {
      setResult({
        ok: false,
        status: "Request failed",
        statusText: "",
        duration: 0,
        url: "",
        body: err && err.message ? err.message : "Request failed."
      });
    }

    setRunning(false);
  };

  const methodColor = method => {
    if (method === "GET") return "#2563eb";
    if (method === "POST") return "#16a34a";
    if (method === "PUT") return "#ca8a04";
    if (method === "PATCH") return "#ea580c";
    if (method === "DELETE") return "#dc2626";
    return "#4b5563";
  };

  const getAuthModeLabel = value => {
    if (value === "none") return "No auth";
    if (value === "bearer") return "Bearer token";
    if (value === "apiKey") return "API key";
    if (value === "basic") return "Basic auth";
    return value;
  };

  const styles =
    ".endpointBlock{font-family:-apple-system,BlinkMacSystemFont,'Segoe UI',sans-serif;color:var(--color-text-default,#111827);border:1px solid var(--color-border-default,#e5e7eb);border-radius:14px;padding:16px;background:var(--color-bg-page,#fff);box-shadow:0 4px 14px rgba(0,0,0,.05);max-width:760px}" +
    ".endpointBlockTop{display:flex;align-items:flex-start;justify-content:space-between;gap:12px;border-bottom:1px solid var(--color-border-default,#e5e7eb);padding-bottom:12px;margin-bottom:12px}" +
    ".endpointBlock h3{margin:0 0 6px 0;font-size:1.15rem}" +
    ".endpointBlock p{margin:0;color:var(--color-text-muted,#6b7280)}" +
    ".endpointMethod{display:inline-block;color:#fff;border-radius:999px;font-size:11px;font-weight:800;padding:4px 8px;margin-right:8px}" +
    ".endpointPath{font-family:ui-monospace,SFMono-Regular,Menlo,Monaco,Consolas,monospace;font-size:13px}" +
    ".endpointPill{display:inline-block;border:1px solid var(--color-border-default,#d1d5db);border-radius:999px;padding:3px 8px;font-size:12px;font-weight:700;background:var(--color-bg-muted,#f9fafb);white-space:nowrap}" +
    ".endpointField{margin-top:12px}" +
    ".endpointField label{display:block;font-weight:700;margin-bottom:5px}" +
    ".endpointInput,.endpointSelect,.endpointTextarea{width:100%;box-sizing:border-box;border:1px solid var(--color-border-default,#d1d5db);border-radius:8px;padding:9px 10px;background:var(--color-bg-page,#fff);color:var(--color-text-default,#111827);font:inherit}" +
    ".endpointTextarea{min-height:120px;font-family:ui-monospace,SFMono-Regular,Menlo,Monaco,Consolas,monospace;font-size:13px}" +
    ".endpointHelp{font-size:12px;color:var(--color-text-muted,#6b7280);margin-top:4px}" +
    ".endpointAuth{margin-top:14px;border:1px solid var(--color-border-default,#e5e7eb);border-radius:12px;padding:12px;background:var(--color-bg-muted,#f9fafb)}" +
    ".endpointAuthTop{display:flex;align-items:center;justify-content:space-between;gap:10px}" +
    ".endpointGrid{display:grid;grid-template-columns:1fr 1fr;gap:10px}" +
    ".endpointControls{display:flex;gap:8px;flex-wrap:wrap;margin-top:14px}" +
    ".endpointButton{border:1px solid var(--color-border-default,#d1d5db);border-radius:8px;padding:9px 13px;font-weight:700;cursor:pointer;background:var(--color-bg-page,#fff);color:var(--color-text-default,#111827)}" +
    ".endpointButton:disabled{opacity:.6;cursor:not-allowed}" +
    ".endpointRun{background:#111827;color:#fff;border-color:#111827}" +
    ".endpointResult{margin-top:16px;border-radius:12px;border:1px solid var(--color-border-default,#e5e7eb);overflow:hidden}" +
    ".endpointResultHead{padding:10px 12px;font-weight:700;background:var(--color-bg-muted,#f3f4f6)}" +
    ".endpointPre{margin:0;padding:12px;overflow:auto;max-height:420px;background:var(--color-bg-muted,#f9fafb);font-family:ui-monospace,SFMono-Regular,Menlo,Monaco,Consolas,monospace;font-size:13px;white-space:pre-wrap}" +
    ".endpointError{color:#dc2626;font-weight:700}" +
    "@media(max-width:700px){.endpointGrid{grid-template-columns:1fr}.endpointBlockTop{flex-direction:column}}";

  return (
    <div className="endpointBlock">
      <style>{styles}</style>

      {loadingSpec && <p>Loading endpoint...</p>}

      {error && <p className="endpointError">{error}</p>}

      {!loadingSpec && !error && endpoint && (
        <div>
          <div className="endpointBlockTop">
            <div>
              <h3>
                <span
                  className="endpointMethod"
                  style={{ background: methodColor(endpoint.method) }}
                >
                  {endpoint.method}
                </span>
                <span className="endpointPath">{endpoint.path}</span>
              </h3>
              {endpoint.summary && <p>{endpoint.summary}</p>}
            </div>
            <span className="endpointPill">{getAuthModeLabel(authMode)}</span>
          </div>

          <div className="endpointField">
            <label>API base URL</label>
            <input
              className="endpointInput"
              value={apiBaseUrlInput}
              onChange={event => setApiBaseUrlInput(event.target.value)}
              placeholder="https://api.example.com"
            />
          </div>

          <div className="endpointAuth">
            <div className="endpointAuthTop">
              <strong>Authorization</strong>
              <span className="endpointPill">{getAuthModeLabel(authMode)}</span>
            </div>

            <div className="endpointField">
              <label>Auth type</label>
              <select
                className="endpointSelect"
                value={authMode}
                onChange={event => setAuthMode(event.target.value)}
              >
                <option value="none">No auth</option>
                <option value="bearer">Bearer token</option>
                <option value="apiKey">API key</option>
                <option value="basic">Basic auth</option>
              </select>
            </div>

            {authMode === "bearer" && (
              <div className="endpointField">
                <label>Bearer token</label>
                <input
                  className="endpointInput"
                  type="password"
                  value={bearerToken}
                  onChange={event => setBearerToken(event.target.value)}
                  placeholder="Paste bearer token"
                />
              </div>
            )}

            {authMode === "apiKey" && (
              <div>
                <div className="endpointGrid">
                  <div className="endpointField">
                    <label>API key location</label>
                    <select
                      className="endpointSelect"
                      value={apiKeyLocation}
                      onChange={event => setApiKeyLocation(event.target.value)}
                    >
                      <option value="header">Header</option>
                      <option value="query">Query param</option>
                    </select>
                  </div>

                  {apiKeyLocation === "header" && (
                    <div className="endpointField">
                      <label>Header name</label>
                      <input
                        className="endpointInput"
                        value={apiKeyHeaderName}
                        onChange={event => setApiKeyHeaderName(event.target.value)}
                        placeholder="x-api-key"
                      />
                    </div>
                  )}

                  {apiKeyLocation === "query" && (
                    <div className="endpointField">
                      <label>Query name</label>
                      <input
                        className="endpointInput"
                        value={apiKeyQueryName}
                        onChange={event => setApiKeyQueryName(event.target.value)}
                        placeholder="api_key"
                      />
                    </div>
                  )}
                </div>

                <div className="endpointField">
                  <label>API key</label>
                  <input
                    className="endpointInput"
                    type="password"
                    value={apiKeyValue}
                    onChange={event => setApiKeyValue(event.target.value)}
                    placeholder="Paste API key"
                  />
                </div>
              </div>
            )}

            {authMode === "basic" && (
              <div className="endpointGrid">
                <div className="endpointField">
                  <label>Username / API key</label>
                  <input
                    className="endpointInput"
                    type="password"
                    value={basicUsername}
                    onChange={event => setBasicUsername(event.target.value)}
                    placeholder="Username or API key"
                  />
                </div>

                <div className="endpointField">
                  <label>Password</label>
                  <input
                    className="endpointInput"
                    type="password"
                    value={basicPassword}
                    onChange={event => setBasicPassword(event.target.value)}
                    placeholder="Optional"
                  />
                  <div className="endpointHelp">
                    Some APIs use the API key as username and a blank password.
                  </div>
                </div>
              </div>
            )}
          </div>

          {pathParams.map(param => (
            <div className="endpointField" key={"path-" + param.name}>
              <label>
                Path: {param.name}
                {param.required ? " *" : ""}
              </label>
              <input
                className="endpointInput"
                value={pathValues[param.name] || ""}
                onChange={event =>
                  setPathValues(
                    Object.assign({}, pathValues, {
                      [param.name]: event.target.value
                    })
                  )
                }
              />
            </div>
          ))}

          {queryParams.map(param => (
            <div className="endpointField" key={"query-" + param.name}>
              <label>
                Query: {param.name}
                {param.required ? " *" : ""}
              </label>
              <input
                className="endpointInput"
                value={queryValues[param.name] || ""}
                onChange={event =>
                  setQueryValues(
                    Object.assign({}, queryValues, {
                      [param.name]: event.target.value
                    })
                  )
                }
              />
            </div>
          ))}

          {headerParams.map(param => {
            const lowerName = safeString(param.name).toLowerCase();

            if (lowerName === "authorization") return null;
            if (lowerName === "accept") return null;
            if (lowerName === "content-type") return null;

            return (
              <div className="endpointField" key={"header-" + param.name}>
                <label>
                  Header: {param.name}
                  {param.required ? " *" : ""}
                </label>
                <input
                  className="endpointInput"
                  value={headerValues[param.name] || ""}
                  onChange={event =>
                    setHeaderValues(
                      Object.assign({}, headerValues, {
                        [param.name]: event.target.value
                      })
                    )
                  }
                />
              </div>
            );
          })}

          {endpoint.method !== "GET" && endpoint.method !== "HEAD" && (
            <div className="endpointField">
              <label>JSON body</label>
              <textarea
                className="endpointTextarea"
                value={requestBody}
                onChange={event => setRequestBody(event.target.value)}
                placeholder='{"example":"value"}'
              />
            </div>
          )}

          <div className="endpointControls">
            <button
              className="endpointButton"
              onClick={() => setShowAdvanced(!showAdvanced)}
              type="button"
            >
              {showAdvanced ? "Hide advanced" : "Show advanced"}
            </button>

            <button
              className="endpointButton endpointRun"
              onClick={runRequest}
              disabled={running}
            >
              {running ? "Running..." : "Run request"}
            </button>
          </div>

          {showAdvanced && (
            <div className="endpointField">
              <label>Default headers JSON</label>
              <textarea
                className="endpointTextarea"
                value={headersText}
                onChange={event => setHeadersText(event.target.value)}
                placeholder='{"Accept":"application/json"}'
              />
              <div className="endpointHelp">
                Use this for Accept, API version, or other static headers.
              </div>
            </div>
          )}

          {result && (
            <div className="endpointResult">
              <div className="endpointResultHead">
                {result.status} {result.statusText}
                {result.duration ? " · " + result.duration + "ms" : ""}
              </div>
              {result.url && <pre className="endpointPre">{result.url}</pre>}
              <pre className="endpointPre">{result.body}</pre>
            </div>
          )}
        </div>
      )}
    </div>
  );
};
