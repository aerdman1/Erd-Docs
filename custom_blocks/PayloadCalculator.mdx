name: PayloadCalculator
---
import React, { useState } from "react";

export const PayloadCalculator = props => {
  const [type, setType] = useState(props.type === "bulk_cargo" ? "bulk_cargo" : "satellite");
  const [mass, setMass] = useState(Number(props.mass) > 0 ? Number(props.mass) : 4);
  const [orbit, setOrbit] = useState(props.orbit || "sso");
  const [insurance, setInsurance] = useState("standard");
  const [pressurized, setPressurized] = useState(true);
  const [name, setName] = useState("");
  const [tab, setTab] = useState("curl");
  const [copied, setCopied] = useState(false);

  const ORBITS = { leo: ["LEO", 6000], sso: ["SSO", 6800], meo: ["MEO", 11000], geo: ["GEO", 19500] };
  const VEHICLES = [
    { id: "wren", cap: { leo: 900, sso: 700, meo: 300, geo: 150 } },
    { id: "kestrel", cap: { leo: 4000, sso: 3200, meo: 1800, geo: 900 } },
    { id: "kestrel-heavy", cap: { leo: 6400, sso: 5200, meo: 3400, geo: 2100 } },
  ];
  const INS = { none: 0, standard: 0.04, premium: 0.09 };
  const MAX = 6400;
  const effOrbit = type === "bulk_cargo" ? "leo" : orbit;
  const m = Math.max(1, Math.min(MAX, Math.round(mass) || 1));
  const fits = VEHICLES.map(v => ({ ...v, ok: m <= v.cap[effOrbit] }));
  const pick = fits.find(v => v.ok);
  const base = Math.max(25000, m * ORBITS[effOrbit][1]);
  const total = Math.round(base * (1 + INS[insurance]));
  const money = n => "$" + n.toLocaleString("en-US");
  const toSlider = kg => Math.round((Math.log(kg) / Math.log(MAX)) * 1000);
  const fromSlider = v => Math.max(1, Math.round(Math.pow(MAX, v / 1000)));
  const label = name || (type === "satellite" ? "Pixel Owl 3U" : "Hydroponics resupply");

  const payload = type === "satellite"
    ? { type: "satellite", name: label, mass_kg: m, target_orbit: effOrbit }
    : { type: "bulk_cargo", name: label, mass_kg: m, pressurized: pressurized, hazmat_class: "none" };
  const body = { launch_id: "lch_9Kd4Mw", payload: payload, insurance: insurance };
  const json = JSON.stringify(body, null, 2);
  const code = {
    json: json,
    curl: "curl -X POST https://api.us-east.orbit.erdman.example/v1/bookings \\\n  -H \"X-Orbit-Key: $ORBIT_KEY\" \\\n  -H \"Idempotency-Key: $(uuidgen)\" \\\n  -H \"Content-Type: application/json\" \\\n  -d '" + JSON.stringify(body) + "'",
    node: "const booking = await fetch('https://api.us-east.orbit.erdman.example/v1/bookings', {\n  method: 'POST',\n  headers: {\n    'X-Orbit-Key': process.env.ORBIT_KEY,\n    'Idempotency-Key': crypto.randomUUID(),\n    'Content-Type': 'application/json',\n  },\n  body: JSON.stringify(" + json.replace(/\n/g, "\n  ") + "),\n}).then(r => r.json());",
  }[tab];

  const copy = () => {
    let ok = false;
    try {
      const ta = document.createElement("textarea");
      ta.value = code; ta.setAttribute("readonly", ""); ta.style.position = "fixed"; ta.style.opacity = "0";
      document.body.appendChild(ta); ta.select(); ok = document.execCommand("copy"); ta.remove();
    } catch (e) {}
    if (!ok && navigator.clipboard) navigator.clipboard.writeText(code).catch(() => {});
    setCopied(true);
    window.setTimeout(() => setCopied(false), 1400);
  };

  const css = `
.eo-calc{--eo-accent:var(--color-link-primary,#2F7BF5);--eo-border:var(--color-border-default,rgba(0,0,0,.12));--eo-muted:var(--color-text-minimum,#6b7280);
  display:grid;grid-template-columns:minmax(0,1fr) minmax(0,1.1fr);border:1px solid var(--eo-border);border-radius:16px;overflow:hidden;margin:24px 0}
.eo-calc-in{padding:20px 22px;display:flex;flex-direction:column;gap:16px}
.eo-calc-out{padding:20px 22px;background:color-mix(in srgb,var(--eo-accent) 6%,transparent);border-left:1px solid var(--eo-border);display:flex;flex-direction:column;gap:14px;min-width:0}
.eo-calc label,.eo-calc .eo-calc-lbl{display:block;font-size:12px;font-weight:700;letter-spacing:.06em;text-transform:uppercase;color:var(--eo-muted);margin-bottom:6px}
.eo-calc-seg{display:flex;border:1px solid var(--eo-border);border-radius:10px;padding:3px;gap:3px}
.eo-calc-seg button{flex:1;border:0;background:transparent;color:inherit;border-radius:7px;padding:7px 8px;font:inherit;font-size:14px;cursor:pointer}
.eo-calc-seg button[aria-pressed="true"]{background:var(--eo-accent);color:#fff;font-weight:600}
.eo-calc-seg button:disabled{opacity:.4;cursor:not-allowed}
.eo-calc-mass{display:flex;align-items:center;gap:12px}
.eo-calc-mass input[type=range]{flex:1;accent-color:var(--eo-accent)}
.eo-calc-num{width:92px;border:1px solid var(--eo-border);border-radius:8px;padding:6px 8px;background:transparent;color:inherit;font:inherit;text-align:right}
.eo-calc-text{width:100%;box-sizing:border-box;border:1px solid var(--eo-border);border-radius:8px;padding:8px 10px;background:transparent;color:inherit;font:inherit}
.eo-calc-check{display:flex!important;align-items:center;gap:8px;text-transform:none!important;letter-spacing:0!important;font-size:14px!important;font-weight:500!important;color:inherit!important}
.eo-calc-price{font-size:34px;font-weight:800;letter-spacing:-.01em;line-height:1}
.eo-calc-break{font-size:13px;color:var(--eo-muted);margin-top:4px}
.eo-calc-veh{display:grid;grid-template-columns:repeat(3,1fr);gap:8px}
.eo-calc-v{border:1px solid var(--eo-border);border-radius:10px;padding:8px 10px;font-size:13px;background:var(--color-bg-page,transparent);position:relative}
.eo-calc-v b{display:block;font-size:14px}
.eo-calc-v[data-ok="false"]{opacity:.5}
.eo-calc-v[data-pick="true"]{border-color:var(--eo-accent);box-shadow:0 0 0 3px color-mix(in srgb,var(--eo-accent) 18%,transparent)}
.eo-calc-v em{position:absolute;top:-9px;right:8px;font-style:normal;font-size:10px;font-weight:800;letter-spacing:.06em;background:var(--eo-accent);color:#fff;padding:1px 6px;border-radius:999px}
.eo-calc-err{border-radius:10px;padding:10px 12px;background:rgba(240,68,56,.12);border-left:4px solid #F04438;font-size:14px}
.eo-calc-code{border-radius:12px;overflow:hidden;background:#0B0F1E;color:#E6EDF7;min-width:0}
.eo-calc-tabs{display:flex;align-items:center;gap:2px;padding:6px 8px;border-bottom:1px solid rgba(255,255,255,.1)}
.eo-calc-tabs button{border:0;background:transparent;color:rgba(230,237,247,.65);font:inherit;font-size:13px;padding:5px 10px;border-radius:6px;cursor:pointer}
.eo-calc-tabs button[aria-selected="true"]{background:rgba(255,255,255,.1);color:#fff}
.eo-calc-copy{margin-left:auto}
.eo-calc-code pre{margin:0!important;padding:12px 14px!important;background:transparent!important;color:inherit!important;font:12.5px/1.55 var(--font-family-mono,ui-monospace,Menlo,monospace);overflow:auto;max-height:260px;white-space:pre}
.eo-calc-fine{font-size:12px;color:var(--eo-muted)}
@media (max-width:760px){.eo-calc{grid-template-columns:1fr}.eo-calc-out{border-left:0;border-top:1px solid var(--eo-border)}}
`;
  return (
    <div className="eo-calc">
      <style>{css}</style>
      <div className="eo-calc-in">
        <div>
          <span className="eo-calc-lbl">Payload</span>
          <div className="eo-calc-seg" role="group" aria-label="Payload type">
            <button type="button" aria-pressed={type === "satellite"} onClick={() => setType("satellite")}>🛰️ Satellite</button>
            <button type="button" aria-pressed={type === "bulk_cargo"} onClick={() => setType("bulk_cargo")}>📦 Bulk cargo</button>
          </div>
        </div>
        <div>
          <label htmlFor="eo-calc-mass">Mass</label>
          <div className="eo-calc-mass">
            <input id="eo-calc-mass" type="range" min="0" max="1000" value={toSlider(m)} onChange={e => setMass(fromSlider(Number(e.target.value)))} aria-label="Mass in kilograms" />
            <span><input className="eo-calc-num" type="number" min="1" max={MAX} value={m} onChange={e => setMass(Number(e.target.value))} aria-label="Mass in kg" /> kg</span>
          </div>
        </div>
        <div>
          <span className="eo-calc-lbl">Target orbit</span>
          <div className="eo-calc-seg" role="group" aria-label="Target orbit">
            {Object.keys(ORBITS).map(o => (
              <button key={o} type="button" aria-pressed={effOrbit === o} disabled={type === "bulk_cargo" && o !== "leo"} onClick={() => setOrbit(o)}>{ORBITS[o][0]}</button>
            ))}
          </div>
          {type === "bulk_cargo" ? <div className="eo-calc-fine" style={{ marginTop: "6px" }}>Bulk cargo flies to stations in LEO.</div> : null}
        </div>
        <div>
          <span className="eo-calc-lbl">Insurance</span>
          <div className="eo-calc-seg" role="group" aria-label="Insurance">
            {Object.keys(INS).map(k => (
              <button key={k} type="button" aria-pressed={insurance === k} onClick={() => setInsurance(k)}>{k[0].toUpperCase() + k.slice(1)}</button>
            ))}
          </div>
        </div>
        <div>
          <label htmlFor="eo-calc-name">Name</label>
          <input id="eo-calc-name" className="eo-calc-text" type="text" value={name} placeholder={label} onChange={e => setName(e.target.value.slice(0, 60))} />
          {type === "bulk_cargo" ? (
            <label className="eo-calc-check" style={{ marginTop: "10px" }}><input type="checkbox" checked={pressurized} onChange={e => setPressurized(e.target.checked)} /> Needs a pressurized hold</label>
          ) : null}
        </div>
      </div>
      <div className="eo-calc-out">
        <div>
          <span className="eo-calc-lbl">Estimated price</span>
          <div className="eo-calc-price" aria-live="polite">{pick ? money(total) : "—"}</div>
          <div className="eo-calc-break">{m.toLocaleString("en-US")} kg × {money(ORBITS[effOrbit][1])}/kg to {ORBITS[effOrbit][0]}{base === 25000 && m * ORBITS[effOrbit][1] < 25000 ? " (minimum charge)" : ""}{INS[insurance] ? " + " + Math.round(INS[insurance] * 100) + "% insurance" : ""}</div>
        </div>
        <div className="eo-calc-veh">
          {fits.map(v => (
            <div key={v.id} className="eo-calc-v" data-ok={v.ok ? "true" : "false"} data-pick={pick && pick.id === v.id ? "true" : "false"}>
              {pick && pick.id === v.id ? <em>BEST FIT</em> : null}
              <b>{v.ok ? "✓" : "✗"} {v.id}</b>
              up to {v.cap[effOrbit].toLocaleString("en-US")} kg
            </div>
          ))}
        </div>
        {pick ? (
          <div className="eo-calc-code">
            <div className="eo-calc-tabs" role="tablist">
              {[["curl", "cURL"], ["node", "Node.js"], ["json", "JSON body"]].map(t => (
                <button key={t[0]} type="button" role="tab" aria-selected={tab === t[0]} onClick={() => setTab(t[0])}>{t[1]}</button>
              ))}
              <button type="button" className="eo-calc-copy" onClick={copy}>{copied ? "✓ Copied" : "Copy"}</button>
            </div>
            <pre>{code}</pre>
          </div>
        ) : (
          <div className="eo-calc-err"><b>422: delta-v exceeded.</b> No vehicle can lift {m.toLocaleString("en-US")} kg to {ORBITS[effOrbit][0]}. Split the payload or pick a lower orbit. <a href="/docs/errors-and-rate-limits#delta-v-exceeded">About this error</a></div>
        )}
        <div className="eo-calc-fine">Prices are made up, like the rest of Orbit. The request is exactly what <a href="/reference/createbooking">Book a payload slot</a> expects.</div>
      </div>
    </div>
  );
};

<PayloadCalculator />
