---
name: BTMarketplace
---
import React, { useMemo, useState } from 'react';

export const BtMarketplace = () => {
  const masterRows = [
    {
      btProds: "AD Bridge",
      extInts: "Apache",
      TitleForInt: "AD Bridge",
      TitleForProd: "Apache HTTP Server",
      html: "https://docs.beyondtrust.com/adb/docs/apache"
    },
    {
      btProds: "AD Bridge; Password Safe",
      extInts: "",
      TitleForInt: "",
      TitleForProd: "AD Bridge and Password Safe",
      html: "https://docs.beyondtrust.com/adb/docs/ps-integration"
    },
    {
      btProds: "AD Bridge",
      extInts: "Elasticsearch; Logstash",
      TitleForInt: "AD Bridge",
      TitleForProd: "Elasticsearch or Logstash",
      html: "https://docs.beyondtrust.com/adb/docs/elasticsearch-logstash-reporting"
    },
    {
      btProds: "AD Bridge",
      extInts: "NFS",
      TitleForInt: "AD Bridge",
      TitleForProd: "NFS",
      html: "https://docs.beyondtrust.com/adb/docs/nfs-integration"
    },
    {
      btProds: "AD Bridge",
      extInts: "Samba",
      TitleForInt: "AD Bridge",
      TitleForProd: "Samba",
      html: "https://docs.beyondtrust.com/adb/docs/samba"
    },
    {
      btProds: "AD Bridge",
      extInts: "Smart cards",
      TitleForInt: "AD Bridge",
      TitleForProd: "Smart cards",
      html: "https://docs.beyondtrust.com/adb/docs/smart-card"
    },
    {
      btProds: "AD Bridge",
      extInts: "SSH",
      TitleForInt: "AD Bridge",
      TitleForProd: "SSH",
      html: "https://docs.beyondtrust.com/adb/docs/ssh"
    },
    {
      btProds: "BeyondInsight for Unix and Linux",
      extInts: "SailPoint",
      TitleForInt: "BeyondInsight for Unix and Linux - Identity IQ",
      TitleForProd: "SailPoint IdentityIQ",
      html: "https://docs.beyondtrust.com/epm-ul/docs/sailpoint-identity-iq"
    },
    {
      btProds: "BeyondInsight for Unix and Linux",
      extInts: "SailPoint",
      TitleForInt: "BeyondInsight for Unix and Linux - IdentityNow",
      TitleForProd: "SailPoint IdentityNow",
      html: "https://docs.beyondtrust.com/epm-ul/docs/identitynow"
    },
    {
      btProds: "Endpoint Privilege Management for Linux Cloud",
      extInts: "Elasticsearch",
      TitleForInt: "Endpoint Privilege Management for Linux Cloud - Unified search",
      TitleForProd: "Elasticsearch - Unified search",
      html: "https://docs.beyondtrust.com/epm-l/docs/unified-search#add-a-new-elasticsearch-siem-connection"
    }
  ];

  // --- helpers --------------------------------------------------

  const splitField = (value) => {
    if (!value) return [];
    return value
      .split(';')
      .map((v) => v.trim())
      .filter(Boolean);
  };

  const buildItems = (rows) => {
    return rows.map((row, index) => {
      const btProducts = splitField(row.btProds);
      const partners = splitField(row.extInts);

      const title =
        row.TitleForInt ||
        row.TitleForProd ||
        [btProducts.join(', '), partners.join(', ')].filter(Boolean).join(' + ');

      return {
        id: index,
        title,
        btProducts,
        partners,
        url: row.html
      };
    });
  };

  const items = useMemo(() => buildItems(masterRows), []);

  const allBtProducts = useMemo(() => {
    const set = new Set();
    items.forEach((item) => {
      item.btProducts.forEach((p) => set.add(p));
    });
    return Array.from(set).sort();
  }, [items]);

  const allPartners = useMemo(() => {
    const set = new Set();
    items.forEach((item) => {
      item.partners.forEach((p) => set.add(p));
    });
    return Array.from(set).sort();
  }, [items]);

  const counts = useMemo(() => {
    const productCounts = {};
    const partnerCounts = {};

    items.forEach((item) => {
      item.btProducts.forEach((p) => {
        productCounts[p] = (productCounts[p] || 0) + 1;
      });
      item.partners.forEach((p) => {
        partnerCounts[p] = (partnerCounts[p] || 0) + 1;
      });
    });

    return { productCounts, partnerCounts };
  }, [items]);

  const [selectedBtProducts, setSelectedBtProducts] = useState([]);
  const [selectedPartners, setSelectedPartners] = useState([]);
  const [search, setSearch] = useState('');

  const toggleFilter = (value, current, setFn) => {
    if (current.includes(value)) {
      setFn(current.filter((v) => v !== value));
    } else {
      setFn([...current, value]);
    }
  };

  const clearAll = () => {
    setSelectedBtProducts([]);
    setSelectedPartners([]);
    setSearch('');
  };

  const matchesFilters = (item) => {
    const searchLower = search.trim().toLowerCase();

    if (searchLower) {
      const haystack = [item.title, item.btProducts.join(' '), item.partners.join(' ')]
        .join(' ')
        .toLowerCase();

      if (!haystack.includes(searchLower)) return false;
    }

    if (selectedBtProducts.length) {
      const hasProduct = item.btProducts.some((p) => selectedBtProducts.includes(p));
      if (!hasProduct) return false;
    }

    if (selectedPartners.length) {
      const hasPartner = item.partners.some((p) => selectedPartners.includes(p));
      if (!hasPartner) return false;
    }

    return true;
  };

  const filteredItems = items.filter(matchesFilters);
  const totalCount = items.length;
  const visibleCount = filteredItems.length;

  const hasAnyFilters = Boolean(
    search.trim() || selectedBtProducts.length || selectedPartners.length
  );

  // --- render ---------------------------------------------------

  return (
    <div className="bg-slate-50 text-slate-900 p-6 md:p-8 rounded-2xl border border-slate-200 shadow-sm font-sans w-full not-prose">
      {/* Header */}
      <div className="mb-6 md:mb-8">
        <div className="flex flex-col lg:flex-row lg:items-end lg:justify-between gap-4">
          <div className="min-w-0">
            <div className="flex items-center gap-3">
              <h1 className="text-2xl md:text-3xl font-semibold tracking-tight m-0 border-none">
                BeyondTrust Marketplace
              </h1>
              <span className="inline-flex items-center rounded-full border border-slate-200 bg-white px-2.5 py-1 text-xs font-semibold text-slate-700">
                {visibleCount} of {totalCount}
              </span>
            </div>
            <p className="text-sm text-slate-600 mt-2 mb-0">
              Filter integrations by BeyondTrust product and partner/technology, or search across everything.
            </p>
          </div>

          <div className="w-full lg:w-[420px]">
            <div className="relative">
              <span className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400">
                <svg width="16" height="16" viewBox="0 0 24 24" fill="none" aria-hidden="true">
                  <path
                    d="M10.5 18a7.5 7.5 0 1 1 0-15 7.5 7.5 0 0 1 0 15Z"
                    stroke="currentColor"
                    strokeWidth="2"
                  />
                  <path
                    d="M16.5 16.5 21 21"
                    stroke="currentColor"
                    strokeWidth="2"
                    strokeLinecap="round"
                  />
                </svg>
              </span>
              <input
                type="search"
                placeholder="Search integrations (product, partner, tech, title)…"
                value={search}
                onChange={(e) => setSearch(e.target.value)}
                className="w-full rounded-xl bg-white border border-slate-200 pl-10 pr-3 py-2.5 text-sm text-slate-900 placeholder:text-slate-400 shadow-sm focus:outline-none focus:ring-2 focus:ring-indigo-500/20 focus:border-indigo-400"
              />
            </div>

            <div className="mt-3 flex items-center justify-between gap-3">
              <div className="text-xs text-slate-500">
                Tip: select multiple filters to narrow results.
              </div>
              {hasAnyFilters && (
                <button
                  type="button"
                  onClick={clearAll}
                  className="text-xs font-semibold text-slate-700 hover:text-slate-900 underline underline-offset-4"
                >
                  Clear all
                </button>
              )}
            </div>
          </div>
        </div>

        {/* Selected chips */}
        {hasAnyFilters && (
          <div className="mt-5 flex flex-wrap gap-2">
            {search.trim() && (
              <span className="inline-flex items-center gap-2 rounded-full border border-slate-200 bg-white px-3 py-1.5 text-xs text-slate-700">
                <span className="font-semibold">Search:</span>
                <span className="max-w-[320px] truncate">{search.trim()}</span>
                <button
                  type="button"
                  onClick={() => setSearch('')}
                  className="rounded-full p-0.5 hover:bg-slate-100"
                  aria-label="Clear search"
                >
                  <svg width="14" height="14" viewBox="0 0 24 24" fill="none" aria-hidden="true">
                    <path
                      d="M7 7 17 17M17 7 7 17"
                      stroke="currentColor"
                      strokeWidth="2"
                      strokeLinecap="round"
                    />
                  </svg>
                </button>
              </span>
            )}

            {selectedBtProducts.map((p) => (
              <span
                key={p}
                className="inline-flex items-center gap-2 rounded-full border border-indigo-200 bg-indigo-50 px-3 py-1.5 text-xs text-indigo-900"
              >
                <span className="font-semibold">Product</span>
                <span className="truncate max-w-[260px]">{p}</span>
                <button
                  type="button"
                  onClick={() => toggleFilter(p, selectedBtProducts, setSelectedBtProducts)}
                  className="rounded-full p-0.5 hover:bg-indigo-100"
                  aria-label={"Remove product filter " + p}
                >
                  <svg width="14" height="14" viewBox="0 0 24 24" fill="none" aria-hidden="true">
                    <path
                      d="M7 7 17 17M17 7 7 17"
                      stroke="currentColor"
                      strokeWidth="2"
                      strokeLinecap="round"
                    />
                  </svg>
                </button>
              </span>
            ))}

            {selectedPartners.map((p) => (
              <span
                key={p}
                className="inline-flex items-center gap-2 rounded-full border border-emerald-200 bg-emerald-50 px-3 py-1.5 text-xs text-emerald-900"
              >
                <span className="font-semibold">Partner</span>
                <span className="truncate max-w-[260px]">{p}</span>
                <button
                  type="button"
                  onClick={() => toggleFilter(p, selectedPartners, setSelectedPartners)}
                  className="rounded-full p-0.5 hover:bg-emerald-100"
                  aria-label={"Remove partner filter " + p}
                >
                  <svg width="14" height="14" viewBox="0 0 24 24" fill="none" aria-hidden="true">
                    <path
                      d="M7 7 17 17M17 7 7 17"
                      stroke="currentColor"
                      strokeWidth="2"
                      strokeLinecap="round"
                    />
                  </svg>
                </button>
              </span>
            ))}
          </div>
        )}
      </div>

      <div className="flex flex-col lg:flex-row gap-6 lg:gap-8">
        {/* Sidebar Filters */}
        <div className="w-full lg:w-80 shrink-0">
          <div className="lg:sticky lg:top-4 space-y-4">
            <div className="rounded-2xl border border-slate-200 bg-white shadow-sm">
              <div className="px-4 py-3 border-b border-slate-100">
                <h4 className="text-sm font-semibold text-slate-900 m-0">BeyondTrust product</h4>
                <p className="text-xs text-slate-500 mt-1 mb-0">Select one or more.</p>
              </div>

              <div className="p-3">
                <div className="flex flex-wrap gap-2 max-h-56 overflow-auto pr-1">
                  {allBtProducts.map((prod) => {
                    const active = selectedBtProducts.includes(prod);
                    return (
                      <button
                        key={prod}
                        type="button"
                        aria-pressed={active}
                        onClick={() => toggleFilter(prod, selectedBtProducts, setSelectedBtProducts)}
                        className={
                          "inline-flex items-center gap-2 rounded-full border px-3 py-1.5 text-xs font-semibold transition " +
                          (active
                            ? "border-indigo-300 bg-indigo-50 text-indigo-900"
                            : "border-slate-200 bg-white text-slate-700 hover:bg-slate-50")
                        }
                        title={prod}
                      >
                        <span className="truncate max-w-[210px]">{prod}</span>
                        <span
                          className={
                            "rounded-full px-2 py-0.5 text-[11px] font-bold " +
                            (active ? "bg-indigo-100 text-indigo-900" : "bg-slate-100 text-slate-600")
                          }
                        >
                          {counts.productCounts[prod] || 0}
                        </span>
                      </button>
                    );
                  })}
                </div>
              </div>
            </div>

            <div className="rounded-2xl border border-slate-200 bg-white shadow-sm">
              <div className="px-4 py-3 border-b border-slate-100">
                <h4 className="text-sm font-semibold text-slate-900 m-0">Partner / technology</h4>
                <p className="text-xs text-slate-500 mt-1 mb-0">Select one or more.</p>
              </div>

              <div className="p-3">
                <div className="flex flex-wrap gap-2 max-h-56 overflow-auto pr-1">
                  {allPartners.map((partner) => {
                    const active = selectedPartners.includes(partner);
                    return (
                      <button
                        key={partner}
                        type="button"
                        aria-pressed={active}
                        onClick={() => toggleFilter(partner, selectedPartners, setSelectedPartners)}
                        className={
                          "inline-flex items-center gap-2 rounded-full border px-3 py-1.5 text-xs font-semibold transition " +
                          (active
                            ? "border-emerald-300 bg-emerald-50 text-emerald-900"
                            : "border-slate-200 bg-white text-slate-700 hover:bg-slate-50")
                        }
                        title={partner}
                      >
                        <span className="truncate max-w-[210px]">{partner}</span>
                        <span
                          className={
                            "rounded-full px-2 py-0.5 text-[11px] font-bold " +
                            (active ? "bg-emerald-100 text-emerald-900" : "bg-slate-100 text-slate-600")
                          }
                        >
                          {counts.partnerCounts[partner] || 0}
                        </span>
                      </button>
                    );
                  })}
                </div>
              </div>
            </div>

            <div className="rounded-2xl border border-slate-200 bg-gradient-to-b from-white to-slate-50 p-4 shadow-sm">
              <div className="flex items-start justify-between gap-3">
                <div>
                  <div className="text-sm font-semibold text-slate-900">Results</div>
                  <div className="text-xs text-slate-600 mt-1">
                    Showing <span className="font-semibold">{visibleCount}</span> of{" "}
                    <span className="font-semibold">{totalCount}</span>
                  </div>
                </div>
                {hasAnyFilters && (
                  <button
                    type="button"
                    onClick={clearAll}
                    className="rounded-xl border border-slate-200 bg-white px-3 py-2 text-xs font-semibold text-slate-700 hover:bg-slate-50"
                  >
                    Reset
                  </button>
                )}
              </div>
            </div>
          </div>
        </div>

        {/* Main Grid */}
        <div className="flex-1 min-w-0">
          {visibleCount === 0 ? (
            <div className="rounded-2xl border border-slate-200 bg-white p-6 shadow-sm">
              <p className="text-sm text-slate-700 m-0 font-semibold">No integrations match the current filters.</p>
              <p className="text-sm text-slate-600 mt-2 mb-0">
                Try clearing a filter, or searching for a broader term.
              </p>
              {hasAnyFilters && (
                <div className="mt-4">
                  <button
                    type="button"
                    onClick={clearAll}
                    className="rounded-xl bg-slate-900 text-white px-4 py-2 text-sm font-semibold hover:bg-slate-800"
                  >
                    Clear all filters
                  </button>
                </div>
              )}
            </div>
          ) : (
            <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-4">
              {filteredItems.map((item) => (
                <a
                  key={item.id}
                  href={item.url}
                  className="group rounded-2xl border border-slate-200 bg-white shadow-sm hover:shadow-md hover:border-slate-300 transition flex flex-col h-full overflow-hidden"
                >
                  <div className="p-5 flex flex-col gap-3 flex-grow">
                    <div className="flex items-start justify-between gap-3">
                      <div className="min-w-0">
                        <h3 className="text-base font-semibold leading-snug text-slate-900 m-0 break-words group-hover:text-indigo-700">
                          {item.title}
                        </h3>

                        {item.btProducts.length > 0 && (
                          <div className="mt-2 flex flex-wrap gap-2">
                            {item.btProducts.map((p) => (
                              <span
                                key={p}
                                className="inline-flex items-center rounded-full bg-slate-100 text-slate-700 px-2.5 py-1 text-[11px] font-semibold"
                              >
                                {p}
                              </span>
                            ))}
                          </div>
                        )}
                      </div>

                      <span className="shrink-0 inline-flex items-center rounded-full border border-slate-200 bg-white px-2.5 py-1 text-[11px] font-bold text-slate-600">
                        Integration
                      </span>
                    </div>

                    {item.partners.length > 0 && (
                      <div className="mt-1">
                        <div className="text-xs font-semibold text-slate-700">Integrates with</div>
                        <div className="mt-2 flex flex-wrap gap-2">
                          {item.partners.map((p) => (
                            <span
                              key={p}
                              className="inline-flex items-center rounded-full bg-emerald-50 text-emerald-900 px-2.5 py-1 text-[11px] font-semibold border border-emerald-100"
                            >
                              {p}
                            </span>
                          ))}
                        </div>
                      </div>
                    )}

                    <div className="mt-auto pt-4">
                      <div className="flex items-center justify-between rounded-xl bg-slate-50 border border-slate-200 px-4 py-3">
                        <div className="text-xs font-semibold text-slate-700">View integration docs</div>
                        <div className="text-sm font-semibold text-indigo-700 group-hover:translate-x-0.5 transition-transform">
                          Open →
                        </div>
                      </div>
                    </div>
                  </div>

                  <div className="h-1 w-full bg-gradient-to-r from-indigo-500 via-sky-500 to-emerald-500" />
                </a>
              ))}
            </div>
          )}
        </div>
      </div>
    </div>
  );
};

<BtMarketplace />
