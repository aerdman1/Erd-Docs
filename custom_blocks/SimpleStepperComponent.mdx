import React, { useState } from "react";

export function SimpleStepperComponent({ steps }) {
  // This component is a simple stepper that shows numbered steps,
  // displays the header and content for the current step, and includes Next/Back buttons.
  
  // useState hook to track the current step index (starting at 0)
  const [currentStep, setCurrentStep] = useState(0);

  // Define the style for the main container holding the stepper.
  const containerStyle = {
    maxWidth: "600px",
    margin: "40px auto",
    padding: "20px",
    border: "1px solid #ccc",
    borderRadius: "8px",
    textAlign: "center",
    fontFamily: "Arial, sans-serif",
  };

  // Define the style for the container that holds the step indicators.
  const indicatorContainerStyle = {
    marginBottom: "20px",
  };

  // A function that returns the style for a step indicator circle.
  // The 'active' parameter is true if the indicator represents the current step.
  const indicatorStyle = (active) => ({
    display: "inline-block",
    width: "30px",
    height: "30px",
    lineHeight: "30px",
    borderRadius: "50%",
    backgroundColor: active ? "#835db9" : "#ddd",
    color: active ? "white" : "black",
    margin: "0 5px",
    fontWeight: "bold",
  });

  // Define the style for the navigation buttons.
  const buttonStyle = {
    padding: "8px 16px",
    margin: "10px",
    border: "none",
    borderRadius: "4px",
    backgroundColor: "#835db9",
    color: "white",
    cursor: "pointer",
  };

  // Handler for the "Back" button. It decrements the current step, ensuring it doesn't go below 0.
  const handlePrev = () => setCurrentStep((prev) => Math.max(prev - 1, 0));

  // Handler for the "Next" button. It increments the current step, ensuring it doesn't exceed the last step.
  const handleNext = () => setCurrentStep((prev) => Math.min(prev + 1, steps.length - 1));

  // The JSX returned below renders:
  // 1. The step indicators (numbered circles)
  // 2. The header and content for the current step
  // 3. Navigation buttons ("Back" and "Next")
  return (
    <div style={containerStyle}>
      <div style={indicatorContainerStyle}>
        {steps.map((_, index) => (
          <span key={index} style={indicatorStyle(index === currentStep)}>
            {index + 1}
          </span>
        ))}
      </div>
      <h2>{steps[currentStep].header}</h2>
      <div>{steps[currentStep].content}</div>
      <div>
        <button
          type="button"
          style={buttonStyle}
          onClick={handlePrev}
          disabled={currentStep === 0}
        >
          Back
        </button>
        <button
          type="button"
          style={buttonStyle}
          onClick={handleNext}
          disabled={currentStep === steps.length - 1}
        >
          Next
        </button>
      </div>
    </div>
  );
}

<SimpleStepperComponent
  steps={[
    { header: "Step 1: Plan", content: <p>Plan your documentation and gather resources.</p> },
    { header: "Step 2: Write", content: <p>Write effective and clear documentation.</p> },
    { header: "Step 3: Review", content: <p>Review and refine your content.</p> },
  ]}
/>
