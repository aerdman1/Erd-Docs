import React, { useState, Children, isValidElement } from "react";

export const SimpleStep = ({ header, children }) => (
  <div className="eo-step-body">
    {header ? <h3 className="eo-step-title">{header}</h3> : null}
    <div>{children}</div>
  </div>
);

export const SimpleStepper = rawProps => {
  const props = {};
  Object.keys(rawProps || {}).forEach(k => { props[k.toLowerCase()] = rawProps[k]; });
  props.children = rawProps ? rawProps.children : undefined;
  const vertical = props.vertical === true || props.vertical === "true";
  const steps = Children.toArray(props.children).filter(c => isValidElement(c));
  const [current, setCurrent] = useState(0);
  const last = Math.max(steps.length - 1, 0);
  const go = i => setCurrent(Math.min(Math.max(i, 0), last));
  const css = `
.eo-stepper{--eo-accent:var(--color-link-primary,#2F7BF5);--eo-border:var(--color-border-default,rgba(0,0,0,.12));--eo-muted:var(--color-text-minimum,#6b7280);
  border:1px solid var(--eo-border);border-radius:12px;padding:22px 24px;margin:24px 0;color:var(--color-text-default,inherit)}
.eo-stepper .eo-step-title{margin:0 0 6px;font-size:1.1em}
.eo-stepper-dots{display:flex;align-items:center;gap:0;margin-bottom:18px}
.eo-stepper-dot{flex:none;width:32px;height:32px;border-radius:50%;border:2px solid var(--eo-border);background:transparent;color:var(--eo-muted);font-family:inherit;font-weight:600;font-size:14px;line-height:28px;text-align:center;cursor:pointer;padding:0;transition:all .2s}
.eo-stepper-dot[data-state="done"]{background:var(--eo-accent);border-color:var(--eo-accent);color:#fff}
.eo-stepper-dot[data-state="current"]{border-color:var(--eo-accent);color:var(--eo-accent);box-shadow:0 0 0 4px color-mix(in srgb,var(--eo-accent) 18%,transparent)}
.eo-stepper-dot:focus-visible{outline:2px solid var(--eo-accent);outline-offset:2px}
.eo-stepper-line{flex:1;height:2px;background:var(--eo-border);margin:0 6px;border-radius:2px;overflow:hidden}
.eo-stepper-line>i{display:block;height:100%;background:var(--eo-accent);transition:width .3s}
.eo-stepper-nav{display:flex;justify-content:space-between;align-items:center;margin-top:18px;gap:12px}
.eo-stepper-count{color:var(--eo-muted);font-size:13px}
.eo-stepper-btn{border:1px solid var(--eo-border);background:transparent;color:inherit;border-radius:8px;padding:8px 16px;font-family:inherit;font-weight:600;font-size:14px;cursor:pointer}
.eo-stepper-btn[data-primary]{background:var(--eo-accent);border-color:var(--eo-accent);color:#fff}
.eo-stepper-btn:disabled{opacity:.4;cursor:default}
.eo-stepper .eo-stepper-v{list-style:none!important;margin:0!important;padding:0!important}
.eo-stepper .eo-stepper-v>li{position:relative;padding:0 0 22px 48px!important;margin:0!important;list-style:none!important}
.eo-stepper-v>li:last-child{padding-bottom:0}
.eo-stepper-v>li::before{content:attr(data-n);position:absolute;left:0;top:0;width:30px;height:30px;border-radius:50%;background:var(--eo-accent);color:#fff;font-family:inherit;font-weight:600;font-size:14px;line-height:30px;text-align:center}
.eo-stepper-v>li::after{content:"";position:absolute;left:14px;top:34px;bottom:4px;width:2px;background:var(--eo-border)}
.eo-stepper-v>li:last-child::after{display:none}
`;
  if (!steps.length) return null;
  if (vertical) {
    return (
      <div className="eo-stepper">
        <style>{css}</style>
        <ol className="eo-stepper-v">
          {steps.map((s, i) => <li key={i} data-n={i + 1}>{s}</li>)}
        </ol>
      </div>
    );
  }
  const onKey = e => {
    if (e.key === "ArrowRight") { e.preventDefault(); go(current + 1); }
    if (e.key === "ArrowLeft") { e.preventDefault(); go(current - 1); }
  };
  return (
    <div className="eo-stepper" onKeyDown={onKey}>
      <style>{css}</style>
      <div className="eo-stepper-dots" role="tablist" aria-label="步骤">
        {steps.map((s, i) => {
          const state = i < current ? "done" : i === current ? "current" : "todo";
          const dot = (
            <button key={"d" + i} type="button" role="tab" aria-selected={i === current} className="eo-stepper-dot"
              data-state={state} title={s.props && s.props.header ? String(s.props.header) : "Step " + (i + 1)}
              onClick={() => go(i)}>
              {i < current ? "✓" : i + 1}
            </button>
          );
          if (i === 0) return dot;
          return [
            <span key={"l" + i} className="eo-stepper-line"><i style={{ width: i <= current ? "100%" : "0%" }} /></span>,
            dot,
          ];
        })}
      </div>
      <div aria-live="polite">{steps[current]}</div>
      <div className="eo-stepper-nav">
        <button type="button" className="eo-stepper-btn" onClick={() => go(current - 1)} disabled={current === 0}>← 上一步</button>
        <span className="eo-stepper-count">步骤 {current + 1} 共 {steps.length}</span>
        <button type="button" className="eo-stepper-btn" data-primary="" onClick={() => go(current + 1)} disabled={current === last}>下一步 →</button>
      </div>
    </div>
  );
};

<SimpleStepper>
  <SimpleStep header="规划">
    选择发射任务和运载工具。
  </SimpleStep>

  <SimpleStep header="预订">
    预留舱位并上传舱单。
  </SimpleStep>

  <SimpleStep header="飞行">
    全程跟踪遥测数据，直至进入轨道。
  </SimpleStep>
</SimpleStepper>
