---
name: PageBanner_Advanced
---
import React, { useState, useLayoutEffect, useEffect } from "react";
// Page Banner + Editor (3/16/25)
export const PageBannerAdvanced = ({
  message: propMessage = "Hello! 👋 This is your page-specific announcement banner — customize it to share updates, reminders, or anything important.",
  bannerColor: propBannerColor = "#381D71",
  bannerTextColor: propBannerTextColor = "#ffffff",
  fontSize: propFontSize = "16px",
  fontWeight: propFontWeight = "normal",
}) => {
  // Determine admin status from hub-me script data
  let isAdmin = false;
  if (typeof document !== "undefined") {
    const hubElem = document.getElementById("hub-me");
    if (hubElem) {
      try {
        isAdmin = JSON.parse(hubElem.getAttribute("data-json")).isAdmin;
      } catch (e) {
        console.error("Error parsing hub-me data", e);
      }
    }
  }
  const initialFontWeight = propFontWeight === "normal" ? "400" : propFontWeight;

  // Check if on /custom-components/ page
  const isCustomComponentsPage =
    typeof window !== "undefined" && window.location.pathname.includes("/custom-components/");

  // Initialize state from props
  const [message, setMessage] = useState(propMessage),
    [bannerColor, setBannerColor] = useState(propBannerColor),
    [bannerTextColor, setBannerTextColor] = useState(propBannerTextColor),
    [fontSize, setFontSize] = useState(propFontSize),
    [fontWeight, setFontWeight] = useState(initialFontWeight),
    [preset, setPreset] = useState("custom"), // 'critical', 'warning', 'standard', 'custom'
    [placement, setPlacement] = useState("bottom"), // 'top' or 'bottom'
    [copied, setCopied] = useState(false),
    // For non-custom-components pages, admin toggle controls showing the generator UI.
    // For /custom-components/ pages, we'll show the generator UI regardless.
    [showGenerator, setShowGenerator] = useState(isCustomComponentsPage ? true : false),
    [headerHeight, setHeaderHeight] = useState(0);

  // Update header height
  const updateHeaderHeight = () => {
    requestAnimationFrame(() => {
      const header = document.querySelector("header.rm-Header");
      if (header) setHeaderHeight(header.offsetHeight);
    });
  };

  useLayoutEffect(() => {
    updateHeaderHeight();
    window.addEventListener("resize", updateHeaderHeight);
    return () => window.removeEventListener("resize", updateHeaderHeight);
  }, []);

  // Sync state with props changes
  useEffect(() => {
    setMessage(propMessage);
    setBannerColor(propBannerColor);
    setBannerTextColor(propBannerTextColor);
    setFontSize(propFontSize);
    setFontWeight(propFontWeight === "normal" ? "400" : propFontWeight);
  }, [propMessage, propBannerColor, propBannerTextColor, propFontSize, propFontWeight]);

  const handlePresetChange = (e) => {
    const selectedPreset = e.target.value;
    setPreset(selectedPreset);
    if (selectedPreset === "critical") {
      setBannerColor("#dc2626");
      setBannerTextColor("#ffffff");
    } else if (selectedPreset === "warning") {
      setBannerColor("#facc15");
      setBannerTextColor("#000000");
    } else if (selectedPreset === "standard") {
      setBannerColor("#2563eb");
      setBannerTextColor("#ffffff");
    } else if (selectedPreset === "custom") {
      setBannerColor("#00bcff");
      setBannerTextColor("#ffffff");
    }
  };

  const generateBannerProps = () => {
    let props = `message="${message}"`;
    props += preset !== "custom" ? ` ${preset}={true}` : ` bannerColor="${bannerColor}" bannerTextColor="${bannerTextColor}"`;
    props += ` fontSize="${fontSize}" fontWeight="${fontWeight}" placement="${placement}"`;
    return props;
  };

  const generatedSnippet = `<BannerMessage ${generateBannerProps()} />`;

  const defaultPresets = {
    critical: {
      bannerColor: "#dc2626",
      bannerTextColor: "#ffffff",
      fontSize: "16px",
      fontWeight: "normal",
      message: "Critical Alert: Please check immediately!",
    },
    warning: {
      bannerColor: "#facc15",
      bannerTextColor: "#000000",
      fontSize: "16px",
      fontWeight: "normal",
      message: "Warning Alert: There might be an issue.",
    },
    standard: {
      bannerColor: "#2563eb",
      bannerTextColor: "#ffffff",
      fontSize: "16px",
      fontWeight: "normal",
      message: "Standard Alert: Here is some information.",
    },
    custom: {
      bannerColor: "#00BCFF",
      bannerTextColor: "#ffffff",
      fontSize: "14px",
      fontWeight: "bold",
      message: "Here is my banner message - hope you enjoy!",
    },
  };
  const defaults = defaultPresets[preset];

  // Compute banner styles for public banner
  const bannerStyles = {
    background: bannerColor || defaults.bannerColor,
    color: bannerTextColor || defaults.bannerTextColor,
    fontSize: fontSize || defaults.fontSize,
    fontWeight: fontWeight || defaults.fontWeight,
    width: "100%",
    boxSizing: "border-box",
    borderRadius: "0px",
    padding: "10px",
  };

  const containerStyles = {
    maxWidth: "var(--rm-container-width, 1200px)",
    margin: "0 auto",
    padding: "1rem 1.5rem",
    textAlign: "center",
    boxSizing: "border-box",
    fontFamily: "Inter, sans-serif",
  };

  const uiStyles = {
    container: {
      maxWidth: "800px",
      padding: "2rem",
      background: "#2C303B",
      borderRadius: "12px",
      boxShadow: "0 8px 24px rgba(0, 0, 0, 0.7)",
      fontFamily: "Inter, sans-serif",
    },
    headerRow: {
      display: "flex",
      justifyContent: "space-between",
      alignItems: "center",
      marginBottom: "1.5rem",
      position: "relative",
    },
    heading: { margin: 0, color: "#adb4c1", fontSize: "1.75rem", fontWeight: 600 },
    image: { position: "absolute", top: "-23px", right: "-32px", maxWidth: "71px", height: "auto" },
    label: { display: "block", marginBottom: "0.5rem", fontWeight: 500, color: "#adb4c1", fontSize: "1rem" },
    input: {
      width: "93%",
      minHeight: "100px",
      padding: "0.5rem",
      borderRadius: "6px",
      border: "1px solid #ddd",
      color: "#adb4c1",
      fontSize: "1rem",
      background: "#384248",
      marginTop: "2px",
      transition: "border-color 0.2s ease-in-out",
      resize: "vertical",
      overflowY: "auto",
      whiteSpace: "pre-wrap",
      wordWrap: "break-word",
    },
    select: {
      width: "100%",
      padding: "0.60rem",
      paddingRight: "2.5rem",
      color: "#adb4c1",
      borderRadius: "6px",
      border: "1px solid #ddd",
      fontSize: "1rem",
      background:
        '#384248 url("data:image/svg+xml,%3Csvg width=\'10\' height=\'10\' viewBox=\'0 0 10 10\' xmlns=\'http://www.w3.org/2000/svg\'%3E%3Cpolyline points=\'1,3 5,7 9,3\' stroke=\'%23fff\' stroke-width=\'2\' fill=\'none\' stroke-linecap=\'round\' stroke-linejoin=\'round\'/%3E%3C/svg%3E") no-repeat right 0.75rem center',
      backgroundSize: "1em",
      marginTop: "2px",
      transition: "border-color 0.2s ease-in-out",
      appearance: "none",
      WebkitAppearance: "none",
      MozAppearance: "none",
    },
    row: { display: "flex", gap: "1rem", marginBottom: "1rem" },
    fourColumnRow: {
      display: "grid",
      gridTemplateColumns: "repeat(auto-fit, minmax(100px, 1fr))",
      gap: "1rem",
      marginBottom: "1rem",
    },
    colorInput: { width: "100%", height: "45px", padding: 0, border: 0, cursor: "pointer", background: "none" },
    code: {
      background: "#2d2d2d",
      color: "#ffffff",
      padding: "1rem",
      borderRadius: "6px",
      overflowX: "hidden",
      fontFamily: "monospace",
      marginBottom: "1.5rem",
      fontSize: "0.95rem",
    },
    button: {
      padding: "0.75rem 1.5rem",
      cursor: "pointer",
      background: "#1DB954",
      color: "#fff",
      border: "none",
      borderRadius: "30px",
      display: "block",
      margin: "0 auto",
      fontSize: "1rem",
      transition: "transform 0.2s ease, box-shadow 0.2s ease",
    },
    toggleButton: {
      display: "flex",
      opacity: 0.8,
      padding: "6px 12px",
      fontSize: "11px",
      fontWeight: "bold",
      background: "rgb(18, 24, 27)",
      color: "rgb(255, 255, 255)",
      borderRadius: "20px",
      cursor: "pointer",
      maxWidth: "80px",
      border: "1px solid #80808096",
    },
    note: {
      marginTop: "1rem",
      padding: "1rem",
      borderLeft: "4px solid #1DB954",
      backgroundColor: "#000000",
      color: "#ffffff",
      fontSize: "0.9rem",
      display: "flex",
      alignItems: "center",
      maxWidth: "300px",
    },
    sectionHeading: { fontSize: "1.25rem", marginBottom: "1rem", textAlign: "center", color: "#ffffff" },
  };

  // Toggle blur effect when generator is shown
  useEffect(() => {
    const contentBody = document.querySelector("div.content-body.grid-75");
    if (contentBody) {
      showGenerator
        ? contentBody.classList.add("blurred")
        : contentBody.classList.remove("blurred");
    }
  }, [showGenerator]);

  // Create and remove banner container on mount/unmount
  useEffect(() => {
    const headerElement = document.querySelector("header.rm-Header");
    if (!headerElement) return;
    let bannerWrapper = document.createElement("div");
    bannerWrapper.id = "custom-banner-container";
    placement === "top"
      ? headerElement.insertBefore(bannerWrapper, headerElement.firstChild)
      : headerElement.appendChild(bannerWrapper);
    return () => {
      if (bannerWrapper) bannerWrapper.remove();
    };
  }, []);

  // Update banner container styles, content and placement on changes
  useEffect(() => {
    const headerElement = document.querySelector("header.rm-Header"),
      bannerWrapper = document.getElementById("custom-banner-container");
    if (!headerElement || !bannerWrapper) return;
    Object.assign(bannerWrapper.style, bannerStyles);
    placement === "top"
      ? headerElement.insertBefore(bannerWrapper, headerElement.firstChild)
      : headerElement.appendChild(bannerWrapper);
    const bannerContent = document.getElementById("custom-banner-content");
    if (bannerContent && bannerContent.parentNode !== bannerWrapper) {
      bannerWrapper.appendChild(bannerContent);
      bannerContent.style.display = "";
    }
  }, [bannerColor, bannerTextColor, fontSize, fontWeight, message, preset, placement, bannerStyles]);

  // Remove banner container on route changes
  useEffect(() => {
    const handleRouteChange = () => {
      const bannerWrapper = document.getElementById("custom-banner-container");
      if (bannerWrapper) bannerWrapper.remove();
    };
    window.addEventListener("popstate", handleRouteChange);
    return () => {
      window.removeEventListener("popstate", handleRouteChange);
    };
  }, []);

  // Dynamic generator styles using headerHeight
  const generatorStyles = isCustomComponentsPage
    ? {
        position: "relative",
        borderRadius: "12px",
        width: "600px",
        zIndex: 1000,
        background: "#2C303B",
        overflowY: "auto",
        boxShadow: "-5px 0 10px rgba(0,0,0,0.2)",
      }
    : {
        position: "fixed",
        top: `${headerHeight + 50}px`,
        right: "20px",
        borderRadius: "12px",
        width: "600px",
        zIndex: 1000,
        background: "#2C303B",
        overflowY: "auto",
        boxShadow: "-5px 0 10px rgba(0,0,0,0.2)",
        transform: "scale(0.8)",
        transformOrigin: "top right",
      };

  return (
    <>
      {/* CSS in the component */}
      <style>{`
        .content-body.grid-75.blurred { filter: blur(5px) brightness(0.5); }
        .content-body.grid-75.blurred .generator-ui { filter: none; }
      `}</style>
      {/* Banner Injection Content */}
      <div id="custom-banner-content" style={{ width: "100%" }}>
        <div
          style={{
            display: "flex",
            justifyContent: "space-between",
            alignItems: "center",
            padding: "1px 10px",
          }}
        >
          <div style={containerStyles}>{message || defaults.message}</div>
          {/* Only show the Banner Edit button if admin and NOT on /custom-components/ page */}
          {isAdmin && !isCustomComponentsPage && (
            <button
              type="button"
              style={uiStyles.toggleButton}
              onClick={() => {
                updateHeaderHeight();
                setShowGenerator((prev) => !prev);
              }}
            >
              {showGenerator ? (
                <>
                  <i className="fas fa-times" style={{ marginRight: "0.5rem" }}></i>
                  Banner
                </>
              ) : (
                <>
                  <i className="fas fa-edit" style={{ marginRight: "0.5rem" }}></i>
                  Banner
                </>
              )}
            </button>
          )}
        </div>
      </div>
      {/* Admin Generator UI - force display on /custom-components/ pages */}
      {(isCustomComponentsPage || showGenerator) && (
        <div className="generator-ui" style={generatorStyles}>
          <div style={uiStyles.container}>
            <div style={uiStyles.headerRow}>
              <h3 style={uiStyles.heading}>ReadMe Banner Generator!</h3>
              <img
                src="https://owlbertsio-resized.s3.amazonaws.com/Traffic.psd.png"
                alt="Owlbert"
                style={uiStyles.image}
              />
            </div>
            <label style={uiStyles.label}>
              Message:
              <textarea
                value={message}
                onChange={(e) => setMessage(e.target.value)}
                style={uiStyles.input}
              />
            </label>
            <div style={uiStyles.row}>
              <label style={uiStyles.label}>
                Preset:
                <select
                  value={preset}
                  onChange={handlePresetChange}
                  style={uiStyles.select}
                >
                  <option value="custom">
                    Custom (you choose colors)
                  </option>
                  <option value="critical">
                    Critical (Red)
                  </option>
                  <option value="warning">
                    Warning (Yellow)
                  </option>
                  <option value="standard">
                    Standard (Blue)
                  </option>
                </select>
              </label>
              <label style={uiStyles.label}>
                Banner Placement:
                <select
                  value={placement}
                  onChange={(e) => setPlacement(e.target.value)}
                  style={uiStyles.select}
                >
                  <option value="top">
                    Top (first in header)
                  </option>
                  <option value="bottom">
                    Bottom (default)
                  </option>
                </select>
              </label>
            </div>
            {preset === "custom" ? (
              <div style={uiStyles.fourColumnRow}>
                <div>
                  <label style={uiStyles.label}>
                    BG Color 🎨
                  </label>
                  <input
                    type="color"
                    value={bannerColor}
                    onChange={(e) => setBannerColor(e.target.value)}
                    style={uiStyles.colorInput}
                  />
                </div>
                <div>
                  <label style={uiStyles.label}>
                    Text Color 🎨
                  </label>
                  <input
                    type="color"
                    value={bannerTextColor}
                    onChange={(e) => setBannerTextColor(e.target.value)}
                    style={uiStyles.colorInput}
                  />
                </div>
                <div>
                  <label style={uiStyles.label}>
                    Font Size
                  </label>
                  <select
                    value={fontSize}
                    onChange={(e) => setFontSize(e.target.value)}
                    style={uiStyles.select}
                  >
                    {["12px", "13px", "14px", "15px", "16px", "17px", "18px"].map(
                      (size) => (
                        <option key={size} value={size}>
                          {size}
                        </option>
                      )
                    )}
                  </select>
                </div>
                <div>
                  <label style={uiStyles.label}>
                    Font Weight
                  </label>
                  <select
                    value={fontWeight}
                    onChange={(e) => setFontWeight(e.target.value)}
                    style={uiStyles.select}
                  >
                    <option value="300">Light</option>
                    <option value="400">Normal</option>
                    <option value="700">Bold</option>
                  </select>
                </div>
              </div>
            ) : (
              <div style={uiStyles.row}>
                <div>
                  <label style={uiStyles.label}>
                    Font Size
                  </label>
                  <select
                    value={fontSize}
                    onChange={(e) => setFontSize(e.target.value)}
                    style={uiStyles.select}
                  >
                    {["12px", "13px", "14px", "15px", "16px", "17px", "18px"].map(
                      (size) => (
                        <option key={size} value={size}>
                          {size}
                        </option>
                      )
                    )}
                  </select>
                </div>
                <div>
                  <label style={uiStyles.label}>
                    Font Weight
                  </label>
                  <select
                    value={fontWeight}
                    onChange={(e) => setFontWeight(e.target.value)}
                    style={uiStyles.select}
                  >
                    <option value="300">Light</option>
                    <option value="400">Normal</option>
                    <option value="700">Bold</option>
                  </select>
                </div>
              </div>
            )}
            <h4 style={uiStyles.sectionHeading}>Generated Code:</h4>
            <pre style={uiStyles.code}>{generatedSnippet}</pre>
            <div
              style={{
                display: "flex",
                justifyContent: "space-between",
                alignItems: "center",
              }}
            >
              <button
                onClick={() => {
                  navigator.clipboard.writeText(generatedSnippet);
                  setCopied(true);
                  setTimeout(() => setCopied(false), 2000);
                }}
                style={uiStyles.button}
              >
                Copy to Clipboard
              </button>
              <div style={uiStyles.note}>
                <span style={{ marginRight: "0.95rem" }}>ℹ️</span>
                <span>
                  Save this code snippet into your ReadMe page to make it public!
                </span>
              </div>
            </div>
            {copied && (
              <div
                style={{
                  textAlign: "center",
                  marginTop: "1rem",
                  color: "#ffffff",
                }}
              >
                Copied!
              </div>
            )}
          </div>
        </div>
      )}
    </>
  );
};

export default PageBannerAdvanced;
