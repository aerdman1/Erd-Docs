---
import React, { useEffect, useRef, useState } from "react";

export const ContentModal = rawProps => {
  const props = {};
  Object.keys(rawProps || {}).forEach(k => { props[k.toLowerCase()] = rawProps[k]; });
  props.children = rawProps ? rawProps.children : undefined;
  const label = props.label || "Open";
  const title = props.title || "";
  const size = { sm: "480px", md: "720px", lg: "960px", xl: "1200px" }[props.size || "md"] || "720px";
  const closeOnBackdrop = !(props.closeonbackdrop === false || props.closeonbackdrop === "false");
  const showIcon = !(props.showicon === false || props.showicon === "false");
  const variant = props.variant || "primary";
  const [open, setOpen] = useState(false);
  const dialogRef = useRef(null);
  const btnRef = useRef(null);
  const wasOpen = useRef(false);
  const idRef = useRef("eo-modal-" + Math.random().toString(36).slice(2, 9));

  useEffect(() => {
    if (typeof document === "undefined") return undefined;
    if (open) {
      wasOpen.current = true;
      const html = document.documentElement;
      const prev = html.style.overflow;
      html.style.overflow = "hidden";
      const onKey = e => {
        if (e.key === "Escape") { e.stopPropagation(); setOpen(false); }
        if (e.key === "Tab" && dialogRef.current) {
          const f = dialogRef.current.querySelectorAll('a[href],button:not([disabled]),input,select,textarea,[tabindex]:not([tabindex="-1"])');
          if (!f.length) return;
          const first = f[0], last = f[f.length - 1];
          if (e.shiftKey && document.activeElement === first) { e.preventDefault(); last.focus(); }
          else if (!e.shiftKey && document.activeElement === last) { e.preventDefault(); first.focus(); }
        }
      };
      document.addEventListener("keydown", onKey, true);
      window.setTimeout(() => { if (dialogRef.current) dialogRef.current.focus(); }, 0);
      return () => { html.style.overflow = prev; document.removeEventListener("keydown", onKey, true); };
    }
    if (wasOpen.current && btnRef.current) btnRef.current.focus();
    return undefined;
  }, [open]);

  const css = `
.eo-modal{--eo-accent:var(--color-link-primary,#2F7BF5);--eo-border:var(--color-border-default,rgba(0,0,0,.12))}
.eo-modal-btn{display:inline-flex;align-items:center;gap:8px;border-radius:8px;padding:10px 16px;font-family:inherit;font-weight:600;font-size:15px;cursor:pointer;border:1px solid var(--eo-accent);transition:transform .1s,box-shadow .15s}
.eo-modal-btn[data-variant="primary"]{background:var(--eo-accent);color:#fff}
.eo-modal-btn[data-variant="outline"]{background:transparent;color:var(--eo-accent)}
.eo-modal-btn[data-variant="link"]{background:transparent;border-color:transparent;color:var(--eo-accent);padding:0;text-decoration:underline}
.eo-modal-btn:hover{box-shadow:0 4px 14px color-mix(in srgb,var(--eo-accent) 30%,transparent)}
.eo-modal-btn:active{transform:translateY(1px)}
.eo-modal-btn:focus-visible{outline:2px solid var(--eo-accent);outline-offset:2px}
.eo-modal-wrap{position:fixed;inset:0;z-index:100000;display:flex;align-items:center;justify-content:center;padding:24px;background:rgba(5,8,20,.55);backdrop-filter:blur(3px);animation:eo-fade .15s ease both}
.eo-modal-dialog{width:100%;max-height:86vh;display:flex;flex-direction:column;overflow:hidden;outline:none;border-radius:14px;border:1px solid var(--eo-border);
  background:var(--color-bg-page,#fff);color:var(--color-text-default,#111);box-shadow:0 24px 60px rgba(0,0,0,.35);animation:eo-pop .18s cubic-bezier(.2,.9,.3,1.2) both}
.eo-modal-head{display:flex;align-items:center;justify-content:space-between;gap:12px;padding:14px 18px;border-bottom:1px solid var(--eo-border)}
.eo-modal-head h2{margin:0!important;font-size:17px!important;border:0!important;padding:0!important}
.eo-modal-x{border:0;background:transparent;color:inherit;font-size:22px;line-height:1;cursor:pointer;opacity:.6;padding:4px 8px;border-radius:6px}
.eo-modal-x:hover{opacity:1;background:color-mix(in srgb,currentColor 8%,transparent)}
.eo-modal-body{overflow:auto;padding:18px 20px;text-align:left}
.eo-modal-body>:first-child{margin-top:0}
@keyframes eo-fade{from{opacity:0}to{opacity:1}}
@keyframes eo-pop{from{opacity:0;transform:translateY(10px) scale(.98)}to{opacity:1;transform:none}}
@media (prefers-reduced-motion:reduce){.eo-modal-wrap,.eo-modal-dialog{animation:none}}
@media (max-width:640px){.eo-modal-wrap{padding:0}.eo-modal-dialog{max-height:100vh;height:100%;border-radius:0}}
`;
  return (
    <span className="eo-modal">
      <style>{css}</style>
      <button ref={btnRef} type="button" className="eo-modal-btn" data-variant={variant} aria-haspopup="dialog" aria-expanded={open ? "true" : "false"} onClick={() => setOpen(true)}>
        {props.icon ? <span aria-hidden="true">{props.icon}</span> : null}
        {label}
        {showIcon && variant !== "link" ? <span aria-hidden="true" style={{ opacity: .8 }}>↗</span> : null}
      </button>
      {open ? (
        <span className="eo-modal-wrap" onMouseDown={e => { if (closeOnBackdrop && e.target === e.currentTarget) setOpen(false); }}>
          <span className="eo-modal-dialog" role="dialog" aria-modal="true" aria-labelledby={title ? idRef.current : undefined} aria-label={title ? undefined : label}
            ref={dialogRef} tabIndex={-1} style={{ maxWidth: size }}>
            <span className="eo-modal-head" style={title ? undefined : { justifyContent: "flex-end", borderBottom: 0, paddingBottom: 0 }}>
              {title ? <h2 id={idRef.current}>{title}</h2> : null}
              <button type="button" className="eo-modal-x" aria-label="关闭" onClick={() => setOpen(false)}>×</button>
            </span>
            <span className="eo-modal-body" style={{ display: "block" }}>{props.children || props.content}</span>
          </span>
        </span>
      ) : null}
    </span>
  );
};

<ContentModal label="查看有效载荷模式" title="有效载荷">
  有效载荷是指卫星或散装货物。其类型决定了哪些预订字段是必填的。
</ContentModal>
