import React, { useEffect, useRef, useState } from "react";

export const ContentModal = props => {
  const label = props.label || "Open";
  const title = props.title || "Modal";
  const content = props.content;
  const size = props.size === undefined ? "lg" : props.size;
  const buttonClassName = props.buttonClassName || "";
  const closeOnBackdrop =
    props.closeOnBackdrop === undefined ? true : !!props.closeOnBackdrop;
  const showIcon = props.showIcon === undefined ? true : !!props.showIcon;

  const buttonColor = props.buttonColor || "#fe8d53";
  const buttonPadding = props.buttonPadding || "17px 19px";

  const autoIdRef = useRef(
    (typeof crypto !== "undefined" && crypto.randomUUID)
      ? crypto.randomUUID()
      : "cm-" + Math.random().toString(36).slice(2)
  );
  const modalId = props.modalId || autoIdRef.current;

  const [open, setOpen] = useState(false);
  const dialogRef = useRef(null);
  const btnRef = useRef(null);
  const didMountRef = useRef(false);

  const css = [
    ".cm-Backdrop{position:fixed;top:0;right:0;bottom:0;left:0;display:none;z-index:999998!important;}",
    '.cm-Backdrop[aria-hidden="false"]{display:block;}',
    '.cm-Backdrop::before{content:\"\";position:absolute;inset:0;background:rgba(0,0,0,.45);-webkit-backdrop-filter:saturate(1.1) blur(2px);backdrop-filter:saturate(1.1) blur(2px);}',
    ".cm-DialogWrap{position:fixed;inset:0;display:none;align-items:center;justify-content:center;padding:24px;z-index:999999!important;}",
    '.cm-DialogWrap[aria-hidden=\"false\"]{display:flex;}',
    `.cm-Dialog{width:100%;max-height:86vh;overflow:hidden;
      background:var(--color-bg-page, #fff);
      color:var(--color-text-default, #000);
      border-radius:12px;
      box-shadow:0 10px 30px rgba(0,0,0,.25);
      display:flex;flex-direction:column;outline:none;}`,
    ".cm-size-sm{max-width:480px;}",
    ".cm-size-md{max-width:720px;}",
    ".cm-size-lg{max-width:960px;}",
    ".cm-size-xl{max-width:1200px;}",
    `.cm-Header{display:flex;align-items:center;justify-content:space-between;gap:12px;
      padding:14px 18px;
      border-bottom:1px solid rgba(0,0,0,.08);
      background:var(--color-bg-page, #fff);
      color:var(--color-text-default, #000);
      position:sticky;top:0;z-index:1;}`,
    ".cm-Header h2{margin:0;font-size:16px;line-height:1.2;font-weight:600;}",
    ".cm-CloseBtn{border:none;background:transparent;font-size:20px;line-height:1;cursor:pointer;color:inherit;opacity:.6;}",
    ".cm-CloseBtn:hover{opacity:1;}",
    `.cm-Body{overflow:auto;padding:16px 18px;
      background:var(--color-bg-page, #fff);
      color:var(--color-text-default, #000);
      text-align:left;}`,
    ".cm-Body .placeholder{padding:50px 0;color:#666;font-style:italic;text-align:center;}",
    ".modal-button{display:inline-flex;align-items:center;gap:8px;cursor:pointer;}",
    `.cm-TriggerBtnDefault{line-height:20px;color:#fff;background-color:${buttonColor};
      justify-content:space-evenly;display:inline-flex;box-sizing:border-box;
      border:2px solid ${buttonColor};border-radius:6px;height:46px;
      padding:${buttonPadding};align-items:center;cursor:pointer;transition:opacity .2s ease;}`,
    ".cm-TriggerBtnDefault:hover{opacity:0.85;}",
    ".modal{display:contents;}"
  ].join("");

  // Lock page scroll while open
  useEffect(() => {
    if (typeof document === "undefined") return;
    const html = document.documentElement;
    const prev = html.style.overflow;
    if (open) html.style.overflow = "hidden";
    return () => { html.style.overflow = prev; };
  }, [open]);

  // Focus management
  useEffect(() => {
    if (!didMountRef.current) { didMountRef.current = true; return; }
    if (open) {
      const el = dialogRef.current;
      const firstFocusable = el
        ? el.querySelector('button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])')
        : null;
      if (firstFocusable && firstFocusable.focus) firstFocusable.focus();
      else if (el && el.focus) el.focus();
    } else {
      if (btnRef.current && btnRef.current.focus) btnRef.current.focus();
    }
  }, [open]);

  // ESC to close
  useEffect(() => {
    if (!open || typeof window === "undefined") return;
    function onKey(e) { if (e.key === "Escape") setOpen(false); }
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [open]);

  function handleBackdropMouseDown(e) {
    if (!closeOnBackdrop) return;
    const target = e?.target;
    const isBackdrop = target?.getAttribute?.("data-cm-backdrop") === "1";
    if (isBackdrop) setOpen(false);
  }

  const sizeClass =
    size === "sm" ? "cm-size-sm"
    : size === "md" ? "cm-size-md"
    : size === "xl" ? "cm-size-xl"
    : "cm-size-lg";

  const modalIdAttr = modalId || undefined;
  const ariaLabelledBy = modalId ? modalId + "-title" : undefined;
  const ariaExpanded = open ? "true" : "false";
  const triggerClass = "modal-button " + (buttonClassName || "cm-TriggerBtnDefault");
  const dialogClass = "cm-Dialog " + sizeClass;

  return (
    <div data-cm-root="">
      <style id="content-modal-styles">{css}</style>

      <button
        ref={btnRef}
        type="button"
        className={triggerClass}
        aria-haspopup="dialog"
        aria-expanded={ariaExpanded}
        onClick={() => setOpen(true)}
        data-modal-id={modalIdAttr}
        title={title}
      >
        {label}
        {showIcon && <i className="fa-solid fa-up-right-from-square" aria-hidden="true" />}
      </button>

      <div
        className="cm-Backdrop"
        aria-hidden={open ? "false" : "true"}
        style={{ display: open ? undefined : "none" }}
      />

      <div
        className="cm-DialogWrap"
        aria-hidden={open ? "false" : "true"}
        data-cm-backdrop="1"
        onMouseDown={handleBackdropMouseDown}
        style={{ display: open ? undefined : "none" }}
      >
        <div
          className={dialogClass}
          role="dialog"
          aria-modal="true"
          aria-labelledby={ariaLabelledBy}
          ref={dialogRef}
          tabIndex={-1}
          id={modalIdAttr}
        >
          <div className="cm-Header modal-header">
            <h2 id={ariaLabelledBy}>{title}</h2>
            <button
              className="cm-CloseBtn close"
              type="button"
              aria-label="Close"
              onClick={() => setOpen(false)}
            >
              ×
            </button>
          </div>

          <div className="cm-Body">
            {props.children
              ? props.children
              : content || <div className="placeholder">Paste your content here</div>}
          </div>
        </div>
      </div>
    </div>
  );
};
