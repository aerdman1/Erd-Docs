import React, { useState, useEffect } from 'react';

export const BannerMessage = ({
  message: propMessage = 'This is my banner message!',
  bannerColor: propBannerColor = '#00bcff',
  bannerTextColor: propBannerTextColor = '#ffffff',
  fontSize: propFontSize = '16px',
  fontWeight: propFontWeight = 'normal',
}) => {
  // Check if the user is an admin based on the hub-me script data
  let isAdmin = false;
  if (typeof document !== 'undefined') {
    const hubElem = document.getElementById('hub-me');
    if (hubElem) {
      try {
        const hubData = JSON.parse(hubElem.getAttribute('data-json'));
        isAdmin = hubData.isAdmin;
      } catch (e) {
        console.error('Error parsing hub-me data', e);
      }
    }
  }

  // Map incoming fontWeight "normal" to "400" for our select
  const initialFontWeight = propFontWeight === 'normal' ? '400' : propFontWeight;

  // Use incoming props to initialize state
  const [message, setMessage] = useState(propMessage);
  const [bannerColor, setBannerColor] = useState(propBannerColor);
  const [bannerTextColor, setBannerTextColor] = useState(propBannerTextColor);
  const [fontSize, setFontSize] = useState(propFontSize);
  const [fontWeight, setFontWeight] = useState(initialFontWeight);
  const [preset, setPreset] = useState('custom'); // 'critical', 'warning', 'standard', 'custom'
  const [placement, setPlacement] = useState('bottom'); // 'top' or 'bottom'
  const [copied, setCopied] = useState(false);
  // New state to toggle the popup editor
  const [isPopupOpen, setIsPopupOpen] = useState(false);

  // Sync state with props if they change externally
  useEffect(() => {
    setMessage(propMessage);
    setBannerColor(propBannerColor);
    setBannerTextColor(propBannerTextColor);
    setFontSize(propFontSize);
    setFontWeight(propFontWeight === 'normal' ? '400' : propFontWeight);
  }, [propMessage, propBannerColor, propBannerTextColor, propFontSize, propFontWeight]);

  const handlePresetChange = (e) => {
    const selectedPreset = e.target.value;
    setPreset(selectedPreset);
    if (selectedPreset === 'critical') {
      setBannerColor('#dc2626');
      setBannerTextColor('#ffffff');
    } else if (selectedPreset === 'warning') {
      setBannerColor('#facc15');
      setBannerTextColor('#000000');
    } else if (selectedPreset === 'standard') {
      setBannerColor('#2563eb');
      setBannerTextColor('#ffffff');
    } else if (selectedPreset === 'custom') {
      setBannerColor('#00bcff');
      setBannerTextColor('#ffffff');
    }
  };

  const generateBannerProps = () => {
    let props = `message="${message}"`;
    if (preset !== 'custom') {
      props += ` ${preset}={true}`;
    } else {
      props += ` bannerColor="${bannerColor}" bannerTextColor="${bannerTextColor}"`;
    }
    props += ` fontSize="${fontSize}" fontWeight="${fontWeight}"`;
    props += ` placement="${placement}"`;
    return props;
  };

  // Generated snippet now shows the unified component's name
  const generatedSnippet = `<BannerMessage ${generateBannerProps()} />`;

  const defaultPresets = {
    critical: {
      bannerColor: "#dc2626",
      bannerTextColor: "#ffffff",
      fontSize: "16px",
      fontWeight: "normal",
      message: "Critical Alert: Please check immediately!"
    },
    warning: {
      bannerColor: "#facc15",
      bannerTextColor: "#000000",
      fontSize: "16px",
      fontWeight: "normal",
      message: "Warning Alert: There might be an issue."
    },
    standard: {
      bannerColor: "#2563eb",
      bannerTextColor: "#ffffff",
      fontSize: "16px",
      fontWeight: "normal",
      message: "Standard Alert: Here is some information."
    },
    custom: {
      bannerColor: "#00BCFF",    
      bannerTextColor: "#ffffff",
      fontSize: "14px",
      fontWeight: "bold",
      message: "Here is my banner message - hope you enjoy!"
    }
  };

  const defaults = defaultPresets[preset];

  // Compute banner styles for the public injected banner
  const bannerStyles = {
    background: bannerColor || defaults.bannerColor,
    color: bannerTextColor || defaults.bannerTextColor,
    fontSize: fontSize || defaults.fontSize,
    fontWeight: fontWeight || defaults.fontWeight,
    width: "100%",
    boxSizing: "border-box",
    borderRadius: "0px",
    padding: "10px"
  };

  // Container styles for public banner content (injected into header)
  const containerStyles = {
    maxWidth: "var(--rm-container-width, 1200px)",
    margin: "0 auto",
    padding: "1rem 1.5rem",
    textAlign: "center",
    boxSizing: "border-box",
    fontFamily: 'Inter, sans-serif'
  };

  // Inline styles for Admin Generator UI (all styles inside the component)
  const uiStyles = {
    container: {
      maxWidth: '800px',
      margin: '2rem auto',
      padding: '2rem',
      background: '#2C303B',
      borderRadius: '12px',
      boxShadow: '0 8px 24px rgba(0, 0, 0, 0.1)',
      fontFamily: 'Inter, sans-serif',
      display: isAdmin ? 'block' : 'none',
      position: 'relative'
    },
    headerRow: {
      display: 'flex',
      justifyContent: 'space-between',
      alignItems: 'center',
      marginBottom: '1.5rem',
      position: 'relative'
    },
    heading: {
      margin: 0,
      color: '#fff',
      fontSize: '1.75rem',
      fontWeight: 600
    },
    image: {
      position: 'absolute',
      top: '-23px',
      right: '-32px',
      maxWidth: '71px',
      height: 'auto'
    },
    label: {
      display: 'block',
      marginBottom: '0.5rem',
      fontWeight: 500,
      color: '#fff',
      fontSize: '1rem'
    },
    input: {
      width: '93%',
      padding: '0.40rem',
      borderRadius: '6px',
      border: '1px solid #ddd',
      color: '#fff',
      fontSize: '1rem',
      background: '#384248',
      marginTop: '2px',
      transition: 'border-color 0.2s ease-in-out'
    },
    // Modified select style: remove native appearance and add a custom down arrow
    select: {
      width: '100%',
      padding: '0.60rem',
      paddingRight: '2.5rem',
      color: '#fff',
      borderRadius: '6px',
      border: '1px solid #ddd',
      fontSize: '1rem',
      background: `#384248 url("data:image/svg+xml,%3Csvg width='10' height='10' viewBox='0 0 10 10' xmlns='http://www.w3.org/2000/svg'%3E%3Cpolyline points='1,3 5,7 9,3' stroke='%23fff' stroke-width='2' fill='none' stroke-linecap='round' stroke-linejoin='round'/%3E%3C/svg%3E") no-repeat right 0.75rem center`,
      backgroundSize: '1em',
      marginTop: '2px',
      transition: 'border-color 0.2s ease-in-out',
      appearance: 'none',
      WebkitAppearance: 'none',
      MozAppearance: 'none'
    },
    row: {
      display: 'flex',
      gap: '1rem',
      marginBottom: '1rem'
    },
    fourColumnRow: {
      display: 'grid',
      gridTemplateColumns: 'repeat(auto-fit, minmax(150px, 1fr))',
      gap: '1rem',
      marginBottom: '1rem'
    },
    colorInput: {
      width: '100%',
      height: '45px',
      padding: 0,
      border: 0,
      cursor: 'pointer',
      background: 'none'
    },
    code: {
      background: '#2d2d2d',
      color: '#ffffff',
      padding: '1rem',
      borderRadius: '6px',
      overflowX: 'auto',
      fontFamily: 'monospace',
      marginBottom: '1.5rem',
      fontSize: '0.95rem'
    },
    button: {
      padding: '0.75rem 1.5rem',
      cursor: 'pointer',
      background: '#1DB954',
      color: '#fff',
      border: 'none',
      borderRadius: '30px',
      display: 'block',
      margin: '0 auto',
      fontSize: '1rem',
      transition: 'transform 0.2s ease, box-shadow 0.2s ease'
    },
    note: {
      marginTop: '1rem',
      padding: '1rem',
      borderLeft: '4px solid #1DB954',
      backgroundColor: '#000000',
      color: '#ffffff',
      fontSize: '0.9rem',
      display: 'flex',
      alignItems: 'center'
    },
    sectionHeading: {
      fontSize: '1.25rem',
      marginBottom: '1rem',
      textAlign: 'center',
      color: '#ffffff'
    }
  };

// ── Effect 1: Create banner container on mount and remove on unmount ──
useEffect(() => {
  const headerElement = document.querySelector('header.rm-Header');
  if (!headerElement) return;

  // Create bannerWrapper only if it doesn't already exist
  let bannerWrapper = document.getElementById('custom-banner-container');
  if (!bannerWrapper) {
    bannerWrapper = document.createElement('div');
    bannerWrapper.id = 'custom-banner-container';
  }
  try {
    if (placement === 'top') {
      // Check if there's a valid first element child
      if (headerElement.firstElementChild) {
        headerElement.insertBefore(bannerWrapper, headerElement.firstElementChild);
      } else {
        headerElement.appendChild(bannerWrapper);
      }
    } else {
      headerElement.appendChild(bannerWrapper);
    }
  } catch (e) {
    console.error("Error inserting bannerWrapper:", e);
  }
  return () => {
    if (bannerWrapper) bannerWrapper.remove();
  };
}, []);


  // ── Effect 2: Update banner container's styles, content, and placement on changes ──
  useEffect(() => {
    const headerElement = document.querySelector('header.rm-Header');
    if (!headerElement) return;
    const bannerWrapper = document.getElementById('custom-banner-container');
    if (!bannerWrapper) return;
    Object.assign(bannerWrapper.style, bannerStyles);
    // Only reinsert if bannerWrapper isn't already a child of headerElement
    if (placement === 'top' && bannerWrapper.parentNode !== headerElement) {
      headerElement.insertBefore(bannerWrapper, headerElement.firstElementChild);
    } else if (placement === 'bottom' && bannerWrapper.parentNode !== headerElement) {
      headerElement.appendChild(bannerWrapper);
    }
    const bannerContent = document.getElementById('custom-banner-content');
    if (bannerContent && bannerContent.parentNode !== bannerWrapper) {
      bannerWrapper.appendChild(bannerContent);
    }
    if (bannerContent) {
      bannerContent.style.display = '';
    }
  }, [bannerColor, bannerTextColor, fontSize, fontWeight, message, preset, placement, bannerStyles]);

  // ── Effect 3: Listen for route changes and remove the banner container ──
  useEffect(() => {
    const handleRouteChange = () => {
      const bannerWrapper = document.getElementById('custom-banner-container');
      if (bannerWrapper) bannerWrapper.remove();
    };
    window.addEventListener('popstate', handleRouteChange);
    return () => {
      window.removeEventListener('popstate', handleRouteChange);
    };
  }, []);

  return (
    <>
      {/* Popup Admin Generator UI - only render if user is admin */}
      {isAdmin && (
        <>
          {/* Toggle Button - fixed in bottom right */}
          <div style={{
            position: 'fixed',
            bottom: '20px',
            right: '20px',
            zIndex: 9999,
          }}>
            <button
              onClick={() => setIsPopupOpen(!isPopupOpen)}
              style={{
                padding: '10px 20px',
                borderRadius: '20px',
                background: '#1DB954',
                color: '#fff',
                border: 'none',
                cursor: 'pointer',
                fontSize: '1rem'
              }}
            >
              {isPopupOpen ? 'Close Banner Editor' : 'Open Banner Editor'}
            </button>
          </div>
          {/* Popup container */}
          {isPopupOpen && (
            <div style={{
              position: 'fixed',
              bottom: '80px', // adjust so it sits above the toggle button
              right: '20px',
              zIndex: 9999,
              background: '#2C303B',
              borderRadius: '12px',
              boxShadow: '0 8px 24px rgba(0, 0, 0, 0.1)',
              overflow: 'auto',
              maxHeight: '80vh'
            }}>
              <div style={{ transform: 'scale(0.8)', transformOrigin: 'top left' }}>
                <div style={uiStyles.container}>
                  <div style={uiStyles.headerRow}>
                    <h3 style={uiStyles.heading}>Banner Generator</h3>
                    <img 
                      src="https://owlbertsio-resized.s3.amazonaws.com/Traffic.psd.png" 
                      alt="Owlbert" 
                      style={uiStyles.image}
                    />
                  </div>
                  <label style={uiStyles.label}>
                    Message:
                    <input
                      type="text"
                      value={message}
                      onChange={(e) => setMessage(e.target.value)}
                      style={uiStyles.input}
                    />
                  </label>
                  <div style={uiStyles.row}>
                    <label style={uiStyles.label}>
                      Preset:
                      <select value={preset} onChange={handlePresetChange} style={uiStyles.select}>
                        <option value="custom">Custom (you choose colors)</option>
                        <option value="critical">Critical (Red)</option>
                        <option value="warning">Warning (Yellow)</option>
                        <option value="standard">Standard (Blue)</option>
                      </select>
                    </label>
                    <label style={uiStyles.label}>
                      Banner Placement:
                      <select
                        value={placement}
                        onChange={(e) => setPlacement(e.target.value)}
                        style={uiStyles.select}
                      >
                        <option value="top">Top (first in header)</option>
                        <option value="bottom">Bottom (default)</option>
                      </select>
                    </label>
                  </div>
                  {preset === 'custom' ? (
                    <div style={uiStyles.fourColumnRow}>
                      <div>
                        <label style={uiStyles.label}>BG Color 🎨</label>
                        <input
                          type="color"
                          value={bannerColor}
                          onChange={(e) => setBannerColor(e.target.value)}
                          style={uiStyles.colorInput}
                        />
                      </div>
                      <div>
                        <label style={uiStyles.label}>Text Color 🎨</label>
                        <input
                          type="color"
                          value={bannerTextColor}
                          onChange={(e) => setBannerTextColor(e.target.value)}
                          style={uiStyles.colorInput}
                        />
                      </div>
                      <div>
                        <label style={uiStyles.label}>Font Size</label>
                        <input
                          type="text"
                          value={fontSize}
                          onChange={(e) => setFontSize(e.target.value)}
                          style={uiStyles.input}
                        />
                      </div>
                      <div>
                        <label style={uiStyles.label}>Font Weight</label>
                        <select
                          value={fontWeight}
                          onChange={(e) => setFontWeight(e.target.value)}
                          style={uiStyles.select}
                        >
                          <option value="300">Light</option>
                          <option value="400">Normal</option>
                          <option value="700">Bold</option>
                        </select>
                      </div>
                    </div>
                  ) : (
                    <div style={uiStyles.row}>
                      <div>
                        <label style={uiStyles.label}>Font Size</label>
                        <input
                          type="text"
                          value={fontSize}
                          onChange={(e) => setFontSize(e.target.value)}
                          style={uiStyles.input}
                        />
                      </div>
                      <div>
                        <label style={uiStyles.label}>Font Weight</label>
                        <select
                          value={fontWeight}
                          onChange={(e) => setFontWeight(e.target.value)}
                          style={uiStyles.select}
                        >
                          <option value="300">Light</option>
                          <option value="400">Normal</option>
                          <option value="700">Bold</option>
                        </select>
                      </div>
                    </div>
                  )}
                  <h4 style={uiStyles.sectionHeading}>Generated Code:</h4>
                  <pre style={uiStyles.code}>{generatedSnippet}</pre>
                  <button
                    onClick={() => {
                      navigator.clipboard.writeText(generatedSnippet);
                      setCopied(true);
                      setTimeout(() => setCopied(false), 2000);
                    }}
                    style={uiStyles.button}
                  >
                    Copy to Clipboard
                  </button>
                  {copied && (
                    <div style={{ textAlign: 'center', marginTop: '1rem', color: '#ffffff' }}>
                      Copied!
                    </div>
                  )}
                  <div style={uiStyles.note}>
                    <span style={{ marginRight: '0.5rem' }}>ℹ️</span>
                    <span>
                      Copy the code above, click "Edit," then paste it on the page and save. It will go live for the public.
                    </span>
                  </div>
                </div>
              </div>
            </div>
          )}
        </>
      )}
      
      {/* Banner Injection Content */}
      <div id="custom-banner-content" style={{ display: 'none', width: '100%' }}>
        <div style={containerStyles}>
          {message || defaults.message}
        </div>
      </div>
    </>
  );
};

export default BannerMessage;
