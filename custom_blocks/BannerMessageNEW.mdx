import React, { useState, useEffect } from "react";

export const BannerMessage = ({
  message: propMessage = "Hello! 👋 This is your page-specific announcement banner — customize it to share updates, reminders, or anything important.",
  bannerColor: propBannerColor = "#008EF5",
  bannerTextColor: propBannerTextColor = "#ffffff",
  fontSize: propFontSize = "16px",
  fontWeight: propFontWeight = "normal",
}) => {
  let isAdmin = false;
  if (typeof document !== "undefined") {
    const hubElem = document.getElementById("hub-me");
    if (hubElem) {
      try {
        const hubData = JSON.parse(hubElem.getAttribute("data-json"));
        isAdmin = hubData.isAdmin;
      } catch (e) {
        console.error("Error parsing hub-me data", e);
      }
    }
  }

  const initialFontWeight = propFontWeight === "normal" ? "400" : propFontWeight;

  const [message, setMessage] = useState(propMessage);
  const [bannerColor, setBannerColor] = useState(propBannerColor);
  const [bannerTextColor, setBannerTextColor] = useState(propBannerTextColor);
  const [fontSize, setFontSize] = useState(propFontSize);
  const [fontWeight, setFontWeight] = useState(initialFontWeight);
  const [preset, setPreset] = useState("custom");
  const [placement, setPlacement] = useState("bottom");
  const [copied, setCopied] = useState(false);
  const [showGenerator, setShowGenerator] = useState(false);
  const [headerHeight, setHeaderHeight] = useState(0);

  const updateHeaderHeight = () => {
    const header = document.querySelector("header.rm-Header");
    if (header) {
      setHeaderHeight(header.offsetHeight);
    }
  };

  useEffect(() => {
    updateHeaderHeight();

    let resizeTimer;
    const handleResize = () => {
      clearTimeout(resizeTimer);
      resizeTimer = setTimeout(() => {
        updateHeaderHeight();
      }, 100);
    };

    window.addEventListener("resize", handleResize);
    return () => {
      clearTimeout(resizeTimer);
      window.removeEventListener("resize", handleResize);
    };
  }, []);

  useEffect(() => {
    if (showGenerator) {
      updateHeaderHeight();
    }
  }, [showGenerator]);

  useEffect(() => {
    setMessage(propMessage);
    setBannerColor(propBannerColor);
    setBannerTextColor(propBannerTextColor);
    setFontSize(propFontSize);
    setFontWeight(propFontWeight === "normal" ? "400" : propFontWeight);
  }, [propMessage, propBannerColor, propBannerTextColor, propFontSize, propFontWeight]);

  const handlePresetChange = (e) => {
    const selectedPreset = e.target.value;
    setPreset(selectedPreset);
    if (selectedPreset === "critical") {
      setBannerColor("#dc2626");
      setBannerTextColor("#ffffff");
    } else if (selectedPreset === "warning") {
      setBannerColor("#facc15");
      setBannerTextColor("#000000");
    } else if (selectedPreset === "standard") {
      setBannerColor("#2563eb");
      setBannerTextColor("#ffffff");
    } else if (selectedPreset === "custom") {
      setBannerColor("#00bcff");
      setBannerTextColor("#ffffff");
    }
  };

  const bannerStyles = {
    background: bannerColor,
    color: bannerTextColor,
    fontSize,
    fontWeight,
    width: "100%",
    boxSizing: "border-box",
    borderRadius: "0px",
    padding: "10px",
  };

  const containerStyles = {
    maxWidth: "var(--rm-container-width, 1200px)",
    margin: "0 auto",
    padding: "1rem 1.5rem",
    textAlign: "center",
    boxSizing: "border-box",
    fontFamily: "Inter, sans-serif",
  };

  const uiStyles = {
    toggleButton: {
      display: "flex",
      opacity: ".8",
      margin: "0px -20px 0px 20px",
      padding: "6px 12px",
      fontSize: "11px",
      fontWeight: "bold",
      background: "rgb(15, 105, 189)",
      color: "rgb(255, 255, 255)",
      border: "none",
      borderRadius: "10px 0px 0px 10px",
      cursor: "pointer",
      maxWidth: "80px",
    },
  };

  useEffect(() => {
    const headerElement = document.querySelector("header.rm-Header");
    if (!headerElement) return;
    const bannerWrapper = document.createElement("div");
    bannerWrapper.id = "custom-banner-container";

    if (placement === "top") {
      headerElement.insertBefore(bannerWrapper, headerElement.firstChild);
    } else {
      headerElement.appendChild(bannerWrapper);
    }

    return () => bannerWrapper.remove();
  }, []);

  useEffect(() => {
    const bannerWrapper = document.getElementById("custom-banner-container");
    if (bannerWrapper) {
      Object.assign(bannerWrapper.style, bannerStyles);
    }
  }, [bannerColor, bannerTextColor, fontSize, fontWeight, message, preset, placement]);

  const generatorStyles = {
    position: "fixed",
    top: `${headerHeight + 150}px`,
    right: 0,
    width: "600px",
    height: `calc(100vh - ${headerHeight + 150}px)`,
    zIndex: 1000,
    background: "#2C303B",
    overflowY: "auto",
    boxShadow: "-5px 0 10px rgba(0,0,0,0.2)",
    transform: "scale(0.7)",
    transformOrigin: "top right",
  };

  return (
    <>
      <div id="custom-banner-content" style={{ display: "none", width: "100%" }}>
        <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", padding: "1px 10px" }}>
          <div style={containerStyles}>{message}</div>
          {isAdmin && (
            <button style={uiStyles.toggleButton} onClick={() => setShowGenerator((prev) => !prev)}>
              {showGenerator ? (
                <>
                  <i className="fas fa-times" style={{ marginRight: "0.5rem" }}></i> Banner
                </>
              ) : (
                <>
                  <i className="fas fa-edit" style={{ marginRight: "0.5rem" }}></i> Banner
                </>
              )}
            </button>
          )}
        </div>
      </div>

      {showGenerator && (
        <div className="generator-ui" style={generatorStyles}>
          <div style={{ padding: "20px", color: "white" }}>
            <h3>Banner Generator</h3>
            <p>Customize your banner here...</p>
          </div>
        </div>
      )}
    </>
  );
};

export default BannerMessage;
