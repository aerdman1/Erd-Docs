import { useEffect, useMemo, useState } from "react";

export const RemoteJsonCodeBlock = (props) => {
  const url = props.url || "";
  const jsonPath = props.jsonPath || "front_image";
  const language = props.language || "python";
  const filename = props.filename || "sample.py";
  const variableName = props.variableName || "VALUE";
  const quoteStyle = props.quoteStyle || "double";

  const [loading, setLoading] = useState(Boolean(url));
  const [error, setError] = useState("");
  const [data, setData] = useState(null);
  const [copied, setCopied] = useState(false);
  const [dark, setDark] = useState(
    () =>
      typeof window !== "undefined" &&
      window.matchMedia("(prefers-color-scheme: dark)").matches
  );

  useEffect(() => {
    const mq = window.matchMedia("(prefers-color-scheme: dark)");
    const handler = (e) => setDark(e.matches);
    mq.addEventListener("change", handler);
    return () => mq.removeEventListener("change", handler);
  }, []);

  const t = dark
    ? {
        bg: "#1a1a1a",
        border: "#3a3a3a",
        labelBg: "#111111",
        text: "#e5e7eb",
        muted: "#6b7280",
        varColor: "#a5b4fc",
        strColor: "#6ee7b7",
      }
    : {
        bg: "#ffffff",
        border: "#e5e7eb",
        labelBg: "#f9fafb",
        text: "#111827",
        muted: "#6b7280",
        varColor: "#4f46e5",
        strColor: "#065f46",
      };

  const H = 30; // height of each row

  const getByPath = (obj, path) => {
    if (!obj || !path) return undefined;
    let cur = obj;
    for (const part of String(path).split(".").map((p) => p.trim()).filter(Boolean)) {
      if (cur && typeof cur === "object" && part in cur) cur = cur[part];
      else return undefined;
    }
    return cur;
  };

  const quoteOpen = quoteStyle === "triple" ? '"""' : quoteStyle === "single" ? "'" : '"';

  const quoteValue = (v) => {
    if (quoteStyle === "triple") return `"""${v}"""`;
    if (quoteStyle === "single") return `'${v.replace(/'/g, "\\'")}'`;
    return `"${v.replace(/"/g, '\\"')}"`;
  };

  useEffect(() => {
    let cancelled = false;
    async function run() {
      setError(""); setData(null);
      if (!url) { setLoading(false); setError("Missing url prop."); return; }
      setLoading(true);
      try {
        const res = await fetch(url, { credentials: "omit" });
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        const json = await res.json();
        if (!cancelled) setData(json);
      } catch (e) {
        if (!cancelled) setError(e?.message || "Failed to load.");
      } finally {
        if (!cancelled) setLoading(false);
      }
    }
    run();
    return () => { cancelled = true; };
  }, [url]);

  const extracted = useMemo(() => getByPath(data, jsonPath), [data, jsonPath]);

  const codeText = useMemo(() => {
    if (typeof extracted !== "string") return "";
    return `${variableName} = ${quoteValue(extracted)}`;
  }, [extracted, variableName, quoteStyle]);

  const onCopy = async () => {
    try {
      await navigator.clipboard.writeText(codeText);
      setCopied(true);
      setTimeout(() => setCopied(false), 1500);
    } catch {}
  };

  // Outer wrapper — two rows stacked, each H px
  const wrap = {
    display: "flex",
    flexDirection: "column",
    borderRadius: 6,
    border: `1px solid ${t.border}`,
    background: t.bg,
    fontFamily: "ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, monospace",
    fontSize: 12,
    lineHeight: 1,
    color: t.text,
    overflow: "hidden",
    boxSizing: "border-box",
  };

  // Shared row — exactly H px
  const row = {
    display: "flex",
    alignItems: "center",
    height: H,
    minHeight: H,
    maxHeight: H,
    overflow: "hidden",
    boxSizing: "border-box",
  };

  const CopyIcon = () => (
    <svg width="11" height="11" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.2">
      <rect x="9" y="9" width="13" height="13" rx="2" />
      <path d="M5 15H4a2 2 0 0 1-2-2V4a2 2 0 0 1 2-2h9a2 2 0 0 1 2 2v1" />
    </svg>
  );

  const CheckIcon = () => (
    <svg width="11" height="11" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.8">
      <polyline points="20 6 9 17 4 12" />
    </svg>
  );

  const SpinIcon = () => (
    <>
      <style>{`@keyframes _rcb_spin { to { transform: rotate(360deg); } }`}</style>
      <svg width="11" height="11" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5"
        style={{ animation: "_rcb_spin 1s linear infinite", flexShrink: 0 }}>
        <path d="M21 12a9 9 0 1 1-6.219-8.56" />
      </svg>
    </>
  );

  // Loading / error / empty states use single-row fallback
  if (loading) return (
    <div style={{ ...wrap, flexDirection: "row", height: H, minHeight: H, maxHeight: H, gap: 8, padding: "0 12px", color: t.muted }}>
      <SpinIcon /> Loading {filename}…
    </div>
  );

  if (error) return (
    <div style={{ ...wrap, flexDirection: "row", height: H, minHeight: H, maxHeight: H, padding: "0 12px", gap: 6, color: t.muted }}>
      <span>✕</span> {error}
    </div>
  );

  if (!codeText) return (
    <div style={{ ...wrap, flexDirection: "row", height: H, minHeight: H, maxHeight: H, padding: "0 12px", gap: 4, color: t.muted }}>
      No string found at <code style={{ color: t.text, marginLeft: 4 }}>{jsonPath}</code>
    </div>
  );

  return (
    <div style={wrap}>

      {/* ── Row 1: language badge + filename ── */}
      <div style={{ ...row, borderBottom: `1px solid ${t.border}` }}>

        {/* Language badge */}
        <div style={{
          flexShrink: 0,
          display: "flex",
          alignItems: "center",
          height: "100%",
          padding: "0 10px",
          background: t.labelBg,
          borderRight: `1px solid ${t.border}`,
          color: t.muted,
          fontSize: 11,
          fontWeight: 600,
          letterSpacing: "0.05em",
          textTransform: "uppercase",
          whiteSpace: "nowrap",
        }}>
          {language}
        </div>

        {/* Filename */}
        <div style={{
          display: "flex",
          alignItems: "center",
          height: "100%",
          padding: "0 10px",
          color: t.muted,
          fontSize: 11,
          fontWeight: 400,
          whiteSpace: "nowrap",
          overflow: "hidden",
          textOverflow: "ellipsis",
        }}>
          {filename}
        </div>

      </div>

      {/* ── Row 2: code content + copy button ── */}
      <div style={row}>

        {/* Code: varName = "value" */}
        <div style={{
          flex: 1,
          overflow: "hidden",
          padding: "0 12px",
          whiteSpace: "nowrap",
          display: "flex",
          alignItems: "center",
        }}>
          <span style={{ color: t.varColor, flexShrink: 0 }}>{variableName}</span>
          <span style={{ color: t.muted, margin: "0 5px", flexShrink: 0 }}>=</span>
          <span style={{ color: t.strColor, overflow: "hidden", textOverflow: "ellipsis" }}>
            {quoteOpen}{extracted}{quoteOpen}
          </span>
        </div>

        {/* Copy button */}
        <button
          type="button"
          onClick={onCopy}
          style={{
            flexShrink: 0,
            display: "flex",
            alignItems: "center",
            gap: 4,
            height: "100%",
            padding: "0 10px",
            border: "none",
            borderLeft: `1px solid ${t.border}`,
            background: "transparent",
            color: copied ? t.text : t.muted,
            cursor: "pointer",
            fontSize: 11,
            fontWeight: 600,
            transition: "color 0.12s",
            whiteSpace: "nowrap",
          }}
        >
          {copied ? <CheckIcon /> : <CopyIcon />}
          {copied ? "Copied" : "Copy"}
        </button>

      </div>
    </div>
  );
};

<RemoteJsonCodeBlock
  url="https://raw.githubusercontent.com/aerdman1/aerdman1.github.io/refs/heads/main/sample_front_image.json"
  jsonPath="front_image"
  language="python"
  filename="sample_front_image.py"
  variableName="FRONT_IMAGE"
/>
