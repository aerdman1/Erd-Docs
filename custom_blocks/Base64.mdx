import React, { useEffect, useMemo, useState } from 'react';

export const RemoteJsonDoc = (props) => {
  // Props
  const url = props.url || '';
  const titleKey = props.titleKey || 'title';
  const descriptionKey = props.descriptionKey || 'description';
  const sectionsKey = props.sectionsKey || 'sections';
  const showRaw = props.showRaw !== false; // default true
  const rawLabel = props.rawLabel || 'View raw JSON';
  const maxRawHeight = props.maxRawHeight || 520;

  const [loading, setLoading] = useState(Boolean(url));
  const [error, setError] = useState('');
  const [json, setJson] = useState(null);
  const [rawOpen, setRawOpen] = useState(false);

  // Helpers (scoped inside component, per your MDX pattern)
  const isObject = (v) => {
    return Boolean(v) && typeof v === 'object' && !Array.isArray(v);
  };

  const safeString = (v) => {
    if (v === null || v === undefined) return '';
    if (typeof v === 'string') return v;
    if (typeof v === 'number' || typeof v === 'boolean') return String(v);
    return '';
  };

  const pretty = useMemo(() => {
    try {
      if (!json) return '';
      return JSON.stringify(json, null, 2);
    } catch (e) {
      return '';
    }
  }, [json]);

  useEffect(() => {
    let cancelled = false;

    async function run() {
      setError('');
      setJson(null);

      if (!url) {
        setLoading(false);
        setError('Missing url prop.');
        return;
      }

      setLoading(true);

      try {
        const res = await fetch(url, { credentials: 'omit' });
        if (!res.ok) throw new Error('HTTP ' + res.status);

        const data = await res.json();

        if (cancelled) return;
        setJson(data);
      } catch (e) {
        if (cancelled) return;
        setError(e && e.message ? e.message : 'Failed to load JSON.');
      } finally {
        if (cancelled) return;
        setLoading(false);
      }
    }

    run();

    return function cleanup() {
      cancelled = true;
    };
  }, [url]);

  // “Doc” extraction
  const docTitle = useMemo(() => {
    if (!json || !isObject(json)) return '';
    return safeString(json[titleKey]);
  }, [json, titleKey]);

  const docDescription = useMemo(() => {
    if (!json || !isObject(json)) return '';
    return safeString(json[descriptionKey]);
  }, [json, descriptionKey]);

  const docSections = useMemo(() => {
    if (!json || !isObject(json)) return [];
    const s = json[sectionsKey];
    if (!Array.isArray(s)) return [];
    return s;
  }, [json, sectionsKey]);

  // Render
  return (
    <div className="not-prose w-full rounded-2xl border border-slate-200 bg-white shadow-sm">
      <div className="p-5 border-b border-slate-100">
        <div className="text-sm font-semibold text-slate-900">
          {docTitle || 'Loaded JSON'}
        </div>
        <div className="text-xs text-slate-500 mt-1 break-all">
          {url}
        </div>
      </div>

      <div className="p-5">
        {loading && (
          <div className="text-sm text-slate-600">Loading…</div>
        )}

        {!loading && error && (
          <div className="rounded-xl border border-red-200 bg-red-50 p-4">
            <div className="text-sm font-semibold text-red-900">Could not load JSON</div>
            <div className="text-sm text-red-800 mt-1">{error}</div>
            <div className="text-xs text-red-800 mt-3">
              If this is a GitHub raw URL, make sure the file is public. If it’s private, you’ll need a proxy endpoint that returns JSON with CORS enabled.
            </div>
          </div>
        )}

        {!loading && !error && json && (
          <div className="space-y-5">
            {(docTitle || docDescription) && (
              <div>
                {docTitle && (
                  <div className="text-xl font-semibold text-slate-900">
                    {docTitle}
                  </div>
                )}
                {docDescription && (
                  <div className="text-sm text-slate-700 mt-2">
                    {docDescription}
                  </div>
                )}
              </div>
            )}

            {docSections.length > 0 && (
              <div className="space-y-4">
                {docSections.map((sec, idx) => {
                  const heading = isObject(sec) ? safeString(sec.heading) : '';
                  const body = isObject(sec) ? safeString(sec.body) : '';
                  const items = isObject(sec) && Array.isArray(sec.items) ? sec.items : [];

                  return (
                    <div key={String(idx)} className="rounded-xl border border-slate-200 bg-slate-50 p-4">
                      {heading && (
                        <div className="text-sm font-semibold text-slate-900">
                          {heading}
                        </div>
                      )}
                      {body && (
                        <div className="text-sm text-slate-700 mt-2">
                          {body}
                        </div>
                      )}
                      {items.length > 0 && (
                        <ul className="mt-3 list-disc pl-5 text-sm text-slate-700">
                          {items.map((it, j) => {
                            const label = safeString(it);
                            return <li key={String(j)}>{label}</li>;
                          })}
                        </ul>
                      )}
                    </div>
                  );
                })}
              </div>
            )}

            {showRaw && (
              <div className="rounded-xl border border-slate-200">
                <button
                  type="button"
                  onClick={() => setRawOpen(!rawOpen)}
                  className="w-full flex items-center justify-between gap-3 px-4 py-3 text-sm font-semibold text-slate-800 hover:bg-slate-50"
                >
                  <span>{rawLabel}</span>
                  <span className="text-slate-500">{rawOpen ? 'Hide' : 'Show'}</span>
                </button>

                {rawOpen && (
                  <pre
                    className="m-0 border-t border-slate-200 p-4 overflow-auto text-xs leading-relaxed"
                    style={{ maxHeight: maxRawHeight, whiteSpace: 'pre' }}
                  >
                    <code>{pretty}</code>
                  </pre>
                )}
              </div>
            )}
          </div>
        )}
      </div>
    </div>
  );
};


<RemoteJsonDoc url="https://example.com/path/to/file.json" showRaw={true} />
