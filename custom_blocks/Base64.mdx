import React, { useEffect, useMemo, useState } from 'react';

export const RemoteJsonCodeBlock = props => {
  // -------------------------
  // Props
  // -------------------------
  const url = props.url || '';
  const jsonPath = props.jsonPath || 'front_image'; // key in the JSON
  const language = props.language || 'python';
  const filename = props.filename || 'sample_front_image.py';
  const variableName = props.variableName || 'FRONT_IMAGE';
  const quoteStyle = props.quoteStyle || 'double'; // "double" | "single" | "triple"
  const maxHeight = props.maxHeight || 520;

  const [loading, setLoading] = useState(Boolean(url));
  const [error, setError] = useState('');
  const [data, setData] = useState(null);
  const [copied, setCopied] = useState(false);

  // -------------------------
  // Helpers (scoped)
  // -------------------------
  const getByPath = (obj, path) => {
    if (!obj || !path) return undefined;
    const parts = String(path)
      .split('.')
      .map(p => p.trim())
      .filter(Boolean);

    let cur = obj;
    for (const part of parts) {
      if (cur && typeof cur === 'object' && part in cur) cur = cur[part];
      else return undefined;
    }
    return cur;
  };

  const isString = v => typeof v === 'string';

  const quoteValue = v => {
    if (quoteStyle === 'triple') return `"""${v}"""`;
    if (quoteStyle === 'single') return `'${v.replace(/'/g, "\\'")}'`;
    // default double
    return `"${v.replace(/"/g, '\\"')}"`;
  };

  const buildCode = v => {
    // Python-y assignment (what you asked for originally)
    return `${variableName} = ${quoteValue(v)}`;
  };

  const onCopy = async text => {
    try {
      await navigator.clipboard.writeText(text);
      setCopied(true);
      setTimeout(() => setCopied(false), 1200);
    } catch (e) {
      // If clipboard isn't allowed, do nothing (user can still select/copy)
    }
  };

  // -------------------------
  // Fetch
  // -------------------------
  useEffect(() => {
    let cancelled = false;

    async function run() {
      setError('');
      setData(null);

      if (!url) {
        setLoading(false);
        setError('Missing url prop.');
        return;
      }

      setLoading(true);

      try {
        const res = await fetch(url, { credentials: 'omit' });
        if (!res.ok) throw new Error('HTTP ' + res.status);
        const json = await res.json();

        if (cancelled) return;
        setData(json);
      } catch (e) {
        if (cancelled) return;
        setError(e && e.message ? e.message : 'Failed to load JSON.');
      } finally {
        if (cancelled) return;
        setLoading(false);
      }
    }

    run();

    return () => {
      cancelled = true;
    };
  }, [url]);

  // -------------------------
  // Derived output
  // -------------------------
  const extracted = useMemo(() => {
    return getByPath(data, jsonPath);
  }, [data, jsonPath]);

  const codeText = useMemo(() => {
    if (!isString(extracted)) return '';
    return buildCode(extracted);
  }, [extracted]);

  // -------------------------
  // Render
  // -------------------------
  return (
    <div className="not-prose w-full rounded-2xl border border-slate-200 bg-white shadow-sm">
      <div className="px-5 py-4 border-b border-slate-100 flex items-start justify-between gap-4">
        <div className="min-w-0">
          <div className="text-sm font-semibold text-slate-900">Remote code block</div>
          <div className="text-xs text-slate-500 mt-1 break-all">{url}</div>
        </div>

        {codeText && (
          <button
            type="button"
            onClick={() => onCopy(codeText)}
            className="shrink-0 rounded-xl border border-slate-200 bg-white px-3 py-2 text-xs font-semibold text-slate-700 hover:bg-slate-50"
          >
            {copied ? 'Copied' : 'Copy'}
          </button>
        )}
      </div>

      <div className="p-5">
        {loading && <div className="text-sm text-slate-600">Loading…</div>}

        {!loading && error && (
          <div className="rounded-xl border border-red-200 bg-red-50 p-4">
            <div className="text-sm font-semibold text-red-900">Could not load JSON</div>
            <div className="text-sm text-red-800 mt-1">{error}</div>
          </div>
        )}

        {!loading && !error && !codeText && (
          <div className="rounded-xl border border-amber-200 bg-amber-50 p-4">
            <div className="text-sm font-semibold text-amber-900">No string found</div>
            <div className="text-sm text-amber-900 mt-1">
              Expected <span className="font-mono">{jsonPath}</span> to be a string in the loaded JSON.
            </div>
          </div>
        )}

        {!loading && !error && codeText && (
          <div className="rounded-xl border border-slate-200 overflow-hidden">
            <div className="px-4 py-2 border-b border-slate-200 bg-slate-50 text-xs font-semibold text-slate-700">
              {filename}
            </div>

            <pre
              className="m-0 p-4 overflow-auto text-xs leading-relaxed bg-white"
              style={{ maxHeight, whiteSpace: 'pre' }}
            >
              <code className={'language-' + language}>{codeText}</code>
            </pre>
          </div>
        )}
      </div>
    </div>
  );
};


<RemoteJsonCodeBlock
  url="https://raw.githubusercontent.com/aerdman1/aerdman1.github.io/refs/heads/main/sample_front_image.json"
  jsonPath="front_image"
  language="python"
  filename="sample_front_image.py"
  variableName="FRONT_IMAGE"
/>
