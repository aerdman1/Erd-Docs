  function safeText(v) { if (v === null || v === undefined) return ""; return String(v); }
  function safeJsonParse(s) { try { return JSON.parse(s); } catch (e) { return null; } }

  function absolutizeUrlMaybe(u) {
    var s = safeText(u).trim(); if (!s) return "";
    try { return new URL(s, window.location.origin).href; } catch (e) { return s; }

  function normalizeChangelogLinkMaybe(link) {
    var s = safeText(link).trim(); if (!s) return "";
    s = absolutizeUrlMaybe(s);
    if (!rewriteVersionedChangelogLinks) return s;
    var re = /\/v\d+(?:\.\d+)*\/changelog(\/|$)/i;
    try { var u = new URL(s, window.location.origin); u.pathname = safeText(u.pathname).replace(re, "/changelog$1"); return u.toString(); }
    catch (e2) { return s.replace(re, "/changelog$1"); }
  }

  function canonicalizeForFetch(link) {
    var href = normalizeChangelogLinkMaybe(link); if (!href) return "";
    try {
      var u = new URL(href, window.location.origin);
      if (stripQueryParamsForPostFetch) { u.search = ""; u.hash = ""; }
      return u.toString();
    } catch (e) {
      if (!stripQueryParamsForPostFetch) return href;
      return href.split("#")[0].split("?")[0];
    }
  }

  function stripHtml(html) {
    var s = safeText(html);
    s = s.replace(/<script[\s\S]*?>[\s\S]*?<\/script>/gi, "");
    s = s.replace(/<style[\s\S]*?>[\s\S]*?<\/style>/gi, "");
    s = s.replace(/<br\s*\/?>/gi, "\n"); s = s.replace(/<\/p\s*>/gi, "\n\n");
    s = s.replace(/<\/div\s*>/gi, "\n"); s = s.replace(/<\/li\s*>/gi, "\n");
    s = s.replace(/<\/h[1-6]\s*>/gi, "\n\n"); s = s.replace(/<hr\s*\/?>/gi, "\n\n");
    s = s.replace(/<\/?[^>]+>/g, " "); s = s.replace(/[ \t]+\n/g, "\n");
    s = s.replace(/\n{3,}/g, "\n\n"); s = s.replace(/[ \t]{2,}/g, " ");
    return s.trim();
  }

  function firstChildText(el, selectors) {
    if (!el) return "";
    for (var i = 0; i < selectors.length; i++) { var node = el.querySelector(selectors[i]); if (node && node.textContent) return String(node.textContent).trim(); }
    return "";
  }

  function parsePubDate(s) { var v = safeText(s).trim(); if (!v) return null; var d = new Date(v); if (isNaN(d.getTime())) return null; return d; }

  function coerceDate(v) {
    if (!v) return null;
    try { if (v instanceof Date) return isNaN(v.getTime()) ? null : v; } catch (e) {}
    try { var d = new Date(String(v)); return isNaN(d.getTime()) ? null : d; } catch (e2) { return null; }
  }

  function toTime(v) { var d = coerceDate(v); return d ? d.getTime() : 0; }

  function formatDate(d) {
    var dd = coerceDate(d); if (!dd) return "";
    try { return dd.toLocaleDateString(undefined, { year: "numeric", month: "short", day: "numeric" }); }
    catch (e) { return dd.toDateString(); }
  }

  function titleCase(s) {
    var t = safeText(s).trim(); if (!t) return "";
    return t.split(/[\s_-]+/).filter(Boolean).map(function (w) { return w ? w.charAt(0).toUpperCase() + w.slice(1) : ""; }).join(" ");
  }

  function normalizeTypeRaw(rawType) {
    var t = safeText(rawType).trim().toLowerCase(); if (!t) return { key: "", label: "" };
    if (t === "improved") t = "changed";
    if (t === "knownissue" || t === "known_issue" || t === "known-issue") t = "known issue";
    return { key: t, label: titleCase(t) };
  }

  function normalizeMetaRaw(raw) { var t = safeText(raw).trim(); if (!t) return { key: "", label: "" }; return { key: t.toLowerCase(), label: t }; }

    return out;

  function isKnownTypeKey(rawType) {

  function pickPrefixedValue(values, prefixes) {
    if (!values || !values.length) return "";
    for (var i = 0; i < values.length; i++) {
      var raw = safeText(values[i]).trim(); if (!raw) continue;
      var lower = raw.toLowerCase();
      for (var p = 0; p < prefixes.length; p++) { var pre = prefixes[p]; if (lower.indexOf(pre) === 0) return raw.slice(pre.length).trim(); }
    }
    return "";
  }

  function extractTokenValue(textValue, name) {
    var s = safeText(textValue); if (!s) return "";
    try { var re1 = new RegExp("\\{\\%\\s*" + name + "\\s*[:=]\\s*([^%}]+?)\\s*\\%\\}", "i"); var m1 = s.match(re1); if (m1 && m1[1]) return String(m1[1]).trim(); } catch (e) {}
    try { var re2 = new RegExp("\\[" + name + "\\s*[:=]\\s*([^\\]]+?)\\]", "i"); var m2 = s.match(re2); if (m2 && m2[1]) return String(m2[1]).trim(); } catch (e2) {}
    return "";
  }

  function stripMetaTokens(textValue) {
    var s = safeText(textValue); if (!s) return "";
    s = s.replace(/\{\%\s*(product|version|change)\s*[:=][^%}]+\%\}/gi, "");
    s = s.replace(/\[(product|version|change)\s*[:=][^\]]+\]/gi, "");
    s = s.replace(/[ \t]+\n/g, "\n"); s = s.replace(/\n{3,}/g, "\n\n");
    return s.trim();
  }

    var allowed = { a: true, p: true, br: true, strong: true, em: true, b: true, i: true, code: true, pre: true, ul: true, ol: true, li: true, blockquote: true, h1: true, h2: true, h3: true, h4: true, h5: true, h6: true, hr: true, div: true, span: true, table: true, thead: true, tbody: true, tfoot: true, tr: true, th: true, td: true };
    var doc = new DOMParser().parseFromString("<div>" + safeText(html) + "</div>", "text/html");
    var root = doc.body && doc.body.firstChild ? doc.body.firstChild : null; if (!root) return "";

    function cleanEl(el) {
      if (!el || el.nodeType !== 1) return;
      var tag = (el.tagName || "").toLowerCase();
      if (tag === "script" || tag === "style" || tag === "iframe" || tag === "object" || tag === "embed") { el.parentNode && el.parentNode.removeChild(el); return; }
      if (!allowed[tag]) { var parent = el.parentNode; if (!parent) return; while (el.firstChild) parent.insertBefore(el.firstChild, el); parent.removeChild(el); return; }
      var attrs = el.attributes ? Array.prototype.slice.call(el.attributes) : [];
      for (var i = 0; i < attrs.length; i++) {
        var name = attrs[i].name;
        if (/^on/i.test(name) || name === "style") { el.removeAttribute(name); continue; }
        if ((tag === "td" || tag === "th") && (name === "colspan" || name === "rowspan")) continue;
        if (name === "class") continue;
        if (tag === "a") { if (name === "href" || name === "target" || name === "rel") continue; el.removeAttribute(name); continue; }
        el.removeAttribute(name);
      }
      if (tag === "a") {
        var href = safeText(el.getAttribute("href")).trim(); if (/^javascript:/i.test(href)) el.removeAttribute("href");
        var tgt = safeText(el.getAttribute("target")).trim();
        if (tgt === "_blank") { var rel = safeText(el.getAttribute("rel")).trim(); if (!rel || rel.indexOf("noopener") === -1) el.setAttribute("rel", "noopener noreferrer"); }
      }
      if (tag === "table") {
        var p = el.parentNode;
        if (p && (!p.className || String(p.className).indexOf("rmChangelogFeed__tableWrap") === -1)) {
          var wrap = doc.createElement("div"); wrap.setAttribute("class", "rmChangelogFeed__tableWrap"); p.insertBefore(wrap, el); wrap.appendChild(el);
        }
      }
      var kids = el.childNodes ? Array.prototype.slice.call(el.childNodes) : [];
      for (var k = 0; k < kids.length; k++) { if (kids[k] && kids[k].nodeType === 1) cleanEl(kids[k]); }
    }

    cleanEl(root);
    var out = safeText(root.innerHTML).trim(); if (out.length > 70000) out = out.slice(0, 70000);
    return out;
  }

  function markdownToHtmlSimple(md) {
    var src = safeText(md).replace(/\r\n/g, "\n"); var lines = src.split("\n");
    var htmlParts = [], paraLines = [], listType = "", listItems = [], inCode = false, codeLines = [];
    function flushPara() { if (!paraLines.length) return; htmlParts.push("<p>" + inlineMarkdownToHtml(paraLines.join("\n")).replace(/\n/g, "<br/>") + "</p>"); paraLines = []; }
    function flushList() { if (!listType || !listItems.length) { listType = ""; listItems = []; return; } htmlParts.push("<" + listType + ">" + listItems.map(function (li) { return "<li>" + inlineMarkdownToHtml(li) + "</li>"; }).join("") + "</" + listType + ">"); listType = ""; listItems = []; }
    function flushCode() { if (!codeLines.length) return; htmlParts.push("<pre><code>" + escapeHtml(codeLines.join("\n")) + "</code></pre>"); codeLines = []; }
    for (var i = 0; i < lines.length; i++) {
      var line = String(lines[i] || "");
      if (/^```/.test(line.trim())) { if (inCode) { inCode = false; flushCode(); } else { flushPara(); flushList(); inCode = true; codeLines = []; } continue; }
      if (inCode) { codeLines.push(line); continue; }
      var hm = line.match(/^(#{1,6})\s+(.+)$/);
      if (hm) { flushPara(); flushList(); htmlParts.push("<h" + hm[1].length + ">" + inlineMarkdownToHtml(hm[2].trim()) + "</h" + hm[1].length + ">"); continue; }
      var ulm = line.match(/^\s*[-\*\+]\s+(.+)$/); var olm = line.match(/^\s*(\d+)\.\s+(.+)$/);
      if (ulm && ulm[1]) { flushPara(); if (listType && listType !== "ul") flushList(); listType = "ul"; listItems.push(ulm[1].trim()); continue; }
      if (olm && olm[2]) { flushPara(); if (listType && listType !== "ol") flushList(); listType = "ol"; listItems.push(olm[2].trim()); continue; }
      if (!line.trim()) { flushPara(); flushList(); continue; }
      flushList(); paraLines.push(line);
    }
    flushPara(); flushList(); if (inCode) flushCode();
    return htmlParts.join("");
  }

      var it = items[i]; if (!it || !it.link || it.__hydrated) continue;
      if (needsHydration(it.descriptionText)) toHydrate.push({ idx: i, link: it.link });
      if (toHydrate.length >= hydrateMax) break;
    }
    if (!toHydrate.length) return;

    try { if (hydrateAbortRef.current) hydrateAbortRef.current.abort(); } catch (e0) {}
    var ac = new AbortController(); hydrateAbortRef.current = ac;

    runWithLimit(toHydrate.map(function (x) { return function () { return fetchAndExtractExcerpt(x.link, ac.signal); }; }), hydrateConcurrency)
      .then(function (excerpts) {
        setItems(function (prev) {
          var next = prev.slice(0);
          for (var j = 0; j < toHydrate.length; j++) {
            var ex = excerpts[j]; var target = toHydrate[j]; if (!target) continue;
            var cur = next[target.idx]; if (!cur) continue;
            var updated = Object.assign({}, cur, { __hydrated: true });
            if (ex && ex.html && needsHydration(cur.descriptionText)) { updated.descriptionHtml = safeText(ex.html); updated.descriptionText = stripHtml(ex.html); }
            next[target.idx] = updated;
          }
          return next;
        });
      });

    return function () { try { ac.abort(); } catch (e1) {} };
  }, [hydrateFromPermalink, hydrateMax, hydrateConcurrency, minExcerptChars, excerptMaxChars, modalContentSelector, loading, error, items, showDescriptions, fetchTimeoutMs]);

  // -----------------------------
  // Modal
  // -----------------------------
  function modalCacheKey(link) { return "rm_changelog_post_modal_html_v10__" + safeText(link); }

  function openModalForItem(it) {
    if (!enableModal) return;
    var rawHref = safeText(it && it.link); if (!rawHref) return;
    var hrefForOpen = normalizeChangelogLinkMaybe(rawHref); var hrefForFetch = canonicalizeForFetch(hrefForOpen);

    setModalOpen(true); setModalLoading(true); setModalError(""); setModalHtml(""); setModalMode("html");
    setModalTitle(safeText(it && it.title) || "Changelog post"); setModalHref(hrefForOpen);

    try { if (modalAbortRef.current) modalAbortRef.current.abort(); } catch (e0) {}
    var ac = new AbortController(); modalAbortRef.current = ac;
    var ck = modalCacheKey(hrefForFetch || hrefForOpen);

    // PERF FIX #9: modal cache is now checked properly (was always skipped when modalCacheMinutes=0 and then actively deleted)
    if (Number(modalCacheMinutes) > 0) {
      var cachedRaw = readLocal(ck); if (cachedRaw) {
        var cached = safeJsonParse(cachedRaw);
        if (cached && isModalCacheValid(cached)) { setModalHtml(String(cached.html)); if (cached.title) setModalTitle(String(cached.title)); setModalLoading(false); return; }
      }
    }

    function writeModalCacheMaybe(title, html) { if (!(Number(modalCacheMinutes) > 0)) return; try { writeLocal(ck, JSON.stringify({ ts: Date.now(), html: html, title: title || "" })); } catch (e2) {} }
    function doneHtml(title, html) { if (ac.signal && ac.signal.aborted) return; if (html) { setModalHtml(html); if (title) setModalTitle(title); setModalLoading(false); return true; } return false; }

    var first = Promise.resolve({ html: "", title: "" });
    if (preferRenderedHtmlForModal) first = fetchRenderedChangelogModal(hrefForOpen, ac.signal);

    first
      .then(function (res1) { if (res1 && res1.html) { writeModalCacheMaybe(res1.title, res1.html); return doneHtml(res1.title, res1.html); } return false; })
      .then(function (ok) {
        if (ok) return;
        if (preferApiNextForModal) return fetchApiNextChangelog(hrefForOpen, ac.signal).then(function (res2) { if (res2 && res2.html) { writeModalCacheMaybe(res2.title, res2.html); return doneHtml(res2.title, res2.html); } return false; }).catch(function () { return false; });
        return false;
      })
      .then(function (ok2) {
        if (ok2) return;
        if (fallbackToIframeInModal) { setModalMode("iframe"); setModalLoading(false); return; }
        setModalError("Couldn't load the post content."); setModalLoading(false);
      })
      .catch(function (err) {
        if (err && err.name === "AbortError") return;
        if (fallbackToIframeInModal) { setModalMode("iframe"); setModalLoading(false); return; }
        setModalError(err ? String(err.message || err) : "Unknown error."); setModalLoading(false);
      });
  }

  function closeModal() {
    setModalOpen(false); setModalLoading(false); setModalError(""); setModalHtml(""); setModalHref(""); setModalMode("html");
    try { if (modalAbortRef.current) modalAbortRef.current.abort(); } catch (e) {}
  }

  useEffect(() => {
    if (typeof window === "undefined") return; if (!modalOpen) return;
    function onKey(e) { if (e && e.key === "Escape") closeModal(); }
    try { document.addEventListener("keydown", onKey); } catch (e2) {}
    var prevOverflow = ""; try { prevOverflow = document.body.style.overflow; document.body.style.overflow = "hidden"; } catch (e3) {}
    return function () { try { document.removeEventListener("keydown", onKey); } catch (e4) {} try { document.body.style.overflow = prevOverflow; } catch (e5) {} };
  }, [modalOpen]);

  function onModalFrameLoad() {
    if (!modalFrameRef.current) return;
    try {
      var w = modalFrameRef.current.contentWindow; var d = w && w.document ? w.document : null; if (!d) return;
      var st = d.createElement("style"); st.type = "text/css";
      st.appendChild(d.createTextNode(["header, nav, aside, .content-toc, section.content-toc, .hub-sidebar, .rm-TopNav, .rm-Sidebar { display:none !important; }", ".rm-Guides .content-body, main, article { max-width: 980px !important; margin: 0 auto !important; padding: 18px !important; }", "body { background: #fff !important; }"].join("\n")));
      (d.head || d.documentElement).appendChild(st);
    } catch (e) {}
  }

  // -----------------------------
  // Derived lists
  // -----------------------------
  var typeOptions = useMemo(() => {
    var seen = {}; var out = [{ key: "all", label: "All types" }];
    for (var i = 0; i < items.length; i++) { var k = safeText(items[i].typeKey).trim().toLowerCase(); var l = safeText(items[i].typeLabel).trim(); if (!k || !l || seen[k]) continue; seen[k] = true; out.push({ key: k, label: l }); }
    out.sort(function (a, b) { if (a.key === "all") return -1; if (b.key === "all") return 1; return String(a.label).localeCompare(String(b.label)); });
    return out;
  }, [items]);

  var productOptions = useMemo(() => {
    var seen = {}; var out = [{ key: "all", label: "All products" }];
    for (var i = 0; i < items.length; i++) { var k = safeText(items[i].productKey).trim().toLowerCase(); var l = safeText(items[i].productLabel).trim(); if (!k || !l || seen[k]) continue; seen[k] = true; out.push({ key: k, label: l }); }
    out.sort(function (a, b) { if (a.key === "all") return -1; if (b.key === "all") return 1; return String(a.label).localeCompare(String(b.label)); });
    return out;
  }, [items]);

  var versionOptions = useMemo(() => {
    var seen = {}; var out = [{ key: "all", label: "All versions" }];
    for (var i = 0; i < items.length; i++) { var k = safeText(items[i].versionKey).trim().toLowerCase(); var l = safeText(items[i].versionLabel).trim(); if (!k || !l || seen[k]) continue; seen[k] = true; out.push({ key: k, label: l }); }
    out.sort(function (a, b) { if (a.key === "all") return -1; if (b.key === "all") return 1; return String(a.label).localeCompare(String(b.label)); });
    return out;
  }, [items]);

  var authorOptions = useMemo(() => {
    var seen = {}; var out = [];
    for (var i = 0; i < items.length; i++) { var a = safeText(items[i].author).trim(); if (!a) continue; var k = a.toLowerCase(); if (seen[k]) continue; seen[k] = true; out.push(a); }
    out.sort(function (a1, a2) { return a1.localeCompare(a2); }); return out;
  }, [items]);

  var filtered = useMemo(() => {
    var q = safeText(search).trim().toLowerCase(); var a = safeText(author).trim().toLowerCase(); var t = safeText(typeFilter).trim().toLowerCase();
    var applyProduct = Boolean(lockedProduct) || showProductFilter; var applyVersion = Boolean(lockedVersion) || showVersionFilter;
    var p = safeText(lockedProduct ? lockedProduct : productFilter).trim().toLowerCase(); var v = safeText(lockedVersion ? lockedVersion : versionFilter).trim().toLowerCase();
    var result = [];
    for (var i = 0; i < items.length; i++) {
      var it = items[i]; if (!it) continue;
      if (showTypeFilter && t !== "all") { if (safeText(it.typeKey).toLowerCase() !== t) continue; }
      if (applyProduct && p !== "all") { if (safeText(it.productKey).toLowerCase() !== p) continue; }
      if (applyVersion && v !== "all") { if (safeText(it.versionKey).toLowerCase() !== v) continue; }
      if (showAuthorFilter && a) { if (safeText(it.author).toLowerCase().indexOf(a) === -1) continue; }
      if (q) { var hay = (safeText(it.title) + " " + safeText(it.descriptionText) + " " + safeText(it.typeLabel) + " " + safeText(it.productLabel) + " " + safeText(it.versionLabel)).toLowerCase(); if (hay.indexOf(q) === -1) continue; }
      result.push(it);
    }
    result.sort(function (x, y) { var dx = toTime(x && x.pubDate); var dy = toTime(y && y.pubDate); return sort === "oldest" ? dx - dy : dy - dx; });
    return result;
  }, [items, typeFilter, productFilter, versionFilter, lockedProduct, lockedVersion, search, author, sort, showTypeFilter, showProductFilter, showVersionFilter, showAuthorFilter]);

  useEffect(() => { setRenderCount(initialRenderCount); }, [typeFilter, productFilter, versionFilter, search, author, sort, initialRenderCount]);

  var visibleFiltered = useMemo(() => { return filtered.slice(0, Math.max(0, Math.min(renderCount, filtered.length))); }, [filtered, renderCount]);

  useEffect(() => {
    if (typeof window === "undefined") return; if (loading || error) return; if (!filtered || !filtered.length) return; if (!autoLoadMore) return;
    var target = filtered.length; if (renderCount >= target) return;
    var idleId = null; var timerId = null; var cancelled = false;
    function step() { if (cancelled) return; setRenderCount(function (prev) { var next = prev + Math.max(1, renderBatchSize); return next > target ? target : next; }); }
    if (typeof window.requestIdleCallback === "function") { idleId = window.requestIdleCallback(step, { timeout: Math.max(1, renderIdleTimeoutMs) }); } else { timerId = window.setTimeout(step, 0); }
    return function () {
      cancelled = true;
      if (idleId && typeof window.cancelIdleCallback === "function") { try { window.cancelIdleCallback(idleId); } catch (e) {} }
      if (timerId) { try { window.clearTimeout(timerId); } catch (e2) {} }
    };
  }, [loading, error, filtered.length, renderCount, renderBatchSize, renderIdleTimeoutMs, autoLoadMore]);

  // -----------------------------
  // Render
  // -----------------------------
  var rootStyle = { width: "100%" };
  rootStyle["--rmChangelogFeedModalMaxWidth"] = String(modalMaxWidth) + "px";
  rootStyle["--rmChangelogFeedColumns"] = String(Math.max(1, Math.floor(cardsPerRow || 2)));
  var barStyle = stylesRef.current.barStyle; var groupStyle = stylesRef.current.groupStyle; var labelStyle = stylesRef.current.labelStyle;
  var controlStyle = stylesRef.current.controlStyle; var metaStyle = stylesRef.current.metaStyle; var cardStyle = stylesRef.current.cardStyle;

  function Badge(props2) {
    var kind = safeText(props2.kind).trim().toLowerCase(); var label = safeText(props2.label).trim();
    if (!kind || !label) return null;
    var icon = svgIcon(kind);
    return (
      <span style={{ display: "inline-flex", alignItems: "center", gap: 8, padding: "6px 10px", borderRadius: 999, fontWeight: 800, fontSize: 12, letterSpacing: ".01em", background: typeColor(kind), border: "1px solid " + typeBorder(kind), color: "var(--color-text-default, #000)" }}>
        {icon ? <span aria-hidden="true" style={{ width: 18, height: 18, borderRadius: 6, display: "inline-flex", alignItems: "center", justifyContent: "center", background: "rgba(0,0,0,.12)" }} dangerouslySetInnerHTML={{ __html: icon }} /> : null}
        {label}
      </span>
    );
  }

  function Row(props2) {
    var it = props2.item;
    var title = safeText(it.title) || "Untitled"; var href = safeText(it.link) || "#"; var when = formatDate(it.pubDate); var who = safeText(it.author);
    var typeKey = safeText(it.typeKey); var typeLabel = safeText(it.typeLabel);
    var versionLabelRaw = safeText(it.versionLabel).trim();
    var versionDisplay = versionLabelRaw && versionLabelRaw.toLowerCase().indexOf("v") !== 0 ? "v" + versionLabelRaw : versionLabelRaw;
    var htmlExcerpt = safeText(it.descriptionHtml).trim(); var rawDesc = safeText(it.descriptionText).trim();
    var previewText = truncateText(rawDesc, Math.max(40, Number(previewMaxChars) || 280));
    var textExcerpt = truncateText(rawDesc, excerptMaxChars);
    var anchorId = safeText(it.__id) || stableIdFromLink(href);
    function open() { openModalForItem(it); }
    function SeeMore() {
      if (!showSeeMore) return null;
      if (enableModal) return <button type="button" onClick={open} style={{ padding: 0, marginLeft: 8, border: 0, background: "transparent", cursor: "pointer", color: "var(--color-text-link, #1f78ff)", fontWeight: 900 }} aria-label="See more">See more…</button>;
      if (href && href !== "#") return <a href={href} style={{ marginLeft: 8, fontWeight: 900, color: "var(--color-text-link, #1f78ff)" }}>See more…</a>;
      return null;
    }
    return (
      <div id={anchorId} style={cardStyle}>
        <div style={{ display: "flex", gap: 10, alignItems: "flex-start", justifyContent: "space-between", flexWrap: "wrap" }}>
          {showTypeFilter ? <Badge kind={typeKey} label={typeLabel} /> : null}
          <div style={{ fontSize: 12, opacity: 0.72, paddingTop: 2 }}>
            {when ? when : null}{showVersionOnCards && when && versionDisplay ? " • " : null}{showVersionOnCards && versionDisplay ? versionDisplay : null}
            {showAuthorFilter && (when || (showVersionOnCards && versionDisplay)) && who ? " • " : null}{showAuthorFilter && who ? who : null}
          </div>
        </div>
        <div style={{ marginTop: 10 }}>
          {showTitles ? <button type="button" onClick={open} style={{ padding: 0, margin: 0, border: 0, background: "transparent", textAlign: "left", cursor: enableModal ? "pointer" : "default", color: "var(--color-text-link, #1f78ff)", fontWeight: 900, fontSize: "1.15rem", lineHeight: 1.25 }} aria-label={"Open " + title}>{title}</button> : null}
          {!showTitles ? <div style={{ fontSize: ".98rem", opacity: 0.95, lineHeight: 1.55 }}>{previewText ? previewText : null}{previewText ? <SeeMore /> : null}{!previewText ? <SeeMore /> : null}</div> : null}
        </div>
        {showTitles && showPreview && previewText ? <div style={{ marginTop: 8, fontSize: ".98rem", opacity: 0.9, lineHeight: 1.55, whiteSpace: "pre-line" }}>{previewText}<SeeMore /></div> : null}
        {showTitles && showDescriptions ? (htmlExcerpt ? <div className="rmChangelogFeed__excerpt" style={{ marginTop: 10, fontSize: ".98rem", opacity: 0.95, lineHeight: 1.55 }} dangerouslySetInnerHTML={{ __html: htmlExcerpt }} /> : textExcerpt ? <div style={{ marginTop: 10, fontSize: ".98rem", opacity: 0.9, lineHeight: 1.55, whiteSpace: "pre-line" }}>{textExcerpt}<SeeMore /></div> : null) : null}
      </div>
    );
  }

  return (
    <div style={rootStyle}>
      <style>{styleTextRef.current}</style>
      <div className="rmChangelogFeed">
        <div style={barStyle}>

          {showVersionFilter && !lockedVersion ? <div style={groupStyle}><div style={labelStyle}>Version</div><select value={versionFilter} onChange={e => setVersionFilter(e.target.value)} style={controlStyle}>{versionOptions.map(function (o) { return <option key={o.key} value={o.key}>{o.label}</option>; })}</select></div> : null}

          {showSearch ? <div style={groupStyle}><div style={labelStyle}>Search</div><input value={search} onChange={e => setSearch(e.target.value)} placeholder="Title or content…" style={controlStyle} /></div> : null}

          <div style={metaStyle}>
            {loading ? "Loading…" : null}{!loading && refreshing ? "Refreshing…" : null}{!loading && error ? "Error: " + error : null}
            {!loading && !error ? visibleFiltered.length + " / " + items.length + " shown" + (visibleFiltered.length < filtered.length ? " • loading more…" : "") : null}
          </div>
        </div>

        {loading ? <div style={{ opacity: 0.75 }}>Loading RSS feed…</div> : null}
        {!loading && error ? <div style={{ padding: 12, borderRadius: 12, border: "1px solid rgba(239,68,68,.35)", background: "rgba(239,68,68,.08)" }}><div style={{ fontWeight: 800 }}>Couldn't load RSS</div><div style={{ marginTop: 6, opacity: 0.85, lineHeight: 1.5 }}>{error}<div style={{ marginTop: 8, fontSize: 12, opacity: 0.75 }}>If this URL is cross-origin, the browser may block it due to CORS. Using a same-origin path (like <code>/changelog.rss</code>) is the safest option.</div></div></div> : null}

                <div style={{ opacity: 0.7, fontSize: 12, padding: "6px 2px" }}>
                  {autoLoadMore ? "Loading more posts… (" + visibleFiltered.length + "/" + filtered.length + ")" : "Showing " + visibleFiltered.length + " of " + filtered.length}
                  <button type="button" onClick={function () { if (autoLoadMore) { setRenderCount(filtered.length); return; } setRenderCount(function (prev) { var next = prev + Math.max(1, renderBatchSize); return next > filtered.length ? filtered.length : next; }); }} style={{ marginLeft: 10, border: "1px solid var(--color-border-default, rgba(0,0,0,.12))", background: "var(--color-bg-page, #fff)", borderRadius: 10, padding: "6px 10px", cursor: "pointer", fontWeight: 800, fontSize: 12 }}>{autoLoadMore ? "Load all now" : "Load more"}</button>
                  {!autoLoadMore ? <button type="button" onClick={function () { setRenderCount(filtered.length); }} style={{ marginLeft: 10, border: "1px solid var(--color-border-default, rgba(0,0,0,.12))", background: "var(--color-bg-page, #fff)", borderRadius: 10, padding: "6px 10px", cursor: "pointer", fontWeight: 800, fontSize: 12 }}>Load all</button> : null}
                </div>
              ) : null}
            </div>
          </div>
        ) : null}
      </div>

      {modalOpen ? (
        <div className="rmChangelogFeed__modalOverlay" role="dialog" aria-modal="true" aria-label="Changelog post" onMouseDown={function (e) { if (e && e.target && e.target.className && String(e.target.className).indexOf("rmChangelogFeed__modalOverlay") !== -1) closeModal(); }}>
          <div className="rmChangelogFeed__modal" onMouseDown={function (e) { e && e.stopPropagation && e.stopPropagation(); }}>
            <div className="rmChangelogFeed__modalHeader">
              <div className="rmChangelogFeed__modalTitle">{modalTitle || "Changelog post"}</div>
              <div className="rmChangelogFeed__modalActions">
                {modalHref ? <a href={modalHref} style={{ fontSize: 13, fontWeight: 900, color: "var(--color-text-link, #1f78ff)" }} target="_blank" rel="noopener noreferrer">Open in new tab</a> : null}
                <button type="button" className="rmChangelogFeed__modalBtn" onClick={closeModal}>Close</button>
              </div>
            </div>
            <div className="rmChangelogFeed__modalBody">
              {modalLoading ? <div style={{ opacity: 0.75 }}>Loading full post…</div> : null}
              {!modalLoading && modalError ? <div style={{ padding: 12, borderRadius: 12, border: "1px solid rgba(239,68,68,.35)", background: "rgba(239,68,68,.08)" }}><div style={{ fontWeight: 800 }}>Couldn't load the post</div><div style={{ marginTop: 6, opacity: 0.85, lineHeight: 1.5 }}>{modalError}</div></div> : null}
              {!modalLoading && !modalError && modalMode === "html" && modalHtml ? <div className="rmChangelogFeed__excerpt" dangerouslySetInnerHTML={{ __html: modalHtml }} /> : null}
              {!modalLoading && !modalError && modalMode === "iframe" && modalHref ? <iframe ref={modalFrameRef} src={canonicalizeForFetch(modalHref) || modalHref} title={modalTitle || "Changelog post"} onLoad={onModalFrameLoad} style={{ width: "100%", height: "78vh", border: "1px solid rgba(0,0,0,.12)", borderRadius: 12 }} /> : null}
            </div>
          </div>
        </div>
      ) : null}
    </div>
  );
};
