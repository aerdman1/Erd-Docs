import React, { useEffect, useMemo, useRef, useState } from "react";

export const ChangelogTable = props => {
  var rssUrl = props && props.rssUrl ? String(props.rssUrl) : detectDefaultRssUrl();
  var maxItems = props && props.maxItems ? Number(props.maxItems) : 200;
  var cacheMinutes = props && props.cacheMinutes ? Number(props.cacheMinutes) : 0;
  var fetchTimeoutMs = props && props.fetchTimeoutMs ? Number(props.fetchTimeoutMs) : 5000;
  var hydrateContent = props && typeof props.hydrateContent === "boolean" ? props.hydrateContent : true;
  var hydrateMax = props && props.hydrateMax ? Number(props.hydrateMax) : 50;
  var hydrateConcurrency = props && props.hydrateConcurrency ? Number(props.hydrateConcurrency) : 4;
  var excerptMaxChars = props && props.excerptMaxChars ? Number(props.excerptMaxChars) : 400;

  // Layout toggle
  var defaultLayout = props && props.defaultLayout === "cards" ? "cards" : "table";
  var showLayoutToggle = props && typeof props.showLayoutToggle === "boolean" ? props.showLayoutToggle : true;

  // Column visibility
  var showVersionCol = props && typeof props.showVersionCol === "boolean" ? props.showVersionCol : true;
  var showDomainCol = props && typeof props.showDomainCol === "boolean" ? props.showDomainCol : true;
  var showChangeCol = props && typeof props.showChangeCol === "boolean" ? props.showChangeCol : true;
  var showDescriptionCol = props && typeof props.showDescriptionCol === "boolean" ? props.showDescriptionCol : true;
  var showDateCol = props && typeof props.showDateCol === "boolean" ? props.showDateCol : true;

  // Filter bar visibility
  var showVersionFilter = props && typeof props.showVersionFilter === "boolean" ? props.showVersionFilter : true;
  var showDomainFilter = props && typeof props.showDomainFilter === "boolean" ? props.showDomainFilter : true;
  var showChangeFilter = props && typeof props.showChangeFilter === "boolean" ? props.showChangeFilter : true;
  var showSearch = props && typeof props.showSearch === "boolean" ? props.showSearch : true;

  // Custom column labels
  var versionLabel = props && props.versionLabel ? String(props.versionLabel) : "Version";
  var domainLabel = props && props.domainLabel ? String(props.domainLabel) : "Domain";
  var changeLabel = props && props.changeLabel ? String(props.changeLabel) : "Change";
  var descriptionLabel = props && props.descriptionLabel ? String(props.descriptionLabel) : "Description";
  var dateLabel = props && props.dateLabel ? String(props.dateLabel) : "Date";

  var [items, setItems] = useState([]);
  var [loading, setLoading] = useState(true);
  var [error, setError] = useState("");
  var [versionF, setVersionF] = useState("all");
  var [domainF, setDomainF] = useState("all");
  var [changeF, setChangeF] = useState("all");
  var [search, setSearch] = useState("");
  var [sortCol, setSortCol] = useState("pubDate");
  var [sortDir, setSortDir] = useState("desc");
  var [layout, setLayout] = useState(defaultLayout);

  var abortRef = useRef(null);
  var hydrateAbortRef = useRef(null);

  function safeText(v) { if (v === null || v === undefined) return ""; return String(v); }
  function safeJsonParse(s) { try { return JSON.parse(s); } catch (e) { return null; } }

  function readMeta(name) { try { var el = document.querySelector("meta[name='" + String(name) + "']"); return el ? safeText(el.getAttribute("content")).trim() : ""; } catch (e) { return ""; } }
  function readSsrProp(name) { try { var el = document.getElementById("ssr-props"); if (!el || !el.textContent) return ""; var data = safeJsonParse(el.textContent); return data && data[name] ? safeText(data[name]).trim() : ""; } catch (e) { return ""; } }
  function detectProjectBasePath() { var bp = readMeta("readme-basepath-childProject"); if (bp) return "/" + bp.replace(/^\/+|\/+$/g, ""); var sb = readSsrProp("baseUrl"); if (sb && sb !== "/") return "/" + sb.replace(/^\/+|\/+$/g, ""); return ""; }
  function detectDefaultRssUrl() { var bp = detectProjectBasePath(); return (bp ? bp : "") + "/changelog.rss"; }

  function coerceDate(v) {
    if (!v) return null;
    try { if (v instanceof Date) return isNaN(v.getTime()) ? null : v; } catch (e) {}
    try { var d = new Date(String(v)); return isNaN(d.getTime()) ? null : d; } catch (e2) { return null; }
  }

  function toTime(v) { var d = coerceDate(v); return d ? d.getTime() : 0; }

  function formatDate(d) {
    var dd = coerceDate(d); if (!dd) return "";
    try { return dd.toLocaleDateString(undefined, { year: "numeric", month: "short", day: "numeric" }); }
    catch (e) { return dd.toDateString(); }
  }

  function stripHtml(html) {
    var s = safeText(html);
    s = s.replace(/<script[\s\S]*?>[\s\S]*?<\/script>/gi, "");
    s = s.replace(/<style[\s\S]*?>[\s\S]*?<\/style>/gi, "");
    s = s.replace(/<br\s*\/?>/gi, " "); s = s.replace(/<\/p\s*>/gi, " ");
    s = s.replace(/<[^>]+>/g, " "); s = s.replace(/[ \t]{2,}/g, " ");
    return s.trim();
  }

  function truncateText(s, n) { var t = safeText(s); if (t.length <= n) return t; return t.slice(0, n).trim() + "…"; }

  function extractToken(text, name) {
    var s = safeText(text); if (!s) return "";
    try { var re1 = new RegExp("\\{\\%\\s*" + name + "\\s*[:=]\\s*([^%}]+?)\\s*\\%\\}", "i"); var m1 = s.match(re1); if (m1 && m1[1]) return String(m1[1]).trim(); } catch (e) {}
    try { var re2 = new RegExp("\\[" + name + "\\s*[:=]\\s*([^\\]]+?)\\]", "i"); var m2 = s.match(re2); if (m2 && m2[1]) return String(m2[1]).trim(); } catch (e2) {}
    return "";
  }

  function stripMetaTokens(text) {
    var s = safeText(text); if (!s) return "";
    s = s.replace(/\{\%\s*(domain|version|change)\s*[:=][^%}]+\%\}/gi, "");
    s = s.replace(/\[(domain|version|change)\s*[:=][^\]]+\]/gi, "");
    s = s.replace(/[ \t]{2,}/g, " ");
    return s.trim();
  }

  function normalizeType(raw) {
    var t = safeText(raw).trim().toLowerCase(); if (!t) return "";
    if (t === "improved") t = "changed";
    if (t === "knownissue" || t === "known_issue" || t === "known-issue") t = "known issue";
    return t;
  }

  function titleCase(s) {
    var t = safeText(s).trim(); if (!t) return "";
    return t.split(/[\s_-]+/).filter(Boolean).map(function (w) { return w ? w.charAt(0).toUpperCase() + w.slice(1) : ""; }).join(" ");
  }

  function badgeBg(key) {
    if (key === "added") return "#16a34a"; if (key === "fixed") return "#1d6fb8";
    if (key === "changed") return "#9b1c4a"; if (key === "deprecated") return "#b45309";
    if (key === "removed") return "#b91c1c"; if (key === "known issue") return "#64748b";
    return "#334155";
  }

  function firstChildText(el, sel) {
    if (!el) return ""; var n = el.querySelector(sel); return n ? safeText(n.textContent).trim() : "";
  }

  function canonicalizeLink(link) {
    var s = safeText(link).trim(); if (!s) return "";
    try { var u = new URL(s, window.location.origin); u.search = ""; u.hash = ""; return u.toString(); } catch (e) { return s.split("?")[0].split("#")[0]; }
  }

  function looksLikeReactShell(htmlText) {
    var s = safeText(htmlText); if (!s) return false;
    if (s.indexOf("useReactApp") !== -1 && s.indexOf("userLoggedIn") !== -1) return true;
    if (s.indexOf("__NEXT_DATA__") !== -1 && s.indexOf("rm-Markdown") === -1 && s.indexOf("ChangelogPage_list-post") === -1) return true;
    return false;
  }

  function sanitizeHtmlAllowTables(html) {
    var allowed = { a: true, p: true, br: true, strong: true, em: true, b: true, i: true, code: true, pre: true, ul: true, ol: true, li: true, blockquote: true, h1: true, h2: true, h3: true, h4: true, h5: true, h6: true, hr: true, div: true, span: true, table: true, thead: true, tbody: true, tfoot: true, tr: true, th: true, td: true };
    var doc = new DOMParser().parseFromString("<div>" + safeText(html) + "</div>", "text/html");
    var root = doc.body && doc.body.firstChild ? doc.body.firstChild : null; if (!root) return "";
    function cleanEl(el) {
      if (!el || el.nodeType !== 1) return;
      var tag = (el.tagName || "").toLowerCase();
      if (tag === "script" || tag === "style" || tag === "iframe" || tag === "object" || tag === "embed") { el.parentNode && el.parentNode.removeChild(el); return; }
      if (!allowed[tag]) { var parent = el.parentNode; if (!parent) return; while (el.firstChild) parent.insertBefore(el.firstChild, el); parent.removeChild(el); return; }
      var attrs = el.attributes ? Array.prototype.slice.call(el.attributes) : [];
      for (var i = 0; i < attrs.length; i++) {
        var nm = attrs[i].name;
        if (/^on/i.test(nm) || nm === "style") { el.removeAttribute(nm); continue; }
        if ((tag === "td" || tag === "th") && (nm === "colspan" || nm === "rowspan")) continue;
        if (nm === "class") continue;
        if (tag === "a") { if (nm === "href" || nm === "target" || nm === "rel") continue; el.removeAttribute(nm); continue; }
        el.removeAttribute(nm);
      }
      if (tag === "a") {
        var href = safeText(el.getAttribute("href")).trim(); if (/^javascript:/i.test(href)) el.removeAttribute("href");
        var tgt = safeText(el.getAttribute("target")).trim();
        if (tgt === "_blank") { var rel = safeText(el.getAttribute("rel")).trim(); if (!rel || rel.indexOf("noopener") === -1) el.setAttribute("rel", "noopener noreferrer"); }
      }
      var kids = el.childNodes ? Array.prototype.slice.call(el.childNodes) : [];
      for (var k = 0; k < kids.length; k++) { if (kids[k] && kids[k].nodeType === 1) cleanEl(kids[k]); }
    }
    cleanEl(root);
    var out = safeText(root.innerHTML).trim(); if (out.length > 70000) out = out.slice(0, 70000);
    return out;
  }

  function bestHtmlFromNextData(doc) {
    try {
      var nd = doc.getElementById("__NEXT_DATA__"); if (!nd || !nd.textContent) return "";
      var data = safeJsonParse(nd.textContent); if (!data) return "";
      var best = { score: 0, html: "" };
      function scoreString(s) {
        var x = safeText(s); if (!x || x.indexOf("<") === -1) return 0;
        var score = 0;
        if (x.indexOf("<p") !== -1) score += 6; if (x.indexOf("<h1") !== -1 || x.indexOf("<h2") !== -1 || x.indexOf("<h3") !== -1) score += 6;
        if (x.indexOf("<ul") !== -1 || x.indexOf("<ol") !== -1) score += 4; if (x.indexOf("<table") !== -1) score += 6; if (x.indexOf("rm-Markdown") !== -1) score += 8;
        var len = x.length; if (len > 200 && len < 120000) score += Math.min(20, Math.floor(len / 2000));
        return score;
      }
      function walk(v, depth) {
        if (depth > 12 || v === null || v === undefined) return;
        if (typeof v === "string") { var sc = scoreString(v); if (sc > best.score) { best.score = sc; best.html = v; } return; }
        if (typeof v !== "object") return;
        var keys = Array.isArray(v) ? v.map(function (_, i) { return i; }) : Object.keys(v);
        for (var k = 0; k < keys.length; k++) walk(v[keys[k]], depth + 1);
      }
      walk(data, 0);
      return (best && best.html && best.score >= 6) ? safeText(best.html) : "";
    } catch (e) { return ""; }
  }

  function extractExcerptFromHtml(htmlText) {
    var doc = new DOMParser().parseFromString(safeText(htmlText), "text/html");
    var md = doc.querySelector('.rm-Markdown[data-testid="RDMD"]') || doc.querySelector(".rm-Markdown");
    if (md) {
      var styleEls = md.querySelectorAll ? md.querySelectorAll("style") : [];
      for (var i = 0; i < styleEls.length; i++) { try { styleEls[i].parentNode && styleEls[i].parentNode.removeChild(styleEls[i]); } catch (e) {} }
      return sanitizeHtmlAllowTables(stripMetaTokens(md.innerHTML));
    }
    var ndHtml = bestHtmlFromNextData(doc); if (ndHtml) return sanitizeHtmlAllowTables(stripMetaTokens(ndHtml));
    var post = doc.querySelector(".ChangelogPage_list-post") || doc.querySelector(".rm-Changelog-post") || doc.querySelector("[data-testid='changelog-post']") || doc.querySelector("article") || doc.querySelector("main");
    if (post) return sanitizeHtmlAllowTables(stripMetaTokens(post.innerHTML));
    return "";
  }

  function readLocal(key) { try { return window.localStorage.getItem(key); } catch (e) { return null; } }
  function writeLocal(key, value) { try { window.localStorage.setItem(key, value); } catch (e) {} }

  function fetchOnce(url, signal) {
    return fetch(url, { method: "GET", signal: signal, credentials: "omit", cache: "default", redirect: "follow" })
      .then(function (r) { if (!r || !r.ok) throw new Error("HTTP " + (r ? r.status : "?")); return r.text(); });
  }

  function fetchWithTimeout(url, signal, ms) {
    var local = new AbortController(); var timer = null;
    function onAbort() { try { local.abort(); } catch (e) {} }
    try { if (signal && signal.addEventListener) signal.addEventListener("abort", onAbort, { once: true }); } catch (e) {}
    timer = window.setTimeout(function () { try { local.abort(); } catch (e) {} }, ms);
    return fetchOnce(url, local.signal)
      .then(function (txt) { window.clearTimeout(timer); return txt; })
      .catch(function (err) { window.clearTimeout(timer); throw err; });
  }

  function fetchAndExtractExcerpt(link, signal) {
    var url = canonicalizeLink(link); if (!url) return Promise.resolve("");
    var ck = "rmCLExcerpt_v1__" + url;
    var cached = safeJsonParse(readLocal(ck));
    if (cached && cached.html && cached.ts) return Promise.resolve(String(cached.html));
    return fetchWithTimeout(url, signal, fetchTimeoutMs)
      .then(function (htmlText) {
        if (looksLikeReactShell(htmlText)) return fetchOnce(url, signal).catch(function () { return htmlText; });
        return htmlText;
      })
      .then(function (htmlText2) {
        var html = extractExcerptFromHtml(htmlText2);
        if (html) writeLocal(ck, JSON.stringify({ ts: Date.now(), html: html }));
        return html;
      })
      .catch(function () { return ""; });
  }

  function runWithLimit(tasks, limit) {
    var idx = 0; var results = new Array(tasks.length);
    function worker() {
      if (idx >= tasks.length) return Promise.resolve(); var my = idx++;
      return Promise.resolve().then(tasks[my]).then(function (r) { results[my] = r; }).catch(function () { results[my] = ""; }).then(worker);
    }
    var pool = []; var n = Math.max(1, Math.min(limit || 1, tasks.length));
    for (var i = 0; i < n; i++) pool.push(worker());
    return Promise.all(pool).then(function () { return results; });
  }

  function parseRss(xmlText) {
    var parser = new DOMParser(); var doc = parser.parseFromString(xmlText, "text/xml");
    var pe = doc.getElementsByTagName("parsererror"); if (pe && pe.length) return [];
    var itemEls = doc.querySelectorAll("item"); var out = [];
    var count = Math.min(itemEls.length, maxItems);
    for (var i = 0; i < count; i++) {
      var it = itemEls[i];
      var title = firstChildText(it, "title");
      var link = firstChildText(it, "link");
      var pubDate = coerceDate(firstChildText(it, "pubDate") || firstChildText(it, "date"));
      var desc = firstChildText(it, "content\\:encoded") || firstChildText(it, "encoded") || firstChildText(it, "description");
      var descText = stripHtml(desc);
      var version = extractToken(descText, "version") || extractToken(title, "version");
      var domain = extractToken(descText, "domain") || extractToken(title, "domain");
      var change = normalizeType(extractToken(descText, "change") || extractToken(descText, "type") || firstChildText(it, "type"));
      var body = stripMetaTokens(descText);
      out.push({ title: title, link: link, pubDate: pubDate, version: version, domain: domain, change: change, body: body, contentHtml: "", __hydrated: false });
    }
    return out;
  }

  function bustUrl(url) {
    var sep = url.indexOf("?") === -1 ? "?" : "&";
    return url + sep + "_cb=" + Math.floor(Date.now() / 60000);
  }

  // ── Fetch RSS ──
  useEffect(function () {
    if (typeof window === "undefined") return;
    setLoading(true); setError("");
    try { if (abortRef.current) abortRef.current.abort(); } catch (e) {}
    try { if (hydrateAbortRef.current) hydrateAbortRef.current.abort(); } catch (e) {}
    var ac = new AbortController(); abortRef.current = ac;

    var ck = "rmCLTable_v1__" + rssUrl;
    if (Number(cacheMinutes) > 0) {
      var cached = safeJsonParse(readLocal(ck));
      if (cached && cached.items && Array.isArray(cached.items) && (Date.now() - Number(cached.ts || 0)) < Number(cacheMinutes) * 60000) {
        setItems(cached.items); setLoading(false); return function () {};
      }
    }

    var timer = window.setTimeout(function () { try { ac.abort(); } catch (e) {} }, fetchTimeoutMs);
    fetch(bustUrl(rssUrl), { signal: ac.signal, cache: "no-store", credentials: "omit" })
      .then(function (r) { if (!r.ok) throw new Error("HTTP " + r.status); return r.text(); })
      .then(function (xml) {
        window.clearTimeout(timer);
        var parsed = parseRss(xml);
        setItems(parsed); setLoading(false);
        if (Number(cacheMinutes) > 0) { writeLocal(ck, JSON.stringify({ ts: Date.now(), items: parsed })); }
      })
      .catch(function (err) {
        window.clearTimeout(timer);
        if (err && err.name === "AbortError") return;
        setError(safeText(err && err.message ? err.message : err)); setLoading(false);
      });

    return function () { try { ac.abort(); } catch (e) {} };
  }, [rssUrl, maxItems, cacheMinutes, fetchTimeoutMs]);

  // ── Hydrate content from permalinks ──
  useEffect(function () {
    if (typeof window === "undefined") return;
    if (!hydrateContent) return;
    if (loading || error) return;
    if (!items || !items.length) return;

    var toHydrate = [];
    for (var i = 0; i < items.length; i++) {
      var it = items[i];
      if (!it || !it.link || it.__hydrated) continue;
      toHydrate.push({ idx: i, link: it.link });
      if (toHydrate.length >= hydrateMax) break;
    }
    if (!toHydrate.length) return;

    try { if (hydrateAbortRef.current) hydrateAbortRef.current.abort(); } catch (e) {}
    var ac = new AbortController(); hydrateAbortRef.current = ac;

    runWithLimit(
      toHydrate.map(function (x) { return function () { return fetchAndExtractExcerpt(x.link, ac.signal); }; }),
      hydrateConcurrency
    ).then(function (results) {
      setItems(function (prev) {
        var next = prev.slice(0);
        for (var j = 0; j < toHydrate.length; j++) {
          var target = toHydrate[j]; if (!target) continue;
          var cur = next[target.idx]; if (!cur) continue;
          next[target.idx] = Object.assign({}, cur, { contentHtml: safeText(results[j]), __hydrated: true });
        }
        return next;
      });
    });

    return function () { try { ac.abort(); } catch (e) {} };
  }, [loading, error, items, hydrateContent, hydrateMax, hydrateConcurrency, fetchTimeoutMs]);

  var versions = useMemo(function () {
    var seen = {}; var out = [];
    for (var i = 0; i < items.length; i++) { var v = safeText(items[i].version).trim(); if (v && !seen[v]) { seen[v] = true; out.push(v); } }
    out.sort(); return out;
  }, [items]);

  var domains = useMemo(function () {
    var seen = {}; var out = [];
    for (var i = 0; i < items.length; i++) { var v = safeText(items[i].domain).trim(); if (v && !seen[v]) { seen[v] = true; out.push(v); } }
    out.sort(); return out;
  }, [items]);

  var changes = useMemo(function () {
    var seen = {}; var out = [];
    for (var i = 0; i < items.length; i++) { var v = safeText(items[i].change).trim(); if (v && !seen[v]) { seen[v] = true; out.push(v); } }
    out.sort(); return out;
  }, [items]);

  var rows = useMemo(function () {
    var result = [];
    for (var i = 0; i < items.length; i++) {
      var it = items[i]; if (!it) continue;
      if (versionF !== "all" && safeText(it.version) !== versionF) continue;
      if (domainF !== "all" && safeText(it.domain) !== domainF) continue;
      if (changeF !== "all" && safeText(it.change) !== changeF) continue;
      if (search) {
        var q = safeText(search).trim().toLowerCase();
        var hay = (safeText(it.title) + " " + safeText(it.version) + " " + safeText(it.domain) + " " + safeText(it.change) + " " + safeText(it.body) + " " + stripHtml(safeText(it.contentHtml))).toLowerCase();
        if (hay.indexOf(q) === -1) continue;
      }
      result.push(it);
    }
    result.sort(function (a, b) {
      var va, vb;
      if (sortCol === "pubDate") { va = toTime(a.pubDate); vb = toTime(b.pubDate); }
      else { va = safeText(a[sortCol]).toLowerCase(); vb = safeText(b[sortCol]).toLowerCase(); }
      if (va < vb) return sortDir === "asc" ? -1 : 1;
      if (va > vb) return sortDir === "asc" ? 1 : -1;
      return 0;
    });
    return result;
  }, [items, versionF, domainF, changeF, search, sortCol, sortDir]);

  function toggleSort(col) {
    if (sortCol === col) { setSortDir(sortDir === "asc" ? "desc" : "asc"); }
    else { setSortCol(col); setSortDir("asc"); }
  }

  var colCount = (showVersionCol ? 1 : 0) + (showDomainCol ? 1 : 0) + (showChangeCol ? 1 : 0) + (showDescriptionCol ? 1 : 0) + (showDateCol ? 1 : 0);
  var selectStyle = { padding: "6px 28px 6px 12px", borderRadius: 8, border: "1px solid #e2e8f0", background: "#f8fafc", fontSize: 13, color: "#374151", appearance: "none", WebkitAppearance: "none", cursor: "pointer", fontFamily: "inherit", boxShadow: "0 1px 2px rgba(0,0,0,.04)", fontWeight: 500 };
  var thStyle = { padding: "10px 14px", textAlign: "left", fontWeight: 600, fontSize: 11, color: "#64748b", background: "#f8fafc", borderBottom: "2px solid #e2e8f0", cursor: "pointer", userSelect: "none", whiteSpace: "nowrap", textTransform: "uppercase", letterSpacing: ".07em" };
  var tdStyle = { padding: "12px 14px", verticalAlign: "top", borderBottom: "1px solid #f1f5f9" };

  function DescriptionCell(p) {
    var it = p.item;
    var fullHtml = safeText(it.contentHtml).trim();
    var fallback = safeText(it.body).trim();

    var link = it.link ? <a href={safeText(it.link)} target="_blank" rel="noopener noreferrer" style={{ display: "inline-block", marginTop: 6, fontSize: 12, color: "#2563eb", fontWeight: 600, textDecoration: "none" }}>View full log →</a> : null;

    if (!it.__hydrated) {
      return <div>{fallback ? <span style={{ color: "#374151", fontSize: 13, lineHeight: 1.55 }}>{fallback}</span> : <span style={{ color: "#94a3b8", fontSize: 12 }}>Loading…</span>}{link}</div>;
    }

    if (!fullHtml) {
      return <div>{fallback ? <span style={{ color: "#374151", fontSize: 13, lineHeight: 1.55 }}>{fallback}</span> : <span style={{ color: "#94a3b8", fontSize: 12 }}>—</span>}{link}</div>;
    }

    var fullText = stripHtml(fullHtml);
    var isTruncated = fullText.length > excerptMaxChars;
    var excerptText = isTruncated ? truncateText(fullText, excerptMaxChars) : fullText;

    return (
      <div>
        <div style={{ color: "#374151", fontSize: 13, lineHeight: 1.6 }}>{excerptText}</div>
        {link}
      </div>
    );
  }

  function CardView() {
    if (loading) return <div style={{ padding: 24, textAlign: "center", color: "#94a3b8", fontSize: 13 }}>Loading…</div>;
    if (!loading && !error && rows.length === 0) return <div style={{ padding: 24, textAlign: "center", color: "#94a3b8", fontSize: 13 }}>No entries match your filters.</div>;
    return (
      <div style={{ display: "flex", flexDirection: "column", gap: 16 }}>
        {rows.map(function (it, idx) {
          var accent = it.change ? badgeBg(it.change) : "#e2e8f0";
          return (
            <div key={safeText(it.link) || String(idx)} style={{ borderRadius: 10, border: "1px solid #e2e8f0", background: "#fff", boxShadow: "0 1px 4px rgba(0,0,0,.06)", overflow: "hidden", display: "flex" }}>
              <div style={{ width: 4, flexShrink: 0, background: accent }} />
              <div style={{ padding: "16px 20px", flex: 1, minWidth: 0 }}>
                <div style={{ display: "flex", flexWrap: "wrap", alignItems: "center", gap: 8, marginBottom: 8 }}>
                  {it.change ? <span style={{ display: "inline-block", padding: "3px 10px", borderRadius: 4, fontWeight: 700, fontSize: 11, letterSpacing: ".06em", background: accent, color: "#fff", textTransform: "uppercase", whiteSpace: "nowrap" }}>{it.change}</span> : null}
                  {it.version ? <span style={{ display: "inline-block", padding: "2px 10px", borderRadius: 999, background: "#f1f5f9", border: "1px solid #e2e8f0", color: "#475569", fontSize: 12, fontWeight: 600, whiteSpace: "nowrap" }}>{it.version}</span> : null}
                  {it.domain ? <span style={{ fontSize: 12, color: "#64748b", fontWeight: 500 }}>{it.domain}</span> : null}
                  <span style={{ marginLeft: "auto", fontSize: 11, color: "#94a3b8", fontWeight: 500, whiteSpace: "nowrap" }}>{formatDate(it.pubDate)}</span>
                </div>
                {it.title ? (
                  it.link
                    ? <a href={safeText(it.link)} target="_blank" rel="noopener noreferrer" style={{ display: "block", marginBottom: 8, fontSize: 15, fontWeight: 700, color: "#1e293b", textDecoration: "none", lineHeight: 1.35 }}>{it.title}</a>
                    : <div style={{ marginBottom: 8, fontSize: 15, fontWeight: 700, color: "#1e293b", lineHeight: 1.35 }}>{it.title}</div>
                ) : null}
                <DescriptionCell item={it} />
              </div>
            </div>
          );
        })}
      </div>
    );
  }

  return (
    <div style={{ width: "100%", fontFamily: "system-ui, sans-serif" }}>
      <div style={{ display: "flex", flexWrap: "wrap", gap: 8, marginBottom: 16, alignItems: "center" }}>
        {showVersionCol && showVersionFilter ? <select value={versionF} onChange={function (e) { setVersionF(e.target.value); }} style={selectStyle}><option value="all">{versionLabel + " ⏷"}</option>{versions.map(function (v) { return <option key={v} value={v}>{v}</option>; })}</select> : null}
        {showDomainCol && showDomainFilter ? <select value={domainF} onChange={function (e) { setDomainF(e.target.value); }} style={selectStyle}><option value="all">{domainLabel + " ⏷"}</option>{domains.map(function (v) { return <option key={v} value={v}>{v}</option>; })}</select> : null}
        {showChangeCol && showChangeFilter ? <select value={changeF} onChange={function (e) { setChangeF(e.target.value); }} style={selectStyle}><option value="all">{changeLabel + " ⏷"}</option>{changes.map(function (v) { return <option key={v} value={v}>{titleCase(v)}</option>; })}</select> : null}
        {showSearch ? <input value={search} onChange={function (e) { setSearch(e.target.value); }} placeholder="Search…" style={{ padding: "6px 10px", borderRadius: 8, border: "1px solid #e2e8f0", fontSize: 13, fontFamily: "inherit", color: "#374151", outline: "none", minWidth: 160, background: "#f8fafc", boxShadow: "0 1px 2px rgba(0,0,0,.04)" }} /> : null}
        {(versionF !== "all" || domainF !== "all" || changeF !== "all" || search) ? <button type="button" onClick={function () { setVersionF("all"); setDomainF("all"); setChangeF("all"); setSearch(""); }} style={{ padding: "6px 12px", borderRadius: 8, border: "1px solid #e2e8f0", background: "#fff", fontSize: 13, cursor: "pointer", color: "#6b7280", fontFamily: "inherit", boxShadow: "0 1px 2px rgba(0,0,0,.04)" }}>Clear</button> : null}
        <div style={{ marginLeft: "auto", display: "flex", alignItems: "center", gap: 6 }}>
          {showLayoutToggle ? (
            <div style={{ display: "flex", borderRadius: 8, border: "1px solid #e2e8f0", overflow: "hidden", background: "#f8fafc" }}>
              <button type="button" title="Table view" onClick={function () { setLayout("table"); }} style={{ padding: "5px 9px", border: "none", cursor: "pointer", fontSize: 14, lineHeight: 1, background: layout === "table" ? "#e2e8f0" : "transparent", color: layout === "table" ? "#1e293b" : "#94a3b8", transition: "background .15s" }}>☰</button>
              <button type="button" title="Card view" onClick={function () { setLayout("cards"); }} style={{ padding: "5px 9px", border: "none", cursor: "pointer", fontSize: 14, lineHeight: 1, background: layout === "cards" ? "#e2e8f0" : "transparent", color: layout === "cards" ? "#1e293b" : "#94a3b8", transition: "background .15s" }}>⊞</button>
            </div>
          ) : null}
          <span style={{ fontSize: 12, color: "#94a3b8" }}>{loading ? "Loading…" : error ? "Error" : rows.length + " / " + items.length + " entries"}</span>
        </div>
      </div>

      {error ? <div style={{ padding: "12px 16px", borderRadius: 8, background: "#fef2f2", border: "1px solid #fca5a5", color: "#991b1b", fontSize: 13, marginBottom: 12 }}>{"Couldn't load RSS: " + error}</div> : null}

      {layout === "cards" ? <CardView /> : <div style={{ width: "100%", overflowX: "auto" }}>
        <table style={{ width: "100%", borderCollapse: "collapse", fontSize: 13 }}>
          <thead>
            <tr>
              {showVersionCol ? <th style={thStyle} onClick={function () { toggleSort("version"); }}>{versionLabel}{sortCol === "version" ? (sortDir === "asc" ? " ▲" : " ▼") : " ▽"}</th> : null}
              {showDomainCol ? <th style={thStyle} onClick={function () { toggleSort("domain"); }}>{domainLabel}{sortCol === "product" ? (sortDir === "asc" ? " ▲" : " ▼") : " ▽"}</th> : null}
              {showChangeCol ? <th style={thStyle} onClick={function () { toggleSort("change"); }}>{changeLabel}{sortCol === "change" ? (sortDir === "asc" ? " ▲" : " ▼") : " ▽"}</th> : null}
              {showDescriptionCol ? <th style={thStyle} onClick={function () { toggleSort("body"); }}>{descriptionLabel}{sortCol === "body" ? (sortDir === "asc" ? " ▲" : " ▼") : " ▽"}</th> : null}
              {showDateCol ? <th style={thStyle} onClick={function () { toggleSort("pubDate"); }}>{dateLabel}{sortCol === "pubDate" ? (sortDir === "asc" ? " ▲" : " ▼") : " ▽"}</th> : null}
            </tr>
          </thead>
          <tbody>
            {loading ? <tr><td colSpan={colCount} style={{ padding: 24, textAlign: "center", color: "#94a3b8", fontSize: 13 }}>{"Loading…"}</td></tr> : null}
            {!loading && !error && rows.length === 0 ? <tr><td colSpan={colCount} style={{ padding: 24, textAlign: "center", color: "#94a3b8", fontSize: 13 }}>No entries match your filters.</td></tr> : null}
            {rows.map(function (it, idx) {
              return (
                <tr key={safeText(it.link) || String(idx)} style={{ background: idx % 2 === 0 ? "#fff" : "#fafbfc" }}>
                  {showVersionCol ? <td style={tdStyle}>{it.version ? <span style={{ display: "inline-block", padding: "2px 10px", borderRadius: 999, background: "#f1f5f9", border: "1px solid #e2e8f0", color: "#475569", fontSize: 12, fontWeight: 600, whiteSpace: "nowrap", letterSpacing: ".02em" }}>{it.version}</span> : <span style={{ color: "#cbd5e1" }}>—</span>}</td> : null}
                  {showDomainCol ? <td style={tdStyle}><span style={{ color: "#475569", fontWeight: 500 }}>{it.domain || <span style={{ color: "#cbd5e1" }}>—</span>}</span></td> : null}
                  {showChangeCol ? <td style={tdStyle}>{it.change ? <span style={{ display: "inline-block", padding: "3px 10px", borderRadius: 4, fontWeight: 700, fontSize: 12, letterSpacing: ".06em", background: badgeBg(it.change), color: "#fff", textTransform: "uppercase", whiteSpace: "nowrap" }}>{it.change}</span> : <span style={{ color: "#cbd5e1" }}>{"—"}</span>}</td> : null}
                  {showDescriptionCol ? <td style={{ padding: "12px 14px", verticalAlign: "top", borderBottom: "1px solid #f1f5f9" }}><DescriptionCell item={it} /></td> : null}
                  {showDateCol ? <td style={tdStyle}><span style={{ color: "#94a3b8", fontSize: 11, fontWeight: 500, whiteSpace: "nowrap", letterSpacing: ".02em" }}>{formatDate(it.pubDate)}</span></td> : null}
                </tr>
              );
            })}
          </tbody>
        </table>
      </div>}
    </div>
  );
};

export const ChangelogMount = props => {
  return <ChangelogTable {...props} />;
};
