/**
 * KeywordSearch.jsx
 * ---------------------------------------------------------------------
 * MDX‑friendly search component with prop‑based customization.
 * No build‑time deps; mark.js is lazy‑loaded from a CDN.
 */

import React, { useState, useEffect, useRef } from "react";

export default function KeywordSearch({
  /* ─────────────  Public props (all optional)  ───────────── */
  placeholder   = "Search Page…",
  minChars      = 3,                    // min characters before highlighting
  highlightColor = "#e1e8ff",           // background colour for matches
  showKeywords  = true,                 // toggle keyword chips
  keywords: keywordsProp = [],          // override <meta keywords>
  debounceMs    = 200,                  // debounce keystrokes
}) {
  /* ─────────────────────  State  ───────────────────── */
  const [searchTerm,   setSearchTerm]   = useState("");
  const [matchCount,   setMatchCount]   = useState(0);
  const [keywords,     setKeywords]     = useState([]);
  const [keywordCounts,setKeywordCounts]= useState({});

  /* ────────────────────  Refs / consts  ──────────────────── */
  const markRef     = useRef(null);           // mark.js instance
  const contentRef  = useRef(null);           // .content-body root
  const debounceRef = useRef(null);           // timer for debounce

  const searchSelector =
    ".content-body p, .content-body li, .content-body ul, .content-body ol, .content-body details, .content-body code, .content-body table, .content-body th, .content-body td, .content-body pre, .content-body blockquote, .content-body h1, .content-body h2, .content-body h3, .content-body h4, .content-body h5, .content-body h6";

  /* ─────────────  Helper: lazy‑load mark.js  ───────────── */
  const loadMarkJs = () =>
    new Promise((resolve) => {
      if (window.Mark) return resolve(window.Mark);        // already present
      const script = document.createElement("script");
      script.src = "https://unpkg.com/mark.js/dist/mark.min.js";
      script.onload = () => resolve(window.Mark);
      document.head.appendChild(script);
    });

  /* ──────────────  Inject / update highlight CSS  ────────────── */
  useEffect(() => {
    const id = "ks-highlight-style";
    let styleTag = document.getElementById(id);
    if (!styleTag) {
      styleTag = document.createElement("style");
      styleTag.id = id;
      document.head.appendChild(styleTag);
    }
    styleTag.textContent = `.highlighted-match{background-color:${highlightColor};font-weight:600;border-radius:4px;padding:2px 3px;}`;
  }, [highlightColor]);

  /* ────────────────  One‑time setup  ──────────────── */
  useEffect(() => {
    // cache root
    contentRef.current = document.querySelector(".content-body");

    // load mark.js then create instance
    let markInstance;
    loadMarkJs().then((Mark) => {
      if (!contentRef.current) return;
      markInstance      = new Mark(contentRef.current);
      markRef.current   = markInstance;
    });

    return () => { markInstance && markInstance.unmark(); };
  }, []);

  /* ───────────  Keyword source: prop > meta  ─────────── */
  const updateKeywordsFromMeta = () => {
    if (keywordsProp.length) return;           // prop wins
    const meta = document.querySelector('meta[property="keywords"]');
    if (meta && meta.content.trim()) {
      setKeywords(meta.content.split(",").map((k) => k.trim()));
    }
  };

  // when prop changes
  useEffect(() => {
    if (keywordsProp.length) setKeywords(keywordsProp);
  }, [keywordsProp]);

  /* ─────────────  Initialisation per route  ───────────── */
  const initializeSearchComponent = () => {
    console.log("🔍 Initialising on:", window.location.pathname);
    updateKeywordsFromMeta();
    markRef.current && markRef.current.unmark();
    setMatchCount(0);
    setSearchTerm("");
  };

  /* observe meta changes + SPA navigation */
  useEffect(() => {
    initializeSearchComponent();

    /* watch meta[keywords] */
    const meta = document.querySelector('meta[property="keywords"]');
    let observer;
    if (meta) {
      observer = new MutationObserver(updateKeywordsFromMeta);
      observer.observe(meta, { attributes: true, attributeFilter: ["content"] });
    }

    /* hook history.* and hashchange */
    const onRoute = () => setTimeout(initializeSearchComponent, 500);

    const origPush = history.pushState;
    const origReplace = history.replaceState;
    history.pushState = function () { origPush.apply(history, arguments);  window.dispatchEvent(new Event("pushState")); };
    history.replaceState = function () { origReplace.apply(history, arguments); window.dispatchEvent(new Event("replaceState")); };

    window.addEventListener("popstate", onRoute);
    window.addEventListener("pushState", onRoute);
    window.addEventListener("replaceState", onRoute);
    window.addEventListener("hashchange", onRoute);

    return () => {
      observer && observer.disconnect();
      window.removeEventListener("popstate", onRoute);
      window.removeEventListener("pushState", onRoute);
      window.removeEventListener("replaceState", onRoute);
      window.removeEventListener("hashchange", onRoute);
    };
  }, [keywordsProp]);

  /* ────────────────  Highlight logic  ──────────────── */
  useEffect(() => {
    const mark = markRef.current;
    if (!mark || !contentRef.current) return;

    if (searchTerm.length < minChars) {
      mark.unmark();
      setMatchCount(0);
      return;
    }

    mark.unmark({
      done: () =>
        mark.mark(searchTerm, {
          separateWordSearch: false,
          className: "highlighted-match",
          done: (cnt) => {
            setMatchCount(cnt);
            if (cnt) setTimeout(() => {
              const m = document.querySelector(".highlighted-match");
              m && m.scrollIntoView({ behavior: "smooth", block: "center" });
            }, 300);
          },
        }),
    });
  }, [searchTerm, minChars]);

  /* ────────────  Count keyword occurrences  ──────────── */
  useEffect(() => {
    if (!keywords.length) return;
    const counts = {};

    document.querySelectorAll(searchSelector).forEach((el) => {
      const text = el.textContent || "";
      keywords.forEach((kw) => {
        const re = new RegExp(`\\b${kw}\\b`, "gi");
        const m  = text.match(re);
        if (m) counts[kw] = (counts[kw] || 0) + m.length;
      });
    });

    setKeywordCounts(counts);
  }, [keywords]);

  /* ───────────────  Event handlers  ─────────────── */
  const handleInput = (val) => {
    clearTimeout(debounceRef.current);
    debounceRef.current = setTimeout(() => setSearchTerm(val), debounceMs);
  };

  const handleKeywordClick = (kw) => setSearchTerm(kw);

  /* ───────────────  Inline styles  ─────────────── */
  const containerStyle = { width: "100%", display: "flex", flexDirection: "column", gap: "10px" };
  const searchContainerStyle = { display: "flex", alignItems: "center", width: "100%", gap: "30px", position: "relative" };
  const inputContainerStyle = { flex: "1 1 25%", maxWidth: "25%", position: "relative" };
  const inputStyle = { width: "100%", padding: "8px", fontSize: "14px", border: "1px solid #ccc", borderRadius: "4px" };
  const matchCountStyle = { position: "absolute", top: "-18px", left: "5px", fontSize: "12px", color: "#555", opacity: 0.8 };
  const keywordContainerStyle = { flex: "1 1 75%", maxWidth: "75%", display: "flex", flexWrap: "wrap", gap: "5px", justifyContent: "flex-start", overflow: "hidden", whiteSpace: "nowrap" };
  const keywordButtonStyle = (active) => ({
    backgroundColor: active ? highlightColor : "#f0f0f0",
    padding: "5px 10px",
    borderRadius: "15px",
    border: "1px solid #ccc",
    cursor: "pointer",
    fontSize: "12px",
    color: "#333",
  });

  /* ─────────────────────  Render  ───────────────────── */
  return (
    <div style={containerStyle}>
      <div style={searchContainerStyle}>
        {/* Input */}
        <div style={inputContainerStyle}>
          {matchCount > 0 && (
            <span style={matchCountStyle}>
              {matchCount} {matchCount === 1 ? "result" : "results"}
            </span>
          )}
          <input
            type="text"
            placeholder={placeholder}
            style={inputStyle}
            onChange={(e) => handleInput(e.target.value)}
          />
        </div>

        {/* Keyword chips */}
        {showKeywords && (
          <div style={keywordContainerStyle}>
            {keywords.map((kw) => (
              <button
                key={kw}
                onClick={() => handleKeywordClick(kw)}
                style={keywordButtonStyle(searchTerm === kw)}
              >
                {kw} ({keywordCounts[kw] || 0})
              </button>
            ))}
          </div>
        )}
      </div>
    </div>
  );
}

<KeywordSearch placeholder="Search this guide!" highlightColor="#ffd54f" />
