import React, { useEffect, useRef, useState } from "react";

export const KeywordSearch = rawProps => {
  const props = {};
  Object.keys(rawProps || {}).forEach(k => { props[k.toLowerCase()] = rawProps[k]; });
  props.children = rawProps ? rawProps.children : undefined;
  const rootRef = useRef(null);
  const inputRef = useRef(null);
  const rangesRef = useRef([]);
  const [term, setTerm] = useState("");
  const [count, setCount] = useState(0);
  const [cur, setCur] = useState(0);
  const [chips, setChips] = useState([]);
  const placeholder = props.placeholder || "Find on this page…";
  const canHighlight = typeof window !== "undefined" && window.CSS && CSS.highlights && typeof window.Highlight === "function";

  const scope = () => {
    const me = rootRef.current;
    if (!me) return null;
    return me.closest(".markdown-body") || me.closest("article") || document.querySelector(".rm-Article") || document.body;
  };

  const textNodes = () => {
    const root = scope();
    const me = rootRef.current;
    if (!root) return [];
    const out = [];
    const walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT, {
      acceptNode: n => {
        const p = n.parentElement;
        if (!p || !n.nodeValue || !n.nodeValue.trim()) return NodeFilter.FILTER_REJECT;
        if (me && me.contains(n)) return NodeFilter.FILTER_REJECT;
        if (p.closest("style,script,noscript,[aria-hidden='true']")) return NodeFilter.FILTER_REJECT;
        return NodeFilter.FILTER_ACCEPT;
      },
    });
    while (walker.nextNode()) out.push(walker.currentNode);
    return out;
  };

  const findRanges = q => {
    const needle = q.toLowerCase();
    const ranges = [];
    if (needle.length < 2) return ranges;
    for (const node of textNodes()) {
      const hay = node.nodeValue.toLowerCase();
      let i = hay.indexOf(needle);
      while (i !== -1 && ranges.length < 500) {
        const r = document.createRange();
        r.setStart(node, i);
        r.setEnd(node, i + needle.length);
        ranges.push(r);
        i = hay.indexOf(needle, i + needle.length);
      }
    }
    return ranges;
  };

  const paint = (ranges, idx) => {
    if (!canHighlight) return;
    CSS.highlights.delete("eo-find");
    CSS.highlights.delete("eo-find-current");
    if (!ranges.length) return;
    CSS.highlights.set("eo-find", new window.Highlight(...ranges.filter((r, k) => k !== idx)));
    if (ranges[idx]) CSS.highlights.set("eo-find-current", new window.Highlight(ranges[idx]));
  };

  const reveal = (ranges, idx) => {
    const r = ranges[idx];
    if (!r) return;
    const el = r.startContainer.parentElement;
    const details = el && el.closest("details:not([open])");
    if (details) details.open = true;
    const box = r.getBoundingClientRect ? r.getBoundingClientRect() : null;
    if (box && (box.top < 90 || box.bottom > window.innerHeight - 40)) {
      window.scrollTo({ top: window.scrollY + box.top - window.innerHeight / 3, behavior: "smooth" });
    }
  };

  useEffect(() => {
    const t = window.setTimeout(() => {
      const ranges = findRanges(term.trim());
      rangesRef.current = ranges;
      setCount(ranges.length);
      setCur(0);
      paint(ranges, 0);
      if (ranges.length) reveal(ranges, 0);
    }, 140);
    return () => window.clearTimeout(t);
  }, [term]);

  useEffect(() => {
    const words = String(props.keywords || (document.querySelector('meta[name="keywords"],meta[property="keywords"]') || {}).content || "")
      .split(",").map(s => s.trim()).filter(Boolean).slice(0, 12);
    const t = window.setTimeout(() => setChips(words.map(w => ({ w, n: findRanges(w).length }))), 300);
    return () => {
      window.clearTimeout(t);
      if (canHighlight) { CSS.highlights.delete("eo-find"); CSS.highlights.delete("eo-find-current"); }
    };
  }, [props.keywords]);

  const step = d => {
    const ranges = rangesRef.current;
    if (!ranges.length) return;
    const next = (cur + d + ranges.length) % ranges.length;
    setCur(next);
    paint(ranges, next);
    reveal(ranges, next);
  };

  const css = `
.eo-find{--eo-accent:var(--color-link-primary,#2F7BF5);--eo-border:var(--color-border-default,rgba(0,0,0,.12));--eo-muted:var(--color-text-minimum,#6b7280);margin:20px 0}
.eo-find-bar{display:flex;align-items:center;gap:8px;border:1px solid var(--eo-border);border-radius:12px;padding:6px 8px 6px 12px;background:var(--color-bg-page,transparent);transition:border-color .15s,box-shadow .15s}
.eo-find-bar:focus-within{border-color:var(--eo-accent);box-shadow:0 0 0 3px color-mix(in srgb,var(--eo-accent) 15%,transparent)}
.eo-find-bar input{flex:1;min-width:0;border:0;outline:0;background:transparent;color:inherit;font:inherit;font-size:15px;padding:6px 0}
.eo-find-count{font-size:13px;color:var(--eo-muted);white-space:nowrap;font-variant-numeric:tabular-nums}
.eo-find-btn{border:1px solid var(--eo-border);background:transparent;color:inherit;width:30px;height:30px;border-radius:8px;cursor:pointer;font-size:13px}
.eo-find-btn:disabled{opacity:.35;cursor:default}
.eo-find-chips{display:flex;flex-wrap:wrap;gap:6px;margin-top:10px}
.eo-find-chip{border:1px solid var(--eo-border);background:transparent;color:inherit;border-radius:999px;padding:3px 10px;font-size:13px;cursor:pointer}
.eo-find-chip[aria-pressed="true"]{background:var(--eo-accent);border-color:var(--eo-accent);color:#fff}
.eo-find-chip b{font-weight:600;opacity:.65;margin-left:4px}
::highlight(eo-find){background-color:rgba(250,204,21,.45);color:inherit}
::highlight(eo-find-current){background-color:#f59e0b;color:#000}
`;
  const q = term.trim();
  return (
    <div className="eo-find" ref={rootRef}>
      <style>{css}</style>
      <div className="eo-find-bar">
        <span aria-hidden="true">🔎</span>
        <input ref={inputRef} type="search" value={term} placeholder={placeholder} aria-label={placeholder}
          onChange={e => setTerm(e.target.value)}
          onKeyDown={e => { if (e.key === "Enter") { e.preventDefault(); step(e.shiftKey ? -1 : 1); } if (e.key === "Escape") setTerm(""); }} />
        <span className="eo-find-count" aria-live="polite">{q.length < 2 ? "" : count ? (cur + 1) + " of " + count : "No matches"}</span>
        <button type="button" className="eo-find-btn" aria-label="上一个匹配项" disabled={!count} onClick={() => step(-1)}>↑</button>
        <button type="button" className="eo-find-btn" aria-label="下一个匹配项" disabled={!count} onClick={() => step(1)}>↓</button>
      </div>
      {chips.length ? (
        <div className="eo-find-chips">
          {chips.map(c => (
            <button key={c.w} type="button" className="eo-find-chip" aria-pressed={q.toLowerCase() === c.w.toLowerCase() ? "true" : "false"}
              onClick={() => setTerm(q.toLowerCase() === c.w.toLowerCase() ? "" : c.w)}>
              {c.w}<b>{c.n}</b>
            </button>
          ))}
        </div>
      ) : null}
    </div>
  );
};

<KeywordSearch keywords="launch, booking, manifest" />
