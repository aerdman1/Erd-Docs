import React, { useState, useEffect, useMemo } from "react";

export const function KeywordSearch({ keywords: keywordsProp, children }) {
  // Normalize passed keywords from prop or children into an array
  const providedKeywords = useMemo(() => {
    // Check if a keywords prop was provided and is non-empty
    if (keywordsProp && (Array.isArray(keywordsProp) ? keywordsProp.length : keywordsProp)) {
      return Array.isArray(keywordsProp) ? keywordsProp : [keywordsProp];
    }
    // If children exist, treat them as the provided keywords (you can decide on the shape/format)
    if (children) {
      return Array.isArray(children) ? children : [children];
    }
    return [];
  }, [keywordsProp, children]);

  // State for keywords – initialize with provided keywords if available
  const [keywords, setKeywords] = useState(providedKeywords);

  // Other states for search functionality remain unchanged
  const [searchTerm, setSearchTerm] = useState("");
  const [matchCount, setMatchCount] = useState(0);
  const [keywordCounts, setKeywordCounts] = useState({});

  // Function to update keywords from the meta tag
  const updateKeywordsFromMeta = () => {
    const metaTag = document.querySelector('meta[property="keywords"]');
    if (metaTag && metaTag.content.trim() !== "") {
      const extracted = metaTag.content.split(",").map((kw) => kw.trim());
      setKeywords(extracted);
      console.log("Extracted meta keywords:", extracted);
    }
  };

  // If no keywords provided as props or children, poll for meta tag keywords.
  useEffect(() => {
    if (keywords.length === 0) {
      let attempt = 0;
      const maxAttempts = 30;
      const poll = () => {
        const metaTag = document.querySelector('meta[property="keywords"]');
        if (metaTag && metaTag.content.trim() !== "") {
          updateKeywordsFromMeta();
        } else if (attempt < maxAttempts) {
          attempt++;
          setTimeout(poll, 100);
        } else {
          console.log("Polling: maximum attempts reached. No keywords available from meta.");
        }
      };
      poll();
    }
  }, [keywords]);

  // Monitor for changes in search term to perform highlights in designated page elements (if applicable)
  useEffect(() => {
    // Restore original texts when search term is too short
    if (searchTerm.length < 3) {
      restoreOriginalText();
      setMatchCount(0);
      return;
    }
    let matchCounter = 0;
    const regex = new RegExp(`(${searchTerm})`, "gi");

    document.querySelectorAll(".content-body p").forEach((p) => {
      if (p.dataset.originalText) {
        const modifiedText = p.dataset.originalText.replace(regex, (match) => {
          matchCounter++;
          return `<span class="highlighted-match" style="background-color: #e1e8ff; font-weight: 600; border-radius: 4px; padding: 2px 3px;">${match}</span>`;
        });
        p.innerHTML = modifiedText;
      }
    });

    setMatchCount(matchCounter);
    if (matchCounter > 0) {
      setTimeout(scrollToFirstMatch, 300);
    }
  }, [searchTerm]);

  // Update keyword occurrence counts in target elements
  useEffect(() => {
    if (keywords.length === 0) return;
    const counts = {};
    document.querySelectorAll(".content-body p").forEach((p) => {
      const text = p.innerText;
      keywords.forEach((keyword) => {
        const regex = new RegExp(`\\b${keyword}\\b`, "gi");
        const matches = text.match(regex);
        if (matches) {
          counts[keyword] = (counts[keyword] || 0) + matches.length;
        }
      });
    });
    setKeywordCounts(counts);
  }, [keywords]);

  // Helper to restore paragraphs to their original (non-highlighted) text
  const restoreOriginalText = () => {
    document.querySelectorAll(".content-body p").forEach((p) => {
      if (p.dataset.originalText) {
        p.innerHTML = p.dataset.originalText;
      }
    });
  };

  // Scroll smoothly to the first highlighted match
  const scrollToFirstMatch = () => {
    const matches = document.querySelectorAll(".highlighted-match");
    if (matches.length > 0) {
      matches[0].scrollIntoView({ behavior: "smooth", block: "center" });
      console.log("Scrolled to first match.");
    }
  };

  // When a keyword bubble is clicked, update the search term accordingly
  const handleKeywordClick = (keyword) => {
    setSearchTerm(keyword);
  };

  // Inline style objects for improved readability (these can be extracted if needed)
  const containerStyle = {
    width: "100%",
    display: "flex",
    flexDirection: "column",
    gap: "10px",
  };

  const searchContainerStyle = {
    display: "flex",
    alignItems: "center",
    width: "100%",
    gap: "30px",
    position: "relative",
  };

  const inputContainerStyle = {
    flex: "1 1 25%",
    maxWidth: "25%",
    position: "relative",
  };

  const inputStyle = {
    width: "100%",
    padding: "8px",
    fontSize: "14px",
    border: "1px solid #ccc",
    borderRadius: "4px",
  };

  const matchCountStyle = {
    position: "absolute",
    top: "-18px",
    left: "5px",
    fontSize: "12px",
    color: "#555",
    opacity: "0.8",
  };

  const keywordContainerStyle = {
    flex: "1 1 75%",
    maxWidth: "75%",
    display: "flex",
    flexWrap: "wrap",
    gap: "5px",
    justifyContent: "flex-start",
    overflow: "hidden",
    whiteSpace: "nowrap",
  };

  const keywordButtonStyle = (isActive) => ({
    backgroundColor: isActive ? "#e1e8ff" : "#f0f0f0",
    padding: "5px 10px",
    borderRadius: "15px",
    border: "1px solid #ccc",
    cursor: "pointer",
    fontSize: "12px",
    color: "#333",
  });

  return (
    <div style={containerStyle}>
      {/* Search & Keywords Container */}
      <div style={searchContainerStyle}>
        {/* Search Input */}
        <div style={inputContainerStyle}>
          {matchCount > 0 && (
            <span style={matchCountStyle}>
              {matchCount} {matchCount === 1 ? "result" : "results"}
            </span>
          )}
          <input
            type="text"
            placeholder="Search Page..."
            style={inputStyle}
            value={searchTerm}
            onChange={(e) => setSearchTerm(e.target.value)}
          />
        </div>
        {/* Keyword Bubbles */}
        <div style={keywordContainerStyle}>
          {keywords.map((keyword) => (
            <button
              key={keyword}
              onClick={() => handleKeywordClick(keyword)}
              style={keywordButtonStyle(searchTerm === keyword)}
            >
              {keyword} ({keywordCounts[keyword] || 0})
            </button>
          ))}
        </div>
      </div>
    </div>
  );
}

<KeywordSearch />
