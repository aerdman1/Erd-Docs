import React, { useState, useEffect } from "react";

export default function KeywordSearch({
  isReadMeKeyword = '', 
  customKeyWords = []  // Pass an array of custom keywords
}) {
  // State declarations
  const [searchTerm, setSearchTerm] = useState("");
  const [matchCount, setMatchCount] = useState(0);
  const [metaKeywords, setMetaKeywords] = useState([]); // store meta keywords separately
  const [keywords, setKeywords] = useState([]);
  const [keywordCounts, setKeywordCounts] = useState({});

  // Update metaKeywords from the meta tag if content is available
  const updateKeywords = () => {
    const metaTag = document.querySelector('meta[property="keywords"]');
    if (metaTag) {
      const content = metaTag.content;
      if (content.trim() !== "") {
        const extracted = content.split(",").map((kw) => kw.trim());
        setMetaKeywords(extracted);
        console.log("📌 Extracted Meta Keywords:", extracted);
      } else {
        console.log("Meta tag content is empty.");
        setMetaKeywords([]);
      }
    }
  };

  // Poll for meta tag content until it's non-empty, stopping after maxAttempts
  const pollForMetaTagContent = (attempt = 0) => {
    const maxAttempts = 30; // 30 * 100ms = 3000ms (3 seconds)
    const metaTag = document.querySelector('meta[property="keywords"]');
    if (metaTag && metaTag.content.trim() !== "") {
      updateKeywords();
    } else {
      if (attempt < maxAttempts) {
        console.log(`Polling: meta tag content is still empty, attempt ${attempt + 1}`);
        setTimeout(() => pollForMetaTagContent(attempt + 1), 100);
      } else {
        console.log("Polling: maximum attempts reached. Stopping polling.");
      }
    }
  };

  // Initialize/reinitialize component (keywords & paragraphs)
  const initializeSearchComponent = () => {
    console.log("🔍 Initializing Search Component for route:", window.location.pathname);
    if (isReadMeKeyword === true) {
      pollForMetaTagContent();
    }
    // Reset or store original text for paragraphs
    document.querySelectorAll(".content-body p").forEach((p) => {
      if (p.dataset.originalText) {
        p.innerHTML = p.dataset.originalText;
      } else {
        p.dataset.originalText = p.innerHTML;
      }
    });
  };

  // Merge metaKeywords and customKeyWords into the final keywords list.
  useEffect(() => {
    if (isReadMeKeyword === true) {
      setKeywords([...metaKeywords, ...customKeyWords]);
    } else {
      setKeywords([...customKeyWords]);
    }
  }, [metaKeywords, customKeyWords, isReadMeKeyword]);

  // Set up observers and URL change listeners
  useEffect(() => {
    // Run on initial mount
    initializeSearchComponent();

    // Set up a MutationObserver for the meta tag to detect future changes if enabled
    let observer;
    if (isReadMeKeyword === true) {
      const metaTag = document.querySelector('meta[property="keywords"]');
      if (metaTag) {
        observer = new MutationObserver((mutationsList) => {
          for (const mutation of mutationsList) {
            if (mutation.type === "attributes" && mutation.attributeName === "content") {
              console.log("Meta tag content changed:", metaTag.content);
              updateKeywords();
            }
          }
        });
        observer.observe(metaTag, { attributes: true, attributeFilter: ["content"] });
      }
    }

    // Handler to run initialization on URL changes
    const handleUrlChange = () => {
      // Delay allows new content/meta to load
      setTimeout(() => {
        setSearchTerm(""); // Clear any lingering search term/highlights
        initializeSearchComponent();
      }, 500);
    };

    // Monkey-patch pushState and replaceState to dispatch events
    const originalPushState = history.pushState;
    const originalReplaceState = history.replaceState;
    history.pushState = function () {
      originalPushState.apply(history, arguments);
      window.dispatchEvent(new Event("pushState"));
    };
    history.replaceState = function () {
      originalReplaceState.apply(history, arguments);
      window.dispatchEvent(new Event("replaceState"));
    };

    window.addEventListener("popstate", handleUrlChange);
    window.addEventListener("pushState", handleUrlChange);
    window.addEventListener("replaceState", handleUrlChange);
    window.addEventListener("hashchange", handleUrlChange);

    return () => {
      if (observer) observer.disconnect();
      window.removeEventListener("popstate", handleUrlChange);
      window.removeEventListener("pushState", handleUrlChange);
      window.removeEventListener("replaceState", handleUrlChange);
      window.removeEventListener("hashchange", handleUrlChange);
    };
  }, []);

  // Highlight search matches in paragraphs
  useEffect(() => {
    if (searchTerm.length < 3) {
      restoreOriginalText();
      setMatchCount(0);
      return;
    }

    let matchCounter = 0;
    const regex = new RegExp(`(${searchTerm})`, "gi");

    document.querySelectorAll(".content-body p").forEach((p) => {
      if (p.dataset.originalText) {
        const modifiedText = p.dataset.originalText.replace(regex, (match) => {
          matchCounter++;
          return `<span class="highlighted-match" style="background-color: #e1e8ff; font-weight: 600; border-radius: 4px; padding: 2px 3px;">${match}</span>`;
        });
        p.innerHTML = modifiedText;
      }
    });

    setMatchCount(matchCounter);
    console.log(`✅ Total matches found: ${matchCounter}`);

    if (matchCounter > 0) {
      setTimeout(scrollToFirstMatch, 300);
    }
  }, [searchTerm]);

  // Count keyword occurrences in paragraphs
  useEffect(() => {
    if (keywords.length === 0) return;

    const counts = {};
    document.querySelectorAll(".content-body p").forEach((p) => {
      const text = p.innerText;
      keywords.forEach((keyword) => {
        const regex = new RegExp(`\\b${keyword}\\b`, "gi");
        const matches = text.match(regex);
        if (matches) {
          counts[keyword] = (counts[keyword] || 0) + matches.length;
        }
      });
    });

    setKeywordCounts(counts);
    console.log("📊 Keyword Counts:", counts);
  }, [keywords]);

  // Restore paragraphs to original text (remove highlights)
  const restoreOriginalText = () => {
    document.querySelectorAll(".content-body p").forEach((p) => {
      if (p.dataset.originalText) {
        p.innerHTML = p.dataset.originalText;
      }
    });
  };

  // Scroll to the first highlighted match
  const scrollToFirstMatch = () => {
    const matches = document.querySelectorAll(".highlighted-match");
    if (matches.length > 0) {
      matches[0].scrollIntoView({ behavior: "smooth", block: "center" });
      console.log("📍 Scrolled to first match");
    }
  };

  // Handle clicking on a keyword button
  const handleKeywordClick = (keyword) => {
    setSearchTerm(keyword);
  };

  // Inline style objects for better readability
  const containerStyle = {
    width: "100%",
    display: "flex",
    flexDirection: "column",
    gap: "10px",
  };

  const searchContainerStyle = {
    display: "flex",
    alignItems: "center",
    width: "100%",
    gap: "30px",
    position: "relative",
  };

  const inputContainerStyle = {
    flex: "1 1 25%",
    maxWidth: "25%",
    position: "relative",
  };

  const inputStyle = {
    width: "100%",
    padding: "8px",
    fontSize: "14px",
    border: "1px solid #ccc",
    borderRadius: "4px",
  };

  const matchCountStyle = {
    position: "absolute",
    top: "-18px",
    left: "5px",
    fontSize: "12px",
    color: "#555",
    opacity: "0.8",
  };

  const keywordContainerStyle = {
    flex: "1 1 75%",
    maxWidth: "75%",
    display: "flex",
    flexWrap: "wrap",
    gap: "5px",
    justifyContent: "flex-start",
    overflow: "hidden",
    whiteSpace: "nowrap",
  };

  const keywordButtonStyle = (isActive) => ({
    backgroundColor: isActive ? "#e1e8ff" : "#f0f0f0",
    padding: "5px 10px",
    borderRadius: "15px",
    border: "1px solid #ccc",
    cursor: "pointer",
    fontSize: "12px",
    color: "#333",
  });

  return (
    <div style={containerStyle}>
      {/* Search & Keywords Container */}
      <div style={searchContainerStyle}>
        {/* Search Input */}
        <div style={inputContainerStyle}>
          {matchCount > 0 && (
            <span style={matchCountStyle}>
              {matchCount} {matchCount === 1 ? "result" : "results"}
            </span>
          )}
          <input
            type="text"
            placeholder="Search Page..."
            style={inputStyle}
            value={searchTerm}
            onChange={(e) => setSearchTerm(e.target.value)}
          />
        </div>
        {/* Keyword Buttons */}
        <div style={keywordContainerStyle}>
          {keywords.map((keyword) => (
            <button
              key={keyword}
              onClick={() => handleKeywordClick(keyword)}
              style={keywordButtonStyle(searchTerm === keyword)}
            >
              {keyword} ({keywordCounts[keyword] || 0})
            </button>
          ))}
        </div>
      </div>
    </div>
  );
}

<KeywordSearch isReadMeKeyword={true} />
