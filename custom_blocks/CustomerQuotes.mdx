import React, { useState, useMemo } from "react";

export const CustomerQuotes = ({ children, title = "What Our Customers Say" }) => {
  const quotes = useMemo(() => (Array.isArray(children) ? children : [children]), [children]);
  const [currentQuote, setCurrentQuote] = useState(0);

  const containerStyle = {
    width: "100%",
    minHeight: "50vh",
    padding: "40px 20px",
    // Combine a gradient overlay and a background image.
    backgroundImage: `linear-gradient(135deg, rgba(255,255,255,0.2) 0%, rgba(232,243,255,0.5) 100%), url("//files.readme.io/9dbe441-footer-owls.png")`,
    backgroundSize: "cover",
    backgroundPosition: "center",
    textAlign: "center",
    boxSizing: "border-box",
  };

  const quotesCardStyle = {
    maxWidth: "700px",
    margin: "40px auto 0 auto",
    padding: "30px",
    borderRadius: "12px",
    border: "2px solid #3B89FF",
    backgroundColor: "#ffffff8a",
    boxShadow: "0 4px 10px rgba(0, 0, 0, 0.08)",
  };

  const titleStyle = {
    fontSize: "2em",
    fontWeight: "bold",
    marginBottom: "30px",
    color: "#3B89FF",
  };

  const quoteBoxStyle = {
    padding: "30px",
    margin: "20px auto",
    borderRadius: "10px",
    border: "2px dashed #3B89FF",
    backgroundColor: "#FAFAFA",
    boxShadow: "0px 3px 10px rgba(0, 0, 0, 0.05)",
    transition: "transform 0.3s ease, box-shadow 0.3s ease",
    cursor: "pointer",
    fontSize: "1.4em",
    color: "#333",
  };

  const buttonStyle = {
    padding: "12px 24px",
    border: "none",
    borderRadius: "6px",
    backgroundColor: "#3B89FF",
    color: "white",
    cursor: "pointer",
    fontSize: "1.2em",
    marginTop: "20px",
    transition: "background-color 0.3s ease",
  };

  const handleMouseOverButton = (e) => {
    e.currentTarget.style.backgroundColor = "#2C68C1";
  };

  const handleMouseOutButton = (e) => {
    e.currentTarget.style.backgroundColor = "#3B89FF";
  };

  const handleNextQuote = () => setCurrentQuote((prev) => (prev + 1) % quotes.length);

  return (
    <div style={containerStyle}>
      <div style={quotesCardStyle}>
        {title && <div style={titleStyle}>{title}</div>}
        <div
          style={quoteBoxStyle}
          onClick={handleNextQuote}
          onMouseOver={(e) => {
            e.currentTarget.style.transform = "scale(1.02)";
            e.currentTarget.style.boxShadow = "0px 5px 15px rgba(59, 137, 255, 0.3)";
          }}
          onMouseOut={(e) => {
            e.currentTarget.style.transform = "scale(1)";
            e.currentTarget.style.boxShadow = "0px 3px 10px rgba(0, 0, 0, 0.05)";
          }}
        >
          {quotes[currentQuote]}
        </div>
        <button
          type="button"
          onClick={handleNextQuote}
          onMouseOver={handleMouseOverButton}
          onMouseOut={handleMouseOutButton}
          style={buttonStyle}
          aria-label="Show next customer quote"
        >
          Next Quote
        </button>
      </div>
    </div>
  );
};

export const CustomerQuote = ({ author, children }) => (
  <div style={{ marginBottom: "40px" }}>
    <blockquote
      style={{
        fontStyle: "italic",
        margin: "20px 0",
        paddingLeft: "25px",
        borderLeft: "6px solid #3B89FF",
        fontSize: "1.4em",
        lineHeight: "1.4em",
        color: "#555",
      }}
    >
      {children}
    </blockquote>
    {author && (
      <p style={{ textAlign: "right", margin: "0 15px", fontWeight: "bold", fontSize: "1.1em", color: "#333" }}>
        – {author}
      </p>
    )}
  </div>
);


<CustomerQuotes title="Our Happy Customer Moments">
  <CustomerQuote author="Alice">
    ReadMe.io made our documentation sparkle with creativity!
  </CustomerQuote> 
  <CustomerQuote author="Bob">
    The support and experience are as bold as our ideas!
  </CustomerQuote>
  <CustomerQuote author="Charlie">
    Every time I click, I get a fresh dose of inspiration!
  </CustomerQuote>
</CustomerQuotes>
