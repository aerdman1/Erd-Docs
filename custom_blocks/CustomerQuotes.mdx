import React, { useEffect, useState, Children, isValidElement } from "react";

export const CustomerQuote = ({ author, role, company, avatar, children }) => {
  const initials = String(author || "?").split(/\s+/).map(w => w[0]).join("").slice(0, 2).toUpperCase();
  return (
    <figure className="eo-quote">
      <blockquote className="eo-quote-text">{children}</blockquote>
      <figcaption className="eo-quote-who">
        {avatar ? <img className="eo-quote-avatar" src={avatar} alt="" /> : <span className="eo-quote-avatar" aria-hidden="true">{initials}</span>}
        <span>
          <strong>{author}</strong>
          {role || company ? <span className="eo-quote-role">{[role, company].filter(Boolean).join(", ")}</span> : null}
        </span>
      </figcaption>
    </figure>
  );
};

export const CustomerQuotes = rawProps => {
  const props = {};
  Object.keys(rawProps || {}).forEach(k => { props[k.toLowerCase()] = rawProps[k]; });
  props.children = rawProps ? rawProps.children : undefined;
  const quotes = Children.toArray(props.children).filter(c => isValidElement(c));
  const autoplay = !(props.autoplay === false || props.autoplay === "false");
  const seconds = Number(props.interval) > 0 ? Number(props.interval) : 7;
  const [i, setI] = useState(0);
  const [paused, setPaused] = useState(false);
  const n = quotes.length;
  const go = k => setI(((k % n) + n) % n);

  useEffect(() => {
    if (!autoplay || paused || n < 2 || typeof window === "undefined") return undefined;
    if (window.matchMedia && window.matchMedia("(prefers-reduced-motion: reduce)").matches) return undefined;
    const t = window.setTimeout(() => setI(x => (x + 1) % n), seconds * 1000);
    return () => window.clearTimeout(t);
  }, [i, autoplay, paused, n, seconds]);

  if (!n) return null;
  const css = `
.eo-quotes{--eo-accent:var(--color-link-primary,#2F7BF5);--eo-border:var(--color-border-default,rgba(0,0,0,.12));--eo-muted:var(--color-text-minimum,#6b7280);
  position:relative;margin:28px 0;padding:28px 28px 20px;border:1px solid var(--eo-border);border-radius:16px;overflow:hidden;
  background:radial-gradient(120% 140% at 0% 0%,color-mix(in srgb,var(--eo-accent) 10%,transparent) 0%,transparent 60%)}
.eo-quotes-title{margin:0 0 14px;font-size:13px;letter-spacing:.08em;text-transform:uppercase;color:var(--eo-muted);font-weight:700}
.eo-quotes-stage{position:relative;min-height:150px}
.eo-quotes-slide{animation:eo-quote-in .45s ease both}
@keyframes eo-quote-in{from{opacity:0;transform:translateY(8px)}to{opacity:1;transform:none}}
@media (prefers-reduced-motion:reduce){.eo-quotes-slide{animation:none}}
.eo-quote{margin:0}
.eo-quotes .eo-quote-text{margin:0!important;padding:0 0 0 44px!important;border:0!important;background:none!important;position:relative;font-size:1.25em;line-height:1.5;font-weight:500;font-style:normal!important;color:var(--color-text-default,inherit)!important;quotes:none}
.eo-quotes .eo-quote-text::after{content:none!important}
.eo-quotes .eo-quote-text::before{display:block;content:"“";position:absolute;left:0;top:-14px;font-size:64px;line-height:1;color:var(--eo-accent);font-family:Georgia,serif}
.eo-quote-text p{margin:0}
.eo-quotes .eo-quote-who{display:flex!important;align-items:center;gap:12px;margin:18px 0 0 44px!important;padding:0!important;font-style:normal!important;text-align:left!important;color:inherit!important;font-size:15px}
.eo-quotes .eo-quote-avatar{font-style:normal;width:40px;height:40px;border-radius:50%;display:inline-flex;align-items:center;justify-content:center;object-fit:cover;
  background:linear-gradient(135deg,#19C3F5,#2F7BF5 55%,#7B3FF2);color:#fff;font-weight:700;font-size:14px;flex:none}
.eo-quote-role{display:block;color:var(--eo-muted);font-size:13px}
.eo-quotes-nav{display:flex;align-items:center;justify-content:space-between;margin-top:18px}
.eo-quotes-dots{display:flex;gap:8px}
.eo-quotes-dot{width:9px;height:9px;border-radius:50%;border:0;padding:0;background:var(--eo-border);cursor:pointer}
.eo-quotes-dot[aria-current="true"]{background:var(--eo-accent);width:22px;border-radius:9px;transition:width .2s}
.eo-quotes-arrows button{border:1px solid var(--eo-border);background:transparent;color:inherit;width:34px;height:34px;border-radius:50%;cursor:pointer;margin-left:6px;font-size:15px}
.eo-quotes-arrows button:hover{border-color:var(--eo-accent);color:var(--eo-accent)}
`;
  return (
    <section className="eo-quotes" aria-roledescription="carousel" aria-label={props.title || "Testimonials"}
      onMouseEnter={() => setPaused(true)} onMouseLeave={() => setPaused(false)}
      onFocus={() => setPaused(true)} onBlur={() => setPaused(false)}>
      <style>{css}</style>
      {props.title ? <p className="eo-quotes-title">{props.title}</p> : null}
      <div className="eo-quotes-stage" aria-live={paused ? "polite" : "off"}>
        <div key={i} className="eo-quotes-slide">{quotes[i]}</div>
      </div>
      {n > 1 ? (
        <div className="eo-quotes-nav">
          <div className="eo-quotes-dots">
            {quotes.map((q, k) => (
              <button key={k} type="button" className="eo-quotes-dot" aria-label={"Show quote " + (k + 1)} aria-current={k === i ? "true" : "false"} onClick={() => go(k)} />
            ))}
          </div>
          <div className="eo-quotes-arrows">
            <button type="button" aria-label="上一条引言" onClick={() => go(i - 1)}>←</button>
            <button type="button" aria-label="下一条引言" onClick={() => go(i + 1)}>→</button>
          </div>
        </div>
      ) : null}
    </section>
  );
};

</CustomerQuotes>
