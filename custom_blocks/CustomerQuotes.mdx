import React, { useState, useMemo } from "react";

export const CoolCustomerQuotes = ({ children, title = "What Our Customers Say" }) => {
  // Wrap children to ensure we have an array of quotes.
  const quotes = useMemo(() => (Array.isArray(children) ? children : [children]), [children]);
  
  const [currentQuote, setCurrentQuote] = useState(0);
  // flipPhase: 0 = resting, 1 = rotating out, 2 = rotating in.
  const [flipPhase, setFlipPhase] = useState(0);

  // Container with perspective gives a 3D feel.
  const containerStyle = {
    perspective: "1000px",
    maxWidth: "700px",
    margin: "40px auto",
    textAlign: "center",
  };

  const titleStyle = {
    fontSize: "2.5em",
    fontWeight: "bold",
    marginBottom: "20px",
    color: "#5A3E8B",
  };

  // Determine rotation based on phase.
  // Phase 0: resting (0deg); Phase 1: rotating out (90deg); Phase 2: rotating in (0deg)
  const rotation = flipPhase === 1 ? "rotateY(90deg)" : "rotateY(0deg)";

  const cardStyle = {
    position: "relative",
    backgroundColor: "#fff",
    border: "4px dashed #5A3E8B",
    borderRadius: "15px",
    padding: "40px",
    fontSize: "1.8em",
    fontWeight: "bold",
    boxShadow: "0 4px 8px rgba(0,0,0,0.2)",
    transition: "transform 0.6s ease-in-out",
    transformStyle: "preserve-3d",
    transform: rotation,
    cursor: "pointer",
    minHeight: "150px",
    // Adding a little extra fun with a gradient background when resting.
    backgroundImage: flipPhase === 0
      ? "linear-gradient(135deg, #f6d365 0%, #fda085 100%)"
      : "none",
  };

  // This handler implements the two-phase flip:
  // 1. Start Phase 1 (rotate out) and after 300ms update the quote.
  // 2. Then switch to Phase 2 (rotate in) and after another 300ms revert to resting state.
  const handleFlip = () => {
    if (flipPhase !== 0) return; // Prevent new clicks mid-animation.

    setFlipPhase(1);
    setTimeout(() => {
      // Update the quote when the card is edge-on.
      setCurrentQuote((prev) => (prev + 1) % quotes.length);
      // Begin flip-in.
      setFlipPhase(2);
    }, 300); // Half the duration of the full flip.

    setTimeout(() => {
      // Reset to the resting phase.
      setFlipPhase(0);
    }, 600); // Full duration.
  };

  return (
    <div style={containerStyle}>
      {title && <div style={titleStyle}>{title}</div>}
      <div style={cardStyle} onClick={handleFlip}>
        {quotes[currentQuote]}
      </div>
      <button
        type="button"
        onClick={handleFlip}
        style={{
          padding: "12px 24px",
          border: "none",
          borderRadius: "6px",
          backgroundColor: "#5A3E8B",
          color: "white",
          cursor: "pointer",
          fontSize: "1.3em",
          marginTop: "20px",
        }}
        aria-label="Show next customer quote"
      >
        Flip for Next Quote
      </button>
    </div>
  );
};

export const CoolCustomerQuote = ({ author, children }) => (
  <div>
    <blockquote
      style={{
        fontStyle: "italic",
        margin: "20px 0",
        paddingLeft: "25px",
        borderLeft: "6px solid #5A3E8B",
        fontSize: "1.8em",
      }}
    >
      {children}
    </blockquote>
    {author && (
      <p
        style={{
          textAlign: "right",
          margin: "0 15px",
          fontWeight: "bold",
          fontSize: "1.4em",
          color: "#333",
        }}
      >
        – {author}
      </p>
    )}
  </div>
);

<CoolCustomerQuotes title="Our Happy Customers">
  <CoolCustomerQuote author="Alice">
    ReadMe.io made our documentation sparkle with creativity!
  </CoolCustomerQuote>
  <CoolCustomerQuote author="Bob">
    The experience was as bold and exciting as our ideas!
  </CoolCustomerQuote>
  <CoolCustomerQuote author="Charlie">
    Every flip brought a fresh dose of inspiration—simply magical!
  </CoolCustomerQuote>
</CoolCustomerQuotes>
