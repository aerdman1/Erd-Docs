import React, { useState, useMemo } from "react";

export const CustomerQuotes = ({ children, title = "What Our Customers Say" }) => {
  // If the provided children aren’t already an array, wrap them in one.
  const quotes = useMemo(() => (Array.isArray(children) ? children : [children]), [children]);
  const [currentQuote, setCurrentQuote] = useState(0);

  // Container styles
  const containerStyle = {
    maxWidth: "600px",
    margin: "40px auto",
    padding: "20px",
    border: "2px solid #5A3E8B",
    borderRadius: "8px",
    textAlign: "center",
    backgroundColor: "#f9f9f9",
  };

  // Title styles
  const titleStyle = {
    fontSize: "1.5em",
    fontWeight: "bold",
    marginBottom: "20px",
    color: "#5A3E8B",
  };

  // Quote box styling with a fun hover effect
  const quoteBoxStyle = {
    padding: "20px",
    margin: "20px auto",
    borderRadius: "8px",
    border: "1px solid #ddd",
    backgroundColor: "white",
    boxShadow: "0px 2px 8px rgba(0, 0, 0, 0.1)",
    transition: "transform 0.3s ease, box-shadow 0.3s ease",
    cursor: "pointer",
  };

  const handleNextQuote = () =>
    setCurrentQuote((prev) => (prev + 1) % quotes.length);

  return (
    <div style={containerStyle}>
      {title && <div style={titleStyle}>{title}</div>}
      <div
        style={quoteBoxStyle}
        onClick={handleNextQuote}
        onMouseOver={(e) => {
          e.currentTarget.style.transform = "scale(1.02)";
          e.currentTarget.style.boxShadow = "0px 0px 12px rgba(90, 62, 139, 0.3)";
        }}
        onMouseOut={(e) => {
          e.currentTarget.style.transform = "scale(1)";
          e.currentTarget.style.boxShadow = "0px 2px 8px rgba(0, 0, 0, 0.1)";
        }}
      >
        {quotes[currentQuote]}
      </div>
      <button
        type="button"
        onClick={handleNextQuote}
        style={{
          padding: "10px 20px",
          border: "none",
          borderRadius: "4px",
          backgroundColor: "#5A3E8B",
          color: "white",
          cursor: "pointer",
          fontSize: "1em",
        }}
        aria-label="Show next customer quote"
      >
        Show Next Quote
      </button>
    </div>
  );
};

export const CustomerQuote = ({ author, children }) => (
  <div>
    <blockquote
      style={{
        fontStyle: "italic",
        margin: "10px 0",
        paddingLeft: "20px",
        borderLeft: "4px solid #5A3E8B",
      }}
    >
      {children}
    </blockquote>
    {author && (
      <p style={{ textAlign: "right", margin: "0 10px", fontWeight: "bold" }}>
        – {author}
      </p>
    )}
  </div>
);

<CustomerQuotes title="Hear from our happy customers!">
  <CustomerQuote author="Alice">
    ReadMe.io transformed the way we handle documentation!
  </CustomerQuote>
  <CustomerQuote author="Bob">
    The user experience is simply brilliant and fun.
  </CustomerQuote>
  <CustomerQuote author="Charlie">
    We love how customizable and intuitive ReadMe.io is.
  </CustomerQuote>
</CustomerQuotes>
