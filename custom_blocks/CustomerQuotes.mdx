import React, { useState, useMemo } from "react";

export const CustomerQuotes = ({ children, title = "What Our Customers Say" }) => {
  // Ensure children is always an array
  const quotes = useMemo(() => (Array.isArray(children) ? children : [children]), [children]);
  const [currentQuote, setCurrentQuote] = useState(0);

  // Container styling with larger and lively typography
  const containerStyle = {
    maxWidth: "700px",
    margin: "40px auto",
    padding: "30px",
    border: "2px solid #5A3E8B",
    borderRadius: "10px",
    textAlign: "center",
    backgroundColor: "#f0f8ff",
  };

  // Title styling with bigger font and vibrant color
  const titleStyle = {
    fontSize: "2em",
    fontWeight: "bold",
    marginBottom: "30px",
    color: "#5A3E8B",
  };

  // Quote box styling with enlarged text and a playful hover effect
  const quoteBoxStyle = {
    padding: "30px",
    margin: "20px auto",
    borderRadius: "10px",
    border: "2px dashed #5A3E8B",
    backgroundColor: "white",
    boxShadow: "0px 3px 10px rgba(0, 0, 0, 0.15)",
    transition: "transform 0.3s ease, box-shadow 0.3s ease",
    cursor: "pointer",
    fontSize: "1.4em",
  };

  const handleNextQuote = () => setCurrentQuote((prev) => (prev + 1) % quotes.length);

  return (
    <div style={containerStyle}>
      {title && <div style={titleStyle}>{title}</div>}
      <div
        style={quoteBoxStyle}
        onClick={handleNextQuote}
        onMouseOver={(e) => {
          e.currentTarget.style.transform = "scale(1.03)";
          e.currentTarget.style.boxShadow = "0px 0px 15px rgba(90, 62, 139, 0.4)";
        }}
        onMouseOut={(e) => {
          e.currentTarget.style.transform = "scale(1)";
          e.currentTarget.style.boxShadow = "0px 3px 10px rgba(0, 0, 0, 0.15)";
        }}
      >
        {quotes[currentQuote]}
      </div>
      <button
        type="button"
        onClick={handleNextQuote}
        style={{
          padding: "12px 24px",
          border: "none",
          borderRadius: "6px",
          backgroundColor: "#5A3E8B",
          color: "white",
          cursor: "pointer",
          fontSize: "1.2em",
          marginTop: "20px",
        }}
        aria-label="Show next customer quote"
      >
        Next Quote
      </button>
    </div>
  );
};

export const CustomerQuote = ({ author, children }) => (
  <div>
    <blockquote
      style={{
        fontStyle: "italic",
        margin: "20px 0",
        paddingLeft: "25px",
        borderLeft: "6px solid #5A3E8B",
        fontSize: "1.5em",
      }}
    >
      {children}
    </blockquote>
    {author && (
      <p style={{ textAlign: "right", margin: "0 15px", fontWeight: "bold", fontSize: "1.2em", color: "#333" }}>
        – {author}
      </p>
    )}
  </div>
);

<CustomerQuotes title="Our Happy Customer Moments">
  <CustomerQuote author="Alice">
    ReadMe.io made our documentation sparkle with creativity!
  </CustomerQuote>
  <CustomerQuote author="Bob">
    The support and experience are as bold as our ideas!
  </CustomerQuote>
  <CustomerQuote author="Charlie">
    Every time I click, I get a fresh dose of inspiration!
  </CustomerQuote>
</CustomerQuotes>
