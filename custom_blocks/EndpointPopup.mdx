import React, { useEffect, useState } from 'react';

export const EndpointPopup = ({ 
  endpoint, 
  displayMode = 'popup', 
  linkText, 
  specFile = "ErdSpec1.json" 
}) => {
  const [endpointData, setEndpointData] = useState(null);
  const [error, setError] = useState(null);
  const [loading, setLoading] = useState(false);
  const [isHovered, setIsHovered] = useState(false);

  // Default link text to the endpoint value if none is provided.
  const displayLinkText = linkText || endpoint;
  // Build the full spec URL using the specFile prop.
  const fullSpecUrl = `https://erdtest123.readme.io/openapi/${specFile}`;

  useEffect(() => {
    if (!endpoint) {
      setError('Endpoint not provided.');
      return;
    }
    fetchEndpointExample();
  }, [endpoint, fullSpecUrl]);

  const fetchEndpointExample = async () => {
    setLoading(true);
    setError(null);
    try {
      const response = await fetch(fullSpecUrl);
      if (!response.ok) {
        throw new Error("This endpoint may be private or restricted.");
      }
      const spec = await response.json();
      const operations = spec.paths[endpoint];
      if (!operations) {
        throw new Error(`No endpoint found in spec for path ${endpoint}`);
      }

      let chosenOperation = null;
      let chosenMethod = null;
      for (const [method, operation] of Object.entries(operations)) {
        const example = extractResponseExample(operation, spec.components);
        if (example) {
          chosenOperation = operation;
          chosenMethod = method.toUpperCase();
          break;
        }
      }

      if (!chosenOperation) {
        const [method, operation] = Object.entries(operations)[0];
        chosenOperation = operation;
        chosenMethod = method.toUpperCase();
      }

      const summary = chosenOperation.summary || '';
      const description = chosenOperation.description || '';
      const example = extractResponseExample(chosenOperation, spec.components);

      setEndpointData({
        path: endpoint,
        method: chosenMethod,
        summary,
        description,
        example,
      });
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };

  // Helper: Recursively resolve an example from a schema.
  const resolveExampleFromSchema = (schema, components) => {
    if (!schema) return null;
    if (schema.example !== undefined) {
      return schema.example;
    }
    if (schema.$ref) {
      const refParts = schema.$ref.split('/');
      const schemaName = refParts[refParts.length - 1];
      const resolved = components.schemas[schemaName];
      return resolveExampleFromSchema(resolved, components);
    }
    if (schema.type === 'object' && schema.properties) {
      const obj = {};
      Object.entries(schema.properties).forEach(([prop, propSchema]) => {
        obj[prop] = resolveExampleFromSchema(propSchema, components);
      });
      return obj;
    }
    if (schema.type === 'array' && schema.items) {
      return [resolveExampleFromSchema(schema.items, components)];
    }
    return null;
  };

  // Helper: Extract an example from an operation’s responses.
  const extractResponseExample = (operation, components) => {
    if (!operation.responses) return null;
    let response = operation.responses['200'] || operation.responses['default'];
    if (!response) {
      response = Object.values(operation.responses)[0];
    }
    if (response && response.content) {
      const media = response.content['application/json'];
      if (media) {
        if (media.example) {
          return media.example;
        }
        if (media.examples) {
          const firstExample = Object.values(media.examples)[0];
          if (firstExample && firstExample.value) return firstExample.value;
        }
        if (media.schema) {
          return resolveExampleFromSchema(media.schema, components);
        }
      }
    }
    return null;
  };

  // Base card styling.
  const baseCardStyle = {
    border: '1px solid #ddd',
    background: '#fff',
    padding: '16px',
    borderRadius: '8px',
    boxShadow: '0px 0px 10px rgba(0,0,0,0.1)',
  };

  // Use full width for inline mode; fixed 400px width for popup mode.
  const cardWidthStyle = displayMode === 'inline'
    ? { width: '100%' }
    : { width: '400px' };

  // For popup mode, the card is absolutely positioned so that it overlays content without moving it.
  const popupPositionStyle = displayMode === 'popup'
    ? { position: 'absolute', top: '100%', left: '0', zIndex: 1000, marginTop: '4px' }
    : {};

  const cardStyle = { ...baseCardStyle, ...cardWidthStyle, ...popupPositionStyle };

  // Custom link styling so it stands out.
  const customLinkStyle = {
    textDecoration: 'none',
    cursor: 'pointer',
    color: '#1a73e8',
    backgroundColor: '#e8f0fe',
    borderRadius: '4px',
    padding: '2px 4px',
    fontWeight: '600',
  };

  // Render the content card.
  const renderedContent = endpointData && (
    <div style={cardStyle}>
      <h3>
        <a
          href={`https://erdtest123.readme.io/reference/${endpoint.replace('/', '')}`}
          style={customLinkStyle}
        >
          {endpointData.method} {endpointData.path}
        </a>
      </h3>
      {endpointData.summary && <p><strong>Summary:</strong> {endpointData.summary}</p>}
      {endpointData.description && <p><strong>Description:</strong> {endpointData.description}</p>}
      {endpointData.example ? (
        <div>
          <strong>Example Response:</strong>
          <pre
            style={{
              whiteSpace: 'pre-wrap',
              padding: '8px',
              borderRadius: '4px',
              marginTop: '8px'
            }}
          >
            {JSON.stringify(endpointData.example, null, 2)}
          </pre>
        </div>
      ) : (
        <p>No example available.</p>
      )}
    </div>
  );

  if (displayMode === 'popup') {
    return (
      <span
        style={{ display: 'inline-block', position: 'relative' }}
        onMouseEnter={() => setIsHovered(true)}
        onMouseLeave={() => setIsHovered(false)}
      >
        <a
          href={`https://erdtest123.readme.io/reference/${endpoint.replace('/', '')}`}
          style={customLinkStyle}
        >
          {displayLinkText}
        </a>
        {isHovered && (
          <>
            {loading && (
              <div
                style={{
                  ...cardStyle,
                  position: 'absolute',
                  top: '100%',
                  left: '0',
                  zIndex: 1000,
                  marginTop: '4px'
                }}
              >
                Loading endpoint example...
              </div>
            )}
            {error && (
              <div
                style={{
                  ...cardStyle,
                  color: 'red',
                  position: 'absolute',
                  top: '100%',
                  left: '0',
                  zIndex: 1000,
                  marginTop: '4px'
                }}
              >
                {error}
              </div>
            )}
            {!loading && !error && renderedContent}
          </>
        )}
      </span>
    );
  }

  return (
    <div style={{ marginBottom: '16px', width: '100%' }}>
      {loading && <div>Loading endpoint example...</div>}
      {error && <div style={{ color: 'red' }}>{error}</div>}
      {!loading && !error && renderedContent}
    </div>
  );
};

## API Endpoint Popup Examples

### Popup Version (Link)

When you want to keep your page uncluttered, you can use the "popup" display mode. In this mode, the endpoint is shown as a hyperlink. When a user hovers over this <EndpointPopup endpoint="/pet" specFile="ErdSpec2.json" linkText="Pet Endpoint"/>, it displays detailed information about the endpoint in a popup.

Hover over the link above to see a quick summary, description, and an example response for the `/pet` endpoint. This version is ideal if you want to provide more context only when needed.

### Inline Version

If you prefer to show the API details right away without requiring any interaction, you can set the component to the "inline" mode. This will render both the hyperlink and all the endpoint details directly on the page. For example:

<EndpointPopup endpoint="/pet" specFile="ErdSpec2.json" displayMode="inline" />

With the inline version, readers see the complete endpoint information immediately. This is useful in sections where full details are critical to understanding the API behavior.
